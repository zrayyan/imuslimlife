بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ

  1. كٓهيعٓصٓ ۝١

    کہیعص

  2. ذِكْرُ رَحْمَتِ رَبِّكَ عَبْدَهُۥ زَكَرِيَّآ ۝٢

    یہ ہے تیرے پروردگار کی اس مہربانی کا ذکر جو اس نے اپنے بندے زکریا پر کی تھی

  3. إِذْ نَادَىٰ رَبَّهُۥ نِدَآءً خَفِيًّا ۝٣

    جبکہ اس نے اپنے رب سے چپکے چپکے دعا کی تھی

  4. قَالَ رَبِّ إِنِّى وَهَنَ ٱلْعَظْمُ مِنِّى وَٱشْتَعَلَ ٱلرَّأْسُ شَيْبًا وَلَمْ أَكُنۢ بِدُعَآئِكَ رَبِّ شَقِيًّا ۝٤

    کہ اے میرے پروردگار! میری ہڈیاں کمزور ہوگئی ہیں اور سر بڑھاپے کی وجہ سے بھڑک اٹھا ہے، لیکن میں کبھی بھی تجھ سے دعا کر کے محروم نہیں رہا

  5. وَإِنِّى خِفْتُ ٱلْمَوَٲلِىَ مِن وَرَآءِى وَكَانَتِ ٱمْرَأَتِى عَاقِرًا فَهَبْ لِى مِن لَّدُنكَ وَلِيًّا ۝٥

    مجھے اپنے مرنے کے بعد اپنے قرابت والوں کا ڈر ہے، میری بیوی بھی بانجھ ہے پس تو مجھے اپنے پاس سے وارث عطا فرما

  6. يَرِثُنِى وَيَرِثُ مِنْ ءَالِ يَعْقُوبَ‌ۖ وَٱجْعَلْهُ رَبِّ رَضِيًّا ۝٦

    جو میرا بھی وارث ہو اور یعقوب (علیہ السلام) کے خاندان کا بھی جانشین اور میرے رب! تو اسے مقبول بنده بنا لے

  7. يَـٰزَكَرِيَّآ إِنَّا نُبَشِّرُكَ بِغُلَـٰمٍ ٱسْمُهُۥ يَحْيَىٰ لَمْ نَجْعَل لَّهُۥ مِن قَبْلُ سَمِيًّا ۝٧

    اے زکریا! ہم تجھے ایک بچے کی خوشخبری دیتے ہیں جس کا نام یحيٰ ہے، ہم نے اس سے پہلے اس کا ہم نام بھی کسی کو نہیں کیا

  8. قَالَ رَبِّ أَنَّىٰ يَكُونُ لِى غُلَـٰمٌ وَكَانَتِ ٱمْرَأَتِى عَاقِرًا وَقَدْ بَلَغْتُ مِنَ ٱلْكِبَرِ عِتِيًّا ۝٨

    زکریا (علیہ السلام) کہنے لگے میرے رب! میرے ہاں لڑکا کیسے ہوگا، جبکہ میری بیوی بانجھ اور میں خود بڑھاپے کے انتہائی ضعف کو پہنچ چکا ہوں

  9. قَالَ كَذَٲلِكَ قَالَ رَبُّكَ هُوَ عَلَىَّ هَيِّنٌ وَقَدْ خَلَقْتُكَ مِن قَبْلُ وَلَمْ تَكُ شَيْــًٔا ۝٩

    ارشاد ہوا کہ وعده اسی طرح ہو چکا، تیرے رب نے فرما دیا ہے کہ مجھ پر تو یہ بالکل آسان ہے اور تو خود جبکہ کچھ نہ تھا میں تجھے پیدا کر چکا ہوں

  10. قَالَ رَبِّ ٱجْعَل لِّىٓ ءَايَةً‌ۚ قَالَ ءَايَتُكَ أَلَّا تُكَلِّمَ ٱلنَّاسَ ثَلَـٰثَ لَيَالٍ سَوِيًّا ۝١٠

    کہنے لگے میرے پروردگار میرے لئے کوئی علامت مقرر فرما دے، ارشاد ہوا کہ تیرے لئے علامت یہ ہے کہ باوجود بھلا چنگا ہونے کے تو تین راتوں تک کسی شخص سے بول نہ سکے گا

  11. فَخَرَجَ عَلَىٰ قَوْمِهِۦ مِنَ ٱلْمِحْرَابِ فَأَوْحَىٰٓ إِلَيْهِمْ أَن سَبِّحُواْ بُكْرَةً وَعَشِيًّا ۝١١

    اب زکریا (علیہ السلام) اپنے حجرے سے نکل کر اپنی قوم کے پاس آکر انہیں اشاره کرتے ہیں کہ تم صبح وشام اللہ تعالیٰ کی تسبیح بیان کرو

  12. يَـٰيَحْيَىٰ خُذِ ٱلْكِتَـٰبَ بِقُوَّةٍ‌ۖ وَءَاتَيْنَـٰهُ ٱلْحُكْمَ صَبِيًّا ۝١٢

    اے یحيٰ! میری کتاب کو مضبوطی سے تھام لے اور ہم نے اسے لڑکپن ہی سے دانائی عطا فرما دی

  13. وَحَنَانًا مِّن لَّدُنَّا وَزَكَوٲةً‌ۖ وَكَانَ تَقِيًّا ۝١٣

    اور اپنے پاس سے شفقت اور پاکیزگی بھی، وه پرہیزگار شخص تھا

  14. وَبَرَّۢا بِوَٲلِدَيْهِ وَلَمْ يَكُن جَبَّارًا عَصِيًّا ۝١٤

    اور اپنے ماں باپ سے نیک سلوک کرنے واﻻ تھا وه سرکش اور گناه گار نہ تھا

  15. وَسَلَـٰمٌ عَلَيْهِ يَوْمَ وُلِدَ وَيَوْمَ يَمُوتُ وَيَوْمَ يُبْعَثُ حَيًّا ۝١٥

    اس پر سلام ہے جس دن وه پیدا ہوا اور جس دن وه مرے اور جس دن وه زنده کرکے اٹھایا جائے

  16. وَٱذْكُرْ فِى ٱلْكِتَـٰبِ مَرْيَمَ إِذِ ٱنتَبَذَتْ مِنْ أَهْلِهَا مَكَانًا شَرْقِيًّا ۝١٦

    اس کتاب میں مریم کا بھی واقعہ بیان کر۔ جبکہ وه اپنے گھر کے لوگوں سے علیحده ہو کر مشرقی جانب آئیں

  17. فَٱتَّخَذَتْ مِن دُونِهِمْ حِجَابًا فَأَرْسَلْنَآ إِلَيْهَا رُوحَنَا فَتَمَثَّلَ لَهَا بَشَرًا سَوِيًّا ۝١٧

    اور ان لوگوں کی طرف سے پرده کر لیا، پھر ہم نے اس کے پاس اپنی روح (جبرائیل علیہ السلام) کو بھیجا پس وه اس کے سامنے پورا آدمی بن کر ﻇاہر ہوا

  18. قَالَتْ إِنِّىٓ أَعُوذُ بِٱلرَّحْمَـٰنِ مِنكَ إِن كُنتَ تَقِيًّا ۝١٨

    یہ کہنے لگیں میں تجھ سے رحمٰن کی پناه مانگتی ہوں اگر تو کچھ بھی اللہ سے ڈرنے واﻻ ہے

  19. قَالَ إِنَّمَآ أَنَا۟ رَسُولُ رَبِّكِ لِأَهَبَ لَكِ غُلَـٰمًا زَكِيًّا ۝١٩

    اس نے جواب دیا کہ میں تو اللہ کا بھیجا ہوا قاصد ہوں، تجھے ایک پاکیزه لڑکا دینے آیا ہوں

  20. قَالَتْ أَنَّىٰ يَكُونُ لِى غُلَـٰمٌ وَلَمْ يَمْسَسْنِى بَشَرٌ وَلَمْ أَكُ بَغِيًّا ۝٢٠

    کہنے لگیں بھلا میرے ہاں بچہ کیسے ہو سکتا ہے؟ مجھے تو کسی انسان کا ہاتھ تک نہیں لگا اور نہ میں بدکار ہوں

  21. قَالَ كَذَٲلِكِ قَالَ رَبُّكِ هُوَ عَلَىَّ هَيِّنٌ‌ۖ وَلِنَجْعَلَهُۥٓ ءَايَةً لِّلنَّاسِ وَرَحْمَةً مِّنَّا‌ۚ وَكَانَ أَمْرًا مَّقْضِيًّا ۝٢١

    اس نے کہا بات تو یہی ہے۔ لیکن تیرے پروردگار کا ارشاد ہے کہ وه مجھ پر بہت ہی آسان ہے ہم تو اسے لوگوں کے لئے ایک نشانی بنا دیں گے اور اپنی خاص رحمت، یہ تو ایک طے شده بات ہے

  22. ۞ فَحَمَلَتْهُ فَٱنتَبَذَتْ بِهِۦ مَكَانًا قَصِيًّا ۝٢٢

    پس وه حمل سے ہو گئیں اوراسی وجہ سے وه یکسو ہو کر ایک دور کی جگہ چلی گئیں

  23. فَأَجَآءَهَا ٱلْمَخَاضُ إِلَىٰ جِذْعِ ٱلنَّخْلَةِ قَالَتْ يَـٰلَيْتَنِى مِتُّ قَبْلَ هَـٰذَا وَكُنتُ نَسْيًا مَّنسِيًّا ۝٢٣

    پھر درِد زه ایک کھجور کے تنے کے نیچے لے آیا، بولی کاش! میں اس سے پہلے ہی مر گئی ہوتی اور لوگوں کی یاد سے بھی بھولی بسری ہو جاتی

  24. فَنَادَٮٰهَا مِن تَحْتِهَآ أَلَّا تَحْزَنِى قَدْ جَعَلَ رَبُّكِ تَحْتَكِ سَرِيًّا ۝٢٤

    اتنے میں اسے نیچے سے ہی آواز دی کہ آزرده خاطر نہ ہو، تیرے رب نے تیرے پاؤں تلے ایک چشمہ جاری کر دیا ہے

  25. وَهُزِّىٓ إِلَيْكِ بِجِذْعِ ٱلنَّخْلَةِ تُسَـٰقِطْ عَلَيْكِ رُطَبًا جَنِيًّا ۝٢٥

    اور اس کھجور کے تنے کو اپنی طرف ہلا، یہ تیرے سامنے تروتازه پکی کھجوریں گرا دے گا

  26. فَكُلِى وَٱشْرَبِى وَقَرِّى عَيْنًا‌ۖ فَإِمَّا تَرَيِنَّ مِنَ ٱلْبَشَرِ أَحَدًا فَقُولِىٓ إِنِّى نَذَرْتُ لِلرَّحْمَـٰنِ صَوْمًا فَلَنْ أُكَلِّمَ ٱلْيَوْمَ إِنسِيًّا ۝٢٦

    اب چین سے کھا پی اور آنکھیں ٹھنڈی رکھ، اگر تجھے کوئی انسان نظر پڑ جائے تو کہہ دینا کہ میں نے اللہ رحمنٰ کے نام کا روزه مان رکھا ہے۔ میں آج کسی شخص سے بات نہ کروں گی

  27. فَأَتَتْ بِهِۦ قَوْمَهَا تَحْمِلُهُۥ‌ۖ قَالُواْ يَـٰمَرْيَمُ لَقَدْ جِئْتِ شَيْــًٔا فَرِيًّا ۝٢٧

    اب حضرت عیسیٰ (علیہ السلام) کو لئے ہوئے وه اپنی قوم کے پاس آئیں۔ سب کہنے لگے مریم تو نے بڑی بری حرکت کی

  28. يَـٰٓأُخْتَ هَـٰرُونَ مَا كَانَ أَبُوكِ ٱمْرَأَ سَوْءٍ وَمَا كَانَتْ أُمُّكِ بَغِيًّا ۝٢٨

    اے ہارون کی بہن! نہ تو تیرا باپ برا آدمی تھا اور نہ تیری ماں بدکار تھی

  29. فَأَشَارَتْ إِلَيْهِ‌ۖ قَالُواْ كَيْفَ نُكَلِّمُ مَن كَانَ فِى ٱلْمَهْدِ صَبِيًّا ۝٢٩

    مریم نے اپنے بچے کی طرف اشاره کیا۔ سب کہنے لگے کہ لو بھلا ہم گود کے بچے سے باتیں کیسے کریں؟

  30. قَالَ إِنِّى عَبْدُ ٱللَّهِ ءَاتَـٰنِىَ ٱلْكِتَـٰبَ وَجَعَلَنِى نَبِيًّا ۝٣٠

    بچہ بول اٹھا کہ میں اللہ تعالیٰ کا بنده ہوں۔ اس نے مجھے کتاب عطا فرمائی اور مجھے اپنا پیغمبر بنایا ہے

  31. وَجَعَلَنِى مُبَارَكًا أَيْنَ مَا كُنتُ وَأَوْصَـٰنِى بِٱلصَّلَوٲةِ وَٱلزَّكَوٲةِ مَا دُمْتُ حَيًّا ۝٣١

    اور اس نے مجھے بابرکت کیا ہے جہاں بھی میں ہوں، اور اس نے مجھے نماز اور زکوٰة کا حکم دیا ہے جب تک بھی میں زنده رہوں

  32. وَبَرَّۢا بِوَٲلِدَتِى وَلَمْ يَجْعَلْنِى جَبَّارًا شَقِيًّا ۝٣٢

    اور اس نے مجھے اپنی والده کا خدمت گزار بنایا ہے اور مجھے سرکش اور بدبخت نہیں کیا

  33. وَٱلسَّلَـٰمُ عَلَىَّ يَوْمَ وُلِدتُّ وَيَوْمَ أَمُوتُ وَيَوْمَ أُبْعَثُ حَيًّا ۝٣٣

    اور مجھ پر میری پیدائش کے دن اور میری موت کے دن اور جس دن کہ میں دوباره زنده کھڑا کیا جاؤں گا، سلام ہی سلام ہے

  34. ذَٲلِكَ عِيسَى ٱبْنُ مَرْيَمَ‌ۚ قَوْلَ ٱلْحَقِّ ٱلَّذِى فِيهِ يَمْتَرُونَ ۝٣٤

    یہ ہے صحیح واقعہ عیسیٰ بن مریم (علیہ السلام) کا، یہی ہے وه حق بات جس میں لوگ شک وشبہ میں مبتلا ہیں

  35. مَا كَانَ لِلَّهِ أَن يَتَّخِذَ مِن وَلَدٍ‌ۖ سُبْحَـٰنَهُۥٓ‌ۚ إِذَا قَضَىٰٓ أَمْرًا فَإِنَّمَا يَقُولُ لَهُۥ كُن فَيَكُونُ ۝٣٥

    اللہ تعالیٰ کے لئے اوﻻد کا ہونا ﻻئق نہیں، وه تو بالکل پاک ذات ہے، وه تو جب کسی کام کے سر انجام دینے کا اراده کرتا ہے تو اسے کہہ دیتا ہے کہ ہو جا، وه اسی وقت ہو جاتا ہے

  36. وَإِنَّ ٱللَّهَ رَبِّى وَرَبُّكُمْ فَٱعْبُدُوهُ‌ۚ هَـٰذَا صِرَٲطٌ مُّسْتَقِيمٌ ۝٣٦

    میرا اور تم سب کا پروردگار صرف اللہ تعالیٰ ہی ہے۔ تم سب اسی کی عبادت کرو، یہی سیدھی راه ہے

  37. فَٱخْتَلَفَ ٱلْأَحْزَابُ مِنۢ بَيْنِهِمْ‌ۖ فَوَيْلٌ لِّلَّذِينَ كَفَرُواْ مِن مَّشْهَدِ يَوْمٍ عَظِيمٍ ۝٣٧

    پھر یہ فرقے آپس میں اختلاف کرنے لگے، پس کافروں کے لئے ویل ہے ایک بڑے (سخت) دن کی حاضری سے

  38. أَسْمِعْ بِهِمْ وَأَبْصِرْ يَوْمَ يَأْتُونَنَا‌ۖ لَـٰكِنِ ٱلظَّـٰلِمُونَ ٱلْيَوْمَ فِى ضَلَـٰلٍ مُّبِينٍ ۝٣٨

    کیا خوب دیکھنے سننے والے ہوں گے اس دن جبکہ ہمارے سامنے حاضر ہوں گے، لیکن آج تو یہ ﻇالم لوگ صریح گمراہی میں پڑے ہوئے ہیں

  39. وَأَنذِرْهُمْ يَوْمَ ٱلْحَسْرَةِ إِذْ قُضِىَ ٱلْأَمْرُ وَهُمْ فِى غَفْلَةٍ وَهُمْ لَا يُؤْمِنُونَ ۝٣٩

    تو انہیں اس رنج وافسوس کے دن کا ڈر سنا دے جبکہ کام انجام کو پہنچا دیا جائے گا، اور یہ لوگ غفلت اور بے ایمانی میں ہی ره جائیں گے

  40. إِنَّا نَحْنُ نَرِثُ ٱلْأَرْضَ وَمَنْ عَلَيْهَا وَإِلَيْنَا يُرْجَعُونَ ۝٤٠

    خود زمین کے اور تمام زمین والوں کے وارث ہم ہی ہوں گے اور سب لوگ ہماری ہی طرف لوٹا کر ﻻئے جائیں گے

  41. وَٱذْكُرْ فِى ٱلْكِتَـٰبِ إِبْرَٲهِيمَ‌ۚ إِنَّهُۥ كَانَ صِدِّيقًا نَّبِيًّا ۝٤١

    اس کتاب میں ابراہیم (علیہ السلام) کا قصہ بیان کر، بیشک وه بڑی سچائی والے پیغمبر تھے

  42. إِذْ قَالَ لِأَبِيهِ يَـٰٓأَبَتِ لِمَ تَعْبُدُ مَا لَا يَسْمَعُ وَلَا يُبْصِرُ وَلَا يُغْنِى عَنكَ شَيْــًٔا ۝٤٢

    جبکہ انہوں نے اپنے باپ سے کہا کہ ابّا جان! آپ ان کی پوجا پاٹ کیوں کر رہے ہیں جو نہ سنیں نہ دیکھیں؟ نہ آپ کو کچھ بھی فائده پہنچا سکیں

  43. يَـٰٓأَبَتِ إِنِّى قَدْ جَآءَنِى مِنَ ٱلْعِلْمِ مَا لَمْ يَأْتِكَ فَٱتَّبِعْنِىٓ أَهْدِكَ صِرَٲطًا سَوِيًّا ۝٤٣

    میرے مہربان باپ! آپ دیکھیئے میرے پاس وه علم آیا ہے جو آپ کے پاس آیا ہی نہیں، تو آپ میری ہی مانیں میں بالکل سیدھی راه کی طرف آپ کی رہبری کروں گا

  44. يَـٰٓأَبَتِ لَا تَعْبُدِ ٱلشَّيْطَـٰنَ‌ۖ إِنَّ ٱلشَّيْطَـٰنَ كَانَ لِلرَّحْمَـٰنِ عَصِيًّا ۝٤٤

    میرے ابّا جان آپ شیطان کی پرستش سے باز آجائیں شیطان تو رحم وکرم والے اللہ تعالیٰ کا بڑا ہی نافرمان ہے

  45. يَـٰٓأَبَتِ إِنِّىٓ أَخَافُ أَن يَمَسَّكَ عَذَابٌ مِّنَ ٱلرَّحْمَـٰنِ فَتَكُونَ لِلشَّيْطَـٰنِ وَلِيًّا ۝٤٥

    ابّا جان! مجھے خوف لگا ہوا کہ کہیں آپ پر کوئی عذاب الٰہی نہ آپڑے کہ آپ شیطان کے ساتھی بن جائیں

  46. قَالَ أَرَاغِبٌ أَنتَ عَنْ ءَالِهَتِى يَـٰٓإِبْرَٲهِيمُ‌ۖ لَئِن لَّمْ تَنتَهِ لَأَرْجُمَنَّكَ‌ۖ وَٱهْجُرْنِى مَلِيًّا ۝٤٦

    اس نے جواب دیا کہ اے ابراہیم! کیا تو ہمارے معبودوں سے روگردانی کر رہا ہے۔ سن اگر تو باز نہ آیا تو میں تجھے پتھروں سے مار ڈالوں گا، جا ایک مدت دراز تک مجھ سے الگ ره

  47. قَالَ سَلَـٰمٌ عَلَيْكَ‌ۖ سَأَسْتَغْفِرُ لَكَ رَبِّىٓ‌ۖ إِنَّهُۥ كَانَ بِى حَفِيًّا ۝٤٧

    کہا اچھا تم پر سلام ہو، میں تو اپنے پروردگار سے تمہاری بخشش کی دعا کرتا رہوں گا، وه مجھ پر حد درجہ مہربان ہے

  48. وَأَعْتَزِلُكُمْ وَمَا تَدْعُونَ مِن دُونِ ٱللَّهِ وَأَدْعُواْ رَبِّى عَسَىٰٓ أَلَّآ أَكُونَ بِدُعَآءِ رَبِّى شَقِيًّا ۝٤٨

    میں تو تمہیں بھی اور جن جن کو تم اللہ تعالیٰ کے سوا پکارتے ہو انہیں بھی سب کو چھوڑ رہا ہوں۔ صرف اپنے پروردگار کو پکارتا رہوں گا، مجھے یقین ہے کہ میں اپنے پروردگار سے دعا مانگ کر محروم نہ رہوں گا

  49. فَلَمَّا ٱعْتَزَلَهُمْ وَمَا يَعْبُدُونَ مِن دُونِ ٱللَّهِ وَهَبْنَا لَهُۥٓ إِسْحَـٰقَ وَيَعْقُوبَ‌ۖ وَكُلاًّ جَعَلْنَا نَبِيًّا ۝٤٩

    جب ابراہیم (علیہ السلام) ان سب کو اور اللہ کے سوا ان کے سب معبودوں کو چھوڑ چکے تو ہم نے انہیں اسحاق ویعقوب (علیہما السلام) عطا فرمائے، اور دونوں کو نبی بنا دیا

  50. وَوَهَبْنَا لَهُم مِّن رَّحْمَتِنَا وَجَعَلْنَا لَهُمْ لِسَانَ صِدْقٍ عَلِيًّا ۝٥٠

    اور ان سب کو ہم نے اپنی بہت سی رحمتیں عطا فرمائیں اور ہم نے ان کے ذکر جمیل کو بلند درجے کا کر دیا

  51. وَٱذْكُرْ فِى ٱلْكِتَـٰبِ مُوسَىٰٓ‌ۚ إِنَّهُۥ كَانَ مُخْلَصًا وَكَانَ رَسُولاً نَّبِيًّا ۝٥١

    اس قرآن میں موسیٰ (علیہ السلام) کا ذکر بھی کر، جو چنا ہوا اور رسول اور نبی تھا

  52. وَنَـٰدَيْنَـٰهُ مِن جَانِبِ ٱلطُّورِ ٱلْأَيْمَنِ وَقَرَّبْنَـٰهُ نَجِيًّا ۝٥٢

    ہم نے اسے طور کی دائیں جانب سے ندا کی اور راز گوئی کرتے ہوئے اسے قریب کر لیا

  53. وَوَهَبْنَا لَهُۥ مِن رَّحْمَتِنَآ أَخَاهُ هَـٰرُونَ نَبِيًّا ۝٥٣

    اور اپنی خاص مہربانی سے اس کے بھائی کو نبی بنا کر عطا فرمایا

  54. وَٱذْكُرْ فِى ٱلْكِتَـٰبِ إِسْمَـٰعِيلَ‌ۚ إِنَّهُۥ كَانَ صَادِقَ ٱلْوَعْدِ وَكَانَ رَسُولاً نَّبِيًّا ۝٥٤

    اس کتاب میں اسماعیل (علیہ السلام) کا واقعہ بھی بیان کر، وه بڑا ہی وعدے کا سچا تھا اور تھا بھی رسول اور نبی

  55. وَكَانَ يَأْمُرُ أَهْلَهُۥ بِٱلصَّلَوٲةِ وَٱلزَّكَوٲةِ وَكَانَ عِندَ رَبِّهِۦ مَرْضِيًّا ۝٥٥

    وه اپنے گھر والوں کو برابر نماز اور زکوٰة کاحکم دیتا تھا، اور تھا بھی اپنے پروردگار کی بارگاه میں پسندیده اور مقبول

  56. وَٱذْكُرْ فِى ٱلْكِتَـٰبِ إِدْرِيسَ‌ۚ إِنَّهُۥ كَانَ صِدِّيقًا نَّبِيًّا ۝٥٦

    اور اس کتاب میں ادریس (علیہ السلام) کا بھی ذکر کر، وه بھی نیک کردار پیغمبر تھا

  57. وَرَفَعْنَـٰهُ مَكَانًا عَلِيًّا ۝٥٧

    ہم نےاسے بلند مقام پر اٹھا لیا

  58. أُوْلَـٰٓئِكَ ٱلَّذِينَ أَنْعَمَ ٱللَّهُ عَلَيْهِم مِّنَ ٱلنَّبِيِّــۧنَ مِن ذُرِّيَّةِ ءَادَمَ وَمِمَّنْ حَمَلْنَا مَعَ نُوحٍ وَمِن ذُرِّيَّةِ إِبْرَٲهِيمَ وَإِسْرَٲٓءِيلَ وَمِمَّنْ هَدَيْنَا وَٱجْتَبَيْنَآ‌ۚ إِذَا تُتْلَىٰ عَلَيْهِمْ ءَايَـٰتُ ٱلرَّحْمَـٰنِ خَرُّواْ سُجَّدًا وَبُكِيًّا ۩ ۝٥٨

    یہی وه انبیا ہیں جن پر اللہ تعالیٰ نے فضل وکرم کیا جو اوﻻد آدم میں سے ہیں اور ان لوگوں کی نسل سے ہیں جنہیں ہم نے نوح (علیہ السلام) کے ساتھ کشتی میں چڑھا لیا تھا، اور اوﻻد ابراہیم ویعقوب سے اور ہماری طرف سے راه یافتہ اور ہمارے پسندیده لوگوں میں سے۔ ان کے سامنے جب اللہ رحمان کی آیتوں کی تلاوت کی جاتی تھی یہ سجده کرتے اور روتے گڑگڑاتے گر پڑتے تھے

  59. ۞ فَخَلَفَ مِنۢ بَعْدِهِمْ خَلْفٌ أَضَاعُواْ ٱلصَّلَوٲةَ وَٱتَّبَعُواْ ٱلشَّهَوَٲتِ‌ۖ فَسَوْفَ يَلْقَوْنَ غَيًّا ۝٥٩

    پھر ان کے بعد ایسے ناخلف پیدا ہوئے کہ انہوں نے نماز ضائع کردی اور نفسانی خواہشوں کے پیچھے پڑ گئے، سو ان کا نقصان ان کے آگے آئے گا

  60. إِلَّا مَن تَابَ وَءَامَنَ وَعَمِلَ صَـٰلِحًا فَأُوْلَـٰٓئِكَ يَدْخُلُونَ ٱلْجَنَّةَ وَلَا يُظْلَمُونَ شَيْــًٔا ۝٦٠

    بجز ان کے جو توبہ کر لیں اور ایمان ﻻئیں اور نیک عمل کریں۔ ایسے لوگ جنت میں جائیں گے اور ان کی ذرا سی بھی حق تلفی نہ کی جائے گی

  61. جَنَّـٰتِ عَدْنٍ ٱلَّتِى وَعَدَ ٱلرَّحْمَـٰنُ عِبَادَهُۥ بِٱلْغَيْبِ‌ۚ إِنَّهُۥ كَانَ وَعْدُهُۥ مَأْتِيًّا ۝٦١

    ہمیشگی والی جنتوں میں جن کا غائبانہ وعده اللہ مہربان نے اپنے بندوں سے کیا ہے۔ بیشک اس کا وعده پورا ہونے واﻻ ہی ہے

  62. لَّا يَسْمَعُونَ فِيهَا لَغْوًا إِلَّا سَلَـٰمًا‌ۖ وَلَهُمْ رِزْقُهُمْ فِيهَا بُكْرَةً وَعَشِيًّا ۝٦٢

    وه لوگ وہاں کوئی لغو بات نہ سنیں گے صرف سلام ہی سلام سنیں گے، ان کے لئے وہاں صبح شام ان کا رزق ہوگا

  63. تِلْكَ ٱلْجَنَّةُ ٱلَّتِى نُورِثُ مِنْ عِبَادِنَا مَن كَانَ تَقِيًّا ۝٦٣

    یہ ہے وه جنت جس کا وارث ہم اپنے بندوں میں سے انہیں بناتے ہیں جو متقی ہوں

  64. وَمَا نَتَنَزَّلُ إِلَّا بِأَمْرِ رَبِّكَ‌ۖ لَهُۥ مَا بَيْنَ أَيْدِينَا وَمَا خَلْفَنَا وَمَا بَيْنَ ذَٲلِكَ‌ۚ وَمَا كَانَ رَبُّكَ نَسِيًّا ۝٦٤

    ہم بغیر تیرے رب کے حکم کے اتر نہیں سکتے، ہمارے آگے پیچھے اور ان کے درمیان کی کل چیزیں اسی کی ملکیت میں ہیں، تیرا پروردگار بھولنے واﻻ نہیں

  65. رَّبُّ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا فَٱعْبُدْهُ وَٱصْطَبِرْ لِعِبَـٰدَتِهِۦ‌ۚ هَلْ تَعْلَمُ لَهُۥ سَمِيًّا ۝٦٥

    آسمانوں کا، زمین کا اور جو کچھ ان کے درمیان ہے سب کا رب وہی ہے تو اسی کی بندگی کر اور اس کی عبادت پر جم جا۔ کیا تیرے علم میں اس کا ہمنام ہم پلہ کوئی اور بھی ہے؟

  66. وَيَقُولُ ٱلْإِنسَـٰنُ أَءِذَا مَا مِتُّ لَسَوْفَ أُخْرَجُ حَيًّا ۝٦٦

    انسان کہتا ہے کہ جب میں مرجاؤں گا تو کیا پھر زنده کر کے نکالا جاؤں گا؟

  67. أَوَلَا يَذْكُرُ ٱلْإِنسَـٰنُ أَنَّا خَلَقْنَـٰهُ مِن قَبْلُ وَلَمْ يَكُ شَيْــًٔا ۝٦٧

    کیا یہ انسان اتنا بھی یاد نہیں رکھتا کہ ہم نے اسے اس سے پہلے پیدا کیا حاﻻنکہ وه کچھ بھی نہ تھا

  68. فَوَرَبِّكَ لَنَحْشُرَنَّهُمْ وَٱلشَّيَـٰطِينَ ثُمَّ لَنُحْضِرَنَّهُمْ حَوْلَ جَهَنَّمَ جِثِيًّا ۝٦٨

    تیرے پروردگار کی قسم! ہم انہیں اور شیطانوں کو جمع کر کے ضرور ضرور جہنم کے اردگرد گھٹنوں کے بل گرے ہوئے حاضر کر دیں گے

  69. ثُمَّ لَنَنزِعَنَّ مِن كُلِّ شِيعَةٍ أَيُّهُمْ أَشَدُّ عَلَى ٱلرَّحْمَـٰنِ عِتِيًّا ۝٦٩

    ہم پھر ہر ہر گروه سے انہیں الگ نکال کھڑا کریں گے جو اللہ رحمنٰ سے بہت اکڑے اکڑے پھرتے تھے

  70. ثُمَّ لَنَحْنُ أَعْلَمُ بِٱلَّذِينَ هُمْ أَوْلَىٰ بِهَا صِلِيًّا ۝٧٠

    پھر ہم انہیں بھی خوب جانتے ہیں جو جہنم کے داخلے کے زیاده سزاوار ہیں

  71. وَإِن مِّنكُمْ إِلَّا وَارِدُهَا‌ۚ كَانَ عَلَىٰ رَبِّكَ حَتْمًا مَّقْضِيًّا ۝٧١

    تم میں سے ہر ایک وہاں ضرور وارد ہونے واﻻ ہے، یہ تیرے پروردگار کے ذمے قطعی، فیصل شده امر ہے

  72. ثُمَّ نُنَجِّى ٱلَّذِينَ ٱتَّقَواْ وَّنَذَرُ ٱلظَّـٰلِمِينَ فِيهَا جِثِيًّا ۝٧٢

    پھر ہم پرہیزگاروں کو تو بچالیں گے اور نافرمانوں کو اسی میں گھٹنوں کے بل گرا ہوا چھوڑ دیں گے

  73. وَإِذَا تُتْلَىٰ عَلَيْهِمْ ءَايَـٰتُنَا بَيِّنَـٰتٍ قَالَ ٱلَّذِينَ كَفَرُواْ لِلَّذِينَ ءَامَنُوٓاْ أَىُّ ٱلْفَرِيقَيْنِ خَيْرٌ مَّقَامًا وَأَحْسَنُ نَدِيًّا ۝٧٣

    جب ان کے سامنے ہماری روشن آیتیں تلاوت کی جاتی ہیں تو کافر مسلمانوں سے کہتے ہیں بتاؤ ہم تم دونوں جماعتوں میں سے کس کا مرتبہ زیاده ہے؟ اور کس کی مجلس شاندار ہے؟

  74. وَكَمْ أَهْلَكْنَا قَبْلَهُم مِّن قَرْنٍ هُمْ أَحْسَنُ أَثَـٰثًا وَرِءْيًا ۝٧٤

    ہم تو ان سے پہلے بہت سی جماعتوں کو غارت کر چکے ہیں جو ساز وسامان اور نام ونمود میں ان سے بڑھ چڑھ کر تھیں

  75. قُلْ مَن كَانَ فِى ٱلضَّلَـٰلَةِ فَلْيَمْدُدْ لَهُ ٱلرَّحْمَـٰنُ مَدًّا‌ۚ حَتَّىٰٓ إِذَا رَأَوْاْ مَا يُوعَدُونَ إِمَّا ٱلْعَذَابَ وَإِمَّا ٱلسَّاعَةَ فَسَيَعْلَمُونَ مَنْ هُوَ شَرٌّ مَّكَانًا وَأَضْعَفُ جُندًا ۝٧٥

    کہہ دیجئے! جو گمراہی میں ہوتا اللہ رحمنٰ اس کو خوب لمبی مہلت دیتا ہے، یہاں تک کہ وه ان چیزوں کو دیکھ لیں جنکا وعده کیے جاتے ہیں یعنی عذاب یا قیامت کو، اس وقت ان کو صحیح طور پر معلوم ہو جائے گا کہ کون برے مرتبے واﻻ اور کس کا جتھا کمزور ہے

  76. وَيَزِيدُ ٱللَّهُ ٱلَّذِينَ ٱهْتَدَوْاْ هُدًى‌ۗ وَٱلْبَـٰقِيَـٰتُ ٱلصَّـٰلِحَـٰتُ خَيْرٌ عِندَ رَبِّكَ ثَوَابًا وَخَيْرٌ مَّرَدًّا ۝٧٦

    اور ہدایت یافتہ لوگوں کو اللہ تعالیٰ ہدایت میں بڑھاتا ہے، اور باقی رہنے والی نیکیاں تیرے رب کے نزدیک ﺛواب کے لحاظ سے اور انجام کے لحاظ سے بہت ہی بہتر ہیں

  77. أَفَرَءَيْتَ ٱلَّذِى كَفَرَ بِـَٔـايَـٰتِنَا وَقَالَ لَأُوتَيَنَّ مَالاً وَوَلَدًا ۝٧٧

    کیا تو نے اسے بھی دیکھا جس نے ہماری آیتوں سے کفر کیا اور کہا کہ مجھے تو مال واوﻻد ضرور ہی دی جائے گی

  78. أَطَّلَعَ ٱلْغَيْبَ أَمِ ٱتَّخَذَ عِندَ ٱلرَّحْمَـٰنِ عَهْدًا ۝٧٨

    کیا وه غیب پر مطلع ہے یا اللہ کا کوئی وعده لے چکا ہے؟

  79. كَلَّا‌ۚ سَنَكْتُبُ مَا يَقُولُ وَنَمُدُّ لَهُۥ مِنَ ٱلْعَذَابِ مَدًّا ۝٧٩

    ہرگز نہیں، یہ جو بھی کہہ رہا ہے ہم اسے ضرور لکھ لیں گے، اور اس کے لئے عذاب بڑھائے چلے جائیں گے

  80. وَنَرِثُهُۥ مَا يَقُولُ وَيَأْتِينَا فَرْدًا ۝٨٠

    یہ جن چیزوں کو کہہ رہا ہے اسے ہم اس کے بعد لے لیں گے۔ اور یہ تو بالکل اکیلا ہی ہمارے سامنے حاضر ہوگا

  81. وَٱتَّخَذُواْ مِن دُونِ ٱللَّهِ ءَالِهَةً لِّيَكُونُواْ لَهُمْ عِزًّا ۝٨١

    انہوں نے اللہ کے سوا دوسرے معبود بنا رکھے ہیں کہ وه ان کے لئے باعﺚ عزت ہوں

  82. كَلَّا‌ۚ سَيَكْفُرُونَ بِعِبَادَتِهِمْ وَيَكُونُونَ عَلَيْهِمْ ضِدًّا ۝٨٢

    لیکن ایسا ہرگز ہونا نہیں۔ وه تو ان کی پوجا سے منکر ہو جائیں گے، اور الٹے ان کے دشمن بن جائیں گے

  83. أَلَمْ تَرَ أَنَّآ أَرْسَلْنَا ٱلشَّيَـٰطِينَ عَلَى ٱلْكَـٰفِرِينَ تَؤُزُّهُمْ أَزًّا ۝٨٣

    کیا تو نے نہیں دیکھا کہ ہم کافروں کے پاس شیطانوں کو بھیجتے ہیں جو انہیں خوب اکساتے ہیں

  84. فَلَا تَعْجَلْ عَلَيْهِمْ‌ۖ إِنَّمَا نَعُدُّ لَهُمْ عَدًّا ۝٨٤

    تو ان کے بارے میں جلدی نہ کر، ہم تو خود ہی ان کے لئے مدت شماری کر رہے ہیں

  85. يَوْمَ نَحْشُرُ ٱلْمُتَّقِينَ إِلَى ٱلرَّحْمَـٰنِ وَفْدًا ۝٨٥

    جس دن ہم پرہیزگاروں کو اللہ رحمان کی طرف بطور مہمان جمع کریں گے

  86. وَنَسُوقُ ٱلْمُجْرِمِينَ إِلَىٰ جَهَنَّمَ وِرْدًا ۝٨٦

    اور گناه گاروں کوسخت پیاس کی حالت میں جہنم کی طرف ہانک لے جائیں گے

  87. لَّا يَمْلِكُونَ ٱلشَّفَـٰعَةَ إِلَّا مَنِ ٱتَّخَذَ عِندَ ٱلرَّحْمَـٰنِ عَهْدًا ۝٨٧

    کسی کو شفاعت کا اختیار نہ ہوگا سوائے ان کے جنہوں نے اللہ تعالیٰ کی طرف سے کوئی قول قرار لے لیا ہے

  88. وَقَالُواْ ٱتَّخَذَ ٱلرَّحْمَـٰنُ وَلَدًا ۝٨٨

    ان کا قول تو یہ ہے کہ اللہ رحمٰن نے بھی اوﻻد اختیار کی ہے

  89. لَّقَدْ جِئْتُمْ شَيْــًٔا إِدًّا ۝٨٩

    یقیناً تم بہت بری اور بھاری چیز ﻻئے ہو

  90. تَكَادُ ٱلسَّمَـٰوَٲتُ يَتَفَطَّرْنَ مِنْهُ وَتَنشَقُّ ٱلْأَرْضُ وَتَخِرُّ ٱلْجِبَالُ هَدًّا ۝٩٠

    قریب ہے کہ اس قول کی وجہ سے آسمان پھٹ جائیں اور زمین شق ہوجائے اور پہاڑ ریزے ریزے ہو جائیں

  91. أَن دَعَوْاْ لِلرَّحْمَـٰنِ وَلَدًا ۝٩١

    کہ وه رحمان کی اوﻻد ﺛابت کرنے بیٹھے

  92. وَمَا يَنۢبَغِى لِلرَّحْمَـٰنِ أَن يَتَّخِذَ وَلَدًا ۝٩٢

    شان رحمٰن کے ﻻئق نہیں کہ وه اوﻻد رکھے

  93. إِن كُلُّ مَن فِى ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ إِلَّآ ءَاتِى ٱلرَّحْمَـٰنِ عَبْدًا ۝٩٣

    آسمان وزمین میں جو بھی ہیں سب کے سب اللہ کے غلام بن کر ہی آنے والے ہیں

  94. لَّقَدْ أَحْصَـٰهُمْ وَعَدَّهُمْ عَدًّا ۝٩٤

    ان سب کو اس نے گھیر رکھا ہے اور سب کو پوری طرح گن بھی رکھا ہے

  95. وَكُلُّهُمْ ءَاتِيهِ يَوْمَ ٱلْقِيَـٰمَةِ فَرْدًا ۝٩٥

    یہ سارے کے سارے قیامت کے دن اکیلے اس کے پاس حاضر ہونے والے ہیں

  96. إِنَّ ٱلَّذِينَ ءَامَنُواْ وَعَمِلُواْ ٱلصَّـٰلِحَـٰتِ سَيَجْعَلُ لَهُمُ ٱلرَّحْمَـٰنُ وُدًّا ۝٩٦

    بیشک جو ایمان لائے ہیں اور جنہوں نے شائستہ اعمال کیے ہیں ان کے لیے اللہ رحمٰن محبت پیدا کردے گا

  97. فَإِنَّمَا يَسَّرْنَـٰهُ بِلِسَانِكَ لِتُبَشِّرَ بِهِ ٱلْمُتَّقِينَ وَتُنذِرَ بِهِۦ قَوْمًا لُّدًّا ۝٩٧

    ہم نے اس قرآن کو تیری زبان میں بہت ہی آسان کر دیا ہے کہ تو اس کے ذریعہ سے پرہیزگاروں کو خوشخبری دے اور جھگڑالو لوگوں کو ڈرا دے

  98. وَكَمْ أَهْلَكْنَا قَبْلَهُم مِّن قَرْنٍ هَلْ تُحِسُّ مِنْهُم مِّنْ أَحَدٍ أَوْ تَسْمَعُ لَهُمْ رِكْزَۢا ۝٩٨

    ہم نے ان سے پہلے بہت سی جماعتیں تباہ کر دی ہیں، کیا ان میں سے ایک کی بھی آہٹ تو پاتا ہے یا ان کی آواز کی بھنک بھی تیرے کان میں پڑتی ہے؟
