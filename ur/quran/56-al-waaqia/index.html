بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ

  1. إِذَا وَقَعَتِ ٱلْوَاقِعَةُ ۝١

    جب قیامت قائم ہو جائے گی

  2. لَيْسَ لِوَقْعَتِهَا كَاذِبَةٌ ۝٢

    جس کے واقع ہونے میں کوئی جھوٹ نہیں

  3. خَافِضَةٌ رَّافِعَةٌ ۝٣

    وه پست کرنے والی اور بلند کرنے والی ہوگی

  4. إِذَا رُجَّتِ ٱلْأَرْضُ رَجًّا ۝٤

    جبکہ زمین زلزلہ کے ساتھ ہلا دی جائے گی

  5. وَبُسَّتِ ٱلْجِبَالُ بَسًّا ۝٥

    اور پہاڑ بالکل ریزه ریزه کر دیے جائیں گے

  6. فَكَانَتْ هَبَآءً مُّنۢبَثًّا ۝٦

    پھر وه مثل پراگنده غبار کے ہو جائیں گے

  7. وَكُنتُمْ أَزْوَٲجًا ثَلَـٰثَةً ۝٧

    اور تم تین جماعتوں میں ہو جاؤ گے

  8. فَأَصْحَـٰبُ ٱلْمَيْمَنَةِ مَآ أَصْحَـٰبُ ٱلْمَيْمَنَةِ ۝٨

    پس داہنے ہاتھ والے کیسے اچھے ہیں داہنے ہاتھ والے

  9. وَأَصْحَـٰبُ ٱلْمَشْــَٔمَةِ مَآ أَصْحَـٰبُ ٱلْمَشْــَٔمَةِ ۝٩

    اور بائیں ہاتھ والے کیا حال ہے بائیں ہاتھ والوں کا

  10. وَٱلسَّـٰبِقُونَ ٱلسَّـٰبِقُونَ ۝١٠

    اور جو آگے والے ہیں وه تو آگے والے ہی ہیں

  11. أُوْلَـٰٓئِكَ ٱلْمُقَرَّبُونَ ۝١١

    وه بالکل نزدیکی حاصل کیے ہوئے ہیں

  12. فِى جَنَّـٰتِ ٱلنَّعِيمِ ۝١٢

    نعمتوں والی جنتوں میں ہیں

  13. ثُلَّةٌ مِّنَ ٱلْأَوَّلِينَ ۝١٣

    (بہت بڑا) گروه تو اگلے لوگوں میں سے ہوگا

  14. وَقَلِيلٌ مِّنَ ٱلْأَخِرِينَ ۝١٤

    اور تھوڑے سے پچھلے لوگوں میں سے

  15. عَلَىٰ سُرُرٍ مَّوْضُونَةٍ ۝١٥

    یہ لوگ سونے کے تاروں سے بنے ہوئے تختوں پر

  16. مُّتَّكِــِٔينَ عَلَيْهَا مُتَقَـٰبِلِينَ ۝١٦

    ایک دوسرے کےسامنے تکیہ لگائے بیٹھے ہوں گے

  17. يَطُوفُ عَلَيْهِمْ وِلْدَٲنٌ مُّخَلَّدُونَ ۝١٧

    ان کے پاس ایسے لڑکے جو ہمیشہ (لڑکے ہی) رہیں گے آمدورفت کریں گے

  18. بِأَكْوَابٍ وَأَبَارِيقَ وَكَأْسٍ مِّن مَّعِينٍ ۝١٨

    آبخورے اور جگ لے کر اور ایسا جام لے کر جو بہتی ہوئی شراب سے پر ہو

  19. لَّا يُصَدَّعُونَ عَنْهَا وَلَا يُنزِفُونَ ۝١٩

    جس سے نہ سر میں درد ہو نہ عقل میں فتور آئے

  20. وَفَـٰكِهَةٍ مِّمَّا يَتَخَيَّرُونَ ۝٢٠

    اور ایسے میوے لیے ہوئے جو ان کی پسند کے ہوں

  21. وَلَحْمِ طَيْرٍ مِّمَّا يَشْتَهُونَ ۝٢١

    اور پرندوں کے گوشت جو انہیں مرغوب ہوں

  22. وَحُورٌ عِينٌ ۝٢٢

    اور بڑی بڑی آنکھوں والی حوریں

  23. كَأَمْثَـٰلِ ٱللُّؤْلُوِٕ ٱلْمَكْنُونِ ۝٢٣

    جو چھپے ہوئے موتیوں کی طرح ہیں

  24. جَزَآءَۢ بِمَا كَانُواْ يَعْمَلُونَ ۝٢٤

    یہ صلہ ہے ان کے اعمال کا

  25. لَا يَسْمَعُونَ فِيهَا لَغْوًا وَلَا تَأْثِيمًا ۝٢٥

    نہ وہاں بکواس سنیں گے اور نہ گناه کی بات

  26. إِلَّا قِيلاً سَلَـٰمًا سَلَـٰمًا ۝٢٦

    صرف سلام ہی سلام کی آواز ہوگی

  27. وَأَصْحَـٰبُ ٱلْيَمِينِ مَآ أَصْحَـٰبُ ٱلْيَمِينِ ۝٢٧

    اور داہنے ہاتھ والے کیا ہی اچھے ہیں داہنے ہاتھ والے

  28. فِى سِدْرٍ مَّخْضُودٍ ۝٢٨

    وه بغیرکانٹوں کی بیریوں

  29. وَطَلْحٍ مَّنضُودٍ ۝٢٩

    اور تہ بہ تہ کیلوں

  30. وَظِلٍّ مَّمْدُودٍ ۝٣٠

    اور لمبے لمبے سایوں

  31. وَمَآءٍ مَّسْكُوبٍ ۝٣١

    اور بہتے ہوئے پانیوں

  32. وَفَـٰكِهَةٍ كَثِيرَةٍ ۝٣٢

    اور بکثرت پھلوں میں

  33. لَّا مَقْطُوعَةٍ وَلَا مَمْنُوعَةٍ ۝٣٣

    جو نہ ختم ہوں نہ روک لیے جائیں

  34. وَفُرُشٍ مَّرْفُوعَةٍ ۝٣٤

    اور اونچے اونچے فرشوں میں ہوں گے

  35. إِنَّآ أَنشَأْنَـٰهُنَّ إِنشَآءً ۝٣٥

    ہم نے ان کی (بیویوں کو) خاص طور پر بنایا ہے

  36. فَجَعَلْنَـٰهُنَّ أَبْكَارًا ۝٣٦

    اور ہم نے انہیں کنواریاں بنا دیا ہے

  37. عُرُبًا أَتْرَابًا ۝٣٧

    محبت والیاں اور ہم عمر ہیں

  38. لِّأَصْحَـٰبِ ٱلْيَمِينِ ۝٣٨

    دائیں ہاتھ والوں کے لیے ہیں

  39. ثُلَّةٌ مِّنَ ٱلْأَوَّلِينَ ۝٣٩

    جم غفیر ہے اگلوں میں سے

  40. وَثُلَّةٌ مِّنَ ٱلْأَخِرِينَ ۝٤٠

    اور بہت بڑی جماعت ہے پچھلوں میں سے

  41. وَأَصْحَـٰبُ ٱلشِّمَالِ مَآ أَصْحَـٰبُ ٱلشِّمَالِ ۝٤١

    اور بائیں ہاتھ والے کیا ہیں بائیں ہاتھ والے

  42. فِى سَمُومٍ وَحَمِيمٍ ۝٤٢

    گرم ہوا اور گرم پانی میں (ہوں گے)

  43. وَظِلٍّ مِّن يَحْمُومٍ ۝٤٣

    اورسیاه دھوئیں کے سائے میں

  44. لَّا بَارِدٍ وَلَا كَرِيمٍ ۝٤٤

    جو نہ ٹھنڈا ہے نہ فرحت بخش

  45. إِنَّهُمْ كَانُواْ قَبْلَ ذَٲلِكَ مُتْرَفِينَ ۝٤٥

    بیشک یہ لوگ اس سے پہلے بہت نازوں میں پلے ہوئے تھے

  46. وَكَانُواْ يُصِرُّونَ عَلَى ٱلْحِنثِ ٱلْعَظِيمِ ۝٤٦

    اور بڑے بڑے گناہوں پر اصرار کرتے تھے

  47. وَكَانُواْ يَقُولُونَ أَئِذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَـٰمًا أَءِنَّا لَمَبْعُوثُونَ ۝٤٧

    اور کہتے تھے کہ کیا جب ہم مر جائیں گے اور مٹی اور ہڈی ہو جائیں گے تو کیا ہم پھر دوباره اٹھا کھڑے کیے جائیں گے

  48. أَوَءَابَآؤُنَا ٱلْأَوَّلُونَ ۝٤٨

    اور کیا ہمارے اگلے باپ دادا بھی؟

  49. قُلْ إِنَّ ٱلْأَوَّلِينَ وَٱلْأَخِرِينَ ۝٤٩

    آپ کہہ دیجئے کہ یقیناً سب اگلے اور پچھلے

  50. لَمَجْمُوعُونَ إِلَىٰ مِيقَـٰتِ يَوْمٍ مَّعْلُومٍ ۝٥٠

    ضرور جمع کئے جائیں گے ایک مقرر دن کے وقت

  51. ثُمَّ إِنَّكُمْ أَيُّهَا ٱلضَّآلُّونَ ٱلْمُكَذِّبُونَ ۝٥١

    پھر تم اے گمراہو جھٹلانے والو!

  52. لَأَكِلُونَ مِن شَجَرٍ مِّن زَقُّومٍ ۝٥٢

    البتہ کھانے والے ہو تھوہر کا درخت

  53. فَمَالِــُٔونَ مِنْهَا ٱلْبُطُونَ ۝٥٣

    اور اسی سے پیٹ بھرنے والے ہو

  54. فَشَـٰرِبُونَ عَلَيْهِ مِنَ ٱلْحَمِيمِ ۝٥٤

    پھر اس پر گرم کھولتا پانی پینے والے ہو

  55. فَشَـٰرِبُونَ شُرْبَ ٱلْهِيمِ ۝٥٥

    پھر پینے والے بھی پیاسے اونٹوں کی طرح

  56. هَـٰذَا نُزُلُهُمْ يَوْمَ ٱلدِّينِ ۝٥٦

    قیامت کے دن ان کی مہمانی یہ ہے

  57. نَحْنُ خَلَقْنَـٰكُمْ فَلَوْلَا تُصَدِّقُونَ ۝٥٧

    ہم ہی نے تم سب کو پیدا کیا ہے پھر تم کیوں باور نہیں کرتے؟

  58. أَفَرَءَيْتُم مَّا تُمْنُونَ ۝٥٨

    اچھا پھر یہ تو بتلاؤ کہ جو منی تم ٹپکاتے ہو

  59. ءَأَنتُمْ تَخْلُقُونَهُۥٓ أَمْ نَحْنُ ٱلْخَـٰلِقُونَ ۝٥٩

    کیا اس کا (انسان) تم بناتے ہو یا پیدا کرنے والے ہم ہی ہیں؟

  60. نَحْنُ قَدَّرْنَا بَيْنَكُمُ ٱلْمَوْتَ وَمَا نَحْنُ بِمَسْبُوقِينَ ۝٦٠

    ہم ہی نے تم میں موت کو متعین کر دیا ہے اور ہم اس سے ہارے ہوئے نہیں ہیں

  61. عَلَىٰٓ أَن نُّبَدِّلَ أَمْثَـٰلَكُمْ وَنُنشِئَكُمْ فِى مَا لَا تَعْلَمُونَ ۝٦١

    کہ تمہاری جگہ تم جیسے اور پیدا کر دیں اور تمہیں نئے سرے سے اس عالم میں پیدا کریں جس سے تم (بالکل) بےخبر ہو

  62. وَلَقَدْ عَلِمْتُمُ ٱلنَّشْأَةَ ٱلْأُولَىٰ فَلَوْلَا تَذَكَّرُونَ ۝٦٢

    تمہیں یقینی طور پر پہلی دفعہ کی پیدائش معلوم ہی ہے پھر کیوں عبرت حاصل نہیں کرتے؟

  63. أَفَرَءَيْتُم مَّا تَحْرُثُونَ ۝٦٣

    اچھا پھر یہ بھی بتلاؤ کہ تم جو کچھ بوتے ہو

  64. ءَأَنتُمْ تَزْرَعُونَهُۥٓ أَمْ نَحْنُ ٱلزَّٲرِعُونَ ۝٦٤

    اسے تم ہی اگاتے ہو یا ہم اگانے والے ہیں

  65. لَوْ نَشَآءُ لَجَعَلْنَـٰهُ حُطَـٰمًا فَظَلْتُمْ تَفَكَّهُونَ ۝٦٥

    اگر ہم چاہیں تواسے ریزه ریزه کر ڈالیں اور تم حیرت کے ساتھ باتیں بناتے ہی ره جاؤ

  66. إِنَّا لَمُغْرَمُونَ ۝٦٦

    کہ ہم پر تو تاوان ہی پڑ گیا

  67. بَلْ نَحْنُ مَحْرُومُونَ ۝٦٧

    بلکہ ہم بالکل محروم ہی ره گئے

  68. أَفَرَءَيْتُمُ ٱلْمَآءَ ٱلَّذِى تَشْرَبُونَ ۝٦٨

    اچھا یہ بتاؤ کہ جس پانی کو تم پیتے ہو

  69. ءَأَنتُمْ أَنزَلْتُمُوهُ مِنَ ٱلْمُزْنِ أَمْ نَحْنُ ٱلْمُنزِلُونَ ۝٦٩

    اسے بادلوں سے بھی تم ہی اتارتے ہو یا ہم برساتے ہیں؟

  70. لَوْ نَشَآءُ جَعَلْنَـٰهُ أُجَاجًا فَلَوْلَا تَشْكُرُونَ ۝٧٠

    اگر ہماری منشا ہو تو ہم اسے کڑوا زہر کردیں پھر تم ہماری شکرگزاری کیوں نہیں کرتے؟

  71. أَفَرَءَيْتُمُ ٱلنَّارَ ٱلَّتِى تُورُونَ ۝٧١

    اچھا ذرا یہ بھی بتاؤ کہ جو آگ تم سلگاتے ہو

  72. ءَأَنتُمْ أَنشَأْتُمْ شَجَرَتَهَآ أَمْ نَحْنُ ٱلْمُنشِــُٔونَ ۝٧٢

    اس کے درخت کو تم نے پیدا کیا ہے یا ہم اس کے پیدا کرنے والے ہیں؟

  73. نَحْنُ جَعَلْنَـٰهَا تَذْكِرَةً وَمَتَـٰعًا لِّلْمُقْوِينَ ۝٧٣

    ہم نے اسے سبب نصیحت اور مسافروں کے فائدے کی چیز بنایا ہے

  74. فَسَبِّحْ بِٱسْمِ رَبِّكَ ٱلْعَظِيمِ ۝٧٤

    پس اپنے بہت بڑے رب کے نام کی تسبیح کیا کرو

  75. ۞ فَلَآ أُقْسِمُ بِمَوَٲقِعِ ٱلنُّجُومِ ۝٧٥

    پس میں قسم کھاتا ہوں ستاروں کے گرنے کی

  76. وَإِنَّهُۥ لَقَسَمٌ لَّوْ تَعْلَمُونَ عَظِيمٌ ۝٧٦

    اور اگرتمہیں علم ہو تو یہ بہت بڑی قسم ہے

  77. إِنَّهُۥ لَقُرْءَانٌ كَرِيمٌ ۝٧٧

    کہ بیشک یہ قرآن بہت بڑی عزت واﻻ ہے

  78. فِى كِتَـٰبٍ مَّكْنُونٍ ۝٧٨

    جو ایک محفوظ کتاب میں درج ہے

  79. لَّا يَمَسُّهُۥٓ إِلَّا ٱلْمُطَهَّرُونَ ۝٧٩

    جسے صرف پاک لوگ ہی چھو سکتے ہیں

  80. تَنزِيلٌ مِّن رَّبِّ ٱلْعَـٰلَمِينَ ۝٨٠

    یہ رب العالمین کی طرف سےاترا ہوا ہے

  81. أَفَبِهَـٰذَا ٱلْحَدِيثِ أَنتُم مُّدْهِنُونَ ۝٨١

    پس کیا تم ایسی بات کو سرسری (اور معمولی) سمجھ رہے ہو؟

  82. وَتَجْعَلُونَ رِزْقَكُمْ أَنَّكُمْ تُكَذِّبُونَ ۝٨٢

    اور اپنے حصے میں یہی لیتے ہو کہ جھٹلاتے پھرو

  83. فَلَوْلَآ إِذَا بَلَغَتِ ٱلْحُلْقُومَ ۝٨٣

    پس جبکہ روح نرخرے تک پہنچ جائے

  84. وَأَنتُمْ حِينَئِذٍ تَنظُرُونَ ۝٨٤

    اور تم اس وقت آنکھوں سے دیکھتے رہو

  85. وَنَحْنُ أَقْرَبُ إِلَيْهِ مِنكُمْ وَلَـٰكِن لَّا تُبْصِرُونَ ۝٨٥

    ہم اس شخص سے بہ نسبت تمہارے بہت زیاده قریب ہوتے ہیں لیکن تم نہیں دیکھ سکتے

  86. فَلَوْلَآ إِن كُنتُمْ غَيْرَ مَدِينِينَ ۝٨٦

    پس اگر تم کسی کے زیرفرمان نہیں

  87. تَرْجِعُونَهَآ إِن كُنتُمْ صَـٰدِقِينَ ۝٨٧

    اور اس قول میں سچے ہو تو (ذرا) اس روح کو تو لوٹاؤ

  88. فَأَمَّآ إِن كَانَ مِنَ ٱلْمُقَرَّبِينَ ۝٨٨

    پس جو کوئی بارگاه الٰہی سے قریب کیا ہوا ہوگا

  89. فَرَوْحٌ وَرَيْحَانٌ وَجَنَّتُ نَعِيمٍ ۝٨٩

    اسے تو راحت ہے اور غذائیں ہیں اور آرام والی جنت ہے

  90. وَأَمَّآ إِن كَانَ مِنْ أَصْحَـٰبِ ٱلْيَمِينِ ۝٩٠

    اور جو شحص داہنے (ہاتھ) والوں میں سے ہے

  91. فَسَلَـٰمٌ لَّكَ مِنْ أَصْحَـٰبِ ٱلْيَمِينِ ۝٩١

    تو بھی سلامتی ہے تیرے لیے کہ تو داہنے والوں میں سے ہے

  92. وَأَمَّآ إِن كَانَ مِنَ ٱلْمُكَذِّبِينَ ٱلضَّآلِّينَ ۝٩٢

    لیکن اگر کوئی جھٹلانے والوں گمراہوں میں سے ہے

  93. فَنُزُلٌ مِّنْ حَمِيمٍ ۝٩٣

    تو کھولتے ہوئے گرم پانی کی مہمانی ہے

  94. وَتَصْلِيَةُ جَحِيمٍ ۝٩٤

    اور دوزخ میں جانا ہے

  95. إِنَّ هَـٰذَا لَهُوَ حَقُّ ٱلْيَقِينِ ۝٩٥

    یہ خبر سراسر حق اور قطعاً یقینی ہے

  96. فَسَبِّحْ بِٱسْمِ رَبِّكَ ٱلْعَظِيمِ ۝٩٦

    پس تواپنے عظیم الشان پروردگار کی تسبیح کر
