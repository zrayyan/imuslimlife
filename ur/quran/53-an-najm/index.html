بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ

  1. وَٱلنَّجْمِ إِذَا هَوَىٰ ۝١

    قسم ہے ستارے کی جب وه گرے

  2. مَا ضَلَّ صَاحِبُكُمْ وَمَا غَوَىٰ ۝٢

    کہ تمہارے ساتھی نے نہ راه گم کی ہے نہ وه ٹیڑھی راه پر ہے

  3. وَمَا يَنطِقُ عَنِ ٱلْهَوَىٰٓ ۝٣

    اور نہ وه اپنی خواہش سے کوئی بات کہتے ہیں

  4. إِنْ هُوَ إِلَّا وَحْىٌ يُوحَىٰ ۝٤

    وه تو صرف وحی ہے جو اتاری جاتی ہے

  5. عَلَّمَهُۥ شَدِيدُ ٱلْقُوَىٰ ۝٥

    اسے پوری طاقت والے فرشتے نے سکھایا ہے

  6. ذُو مِرَّةٍ فَٱسْتَوَىٰ ۝٦

    جو زور آور ہے پھر وه سیدھا کھڑا ہو گیا

  7. وَهُوَ بِٱلْأُفُقِ ٱلْأَعْلَىٰ ۝٧

    اور وه بلند آسمان کے کناروں پر تھا

  8. ثُمَّ دَنَا فَتَدَلَّىٰ ۝٨

    پھر نزدیک ہوا اور اتر آیا

  9. فَكَانَ قَابَ قَوْسَيْنِ أَوْ أَدْنَىٰ ۝٩

    پس وه دو کمانوں کے بقدر فاصلہ ره گیا بلکہ اس سے بھی کم

  10. فَأَوْحَىٰٓ إِلَىٰ عَبْدِهِۦ مَآ أَوْحَىٰ ۝١٠

    پس اس نے اللہ کے بندے کو وحی پہنچائی جو بھی پہنچائی

  11. مَا كَذَبَ ٱلْفُؤَادُ مَا رَأَىٰٓ ۝١١

    دل نے جھوٹ نہیں کہا جسے (پیغمبر نے) دیکھا

  12. أَفَتُمَـٰرُونَهُۥ عَلَىٰ مَا يَرَىٰ ۝١٢

    کیا تم جھگڑا کرتے ہو اس پر جو (پیغمبر) دیکھتے ہیں

  13. وَلَقَدْ رَءَاهُ نَزْلَةً أُخْرَىٰ ۝١٣

    اسے تو ایک مرتبہ اور بھی دیکھا تھا

  14. عِندَ سِدْرَةِ ٱلْمُنتَهَىٰ ۝١٤

    سدرةالمنتہیٰ کے پاس

  15. عِندَهَا جَنَّةُ ٱلْمَأْوَىٰٓ ۝١٥

    اسی کے پاس جنہ الماویٰ ہے

  16. إِذْ يَغْشَى ٱلسِّدْرَةَ مَا يَغْشَىٰ ۝١٦

    جب کہ سدره کو چھپائے لیتی تھی وه چیز جو اس پر چھا رہی تھی

  17. مَا زَاغَ ٱلْبَصَرُ وَمَا طَغَىٰ ۝١٧

    نہ تو نگاه بہکی نہ حد سے بڑھی

  18. لَقَدْ رَأَىٰ مِنْ ءَايَـٰتِ رَبِّهِ ٱلْكُبْرَىٰٓ ۝١٨

    یقیناً اس نے اپنے رب کی بڑی بڑی نشانیوں میں سے بعض نشانیاں دیکھ لیں

  19. أَفَرَءَيْتُمُ ٱللَّـٰتَ وَٱلْعُزَّىٰ ۝١٩

    کیا تم نے ﻻت اور عزیٰ کو دیکھا

  20. وَمَنَوٲةَ ٱلثَّالِثَةَ ٱلْأُخْرَىٰٓ ۝٢٠

    اور منات تیسرے پچھلے کو

  21. أَلَكُمُ ٱلذَّكَرُ وَلَهُ ٱلْأُنثَىٰ ۝٢١

    کیا تمہارے لیے لڑکے اور اللہ کے لیے لڑکیاں ہیں؟

  22. تِلْكَ إِذًا قِسْمَةٌ ضِيزَىٰٓ ۝٢٢

    یہ تو اب بڑی بےانصافی کی تقسیم ہے

  23. إِنْ هِىَ إِلَّآ أَسْمَآءٌ سَمَّيْتُمُوهَآ أَنتُمْ وَءَابَآؤُكُم مَّآ أَنزَلَ ٱللَّهُ بِهَا مِن سُلْطَـٰنٍ‌ۚ إِن يَتَّبِعُونَ إِلَّا ٱلظَّنَّ وَمَا تَهْوَى ٱلْأَنفُسُ‌ۖ وَلَقَدْ جَآءَهُم مِّن رَّبِّهِمُ ٱلْهُدَىٰٓ ۝٢٣

    دراصل یہ صرف نام ہیں جو تم نے اور تمہارے باپ دادوں نے ان کے رکھ لیے ہیں اللہ نے ان کی کوئی دلیل نہیں اتاری۔ یہ لوگ تو صرف اٹکل کے اور اپنی نفسانی خواہشوں کے پیچھے پڑے ہوئے ہیں۔ اور یقیناً ان کے رب کی طرف سے ان کے پاس ہدایت آ چکی ہے

  24. أَمْ لِلْإِنسَـٰنِ مَا تَمَنَّىٰ ۝٢٤

    کیا ہر شخص جو آرزو کرے اسے میسر ہے؟

  25. فَلِلَّهِ ٱلْأَخِرَةُ وَٱلْأُولَىٰ ۝٢٥

    اللہ ہی کے ہاتھ ہے یہ جہان اور وه جہان

  26. ۞ وَكَم مِّن مَّلَكٍ فِى ٱلسَّمَـٰوَٲتِ لَا تُغْنِى شَفَـٰعَتُهُمْ شَيْــًٔا إِلَّا مِنۢ بَعْدِ أَن يَأْذَنَ ٱللَّهُ لِمَن يَشَآءُ وَيَرْضَىٰٓ ۝٢٦

    اور بہت سے فرشتے آسمانوں میں ہیں جن کی سفارش کچھ بھی نفع نہیں دے سکتی مگر یہ اور بات ہے کہ اللہ تعالیٰ اپنی خوشی اور اپنی چاہت سے جس کے لیے چاہے اجازت دے دے

  27. إِنَّ ٱلَّذِينَ لَا يُؤْمِنُونَ بِٱلْأَخِرَةِ لَيُسَمُّونَ ٱلْمَلَـٰٓئِكَةَ تَسْمِيَةَ ٱلْأُنثَىٰ ۝٢٧

    بیشک جو لوگ آخرت پر ایمان نہیں رکھتے وه فرشتوں کا زنانہ نام مقرر کرتے ہیں

  28. وَمَا لَهُم بِهِۦ مِنْ عِلْمٍ‌ۖ إِن يَتَّبِعُونَ إِلَّا ٱلظَّنَّ‌ۖ وَإِنَّ ٱلظَّنَّ لَا يُغْنِى مِنَ ٱلْحَقِّ شَيْــًٔا ۝٢٨

    حاﻻنکہ انہیں اس کا کوئی علم نہیں وه صرف اپنے گمان کے پیچھے پڑے ہوئے ہیں اور بیشک وہم (و گمان) حق کے مقابلے میں کچھ کام نہیں دیتا

  29. فَأَعْرِضْ عَن مَّن تَوَلَّىٰ عَن ذِكْرِنَا وَلَمْ يُرِدْ إِلَّا ٱلْحَيَوٲةَ ٱلدُّنْيَا ۝٢٩

    تو آپ اس سے منھ موڑ لیں جو ہماری یاد سے منھ موڑے اور جن کا اراده بجز زندگانیٴ دنیا کے اور کچھ نہ ہو

  30. ذَٲلِكَ مَبْلَغُهُم مِّنَ ٱلْعِلْمِ‌ۚ إِنَّ رَبَّكَ هُوَ أَعْلَمُ بِمَن ضَلَّ عَن سَبِيلِهِۦ وَهُوَ أَعْلَمُ بِمَنِ ٱهْتَدَىٰ ۝٣٠

    یہی ان کے علم کی انتہا ہے۔ آپ کا رب اس سے خوب واقف ہے جو اس کی راه سے بھٹک گیا ہے اور وہی خوب واقف ہےاس سے بھی جو راه یافتہ ہے

  31. وَلِلَّهِ مَا فِى ٱلسَّمَـٰوَٲتِ وَمَا فِى ٱلْأَرْضِ لِيَجْزِىَ ٱلَّذِينَ أَسَـٰٓـــُٔواْ بِمَا عَمِلُواْ وَيَجْزِىَ ٱلَّذِينَ أَحْسَنُواْ بِٱلْحُسْنَى ۝٣١

    اور اللہ ہی کا ہے جو کچھ آسمانوں میں ہے اور جو کچھ زمین میں ہے تاکہ اللہ تعالیٰ برے عمل کرنے والوں کو ان کے اعمال کا بدلہ دے اور نیک کام کرنے والوں کو اچھا بدلہ عنایت فرمائے

  32. ٱلَّذِينَ يَجْتَنِبُونَ كَبَـٰٓئِرَ ٱلْإِثْمِ وَٱلْفَوَٲحِشَ إِلَّا ٱللَّمَمَ‌ۚ إِنَّ رَبَّكَ وَٲسِعُ ٱلْمَغْفِرَةِ‌ۚ هُوَ أَعْلَمُ بِكُمْ إِذْ أَنشَأَكُم مِّنَ ٱلْأَرْضِ وَإِذْ أَنتُمْ أَجِنَّةٌ فِى بُطُونِ أُمَّهَـٰتِكُمْ‌ۖ فَلَا تُزَكُّوٓاْ أَنفُسَكُمْ‌ۖ هُوَ أَعْلَمُ بِمَنِ ٱتَّقَىٰٓ ۝٣٢

    ان لوگوں کو جو بڑے گناہوں سے بچتے ہیں اور بے حیائی سے بھی۔ سوائے کسی چھوٹے سے گناه کے۔ بیشک تیرا رب بہت کشاده مغفرت واﻻ ہے، وه تمہیں بخوبی جانتا ہے جبکہ اس نے تمہیں زمین سے پیدا کیا اور جبکہ تم اپنی ماؤں کے پیٹ میں بچے تھے۔ پس تم اپنی پاکیزگی آپ بیان نہ کرو، وہی پرہیزگاروں کو خوب جانتا ہے

  33. أَفَرَءَيْتَ ٱلَّذِى تَوَلَّىٰ ۝٣٣

    کیا آپ نے اسے دیکھا جس نے منھ موڑ لیا

  34. وَأَعْطَىٰ قَلِيلاً وَأَكْدَىٰٓ ۝٣٤

    اور بہت کم دیا اور ہاتھ روک لیا

  35. أَعِندَهُۥ عِلْمُ ٱلْغَيْبِ فَهُوَ يَرَىٰٓ ۝٣٥

    کیا اسے علم غیب ہے کہ وه (سب کچھ) دیکھ رہا ہے؟

  36. أَمْ لَمْ يُنَبَّأْ بِمَا فِى صُحُفِ مُوسَىٰ ۝٣٦

    کیا اسے اس چیز کی خبر نہیں دی گئی جو موسیٰ (علیہ السلام) کے

  37. وَإِبْرَٲهِيمَ ٱلَّذِى وَفَّىٰٓ ۝٣٧

    اور وفادار ابراہیم (علیہ السلام) کے صحیفوں میں تھا

  38. أَلَّا تَزِرُ وَازِرَةٌ وِزْرَ أُخْرَىٰ ۝٣٨

    کہ کوئی شخص کسی دوسرے کا بوجھ نہ اٹھائے گا

  39. وَأَن لَّيْسَ لِلْإِنسَـٰنِ إِلَّا مَا سَعَىٰ ۝٣٩

    اور یہ کہ ہر انسان کے لیے صرف وہی ہے جس کی کوشش خود اس نے کی

  40. وَأَنَّ سَعْيَهُۥ سَوْفَ يُرَىٰ ۝٤٠

    اور یہ کہ بیشک اس کی کوشش عنقریب دیکھی جائے گی

  41. ثُمَّ يُجْزَٮٰهُ ٱلْجَزَآءَ ٱلْأَوْفَىٰ ۝٤١

    پھر اسے پورا پورا بدلہ دیا جائے گا

  42. وَأَنَّ إِلَىٰ رَبِّكَ ٱلْمُنتَهَىٰ ۝٤٢

    اور یہ کہ آپ کے رب ہی کی طرف پہنچنا ہے

  43. وَأَنَّهُۥ هُوَ أَضْحَكَ وَأَبْكَىٰ ۝٤٣

    اور یہ کہ وہی ہنساتا ہے اور وہی رﻻتا ہے

  44. وَأَنَّهُۥ هُوَ أَمَاتَ وَأَحْيَا ۝٤٤

    اور یہ کہ وہی مارتا ہے اور جلاتا ہے

  45. وَأَنَّهُۥ خَلَقَ ٱلزَّوْجَيْنِ ٱلذَّكَرَ وَٱلْأُنثَىٰ ۝٤٥

    اور یہ کہ اسی نے جوڑا یعنی نر ماده پیدا کیا ہے

  46. مِن نُّطْفَةٍ إِذَا تُمْنَىٰ ۝٤٦

    نطفہ سے جبکہ وه ٹپکایا جاتا ہے

  47. وَأَنَّ عَلَيْهِ ٱلنَّشْأَةَ ٱلْأُخْرَىٰ ۝٤٧

    اور یہ کہ اسی کے ذمہ دوباره پیدا کرنا ہے

  48. وَأَنَّهُۥ هُوَ أَغْنَىٰ وَأَقْنَىٰ ۝٤٨

    اور یہ کہ وہی مالدار بناتا ہے اور سرمایہ دیتا ہے

  49. وَأَنَّهُۥ هُوَ رَبُّ ٱلشِّعْرَىٰ ۝٤٩

    اور یہ کہ وہی شعریٰ (ستارے) کا رب ہے

  50. وَأَنَّهُۥٓ أَهْلَكَ عَادًا ٱلْأُولَىٰ ۝٥٠

    اور یہ کہ اس نے عاد اول کو ہلاک کیا ہے

  51. وَثَمُودَاْ فَمَآ أَبْقَىٰ ۝٥١

    اور ﺛمود کو بھی (جن میں سے) ایک کو بھی باقی نہ رکھا

  52. وَقَوْمَ نُوحٍ مِّن قَبْلُ‌ۖ إِنَّهُمْ كَانُواْ هُمْ أَظْلَمَ وَأَطْغَىٰ ۝٥٢

    اور اس سے پہلے قوم نوح کو، یقیناً وه بڑے ﻇالم اور سرکش تھے

  53. وَٱلْمُؤْتَفِكَةَ أَهْوَىٰ ۝٥٣

    اور مؤتفکہ (شہر یا الٹی ہوئی بستیوں کو) اسی نے الٹ دیا

  54. فَغَشَّـٰهَا مَا غَشَّىٰ ۝٥٤

    پھر اس پر چھا دیا جو چھایا

  55. فَبِأَىِّ ءَالَآءِ رَبِّكَ تَتَمَارَىٰ ۝٥٥

    پس اے انسان تو اپنے رب کی کس کس نعمت کے بارے میں جھگڑے گا؟

  56. هَـٰذَا نَذِيرٌ مِّنَ ٱلنُّذُرِ ٱلْأُولَىٰٓ ۝٥٦

    یہ (نبی) ڈرانے والے ہیں پہلے ڈرانے والوں میں سے

  57. أَزِفَتِ ٱلْأَزِفَةُ ۝٥٧

    آنے والی گھڑی قریب آ گئی ہے

  58. لَيْسَ لَهَا مِن دُونِ ٱللَّهِ كَاشِفَةٌ ۝٥٨

    اللہ کے سوا اس کا (وقت معین پر کھول) دکھانے واﻻ اور کوئی نہیں

  59. أَفَمِنْ هَـٰذَا ٱلْحَدِيثِ تَعْجَبُونَ ۝٥٩

    پس کیا تم اس بات سے تعجب کرتے ہو

  60. وَتَضْحَكُونَ وَلَا تَبْكُونَ ۝٦٠

    اور ہنس رہے ہو؟ روتے نہیں؟
