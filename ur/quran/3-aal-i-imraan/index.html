بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ

  1. الٓمٓ ۝١

    الم

  2. ٱللَّهُ لَآ إِلَـٰهَ إِلَّا هُوَ ٱلْحَىُّ ٱلْقَيُّومُ ۝٢

    اللہ تعالیٰ وه ہے جس کے سوا کوئی معبود نہیں، جو زنده اور سب کا نگہبان ہے

  3. نَزَّلَ عَلَيْكَ ٱلْكِتَـٰبَ بِٱلْحَقِّ مُصَدِّقًا لِّمَا بَيْنَ يَدَيْهِ وَأَنزَلَ ٱلتَّوْرَٮٰةَ وَٱلْإِنجِيلَ ۝٣

    جس نے آپ پر حق کے ساتھ اس کتاب کو نازل فرمایا ہے، جو اپنے سے پہلے کی تصدیق کرنے والی ہے، اسی نے اس سے پہلے تورات اور انجیل کو اتارا تھا

  4. مِن قَبْلُ هُدًى لِّلنَّاسِ وَأَنزَلَ ٱلْفُرْقَانَ‌ۗ إِنَّ ٱلَّذِينَ كَفَرُواْ بِـَٔـايَـٰتِ ٱللَّهِ لَهُمْ عَذَابٌ شَدِيدٌ‌ۗ وَٱللَّهُ عَزِيزٌ ذُو ٱنتِقَامٍ ۝٤

    اس سے پہلے، لوگوں کو ہدایت کرنے والی بنا کر، اور قرآن بھی اسی نے اتارا، جو لوگ اللہ تعالیٰ کی آیتوں سے کفر کرتے ہیں ان کے لئے سخت عذاب ہے اور اللہ تعالیٰ غالب ہے، بدلہ لینے واﻻ ہے

  5. إِنَّ ٱللَّهَ لَا يَخْفَىٰ عَلَيْهِ شَىْءٌ فِى ٱلْأَرْضِ وَلَا فِى ٱلسَّمَآءِ ۝٥

    یقیناً اللہ تعالیٰ پر زمین وآسمان کی کوئی چیز پوشیده نہیں

  6. هُوَ ٱلَّذِى يُصَوِّرُكُمْ فِى ٱلْأَرْحَامِ كَيْفَ يَشَآءُ‌ۚ لَآ إِلَـٰهَ إِلَّا هُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ۝٦

    وه ماں کے پیٹ میں تمہاری صورتیں جس طرح چاہتا ہے بناتا ہے۔ اس کے سواکوئی معبود برحق نہیں وه غالب ہے، حکمت واﻻ ہے

  7. هُوَ ٱلَّذِىٓ أَنزَلَ عَلَيْكَ ٱلْكِتَـٰبَ مِنْهُ ءَايَـٰتٌ مُّحْكَمَـٰتٌ هُنَّ أُمُّ ٱلْكِتَـٰبِ وَأُخَرُ مُتَشَـٰبِهَـٰتٌ‌ۖ فَأَمَّا ٱلَّذِينَ فِى قُلُوبِهِمْ زَيْغٌ فَيَتَّبِعُونَ مَا تَشَـٰبَهَ مِنْهُ ٱبْتِغَآءَ ٱلْفِتْنَةِ وَٱبْتِغَآءَ تَأْوِيلِهِۦ‌ۗ وَمَا يَعْلَمُ تَأْوِيلَهُۥٓ إِلَّا ٱللَّهُ‌ۗ وَٱلرَّٲسِخُونَ فِى ٱلْعِلْمِ يَقُولُونَ ءَامَنَّا بِهِۦ كُلٌّ مِّنْ عِندِ رَبِّنَا‌ۗ وَمَا يَذَّكَّرُ إِلَّآ أُوْلُواْ ٱلْأَلْبَـٰبِ ۝٧

    وہی اللہ تعالیٰ ہے جس نے تجھ پر کتاب اتاری جس میں واضح مضبوط آیتیں ہیں جو اصل کتاب ہیں اوربعض متشابہ آیتیں ہیں۔ پس جن کے دلوں میں کجی ہے وه تواس کی متشابہ آیتوں کے پیچھے لگ جاتے ہیں، فتنے کی طلب اور ان کی مراد کی جستجو کے لئے، حاﻻنکہ ان کے حقیقی مراد کو سوائے اللہ تعالیٰ کے کوئی نہیں جانتا اور پختہ ومضبوط علم والے یہی کہتے ہیں کہ ہم تو ان پر ایمان ﻻچکے، یہ ہمارے رب کی طرف سے ہیں اور نصیحت تو صرف عقلمند حاصل کرتے ہیں۔

  8. رَبَّنَا لَا تُزِغْ قُلُوبَنَا بَعْدَ إِذْ هَدَيْتَنَا وَهَبْ لَنَا مِن لَّدُنكَ رَحْمَةً‌ۚ إِنَّكَ أَنتَ ٱلْوَهَّابُ ۝٨

    اے ہمارے رب! ہمیں ہدایت دینے کے بعد ہمارے دل ٹیڑھے نہ کردے اور ہمیں اپنے پاس سے رحمت عطا فرما، یقیناً تو ہی بہت بڑی عطا دینے واﻻ ہے

  9. رَبَّنَآ إِنَّكَ جَامِعُ ٱلنَّاسِ لِيَوْمٍ لَّا رَيْبَ فِيهِ‌ۚ إِنَّ ٱللَّهَ لَا يُخْلِفُ ٱلْمِيعَادَ ۝٩

    اے ہمارے رب! تو یقیناً لوگوں کوایک دن جمع کرنے واﻻ ہے جس کے آنے میں کوئی شک نہیں، یقیناً اللہ تعالیٰ وعده خلافی نہیں کرتا

  10. إِنَّ ٱلَّذِينَ كَفَرُواْ لَن تُغْنِىَ عَنْهُمْ أَمْوَٲلُهُمْ وَلَآ أَوْلَـٰدُهُم مِّنَ ٱللَّهِ شَيْــًٔا‌ۖ وَأُوْلَـٰٓئِكَ هُمْ وَقُودُ ٱلنَّارِ ۝١٠

    کافروں کوان کے مال اور ان کی اوﻻد اللہ تعالیٰ (کے عذاب) سے چھڑانے میں کچھ کام نہ آئیں گی، یہ تو جہنم کا ایندھن ہی ہیں

  11. كَدَأْبِ ءَالِ فِرْعَوْنَ وَٱلَّذِينَ مِن قَبْلِهِمْ‌ۚ كَذَّبُواْ بِـَٔـايَـٰتِنَا فَأَخَذَهُمُ ٱللَّهُ بِذُنُوبِهِمْ‌ۗ وَٱللَّهُ شَدِيدُ ٱلْعِقَابِ ۝١١

    جیسا آل فرعون کاحال ہوا، اور ان کا جو ان سے پہلے تھے، انہوں نے ہماری آیتوں کوجھٹلایا، پھر اللہ تعالیٰ نے بھی انہیں ان کے گناہوں پر پکڑ لیا، اور اللہ تعالیٰ سخت عذاب واﻻ ہے

  12. قُل لِّلَّذِينَ كَفَرُواْ سَتُغْلَبُونَ وَتُحْشَرُونَ إِلَىٰ جَهَنَّمَ‌ۚ وَبِئْسَ ٱلْمِهَادُ ۝١٢

    کافروں سے کہہ دیجئے! کہ تم عنقریب مغلوب کئے جاؤ گے اور جہنم کی طرف جمع کئے جاؤ گے اور وه برا ٹھکانا ہے

  13. قَدْ كَانَ لَكُمْ ءَايَةٌ فِى فِئَتَيْنِ ٱلْتَقَتَا‌ۖ فِئَةٌ تُقَـٰتِلُ فِى سَبِيلِ ٱللَّهِ وَأُخْرَىٰ كَافِرَةٌ يَرَوْنَهُم مِّثْلَيْهِمْ رَأْىَ ٱلْعَيْنِ‌ۚ وَٱللَّهُ يُؤَيِّدُ بِنَصْرِهِۦ مَن يَشَآءُ‌ۗ إِنَّ فِى ذَٲلِكَ لَعِبْرَةً لِّأُوْلِى ٱلْأَبْصَـٰرِ ۝١٣

    یقیناً تمہارے لئے عبرت کی نشانی تھی ان دو جماعتوں میں جو گتھ گئی تھیں، ایک جماعت تو اللہ تعالیٰ کی راه میں لڑ رہی تھی اور دوسرا گروه کافروں کا تھا وه انہیں اپنی آنکھوں سے اپنے سے دگنا دیکھتے تھے اور اللہ تعالیٰ جسے چاہے اپنی مدد سے قوی کرتا ہے۔ یقیناً اس میں آنکھوں والوں کے لئے بڑی عبرت ہے

  14. زُيِّنَ لِلنَّاسِ حُبُّ ٱلشَّهَوَٲتِ مِنَ ٱلنِّسَآءِ وَٱلْبَنِينَ وَٱلْقَنَـٰطِيرِ ٱلْمُقَنطَرَةِ مِنَ ٱلذَّهَبِ وَٱلْفِضَّةِ وَٱلْخَيْلِ ٱلْمُسَوَّمَةِ وَٱلْأَنْعَـٰمِ وَٱلْحَرْثِ‌ۗ ذَٲلِكَ مَتَـٰعُ ٱلْحَيَوٲةِ ٱلدُّنْيَا‌ۖ وَٱللَّهُ عِندَهُۥ حُسْنُ ٱلْمَـَٔـابِ ۝١٤

    مرغوب چیزوں کی محبت لوگوں کے لئے مزین کر دی گئی ہے، جیسے عورتیں اور بیٹے اور سونے اور چاندی کے جمع کئے ہوئے خزانے اور نشاندار گھوڑے اور چوپائے اور کھیتی، یہ دنیا کی زندگی کا سامان ہے اور لوٹنے کا اچھا ٹھکانا تو اللہ تعالیٰ ہی کے پاس ہے

  15. ۞ قُلْ أَؤُنَبِّئُكُم بِخَيْرٍ مِّن ذَٲلِكُمْ‌ۚ لِلَّذِينَ ٱتَّقَوْاْ عِندَ رَبِّهِمْ جَنَّـٰتٌ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ خَـٰلِدِينَ فِيهَا وَأَزْوَٲجٌ مُّطَهَّرَةٌ وَرِضْوَٲنٌ مِّنَ ٱللَّهِ‌ۗ وَٱللَّهُ بَصِيرُۢ بِٱلْعِبَادِ ۝١٥

    آپ کہہ دیجئے! کیا میں تمہیں اس سے بہت ہی بہتر چیز بتاؤں؟ تقویٰ والوں کے لئے ان کے رب تعالیٰ کے پاس جنتیں ہیں جن کے نیچے نہریں بہہ رہی ہیں جن میں وه ہمیشہ رہیں گے اور پاکیزه بیویاں اور اللہ تعالیٰ کی رضامندی ہے، سب بندے اللہ تعالیٰ کی نگاه میں ہیں

  16. ٱلَّذِينَ يَقُولُونَ رَبَّنَآ إِنَّنَآ ءَامَنَّا فَٱغْفِرْ لَنَا ذُنُوبَنَا وَقِنَا عَذَابَ ٱلنَّارِ ۝١٦

    جو کہتے ہیں کہ اے ہمارے رب! ہم ایمان ﻻ چکے اس لئے ہمارے گناه معاف فرما اور ہمیں آگ کے عذاب سے بچا

  17. ٱلصَّـٰبِرِينَ وَٱلصَّـٰدِقِينَ وَٱلْقَـٰنِتِينَ وَٱلْمُنفِقِينَ وَٱلْمُسْتَغْفِرِينَ بِٱلْأَسْحَارِ ۝١٧

    جو صبر کرنے والے اور سچ بولنے والے اور فرمانبرداری کرنے والے اور اللہ کی راه میں خرچ کرنے والے اور پچھلی رات کو بخشش مانگنے والے ہیں

  18. شَهِدَ ٱللَّهُ أَنَّهُۥ لَآ إِلَـٰهَ إِلَّا هُوَ وَٱلْمَلَـٰٓئِكَةُ وَأُوْلُواْ ٱلْعِلْمِ قَآئِمَۢا بِٱلْقِسْطِ‌ۚ لَآ إِلَـٰهَ إِلَّا هُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ۝١٨

    اللہ تعالیٰ، فرشتے اور اہل علم اس بات کی گواہی دیتے ہیں کہ اللہ کے سوا کوئی معبود نہیں اور وه عدل کو قائم رکھنے واﻻ ہے، اس غالب اور حکمت والے کے سوا کوئی عبادت کے ﻻئق نہیں

  19. إِنَّ ٱلدِّينَ عِندَ ٱللَّهِ ٱلْإِسْلَـٰمُ‌ۗ وَمَا ٱخْتَلَفَ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ إِلَّا مِنۢ بَعْدِ مَا جَآءَهُمُ ٱلْعِلْمُ بَغْيَۢا بَيْنَهُمْ‌ۗ وَمَن يَكْفُرْ بِـَٔـايَـٰتِ ٱللَّهِ فَإِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ ۝١٩

    بے شک اللہ تعالیٰ کے نزدیک دین اسلام ہی ہے، اور اہل کتاب نے اپنے پاس علم آجانے کے بعد آپس کی سرکشی اور حسد کی بنا پر ہی اختلاف کیا ہے اور اللہ تعالیٰ کی آیتوں کے ساتھ جو بھی کفر کرے اللہ تعالیٰ اس کا جلد حساب لینے واﻻ ہے

  20. فَإِنْ حَآجُّوكَ فَقُلْ أَسْلَمْتُ وَجْهِىَ لِلَّهِ وَمَنِ ٱتَّبَعَنِ‌ۗ وَقُل لِّلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ وَٱلْأُمِّيِّــۧنَ ءَأَسْلَمْتُمْ‌ۚ فَإِنْ أَسْلَمُواْ فَقَدِ ٱهْتَدَواْ‌ۖ وَّإِن تَوَلَّوْاْ فَإِنَّمَا عَلَيْكَ ٱلْبَلَـٰغُ‌ۗ وَٱللَّهُ بَصِيرُۢ بِٱلْعِبَادِ ۝٢٠

    پھر بھی اگر یہ آپ سے جھگڑیں توآپ کہہ دیں کہ میں اور میرے تابعداروں نے اللہ تعالیٰ کے سامنے اپنا سر تسلیم خم کر دیا ہے اور اہل کتاب سے اور انپڑھ لوگوں سے کہہ دیجیئے! کہ کیا تم بھی اطاعت کرتے ہو؟ پس اگر یہ بھی تابعدار بن جائیں تو یقیناً ہداہت والے ہیں اور اگر یہ روگردانی کریں، تو آپ پر صرف پہنچا دینا ہے اور اللہ بندوں کو خوب دیکھ بھال رہا ہے

  21. إِنَّ ٱلَّذِينَ يَكْفُرُونَ بِـَٔـايَـٰتِ ٱللَّهِ وَيَقْتُلُونَ ٱلنَّبِيِّــۧنَ بِغَيْرِ حَقٍّ وَيَقْتُلُونَ ٱلَّذِينَ يَأْمُرُونَ بِٱلْقِسْطِ مِنَ ٱلنَّاسِ فَبَشِّرْهُم بِعَذَابٍ أَلِيمٍ ۝٢١

    جو لوگ اللہ تعالیٰ کی آیتوں سے کفر کرتے ہیں اورناحق نبیوں کو قتل کر ڈالتے ہیں اور جو لوگ عدل وانصاف کی بات کہیں انہیں بھی قتل کر ڈالتے ہیں، تو اے نبی! انہیں دردناک عذاب کی خبردے دیجئے

  22. أُوْلَـٰٓئِكَ ٱلَّذِينَ حَبِطَتْ أَعْمَـٰلُهُمْ فِى ٱلدُّنْيَا وَٱلْأَخِرَةِ وَمَا لَهُم مِّن نَّـٰصِرِينَ ۝٢٢

    ان کے اعمال دنیا وآخرت میں غارت ہیں اور ان کا کوئی مددگار نہیں

  23. أَلَمْ تَرَ إِلَى ٱلَّذِينَ أُوتُواْ نَصِيبًا مِّنَ ٱلْكِتَـٰبِ يُدْعَوْنَ إِلَىٰ كِتَـٰبِ ٱللَّهِ لِيَحْكُمَ بَيْنَهُمْ ثُمَّ يَتَوَلَّىٰ فَرِيقٌ مِّنْهُمْ وَهُم مُّعْرِضُونَ ۝٢٣

    کیا آپ نے انہیں نہیں دیکھا جنہیں ایک حصہ کتاب کا دیا گیا ہے وه اپنے آپس کے فیصلوں کے لئے اللہ تعالیٰ کی کتاب کی طرف بلائے جاتے ہیں، پھر بھی ایک جماعت ان کی منھ پھیر کر لوٹ جاتی ہے

  24. ذَٲلِكَ بِأَنَّهُمْ قَالُواْ لَن تَمَسَّنَا ٱلنَّارُ إِلَّآ أَيَّامًا مَّعْدُودَٲتٍ‌ۖ وَغَرَّهُمْ فِى دِينِهِم مَّا كَانُواْ يَفْتَرُونَ ۝٢٤

    اس کی وجہ ان کا یہ کہنا کہ ہمیں تو گنے چنے چند دن ہی آگ جلائے گی، ان کی گڑھی گڑھائی باتوں نے انہیں ان کے دین کے بارے میں دھوکے میں ڈال رکھا ہے

  25. فَكَيْفَ إِذَا جَمَعْنَـٰهُمْ لِيَوْمٍ لَّا رَيْبَ فِيهِ وَوُفِّيَتْ كُلُّ نَفْسٍ مَّا كَسَبَتْ وَهُمْ لَا يُظْلَمُونَ ۝٢٥

    پس کیا حال ہوگا جبکہ ہم انہیں اس دن جمع کرینگے؟ جس کے آنے میں کوئی شک نہیں اور ہر شخص اپنا اپنا کیا پورا پورا دیاجائے گا اور ان پر ﻇلم نہ کیاجائے گا

  26. قُلِ ٱللَّهُمَّ مَـٰلِكَ ٱلْمُلْكِ تُؤْتِى ٱلْمُلْكَ مَن تَشَآءُ وَتَنزِعُ ٱلْمُلْكَ مِمَّن تَشَآءُ وَتُعِزُّ مَن تَشَآءُ وَتُذِلُّ مَن تَشَآءُ‌ۖ بِيَدِكَ ٱلْخَيْرُ‌ۖ إِنَّكَ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ۝٢٦

    آپ کہہ دیجئے اے اللہ! اے تمام جہان کے مالک! تو جسے چاہے بادشاہی دے اور جس سے چاہے سلطنت چھین لے اور تو جسے چاہے عزت دے اورجسے چاہے ذلت دے، تیرے ہی ہاتھ میں سب بھلائیاں ہیں، بے شک تو ہر چیز پر قادر ہے

  27. تُولِجُ ٱلَّيْلَ فِى ٱلنَّهَارِ وَتُولِجُ ٱلنَّهَارَ فِى ٱلَّيْلِ‌ۖ وَتُخْرِجُ ٱلْحَىَّ مِنَ ٱلْمَيِّتِ وَتُخْرِجُ ٱلْمَيِّتَ مِنَ ٱلْحَىِّ‌ۖ وَتَرْزُقُ مَن تَشَآءُ بِغَيْرِ حِسَابٍ ۝٢٧

    تو ہی رات کو دن میں داخل کرتا ہے اور دن کو رات میں لے جاتا ہے، تو ہی بے جان سے جاندار پیدا کرتا ہے اور تو ہی جاندار سے بے جان پیدا کرتا ہے، تو ہی ہے کہ جسے چاہتا ہے بے شمار روزی دیتا ہے

  28. لَّا يَتَّخِذِ ٱلْمُؤْمِنُونَ ٱلْكَـٰفِرِينَ أَوْلِيَآءَ مِن دُونِ ٱلْمُؤْمِنِينَ‌ۖ وَمَن يَفْعَلْ ذَٲلِكَ فَلَيْسَ مِنَ ٱللَّهِ فِى شَىْءٍ إِلَّآ أَن تَتَّقُواْ مِنْهُمْ تُقَـٰةً‌ۗ وَيُحَذِّرُكُمُ ٱللَّهُ نَفْسَهُۥ‌ۗ وَإِلَى ٱللَّهِ ٱلْمَصِيرُ ۝٢٨

    مومنوں کو چاہئے کہ ایمان والوں کو چھوڑ کر کافروں کو اپنا دوست نہ بنائیں اور جوایسا کرے گا وه اللہ تعالیٰ کی کسی حمایت میں نہیں مگر یہ کہ ان کے شر سے کسی طرح بچاؤ مقصود ہو، اور اللہ تعالیٰ خود تمہیں اپنی ذات سے ڈرا رہا ہے اور اللہ تعالیٰ ہی کی طرف لوٹ جانا ہے

  29. قُلْ إِن تُخْفُواْ مَا فِى صُدُورِكُمْ أَوْ تُبْدُوهُ يَعْلَمْهُ ٱللَّهُ‌ۗ وَيَعْلَمُ مَا فِى ٱلسَّمَـٰوَٲتِ وَمَا فِى ٱلْأَرْضِ‌ۗ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ۝٢٩

    کہہ دیجئے! کہ خواه تم اپنے سینوں کی باتیں چھپاؤ خواه ﻇاہر کرو اللہ تعالیٰ (بہرحال) جانتا ہے، آسمانوں اور زمین میں جو کچھ ہے سب اسے معلوم ہے اور اللہ تعالیٰ ہر چیز پر قادر ہے

  30. يَوْمَ تَجِدُ كُلُّ نَفْسٍ مَّا عَمِلَتْ مِنْ خَيْرٍ مُّحْضَرًا وَمَا عَمِلَتْ مِن سُوٓءٍ تَوَدُّ لَوْ أَنَّ بَيْنَهَا وَبَيْنَهُۥٓ أَمَدَۢا بَعِيدًا‌ۗ وَيُحَذِّرُكُمُ ٱللَّهُ نَفْسَهُۥ‌ۗ وَٱللَّهُ رَءُوفُۢ بِٱلْعِبَادِ ۝٣٠

    جس دن ہر نفس (شخص) اپنی کی ہوئی نیکیوں کو اور اپنی کی ہوئی برائیوں کو موجود پالے گا، آرزو کرے گا کہ کاش! اس کے اور برائیوں کے درمیان بہت ہی دوری ہوتی۔ اللہ تعالیٰ تمہیں اپنی ذات سے ڈرا رہا ہے اور اللہ تعالیٰ اپنے بندوں پر بڑا ہی مہربان ہے

  31. قُلْ إِن كُنتُمْ تُحِبُّونَ ٱللَّهَ فَٱتَّبِعُونِى يُحْبِبْكُمُ ٱللَّهُ وَيَغْفِرْ لَكُمْ ذُنُوبَكُمْ‌ۗ وَٱللَّهُ غَفُورٌ رَّحِيمٌ ۝٣١

    کہہ دیجئے! اگر تم اللہ تعالیٰ سے محبت رکھتے ہو تو میری تابعداری کرو، خود اللہ تعالیٰ تم سے محبت کرے گا اور تمہارے گناه معاف فرما دے گا اور اللہ تعالیٰ بڑا بخشنے واﻻ مہربان ہے

  32. قُلْ أَطِيعُواْ ٱللَّهَ وَٱلرَّسُولَ‌ۖ فَإِن تَوَلَّوْاْ فَإِنَّ ٱللَّهَ لَا يُحِبُّ ٱلْكَـٰفِرِينَ ۝٣٢

    کہہ دیجئے! کہ اللہ تعالیٰ اور رسول کی اطاعت کرو، اگر یہ منھ پھیر لیں تو بے شک اللہ تعالیٰ کافروں سے محبت نہیں کرتا

  33. ۞ إِنَّ ٱللَّهَ ٱصْطَفَىٰٓ ءَادَمَ وَنُوحًا وَءَالَ إِبْرَٲهِيمَ وَءَالَ عِمْرَٲنَ عَلَى ٱلْعَـٰلَمِينَ ۝٣٣

    بے شک اللہ تعالیٰ نے تمام جہان کے لوگوں میں سے آدم (علیہ السلام) کو اور نوح (علیہ السلام) کو، ابراہیم (علیہ السلام) کے خاندان اور عمران کے خاندان کو منتخب فرما لیا

  34. ذُرِّيَّةَۢ بَعْضُهَا مِنۢ بَعْضٍ‌ۗ وَٱللَّهُ سَمِيعٌ عَلِيمٌ ۝٣٤

    کہ یہ سب آپس میں ایک دوسرے کی نسل سے ہیں اور اللہ تعالیٰ سنتا جانتا ہے

  35. إِذْ قَالَتِ ٱمْرَأَتُ عِمْرَٲنَ رَبِّ إِنِّى نَذَرْتُ لَكَ مَا فِى بَطْنِى مُحَرَّرًا فَتَقَبَّلْ مِنِّىٓ‌ۖ إِنَّكَ أَنتَ ٱلسَّمِيعُ ٱلْعَلِيمُ ۝٣٥

    جب عمران کی بیوی نے کہا کہ اے میرے رب! میرے پیٹ میں جو کچھ ہے، اسے میں نے تیرے نام آزاد کرنے کی نذر مانی، تو میری طرف سے قبول فرما! یقیناً تو خوب سننے واﻻ اور پوری طرح جاننے واﻻ ہے

  36. فَلَمَّا وَضَعَتْهَا قَالَتْ رَبِّ إِنِّى وَضَعْتُهَآ أُنثَىٰ وَٱللَّهُ أَعْلَمُ بِمَا وَضَعَتْ وَلَيْسَ ٱلذَّكَرُ كَٱلْأُنثَىٰ‌ۖ وَإِنِّى سَمَّيْتُهَا مَرْيَمَ وَإِنِّىٓ أُعِيذُهَا بِكَ وَذُرِّيَّتَهَا مِنَ ٱلشَّيْطَـٰنِ ٱلرَّجِيمِ ۝٣٦

    جب بچی کو جنا تو کہنے لگیں کہ پروردگار! مجھے تو لڑکی ہوئی، اللہ تعالیٰ کو خوب معلوم ہے کہ کیا اوﻻد ہوئی ہے اور لڑکا لڑکی جیسا نہیں میں نے اس کا نام مریم رکھا، میں اسے اور اس کی اوﻻد کو شیطان مردود سے تیری پناه میں دیتی ہوں

  37. فَتَقَبَّلَهَا رَبُّهَا بِقَبُولٍ حَسَنٍ وَأَنۢبَتَهَا نَبَاتًا حَسَنًا وَكَفَّلَهَا زَكَرِيَّا‌ۖ كُلَّمَا دَخَلَ عَلَيْهَا زَكَرِيَّا ٱلْمِحْرَابَ وَجَدَ عِندَهَا رِزْقًا‌ۖ قَالَ يَـٰمَرْيَمُ أَنَّىٰ لَكِ هَـٰذَا‌ۖ قَالَتْ هُوَ مِنْ عِندِ ٱللَّهِ‌ۖ إِنَّ ٱللَّهَ يَرْزُقُ مَن يَشَآءُ بِغَيْرِ حِسَابٍ ۝٣٧

    پس اسے اس کے پروردگار نے اچھی طرح قبول فرمایا اور اسے بہترین پرورش دی۔ اس کی خیر خبر لینے واﻻ زکریا (علیہ السلام) کو بنایا، جب کبھی زکریا (علیہ السلام) ان کے حجرے میں جاتے ان کے پاس روزی رکھی ہوئی پاتے، وه پوچھتے اے مریم! یہ روزی تمہارے پاس کہاں سے آئی؟ وه جواب دیتیں کہ یہ اللہ تعالیٰ کے پاس سے ہے، بے شک اللہ تعالیٰ جسے چاہے بے شمار روزی دے

  38. هُنَالِكَ دَعَا زَكَرِيَّا رَبَّهُۥ‌ۖ قَالَ رَبِّ هَبْ لِى مِن لَّدُنكَ ذُرِّيَّةً طَيِّبَةً‌ۖ إِنَّكَ سَمِيعُ ٱلدُّعَآءِ ۝٣٨

    اسی جگہ زکریا (علیہ السلام) نے اپنے رب سے دعا کی، کہا کہ اے میرے پروردگار! مجھے اپنے پاس سے پاکیزه اوﻻد عطا فرما، بے شک تو دعا کا سننے واﻻ ہے

  39. فَنَادَتْهُ ٱلْمَلَـٰٓئِكَةُ وَهُوَ قَآئِمٌ يُصَلِّى فِى ٱلْمِحْرَابِ أَنَّ ٱللَّهَ يُبَشِّرُكَ بِيَحْيَىٰ مُصَدِّقَۢا بِكَلِمَةٍ مِّنَ ٱللَّهِ وَسَيِّدًا وَحَصُورًا وَنَبِيًّا مِّنَ ٱلصَّـٰلِحِينَ ۝٣٩

    پس فرشتوں نے انہیں آواز دی، جب کہ وه حجرے میں کھڑے نماز پڑھ رہے تھے، کہ اللہ تعالیٰ تجھے یحيٰ کی یقینی خوشخبری دیتا ہے جو اللہ تعالیٰ کے کلمہ کی تصدیق کرنے واﻻ، سردار، ضابط نفس اور نبی ہے نیک لوگوں میں سے

  40. قَالَ رَبِّ أَنَّىٰ يَكُونُ لِى غُلَـٰمٌ وَقَدْ بَلَغَنِىَ ٱلْكِبَرُ وَٱمْرَأَتِى عَاقِرٌ‌ۖ قَالَ كَذَٲلِكَ ٱللَّهُ يَفْعَلُ مَا يَشَآءُ ۝٤٠

    کہنے لگے اے میرے رب! میرے ہاں بچہ کیسے ہوگا؟ میں بالکل بوڑھا ہو گیا ہوں اور میری بیوی بانجھ ہے، فرمایا، اسی طرح اللہ تعالیٰ جو چاہے کرتا ہے

  41. قَالَ رَبِّ ٱجْعَل لِّىٓ ءَايَةً‌ۖ قَالَ ءَايَتُكَ أَلَّا تُكَلِّمَ ٱلنَّاسَ ثَلَـٰثَةَ أَيَّامٍ إِلَّا رَمْزًا‌ۗ وَٱذْكُر رَّبَّكَ كَثِيرًا وَسَبِّحْ بِٱلْعَشِىِّ وَٱلْإِبْكَـٰرِ ۝٤١

    کہنے لگے پروردگار! میرے لئے اس کی کوئی نشانی مقرر کر دے، فرمایا، نشانی یہ ہے کہ تین دن تک تو لوگوں سے بات نہ کر سکے گا، صرف اشارے سے سمجھائے گا، تو اپنے رب کا ذکر کثرت سے کر اور صبح وشام اسی کی تسبیح بیان کرتا ره!

  42. وَإِذْ قَالَتِ ٱلْمَلَـٰٓئِكَةُ يَـٰمَرْيَمُ إِنَّ ٱللَّهَ ٱصْطَفَـٰكِ وَطَهَّرَكِ وَٱصْطَفَـٰكِ عَلَىٰ نِسَآءِ ٱلْعَـٰلَمِينَ ۝٤٢

    اور جب فرشتوں نے کہا، اے مریم! اللہ تعالیٰ نے تجھے برگزیده کر لیا اور تجھے پاک کر دیا اور سارے جہان کی عورتوں میں سے تیرا انتخاب کر لیا

  43. يَـٰمَرْيَمُ ٱقْنُتِى لِرَبِّكِ وَٱسْجُدِى وَٱرْكَعِى مَعَ ٱلرَّٲكِعِينَ ۝٤٣

    اے مریم! تو اپنے رب کی اطاعت کر اور سجده کر اور رکوع کرنے والوں کے ساتھ رکوع کر

  44. ذَٲلِكَ مِنْ أَنۢبَآءِ ٱلْغَيْبِ نُوحِيهِ إِلَيْكَ‌ۚ وَمَا كُنتَ لَدَيْهِمْ إِذْ يُلْقُونَ أَقْلَـٰمَهُمْ أَيُّهُمْ يَكْفُلُ مَرْيَمَ وَمَا كُنتَ لَدَيْهِمْ إِذْ يَخْتَصِمُونَ ۝٤٤

    یہ غیب کی خبروں میں سے ہے جسے ہم تیری طرف وحی سے پہنچاتے ہیں، تو ان کے پاس نہ تھا جب کہ وه اپنے قلم ڈال رہے تھے کہ مریم کو ان میں سے کون پالے گا؟ اور نہ تو ان کے جھگڑنے کے وقت ان کے پاس تھا

  45. إِذْ قَالَتِ ٱلْمَلَـٰٓئِكَةُ يَـٰمَرْيَمُ إِنَّ ٱللَّهَ يُبَشِّرُكِ بِكَلِمَةٍ مِّنْهُ ٱسْمُهُ ٱلْمَسِيحُ عِيسَى ٱبْنُ مَرْيَمَ وَجِيهًا فِى ٱلدُّنْيَا وَٱلْأَخِرَةِ وَمِنَ ٱلْمُقَرَّبِينَ ۝٤٥

    جب فرشتوں نے کہا اے مریم اللہ تعالیٰ تجھے اپنے ایک کلمے کی خوشخبری دیتا ہے جس کا نام مسیح عیسیٰ بن مریم ہے جو دنیا اور آخرت میں ذیعزت ہے اور وه میرے مقربین میں سے ہے

  46. وَيُكَلِّمُ ٱلنَّاسَ فِى ٱلْمَهْدِ وَكَهْلاً وَمِنَ ٱلصَّـٰلِحِينَ ۝٤٦

    وه لوگوں سے اپنے گہوارے میں باتیں کرے گا اور ادھیڑ عمر میں بھی اور وه نیک لوگوں میں سے ہوگا

  47. قَالَتْ رَبِّ أَنَّىٰ يَكُونُ لِى وَلَدٌ وَلَمْ يَمْسَسْنِى بَشَرٌ‌ۖ قَالَ كَذَٲلِكِ ٱللَّهُ يَخْلُقُ مَا يَشَآءُ‌ۚ إِذَا قَضَىٰٓ أَمْرًا فَإِنَّمَا يَقُولُ لَهُۥ كُن فَيَكُونُ ۝٤٧

    کہنے لگیں الٰہی مجھے لڑکا کیسے ہوگا؟ حاﻻنکہ مجھے تو کسی انسان نے ہاتھ بھی نہیں لگایا، فرشتے نے کہا، اسی طرح اللہ تعالیٰ جو چاہے پیدا کرتا ہے، جب کبھی وه کسی کام کو کرنا جاہتا ہے تو صرف یہ کہہ دیتا ہے کہ ہو جا! تو وه ہو جاتا ہے

  48. وَيُعَلِّمُهُ ٱلْكِتَـٰبَ وَٱلْحِكْمَةَ وَٱلتَّوْرَٮٰةَ وَٱلْإِنجِيلَ ۝٤٨

    اللہ تعالیٰ اسے لکھنا اور حکمت اور توراة اورانجیل سکھائے گا

  49. وَرَسُولاً إِلَىٰ بَنِىٓ إِسْرَٲٓءِيلَ أَنِّى قَدْ جِئْتُكُم بِـَٔـايَةٍ مِّن رَّبِّكُمْ‌ۖ أَنِّىٓ أَخْلُقُ لَكُم مِّنَ ٱلطِّينِ كَهَيْــَٔةِ ٱلطَّيْرِ فَأَنفُخُ فِيهِ فَيَكُونُ طَيْرَۢا بِإِذْنِ ٱللَّهِ‌ۖ وَأُبْرِئُ ٱلْأَكْمَهَ وَٱلْأَبْرَصَ وَأُحْىِ ٱلْمَوْتَىٰ بِإِذْنِ ٱللَّهِ‌ۖ وَأُنَبِّئُكُم بِمَا تَأْكُلُونَ وَمَا تَدَّخِرُونَ فِى بُيُوتِكُمْ‌ۚ إِنَّ فِى ذَٲلِكَ لَأَيَةً لَّكُمْ إِن كُنتُم مُّؤْمِنِينَ ۝٤٩

    اور وه بنی اسرائیل کی طرف رسول ہوگا، کہ میں تمہارے پاس تمہارے رب کی نشانی ﻻیا ہوں، میں تمہارے لئے پرندے کی شکل کی طرح مٹی کا پرنده بناتا ہوں، پھر اس میں پھونک مارتا ہوں تو وه اللہ تعالیٰ کے حکم سے پرنده بن جاتاہے اور اللہ تعالیٰ کے حکم سے میں مادرزاد اندھے کو اور کوڑھی کو اچھا کر دیتا ہوں اور مرددوں کو زندہ کرتا ہوں اور جو کچھ تم کھاؤ اور جو اپنے گھروں میں ذخیره کرو میں تمہیں بتا دیتا ہوں، اس میں تمہارے لئے بڑی نشانی ہے، اگر تم ایمان ﻻنے والے ہو

  50. وَمُصَدِّقًا لِّمَا بَيْنَ يَدَىَّ مِنَ ٱلتَّوْرَٮٰةِ وَلِأُحِلَّ لَكُم بَعْضَ ٱلَّذِى حُرِّمَ عَلَيْكُمْ‌ۚ وَجِئْتُكُم بِـَٔـايَةٍ مِّن رَّبِّكُمْ فَٱتَّقُواْ ٱللَّهَ وَأَطِيعُونِ ۝٥٠

    اور میں توراة کی تصدیق کرنے واﻻ ہوں جو میرے سامنے ہے اور میں اس لئے آیا ہوں کہ تم پر بعض وه چیزیں حلال کروں جو تم پر حرام کر دی گئی ہیں اور میں تمہارے پاس تمہارے رب کی نشانی ﻻیا ہوں، اس لئے تم اللہ تعالیٰ سے ڈرو اور میری فرمانبرداری کرو!

  51. إِنَّ ٱللَّهَ رَبِّى وَرَبُّكُمْ فَٱعْبُدُوهُ‌ۗ هَـٰذَا صِرَٲطٌ مُّسْتَقِيمٌ ۝٥١

    یقین مانو! میرا اور تمہارا رب اللہ ہی ہے، تم سب اسی کی عبادت کرو، یہی سیدھی راه ہے

  52. ۞ فَلَمَّآ أَحَسَّ عِيسَىٰ مِنْهُمُ ٱلْكُفْرَ قَالَ مَنْ أَنصَارِىٓ إِلَى ٱللَّهِ‌ۖ قَالَ ٱلْحَوَارِيُّونَ نَحْنُ أَنصَارُ ٱللَّهِ ءَامَنَّا بِٱللَّهِ وَٱشْهَدْ بِأَنَّا مُسْلِمُونَ ۝٥٢

    مگر جب حضرت عیسیٰ علیہ السلام نے ان کا کفر محسوس کر لیا تو کہنے لگے اللہ تعالیٰ کی راه میں میری مدد کرنے واﻻ کون کون ہے؟ حواریوں نے جواب دیا کہ ہم اللہ تعالیٰ کی راه کے مددگار ہیں، ہم اللہ تعالیٰ پر ایمان ﻻئے اور آپ گواه رہئے کہ ہم تابعدار ہیں

  53. رَبَّنَآ ءَامَنَّا بِمَآ أَنزَلْتَ وَٱتَّبَعْنَا ٱلرَّسُولَ فَٱكْتُبْنَا مَعَ ٱلشَّـٰهِدِينَ ۝٥٣

    اے ہمارے پالنے والے معبود! ہم تیری اتاری ہوئی وحی پر ایمان ﻻئے اور ہم نے تیرے رسول کی اتباع کی، پس تو ہمیں گواہوں میں لکھ لے

  54. وَمَكَرُواْ وَمَكَرَ ٱللَّهُ‌ۖ وَٱللَّهُ خَيْرُ ٱلْمَـٰكِرِينَ ۝٥٤

    اور کافروں نے مکر کیا اور اللہ تعالیٰ نے بھی (مکر) خفیہ تدبیر کی اور اللہ تعالیٰ سب خفیہ تدبیر کرنے والوں سے بہتر ہے

  55. إِذْ قَالَ ٱللَّهُ يَـٰعِيسَىٰٓ إِنِّى مُتَوَفِّيكَ وَرَافِعُكَ إِلَىَّ وَمُطَهِّرُكَ مِنَ ٱلَّذِينَ كَفَرُواْ وَجَاعِلُ ٱلَّذِينَ ٱتَّبَعُوكَ فَوْقَ ٱلَّذِينَ كَفَرُوٓاْ إِلَىٰ يَوْمِ ٱلْقِيَـٰمَةِ‌ۖ ثُمَّ إِلَىَّ مَرْجِعُكُمْ فَأَحْكُمُ بَيْنَكُمْ فِيمَا كُنتُمْ فِيهِ تَخْتَلِفُونَ ۝٥٥

    جب اللہ تعالیٰ نے فرمایا کہ اے عیسیٰ! میں تجھے پورا لینے واﻻ ہوں اور تجھے اپنی جانب اٹھانے واﻻ ہوں اور تجھے کافروں سے پاک کرنے واﻻ ہوں اور تیرے تابعداروں کو کافروں کے اوپر غالب کرنے واﻻ ہوں قیامت کے دن تک، پھر تم سب کا لوٹنا میری ہی طرف ہے میں ہی تمہارے آپس کے تمام تر اختلافات کا فیصلہ کروں گا

  56. فَأَمَّا ٱلَّذِينَ كَفَرُواْ فَأُعَذِّبُهُمْ عَذَابًا شَدِيدًا فِى ٱلدُّنْيَا وَٱلْأَخِرَةِ وَمَا لَهُم مِّن نَّـٰصِرِينَ ۝٥٦

    پھر کافروں کو تو میں دنیا اور آخرت میں سخت تر عذاب دوں گا اور ان کا کوئی مددگار نہ ہوگا

  57. وَأَمَّا ٱلَّذِينَ ءَامَنُواْ وَعَمِلُواْ ٱلصَّـٰلِحَـٰتِ فَيُوَفِّيهِمْ أُجُورَهُمْ‌ۗ وَٱللَّهُ لَا يُحِبُّ ٱلظَّـٰلِمِينَ ۝٥٧

    لیکن ایمان والوں اور نیک اعمال والوں کو اللہ تعالیٰ ان کا ﺛواب پورا پورا دے گا اور اللہ تعالیٰ ﻇالموں سے محبت نہیں کرتا

  58. ذَٲلِكَ نَتْلُوهُ عَلَيْكَ مِنَ ٱلْأَيَـٰتِ وَٱلذِّكْرِ ٱلْحَكِيمِ ۝٥٨

    یہ جسے ہم تیرے سامنے پڑھ رہے ہیں آیتیں ہیں اور حکمت والی نصیحت ہیں

  59. إِنَّ مَثَلَ عِيسَىٰ عِندَ ٱللَّهِ كَمَثَلِ ءَادَمَ‌ۖ خَلَقَهُۥ مِن تُرَابٍ ثُمَّ قَالَ لَهُۥ كُن فَيَكُونُ ۝٥٩

    اللہ تعالیٰ کے نزدیک عیسیٰ (علیہ السلام) کی مثال ہو بہو آدم (علیہ السلام) کی مثال ہے جسے مٹی سے بنا کر کے کہہ دیا کہ ہو جا! پس وه ہو گیا!

  60. ٱلْحَقُّ مِن رَّبِّكَ فَلَا تَكُن مِّنَ ٱلْمُمْتَرِينَ ۝٦٠

    تیرے رب کی طرف سے حق یہی ہے خبردار شک کرنے والوں میں نہ ہونا

  61. فَمَنْ حَآجَّكَ فِيهِ مِنۢ بَعْدِ مَا جَآءَكَ مِنَ ٱلْعِلْمِ فَقُلْ تَعَالَوْاْ نَدْعُ أَبْنَآءَنَا وَأَبْنَآءَكُمْ وَنِسَآءَنَا وَنِسَآءَكُمْ وَأَنفُسَنَا وَأَنفُسَكُمْ ثُمَّ نَبْتَهِلْ فَنَجْعَل لَّعْنَتَ ٱللَّهِ عَلَى ٱلْكَـٰذِبِينَ ۝٦١

    اس لئے جو شخص آپ کے پاس اس علم کے آجانے کے بعد بھی آپ سے اس میں جھگڑے تو آپ کہہ دیں کہ آو ہم تم اپنے اپنے فرزندوں کو اور ہم تم اپنی اپنی عورتوں کواور ہم تم خاص اپنی اپنی جانوں کو بلالیں، پھر ہم عاجزی کے ساتھ التجا کریں اور جھوٹوں پر اللہ کی لعنت کریں

  62. إِنَّ هَـٰذَا لَهُوَ ٱلْقَصَصُ ٱلْحَقُّ‌ۚ وَمَا مِنْ إِلَـٰهٍ إِلَّا ٱللَّهُ‌ۚ وَإِنَّ ٱللَّهَ لَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ۝٦٢

    یقیناً صرف یہی سچا بیان ہے اور کوئی معبود برحق نہیں بجز اللہ تعالیٰ کے اور بے شک غالب اور حکمت واﻻ اللہ تعالیٰ ہی ہے

  63. فَإِن تَوَلَّوْاْ فَإِنَّ ٱللَّهَ عَلِيمُۢ بِٱلْمُفْسِدِينَ ۝٦٣

    پھر بھی اگر قبول نہ کریں تو اللہ تعالیٰ بھی صحیح طور پر فسادیوں کو جاننے واﻻ ہے

  64. قُلْ يَـٰٓأَهْلَ ٱلْكِتَـٰبِ تَعَالَوْاْ إِلَىٰ كَلِمَةٍ سَوَآءِۭ بَيْنَنَا وَبَيْنَكُمْ أَلَّا نَعْبُدَ إِلَّا ٱللَّهَ وَلَا نُشْرِكَ بِهِۦ شَيْــًٔا وَلَا يَتَّخِذَ بَعْضُنَا بَعْضًا أَرْبَابًا مِّن دُونِ ٱللَّهِ‌ۚ فَإِن تَوَلَّوْاْ فَقُولُواْ ٱشْهَدُواْ بِأَنَّا مُسْلِمُونَ ۝٦٤

    آپ کہہ دیجئے کہ اے اہل کتاب! ایسی انصاف والی بات کی طرف آو جو ہم میں تم میں برابر ہے کہ ہم اللہ تعالیٰ کے سوا کسی کی عبادت نہ کریں نہ اس کے ساتھ کسی کو شریک بنائیں، نہ اللہ تعالیٰ کو چھوڑ کر آپس میں ایک دوسرے کو ہی رب بنائیں۔ پس اگر وه منھ پھیر لیں تو تم کہہ دو کہ گواه رہو ہم تو مسلمان ہیں

  65. يَـٰٓأَهْلَ ٱلْكِتَـٰبِ لِمَ تُحَآجُّونَ فِىٓ إِبْرَٲهِيمَ وَمَآ أُنزِلَتِ ٱلتَّوْرَٮٰةُ وَٱلْإِنجِيلُ إِلَّا مِنۢ بَعْدِهِۦٓ‌ۚ أَفَلَا تَعْقِلُونَ ۝٦٥

    اے اہل کتاب! تم ابراہیم کی بابت کیوں جھگڑتے ہو حاﻻنکہ تورات وانجیل تو ان کے بعد نازل کی گئیں، کیا تم پھر بھی نہیں سمجھتے؟

  66. هَـٰٓأَنتُمْ هَـٰٓؤُلَآءِ حَـٰجَجْتُمْ فِيمَا لَكُم بِهِۦ عِلْمٌ فَلِمَ تُحَآجُّونَ فِيمَا لَيْسَ لَكُم بِهِۦ عِلْمٌ‌ۚ وَٱللَّهُ يَعْلَمُ وَأَنتُمْ لَا تَعْلَمُونَ ۝٦٦

    سنو! تم لوگ اس میں جھگڑ چکے جس کا تمہیں علم تھا پھر اب اس بات میں کیوں جھگڑتے ہو جس کا تمہیں علم ہی نہیں؟ اور اللہ تعالیٰ جانتا ہے اور تم نہیں جانتے،

  67. مَا كَانَ إِبْرَٲهِيمُ يَهُودِيًّا وَلَا نَصْرَانِيًّا وَلَـٰكِن كَانَ حَنِيفًا مُّسْلِمًا وَمَا كَانَ مِنَ ٱلْمُشْرِكِينَ ۝٦٧

    ابراہیم تو نہ یہودی تھے نہ نصرانی تھے بلکہ وه تو یک طرفہ (خالص) مسلمان تھے، وه مشرک بھی نہیں تھے،

  68. إِنَّ أَوْلَى ٱلنَّاسِ بِإِبْرَٲهِيمَ لَلَّذِينَ ٱتَّبَعُوهُ وَهَـٰذَا ٱلنَّبِىُّ وَٱلَّذِينَ ءَامَنُواْ‌ۗ وَٱللَّهُ وَلِىُّ ٱلْمُؤْمِنِينَ ۝٦٨

    سب لوگوں سے زیاده ابراہیم سے نزدیک تر وه لوگ ہیں جنہوں نے ان کا کہا مانا اور یہ نبی اور جو لوگ ایمان ﻻئے، مومنوں کا ولی اور سہارا اللہ ہی ہے،

  69. وَدَّت طَّآئِفَةٌ مِّنْ أَهْلِ ٱلْكِتَـٰبِ لَوْ يُضِلُّونَكُمْ وَمَا يُضِلُّونَ إِلَّآ أَنفُسَهُمْ وَمَا يَشْعُرُونَ ۝٦٩

    اہل کتاب کی ایک جماعت چاہتی ہے کہ تمہیں گمراه کر دیں۔ دراصل وه خود اپنے آپ کو گمراه کر رہے ہیں اور سمجھتے نہیں

  70. يَـٰٓأَهْلَ ٱلْكِتَـٰبِ لِمَ تَكْفُرُونَ بِـَٔـايَـٰتِ ٱللَّهِ وَأَنتُمْ تَشْهَدُونَ ۝٧٠

    اے اہل کتاب تم (باوجود قائل ہونے کے پھر بھی) دانستہ اللہ کی آیات کا کیوں کفر کر رہے ہو؟

  71. يَـٰٓأَهْلَ ٱلْكِتَـٰبِ لِمَ تَلْبِسُونَ ٱلْحَقَّ بِٱلْبَـٰطِلِ وَتَكْتُمُونَ ٱلْحَقَّ وَأَنتُمْ تَعْلَمُونَ ۝٧١

    اے اہل کتاب! باوجود جاننے کے حق وباطل کو کیوں خلط ملط کر رہے ہو اور کیوں حق کو چھپا رہے ہو؟

  72. وَقَالَت طَّآئِفَةٌ مِّنْ أَهْلِ ٱلْكِتَـٰبِ ءَامِنُواْ بِٱلَّذِىٓ أُنزِلَ عَلَى ٱلَّذِينَ ءَامَنُواْ وَجْهَ ٱلنَّهَارِ وَٱكْفُرُوٓاْ ءَاخِرَهُۥ لَعَلَّهُمْ يَرْجِعُونَ ۝٧٢

    اور اہل کتاب کی ایک جماعت نے کہا کہ جو کچھ ایمان والوں پر اتارا گیا ہے اس پر دن چڑھے تو ایمان ﻻؤ اور شام کے وقت کافر بن جاؤ، تاکہ یہ لوگ بھی پلٹ جائیں

  73. وَلَا تُؤْمِنُوٓاْ إِلَّا لِمَن تَبِعَ دِينَكُمْ قُلْ إِنَّ ٱلْهُدَىٰ هُدَى ٱللَّهِ أَن يُؤْتَىٰٓ أَحَدٌ مِّثْلَ مَآ أُوتِيتُمْ أَوْ يُحَآجُّوكُمْ عِندَ رَبِّكُمْ‌ۗ قُلْ إِنَّ ٱلْفَضْلَ بِيَدِ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ‌ۗ وَٱللَّهُ وَٲسِعٌ عَلِيمٌ ۝٧٣

    اور سوائے تمہارے دین پر چلنے والوں کے اور کسی کا یقین نہ کرو۔ آپ کہہ دیجئے کہ بے شک ہدایت تو اللہ ہی کی ہدایت ہے (اور یہ بھی کہتے ہیں کہ اس بات کا بھی یقین نہ کرو) کہ کوئی اس جیسا دیا جائے جیسا تم دیئے گئے ہو، یا یہ کہ یہ تم سے تمہارے رب کے پاس جھگڑا کریں گے، آپ کہہ دیجئے کہ فضل تو اللہ تعالیٰ ہی کے ہاتھ میں ہے، وه جسے چاہے اسے دے، اللہ تعالیٰ وسعت واﻻ اورجاننے واﻻ ہے

  74. يَخْتَصُّ بِرَحْمَتِهِۦ مَن يَشَآءُ‌ۗ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ۝٧٤

    وه اپنی رحمت کے ساتھ جسے چاہے مخصوص کر لے اور اللہ تعالیٰ بڑے فضل واﻻ ہے

  75. ۞ وَمِنْ أَهْلِ ٱلْكِتَـٰبِ مَنْ إِن تَأْمَنْهُ بِقِنطَارٍ يُؤَدِّهِۦٓ إِلَيْكَ وَمِنْهُم مَّنْ إِن تَأْمَنْهُ بِدِينَارٍ لَّا يُؤَدِّهِۦٓ إِلَيْكَ إِلَّا مَا دُمْتَ عَلَيْهِ قَآئِمًا‌ۗ ذَٲلِكَ بِأَنَّهُمْ قَالُواْ لَيْسَ عَلَيْنَا فِى ٱلْأُمِّيِّــۧنَ سَبِيلٌ وَيَقُولُونَ عَلَى ٱللَّهِ ٱلْكَذِبَ وَهُمْ يَعْلَمُونَ ۝٧٥

    بعض اہل کتاب تو ایسے ہیں کہ اگر انہیں تو خزانے کا امین بنا دے تو بھی وه تجھے واپس کر دیں اور ان میں سے بعض ایسے بھی ہیں کہ اگر تو انہیں ایک دینار بھی امانت دے تو تجھے ادا نہ کریں۔ ہاں یہ اوربات ہے کہ تو اس کے سر پر ہی کھڑا رہے، یہ اس لئے کہ انہوں نے کہہ رکھا ہے کہ ہم پر ان جاہلوں (غیر یہودی) کے حق کاکوئی گناه نہیں، یہ لوگ باوجود جاننے کے اللہ تعالیٰ پر جھوٹ کہتے ہیں

  76. بَلَىٰ مَنْ أَوْفَىٰ بِعَهْدِهِۦ وَٱتَّقَىٰ فَإِنَّ ٱللَّهَ يُحِبُّ ٱلْمُتَّقِينَ ۝٧٦

    کیوں نہیں (مواخذہ ہوگا) البتہ جو شخص اپنا قرار پورا کرے اور پرہیزگاری کرے، تو اللہ تعالیٰ بھی ایسے پرہیز گاروں سے محبت کرتاہے

  77. إِنَّ ٱلَّذِينَ يَشْتَرُونَ بِعَهْدِ ٱللَّهِ وَأَيْمَـٰنِهِمْ ثَمَنًا قَلِيلاً أُوْلَـٰٓئِكَ لَا خَلَـٰقَ لَهُمْ فِى ٱلْأَخِرَةِ وَلَا يُكَلِّمُهُمُ ٱللَّهُ وَلَا يَنظُرُ إِلَيْهِمْ يَوْمَ ٱلْقِيَـٰمَةِ وَلَا يُزَكِّيهِمْ وَلَهُمْ عَذَابٌ أَلِيمٌ ۝٧٧

    بے شک جو لوگ اللہ تعالیٰ کے عہد اور اپنی قسموں کو تھوڑی قیمت پر بیچ ڈالتے ہیں، ان کے لئے آخرت میں کوئی حصہ نہیں، اللہ تعالیٰ نہ تو ان سے بات چیت کرے گا نہ ان کی طرف قیامت کے دن دیکھے گا، نہ انہیں پاک کرے گا اور ان کے لئے دردناک عذاب ہے

  78. وَإِنَّ مِنْهُمْ لَفَرِيقًا يَلْوُۥنَ أَلْسِنَتَهُم بِٱلْكِتَـٰبِ لِتَحْسَبُوهُ مِنَ ٱلْكِتَـٰبِ وَمَا هُوَ مِنَ ٱلْكِتَـٰبِ وَيَقُولُونَ هُوَ مِنْ عِندِ ٱللَّهِ وَمَا هُوَ مِنْ عِندِ ٱللَّهِ وَيَقُولُونَ عَلَى ٱللَّهِ ٱلْكَذِبَ وَهُمْ يَعْلَمُونَ ۝٧٨

    یقیناًان میں ایسا گروه بھی ہے جو کتاب پڑھتے ہوئے اپنی زبان مروڑتا ہے تاکہ تم اسے کتاب ہی کی عبارت خیال کرو حاﻻنکہ دراصل وه کتاب میں سے نہیں، اور یہ کہتے بھی ہیں کہ وه اللہ تعالیٰ کی طرف سے ہے حاﻻنکہ در اصل وه اللہ تعالی کی طرف سے نہیں، وه تو دانستہ اللہ تعالیٰ پر جھوٹ بولتے ہیں

  79. مَا كَانَ لِبَشَرٍ أَن يُؤْتِيَهُ ٱللَّهُ ٱلْكِتَـٰبَ وَٱلْحُكْمَ وَٱلنُّبُوَّةَ ثُمَّ يَقُولَ لِلنَّاسِ كُونُواْ عِبَادًا لِّى مِن دُونِ ٱللَّهِ وَلَـٰكِن كُونُواْ رَبَّـٰنِيِّــۧنَ بِمَا كُنتُمْ تُعَلِّمُونَ ٱلْكِتَـٰبَ وَبِمَا كُنتُمْ تَدْرُسُونَ ۝٧٩

    کسی ایسے انسان کو جسے اللہ تعالیٰ کتاب وحکمت اور نبوت دے، یہ ﻻئق نہیں کہ پھر بھی وه لوگوں سے کہے کہ تم اللہ تعالیٰ کو چھوڑ کر میرے بندے بن جاؤ، بلکہ وه تو کہے گا کہ تم سب رب کے ہو جاؤ، تمہارے کتاب سکھانے کے باعﺚ اور تمہارے کتاب پڑھنے کے سبب۔

  80. وَلَا يَأْمُرَكُمْ أَن تَتَّخِذُواْ ٱلْمَلَـٰٓئِكَةَ وَٱلنَّبِيِّــۧنَ أَرْبَابًا‌ۗ أَيَأْمُرُكُم بِٱلْكُفْرِ بَعْدَ إِذْ أَنتُم مُّسْلِمُونَ ۝٨٠

    اور یہ نہیں (ہو سکتا) کہ وه تمہیں فرشتوں اورنبیوں کو رب بنا لینے کا حکم کرے، کیا وه تمہارے مسلمان ہونے کے بعد بھی تمہیں کفر کا حکم دے گا

  81. وَإِذْ أَخَذَ ٱللَّهُ مِيثَـٰقَ ٱلنَّبِيِّــۧنَ لَمَآ ءَاتَيْتُكُم مِّن كِتَـٰبٍ وَحِكْمَةٍ ثُمَّ جَآءَكُمْ رَسُولٌ مُّصَدِّقٌ لِّمَا مَعَكُمْ لَتُؤْمِنُنَّ بِهِۦ وَلَتَنصُرُنَّهُۥ‌ۚ قَالَ ءَأَقْرَرْتُمْ وَأَخَذْتُمْ عَلَىٰ ذَٲلِكُمْ إِصْرِى‌ۖ قَالُوٓاْ أَقْرَرْنَا‌ۚ قَالَ فَٱشْهَدُواْ وَأَنَا۟ مَعَكُم مِّنَ ٱلشَّـٰهِدِينَ ۝٨١

    جب اللہ تعالیٰ نے نبیوں سے عہد لیا کہ جو کچھ میں تمہیں کتاب وحکمت دوں پھر تمہارے پاس وه رسول آئے جو تمہارے پاس کی چیز کو سچ بتائے تو تمہارے لئے اس پر ایمان ﻻنا اور اس کی مدد کرنا ضروری ہے۔ فرمایا کہ تم اس کے اقراری ہو اور اس پر میرا ذمہ لے رہے ہو؟ سب نے کہا کہ ہمیں اقرار ہے، فرمایا تو اب گواه رہو اور خود میں بھی تمہارے ساتھ گواہوں میں ہوں

  82. فَمَن تَوَلَّىٰ بَعْدَ ذَٲلِكَ فَأُوْلَـٰٓئِكَ هُمُ ٱلْفَـٰسِقُونَ ۝٨٢

    پس اس کے بعد بھی جو پلٹ جائیں وه یقیناً پورے نافرمان ہیں۔

  83. أَفَغَيْرَ دِينِ ٱللَّهِ يَبْغُونَ وَلَهُۥٓ أَسْلَمَ مَن فِى ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ طَوْعًا وَكَرْهًا وَإِلَيْهِ يُرْجَعُونَ ۝٨٣

    کیا وه اللہ تعالیٰ کے دین کے سوا اور دین کی تلاش میں ہیں؟ حاﻻنکہ تمام آسمانوں والے اور سب زمین والے اللہ تعالیٰ ہی کے فرمانبردار ہیں خوشی سے ہوں یا ناخوشی سے، سب اسی کی طرف لوٹائے جائیں گے

  84. قُلْ ءَامَنَّا بِٱللَّهِ وَمَآ أُنزِلَ عَلَيْنَا وَمَآ أُنزِلَ عَلَىٰٓ إِبْرَٲهِيمَ وَإِسْمَـٰعِيلَ وَإِسْحَـٰقَ وَيَعْقُوبَ وَٱلْأَسْبَاطِ وَمَآ أُوتِىَ مُوسَىٰ وَعِيسَىٰ وَٱلنَّبِيُّونَ مِن رَّبِّهِمْ لَا نُفَرِّقُ بَيْنَ أَحَدٍ مِّنْهُمْ وَنَحْنُ لَهُۥ مُسْلِمُونَ ۝٨٤

    آپ کہہ دیجئے کہ ہم اللہ تعالیٰ پر اور جو کچھ ہم پر اتارا گیا ہے اور جو کچھ ابراہیم (علیہ السلام) اور اسماعیل (علیہ السلام) اور یعقوب (علیہ السلام) اوران کی اوﻻد پر اتارا گیا اور جو کچھ موسیٰ وعیسیٰ (علیہما السلام) اور دوسرے انبیاء (علیہم السلام) اللہ تعالیٰ کی طرف سے دیئے گئے، ان سب پر ایمان ﻻئے، ہم ان میں سے کسی کے درمیان فرق نہیں کرتے اور ہم اللہ تعالیٰ کے فرمانبردار ہیں

  85. وَمَن يَبْتَغِ غَيْرَ ٱلْإِسْلَـٰمِ دِينًا فَلَن يُقْبَلَ مِنْهُ وَهُوَ فِى ٱلْأَخِرَةِ مِنَ ٱلْخَـٰسِرِينَ ۝٨٥

    جو شخص اسلام کے سوا اور دین تلاش کرے، اس کا دین قبول نہ کیا جائے گا اور وه آخرت میں نقصان پانے والوں میں ہوگا

  86. كَيْفَ يَهْدِى ٱللَّهُ قَوْمًا كَفَرُواْ بَعْدَ إِيمَـٰنِهِمْ وَشَهِدُوٓاْ أَنَّ ٱلرَّسُولَ حَقٌّ وَجَآءَهُمُ ٱلْبَيِّنَـٰتُ‌ۚ وَٱللَّهُ لَا يَهْدِى ٱلْقَوْمَ ٱلظَّـٰلِمِينَ ۝٨٦

    اللہ تعالیٰ ان لوگوں کو کیسے ہدایت دے گا جو اپنے ایمان ﻻنے اور رسول کی حقانیت کی گواہی دینے اور اپنے پاس روشن دلیلیں آجانے کے بعد کافر ہو جائیں، اللہ تعالیٰ ایسے بے انصاف لوگوں کو راه راست پرنہیں ﻻتا

  87. أُوْلَـٰٓئِكَ جَزَآؤُهُمْ أَنَّ عَلَيْهِمْ لَعْنَةَ ٱللَّهِ وَٱلْمَلَـٰٓئِكَةِ وَٱلنَّاسِ أَجْمَعِينَ ۝٨٧

    ان کی تو یہی سزا ہے کہ ان پر اللہ تعالیٰ کی اور فرشتوں کی اور تمام لوگوں کی لعنت ہو

  88. خَـٰلِدِينَ فِيهَا لَا يُخَفَّفُ عَنْهُمُ ٱلْعَذَابُ وَلَا هُمْ يُنظَرُونَ ۝٨٨

    جس میں یہ ہمیشہ پڑے رہیں گے، نہ تو ان سے عذاب ہلکا کیا جائے گا نہ انہیں مہلت دی جائے گی

  89. إِلَّا ٱلَّذِينَ تَابُواْ مِنۢ بَعْدِ ذَٲلِكَ وَأَصْلَحُواْ فَإِنَّ ٱللَّهَ غَفُورٌ رَّحِيمٌ ۝٨٩

    مگرجو لوگ اس کے بعد توبہ اور اصلاح کر لیں تو بے شک اللہ تعالیٰ بخشنے واﻻ مہربان ہے

  90. إِنَّ ٱلَّذِينَ كَفَرُواْ بَعْدَ إِيمَـٰنِهِمْ ثُمَّ ٱزْدَادُواْ كُفْرًا لَّن تُقْبَلَ تَوْبَتُهُمْ وَأُوْلَـٰٓئِكَ هُمُ ٱلضَّآلُّونَ ۝٩٠

    بےشک جو لوگ اپنے ایمان ﻻنے کے بعد کفر کریں پھر کفر میں بڑھ جائیں، ان کی توبہ ہرگز ہرگز قبول نہ کی جائے گی، یہی گمراه لوگ ہیں

  91. إِنَّ ٱلَّذِينَ كَفَرُواْ وَمَاتُواْ وَهُمْ كُفَّارٌ فَلَن يُقْبَلَ مِنْ أَحَدِهِم مِّلْءُ ٱلْأَرْضِ ذَهَبًا وَلَوِ ٱفْتَدَىٰ بِهِۦٓ‌ۗ أُوْلَـٰٓئِكَ لَهُمْ عَذَابٌ أَلِيمٌ وَمَا لَهُم مِّن نَّـٰصِرِينَ ۝٩١

    ہاں جو لوگ کفر کریں اور مرتے دم تک کافر رہیں ان میں سے کوئی اگر زمین بھر سونا دے، گو فدیئے میں ہی ہو تو بھی ہرگز قبول نہ کیا جائے گا۔ یہی لوگ ہیں جن کے لئے تکلیف دینے واﻻ عذاب ہے اور جن کا کوئی مددگار نہیں

  92. لَن تَنَالُواْ ٱلْبِرَّ حَتَّىٰ تُنفِقُواْ مِمَّا تُحِبُّونَ‌ۚ وَمَا تُنفِقُواْ مِن شَىْءٍ فَإِنَّ ٱللَّهَ بِهِۦ عَلِيمٌ ۝٩٢

    جب تک تم اپنی پسندیده چیز سے اللہ تعالیٰ کی راه میں خرچ نہ کروگے ہرگز بھلائی نہ پاؤ گے، اور تم جو خرچ کرو اسے اللہ تعالیٰ بخوبی جانتا ہے

  93. ۞ كُلُّ ٱلطَّعَامِ كَانَ حِلاًّ لِّبَنِىٓ إِسْرَٲٓءِيلَ إِلَّا مَا حَرَّمَ إِسْرَٲٓءِيلُ عَلَىٰ نَفْسِهِۦ مِن قَبْلِ أَن تُنَزَّلَ ٱلتَّوْرَٮٰةُ‌ۗ قُلْ فَأْتُواْ بِٱلتَّوْرَٮٰةِ فَٱتْلُوهَآ إِن كُنتُمْ صَـٰدِقِينَ ۝٩٣

    توراة کے نزول سے پہلے (حضرت) یعقوب (علیہ السلام) نے جس چیز کو اپنے اوپر حرام کر لیا تھا اس کے سوا تمام کھانے بنی اسرائیل پر حلال تھے، آپ کہہ دیجئے کہ اگر تم سچے ہو تو توراة لے آو اور پڑھ کر سناؤ

  94. فَمَنِ ٱفْتَرَىٰ عَلَى ٱللَّهِ ٱلْكَذِبَ مِنۢ بَعْدِ ذَٲلِكَ فَأُوْلَـٰٓئِكَ هُمُ ٱلظَّـٰلِمُونَ ۝٩٤

    اس کے بعد بھی جو لوگ اللہ تعالیٰ پر جھوٹ بہتان باندھیں وہی ﻇالم ہیں

  95. قُلْ صَدَقَ ٱللَّهُ‌ۗ فَٱتَّبِعُواْ مِلَّةَ إِبْرَٲهِيمَ حَنِيفًا وَمَا كَانَ مِنَ ٱلْمُشْرِكِينَ ۝٩٥

    کہہ دیجئے کہ اللہ تعالیٰ سچا ہے تم سب ابراہیم حنیف کے ملت کی پیروی کرو، جو مشرک نہ تھے

  96. إِنَّ أَوَّلَ بَيْتٍ وُضِعَ لِلنَّاسِ لَلَّذِى بِبَكَّةَ مُبَارَكًا وَهُدًى لِّلْعَـٰلَمِينَ ۝٩٦

    اللہ تعالیٰ کا پہلا گھر جو لوگوں کے لئے مقرر کیا گیا وہی ہے جو مکہ (شریف) میں ہے جو تمام دنیا کے لئے برکت وہدایت واﻻ ہے

  97. فِيهِ ءَايَـٰتُۢ بَيِّنَـٰتٌ مَّقَامُ إِبْرَٲهِيمَ‌ۖ وَمَن دَخَلَهُۥ كَانَ ءَامِنًا‌ۗ وَلِلَّهِ عَلَى ٱلنَّاسِ حِجُّ ٱلْبَيْتِ مَنِ ٱسْتَطَاعَ إِلَيْهِ سَبِيلاً‌ۚ وَمَن كَفَرَ فَإِنَّ ٱللَّهَ غَنِىٌّ عَنِ ٱلْعَـٰلَمِينَ ۝٩٧

    جس میں کھلی کھلی نشانیاں ہیں، مقام ابراہیم ہے، اس میں جو آ جائے امن واﻻ ہو جاتا ہے اللہ تعالیٰ نے ان لوگوں پر جو اس کی طرف راه پا سکتے ہوں اس گھر کا حج فرض کر دیا گیا ہے۔ اور جو کوئی کفر کرے تو اللہ تعالیٰ (اس سے بلکہ) تمام دنیا سے بے پرواه ہے

  98. قُلْ يَـٰٓأَهْلَ ٱلْكِتَـٰبِ لِمَ تَكْفُرُونَ بِـَٔـايَـٰتِ ٱللَّهِ وَٱللَّهُ شَهِيدٌ عَلَىٰ مَا تَعْمَلُونَ ۝٩٨

    آپ کہہ دیجئے کہ اے اہل کتاب تم اللہ تعالیٰ کی آیتوں کے ساتھ کفر کیوں کرتے ہو؟ جو کچھ تم کرتے ہو، اللہ تعالیٰ اس پر گواه ہے

  99. قُلْ يَـٰٓأَهْلَ ٱلْكِتَـٰبِ لِمَ تَصُدُّونَ عَن سَبِيلِ ٱللَّهِ مَنْ ءَامَنَ تَبْغُونَهَا عِوَجًا وَأَنتُمْ شُهَدَآءُ‌ۗ وَمَا ٱللَّهُ بِغَـٰفِلٍ عَمَّا تَعْمَلُونَ ۝٩٩

    ان اہل کتاب سے کہو کہ تم اللہ تعالیٰ کی راه سے لوگوں کو کیوں روکتے ہو؟ اور اس میں عیب ٹٹولتے ہو، حاﻻنکہ تم خود شاہد ہو، اللہ تعالیٰ تمہارے اعمال سے بے خبر نہیں

  100. يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ إِن تُطِيعُواْ فَرِيقًا مِّنَ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ يَرُدُّوكُم بَعْدَ إِيمَـٰنِكُمْ كَـٰفِرِينَ ۝١٠٠

    اے ایمان والو! اگر تم اہل کتاب کی کسی جماعت کی باتیں مانو گے تو وه تمہیں تمہارے ایمان ﻻنے کے بعد مرتد کافر بنا دیں گے۔

  101. وَكَيْفَ تَكْفُرُونَ وَأَنتُمْ تُتْلَىٰ عَلَيْكُمْ ءَايَـٰتُ ٱللَّهِ وَفِيكُمْ رَسُولُهُۥ‌ۗ وَمَن يَعْتَصِم بِٱللَّهِ فَقَدْ هُدِىَ إِلَىٰ صِرَٲطٍ مُّسْتَقِيمٍ ۝١٠١

    ﴿گو یہ ظاہر ہے کہ﴾ تم کیسے کفر کر سکتے ہو؟ باوجود یہ کہ تم پر اللہ تعالیٰ کی آیتیں پڑھی جاتی ہیں اور تم میں رسول اللہ ﴿صلی اللہ علیہ وسلم﴾ موجود ہیں۔ جو شخص اللہ تعالیٰ (کے دین) کو مضبوط تھام لے تو بلاشبہ اسے راه راست دکھادی گئی

  102. يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ ٱتَّقُواْ ٱللَّهَ حَقَّ تُقَاتِهِۦ وَلَا تَمُوتُنَّ إِلَّا وَأَنتُم مُّسْلِمُونَ ۝١٠٢

    اے ایمان والو! اللہ تعالیٰ سے اتنا ڈرو جتنا اس سے ڈرنا چاہئیے اور دیکھو مرتے دم تک مسلمان ہی رہنا

  103. وَٱعْتَصِمُواْ بِحَبْلِ ٱللَّهِ جَمِيعًا وَلَا تَفَرَّقُواْ‌ۚ وَٱذْكُرُواْ نِعْمَتَ ٱللَّهِ عَلَيْكُمْ إِذْ كُنتُمْ أَعْدَآءً فَأَلَّفَ بَيْنَ قُلُوبِكُمْ فَأَصْبَحْتُم بِنِعْمَتِهِۦٓ إِخْوَٲنًا وَكُنتُمْ عَلَىٰ شَفَا حُفْرَةٍ مِّنَ ٱلنَّارِ فَأَنقَذَكُم مِّنْهَا‌ۗ كَذَٲلِكَ يُبَيِّنُ ٱللَّهُ لَكُمْ ءَايَـٰتِهِۦ لَعَلَّكُمْ تَهْتَدُونَ ۝١٠٣

    اللہ تعالیٰ کی رسی کو سب مل کر مضبوط تھام لو اور پھوٹ نہ ڈالو، اور اللہ تعالیٰ کی اس وقت کی نعمت کو یاد کرو جب تم ایک دوسرے کے دشمن تھے، تو اس نے تمہارے دلوں میں الفت ڈال دی، پس تم اس کی مہربانی سے بھائی بھائی ہوگئے، اور تم آگ کے گڑھے کے کنارے پہنچ چکے تھے تو اس نے تمہیں بچا لیا۔ اللہ تعالیٰ اسی طرح تمہارے لئے اپنی نشانیاں بیان کرتا ہے تاکہ تم ہدایت پاؤ

  104. وَلْتَكُن مِّنكُمْ أُمَّةٌ يَدْعُونَ إِلَى ٱلْخَيْرِ وَيَأْمُرُونَ بِٱلْمَعْرُوفِ وَيَنْهَوْنَ عَنِ ٱلْمُنكَرِ‌ۚ وَأُوْلَـٰٓئِكَ هُمُ ٱلْمُفْلِحُونَ ۝١٠٤

    تم میں سے ایک جماعت ایسی ہونی چاہئے جو بھلائی کی طرف بلائے اور نیک کاموں کا حکم کرے اور برے کاموں سے روکے، اور یہی لوگ فلاح ونجات پانے والے ہیں

  105. وَلَا تَكُونُواْ كَٱلَّذِينَ تَفَرَّقُواْ وَٱخْتَلَفُواْ مِنۢ بَعْدِ مَا جَآءَهُمُ ٱلْبَيِّنَـٰتُ‌ۚ وَأُوْلَـٰٓئِكَ لَهُمْ عَذَابٌ عَظِيمٌ ۝١٠٥

    تم ان لوگوں کی طرح نہ ہو جانا جنہوں نے اپنے پاس روشن دلیلیں آ جانے کے بعد بھی تفرقہ ڈاﻻ، اور اختلاف کیا، انہیں لوگوں کے لئے بڑا عذاب ہے

  106. يَوْمَ تَبْيَضُّ وُجُوهٌ وَتَسْوَدُّ وُجُوهٌ‌ۚ فَأَمَّا ٱلَّذِينَ ٱسْوَدَّتْ وُجُوهُهُمْ أَكَفَرْتُم بَعْدَ إِيمَـٰنِكُمْ فَذُوقُواْ ٱلْعَذَابَ بِمَا كُنتُمْ تَكْفُرُونَ ۝١٠٦

    جس دن بعض چہرے سفید ہوں گے اور بعض سیاه، سیاه چہرے والوں (سے کہا جائے گا) کہ کیا تم نے ایمان ﻻنے کے بعد کفر کیا؟ اب اپنے کفر کا عذاب چکھو

  107. وَأَمَّا ٱلَّذِينَ ٱبْيَضَّتْ وُجُوهُهُمْ فَفِى رَحْمَةِ ٱللَّهِ هُمْ فِيهَا خَـٰلِدُونَ ۝١٠٧

    اور سفید چہرے والے اللہ تعالیٰ کی رحمت میں داخل ہوں گے اور اس میں ہمیشہ رہیں گے

  108. تِلْكَ ءَايَـٰتُ ٱللَّهِ نَتْلُوهَا عَلَيْكَ بِٱلْحَقِّ‌ۗ وَمَا ٱللَّهُ يُرِيدُ ظُلْمًا لِّلْعَـٰلَمِينَ ۝١٠٨

    اے نبی! ہم ان حقانی آیتوں کی تلاوت آپ پر کر رہے ہیں اور اللہ تعالیٰ کا اراده لوگوں پر ﻇلم کرنے کا نہیں

  109. وَلِلَّهِ مَا فِى ٱلسَّمَـٰوَٲتِ وَمَا فِى ٱلْأَرْضِ‌ۚ وَإِلَى ٱللَّهِ تُرْجَعُ ٱلْأُمُورُ ۝١٠٩

    اللہ تعالیٰ ہی کے لئے ہے جو کچھ آسمانوں اور زمین میں ہے اور اللہ تعالیٰ ہی کی طرف تمام کام لوٹائے جاتے ہیں

  110. كُنتُمْ خَيْرَ أُمَّةٍ أُخْرِجَتْ لِلنَّاسِ تَأْمُرُونَ بِٱلْمَعْرُوفِ وَتَنْهَوْنَ عَنِ ٱلْمُنكَرِ وَتُؤْمِنُونَ بِٱللَّهِ‌ۗ وَلَوْ ءَامَنَ أَهْلُ ٱلْكِتَـٰبِ لَكَانَ خَيْرًا لَّهُم‌ۚ مِّنْهُمُ ٱلْمُؤْمِنُونَ وَأَكْثَرُهُمُ ٱلْفَـٰسِقُونَ ۝١١٠

    تم بہترین امت ہو جو لوگوں کے لئے پیدا کی گئی ہے کہ تم نیک باتوں کا حکم کرتے ہو اور بری باتوں سے روکتے ہو، اور اللہ تعالیٰ پر ایمان رکھتے ہو، اگر اہل کتاب بھی ایمان ﻻتے تو ان کے لئے بہتر تھا، ان میں ایمان والے بھی ہیں لیکن اکثر تو فاسق ہیں

  111. لَن يَضُرُّوكُمْ إِلَّآ أَذًى‌ۖ وَإِن يُقَـٰتِلُوكُمْ يُوَلُّوكُمُ ٱلْأَدْبَارَ ثُمَّ لَا يُنصَرُونَ ۝١١١

    یہ تمہیں ستانے کے سوا اور زیاده کچھ ضرر نہیں پہنچا سکتے، اگر لڑائی کا موقعہ آجائے تو پیٹھ موڑ لیں گے، پھر مدد نہ کیے جائیں گے

  112. ضُرِبَتْ عَلَيْهِمُ ٱلذِّلَّةُ أَيْنَ مَا ثُقِفُوٓاْ إِلَّا بِحَبْلٍ مِّنَ ٱللَّهِ وَحَبْلٍ مِّنَ ٱلنَّاسِ وَبَآءُو بِغَضَبٍ مِّنَ ٱللَّهِ وَضُرِبَتْ عَلَيْهِمُ ٱلْمَسْكَنَةُ‌ۚ ذَٲلِكَ بِأَنَّهُمْ كَانُواْ يَكْفُرُونَ بِـَٔـايَـٰتِ ٱللَّهِ وَيَقْتُلُونَ ٱلْأَنۢبِيَآءَ بِغَيْرِ حَقٍّ‌ۚ ذَٲلِكَ بِمَا عَصَواْ وَّكَانُواْ يَعْتَدُونَ ۝١١٢

    ان پر ہر جگہ ذلت کی مار پڑی، اﻻّ یہ کہ اللہ تعالیٰ کی یا لوگوں کی پناه میں ہوں، یہ غضب الٰہی کے مستحق ہو گئے اور ان پر فقیری ڈال دی گئی، یہ اس لئے کہ یہ لوگ اللہ تعالیٰ کی آیتوں سے کفر کرتے تھے اور بے وجہ انبیا کو قتل کرتے تھے، یہ بدلہ ہے ان کی نافرمانیوں اور زیادتیوں کا

  113. ۞ لَيْسُواْ سَوَآءً‌ۗ مِّنْ أَهْلِ ٱلْكِتَـٰبِ أُمَّةٌ قَآئِمَةٌ يَتْلُونَ ءَايَـٰتِ ٱللَّهِ ءَانَآءَ ٱلَّيْلِ وَهُمْ يَسْجُدُونَ ۝١١٣

    یہ سارے کے سارے یکساں نہیں بلکہ ان اہل کتاب میں ایک جماعت (حق پر) قائم رہنے والی بھی ہے جو راتوں کے وقت بھی کلام اللہ کی تلاوت کرتے ہیں اور سجدے بھی کرتے ہیں

  114. يُؤْمِنُونَ بِٱللَّهِ وَٱلْيَوْمِ ٱلْأَخِرِ وَيَأْمُرُونَ بِٱلْمَعْرُوفِ وَيَنْهَوْنَ عَنِ ٱلْمُنكَرِ وَيُسَـٰرِعُونَ فِى ٱلْخَيْرَٲتِ وَأُوْلَـٰٓئِكَ مِنَ ٱلصَّـٰلِحِينَ ۝١١٤

    یہ اللہ تعالیٰ پر اورقیامت کے دن پر ایمان بھی رکھتے ہیں، بھلائیوں کا حکم کرتے ہیں اور برائیوں سے روکتے ہیں اور بھلائی کے کاموں میں جلدی کرتے ہیں۔ یہ نیک بخت لوگوں میں سے ہیں

  115. وَمَا يَفْعَلُواْ مِنْ خَيْرٍ فَلَن يُكْفَرُوهُ‌ۗ وَٱللَّهُ عَلِيمُۢ بِٱلْمُتَّقِينَ ۝١١٥

    یہ جو کچھ بھی بھلائیاں کریں ان کی ناقدری نہ کی جائے گی اور اللہ تعالیٰ پرہیزگاروں کو خوب جانتا ہے

  116. إِنَّ ٱلَّذِينَ كَفَرُواْ لَن تُغْنِىَ عَنْهُمْ أَمْوَٲلُهُمْ وَلَآ أَوْلَـٰدُهُم مِّنَ ٱللَّهِ شَيْــًٔا‌ۖ وَأُوْلَـٰٓئِكَ أَصْحَـٰبُ ٱلنَّارِ‌ۚ هُمْ فِيهَا خَـٰلِدُونَ ۝١١٦

    کافروں کو ان کے مال اور ان کی اوﻻد اللہ کے ہاں کچھ کام نہ آئیں گی، یہ تو جہنمی ہیں جو ہمیشہ اسی میں پڑے رہیں گے

  117. مَثَلُ مَا يُنفِقُونَ فِى هَـٰذِهِ ٱلْحَيَوٲةِ ٱلدُّنْيَا كَمَثَلِ رِيحٍ فِيهَا صِرٌّ أَصَابَتْ حَرْثَ قَوْمٍ ظَلَمُوٓاْ أَنفُسَهُمْ فَأَهْلَكَتْهُ‌ۚ وَمَا ظَلَمَهُمُ ٱللَّهُ وَلَـٰكِنْ أَنفُسَهُمْ يَظْلِمُونَ ۝١١٧

    یہ کفار جو خرچ اخراجات کریں اس کی مثال یہ ہے کہ ایک تند ہوا چلی جس سے پاﻻ تھا جو ﻇالموں کی کھیتی پر پڑا اور اسے تہس نہس کر دیا۔ اللہ تعالیٰ نے ان پر ﻇلم نہیں کیا بلکہ وه خود اپنی جانوں پر ﻇلم کرتے تھے

  118. يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ لَا تَتَّخِذُواْ بِطَانَةً مِّن دُونِكُمْ لَا يَأْلُونَكُمْ خَبَالاً وَدُّواْ مَا عَنِتُّمْ قَدْ بَدَتِ ٱلْبَغْضَآءُ مِنْ أَفْوَٲهِهِمْ وَمَا تُخْفِى صُدُورُهُمْ أَكْبَرُ‌ۚ قَدْ بَيَّنَّا لَكُمُ ٱلْأَيَـٰتِ‌ۖ إِن كُنتُمْ تَعْقِلُونَ ۝١١٨

    اے ایمان والو! تم اپنا دلی دوست ایمان والوں کے سوا اور کسی کو نہ بناؤ۔ (تم تو) نہیں دیکھتے دوسرے لوگ تمہاری تباہی میں کوئی کسر اٹھا نہیں رکھتے، وه تو چاہتے ہیں کہ تم دکھ میں پڑو ان کی عداوت تو خود ان کی زبان سے بھی ﻇاہر ہو چکی ہے اور جو ان کے سینوں میں پوشیده ہے وه بہت زیاده ہے، ہم نے تمہارے لئے آیتیں بیان کر دیں

  119. هَـٰٓأَنتُمْ أُوْلَآءِ تُحِبُّونَهُمْ وَلَا يُحِبُّونَكُمْ وَتُؤْمِنُونَ بِٱلْكِتَـٰبِ كُلِّهِۦ وَإِذَا لَقُوكُمْ قَالُوٓاْ ءَامَنَّا وَإِذَا خَلَوْاْ عَضُّواْ عَلَيْكُمُ ٱلْأَنَامِلَ مِنَ ٱلْغَيْظِ‌ۚ قُلْ مُوتُواْ بِغَيْظِكُمْ‌ۗ إِنَّ ٱللَّهَ عَلِيمُۢ بِذَاتِ ٱلصُّدُورِ ۝١١٩

    اگر عقلمند ہو (تو غور کرو) ہاں تم تو انہیں چاہتے ہو اور وه تم سے محبت نہیں رکھتے، تم پوری کتاب کو مانتے ہو، (وه نہیں مانتے پھر محبت کیسی؟) یہ تمہارے سامنے تو اپنے ایمان کا اقرار کرتے ہیں لیکن تنہائی میں مارے غصہ کے انگلیاں چباتے ہیں کہہ دو کہ اپنے غصہ ہی میں مر جاؤ، اللہ تعالیٰ دلوں کے راز کو بخوبی جانتا ہے

  120. إِن تَمْسَسْكُمْ حَسَنَةٌ تَسُؤْهُمْ وَإِن تُصِبْكُمْ سَيِّئَةٌ يَفْرَحُواْ بِهَا‌ۖ وَإِن تَصْبِرُواْ وَتَتَّقُواْ لَا يَضُرُّكُمْ كَيْدُهُمْ شَيْــًٔا‌ۗ إِنَّ ٱللَّهَ بِمَا يَعْمَلُونَ مُحِيطٌ ۝١٢٠

    تمہیں اگر بھلائی ملے تو یہ ناخوش ہوتے ہیں ہاں! اگر برائی پہنچے تو خوش ہوتے ہیں، تم اگر صبر کرو اور پرہیزگاری کرو تو ان کا مکر تمہیں کچھ نقصان نہ دے گا۔ اللہ تعالیٰ نے انکے اعمال کا احاطہ کر رکھا ہے

  121. وَإِذْ غَدَوْتَ مِنْ أَهْلِكَ تُبَوِّئُ ٱلْمُؤْمِنِينَ مَقَـٰعِدَ لِلْقِتَالِ‌ۗ وَٱللَّهُ سَمِيعٌ عَلِيمٌ ۝١٢١

    اے نبی! اس وقت کو بھی یاد کرو جب صبح ہی صبح آپ اپنے گھر سے نکل کر مسلمانوں کو میدان جنگ میں لڑائی کے مورچوں پر باقاعده بٹھا رہے تھے اللہ تعالیٰ سننے جاننے واﻻ ہے

  122. إِذْ هَمَّت طَّآئِفَتَانِ مِنكُمْ أَن تَفْشَلَا وَٱللَّهُ وَلِيُّهُمَا‌ۗ وَعَلَى ٱللَّهِ فَلْيَتَوَكَّلِ ٱلْمُؤْمِنُونَ ۝١٢٢

    جب تمہاری دو جماعتیں پست ہمتی کا اراده کر چکی تھیں، اللہ تعالیٰ انکا ولی اور مددگار ہے۔ اور اسی کی پاک ذات پر مومنوں کو بھروسہ رکھنا چاہئے

  123. وَلَقَدْ نَصَرَكُمُ ٱللَّهُ بِبَدْرٍ وَأَنتُمْ أَذِلَّةٌ‌ۖ فَٱتَّقُواْ ٱللَّهَ لَعَلَّكُمْ تَشْكُرُونَ ۝١٢٣

    جنگ بدر میں اللہ تعالیٰ نے عین اس وقت تمہاری مدد فرمائی تھی جب کہ تم نہایت گری ہوئی حالت میں تھے، اس لئے اللہ ہی سے ڈرو! (نہ کسی اور سے) تاکہ تمہیں شکر گزاری کی توفیق ہو

  124. إِذْ تَقُولُ لِلْمُؤْمِنِينَ أَلَن يَكْفِيَكُمْ أَن يُمِدَّكُمْ رَبُّكُم بِثَلَـٰثَةِ ءَالَـٰفٍ مِّنَ ٱلْمَلَـٰٓئِكَةِ مُنزَلِينَ ۝١٢٤

    (اور یہ شکر گزاری باعﺚ نصرت وامداد ہو) جب آپ مومنوں کو تسلی دے رہے تھے، کیا آسمان سے تین ہزار فرشتے اتار کر اللہ تعالیٰ کا تمہاری مدد کرنا تمہیں کافی نہ ہوگا،

  125. بَلَىٰٓ‌ۚ إِن تَصْبِرُواْ وَتَتَّقُواْ وَيَأْتُوكُم مِّن فَوْرِهِمْ هَـٰذَا يُمْدِدْكُمْ رَبُّكُم بِخَمْسَةِ ءَالَـٰفٍ مِّنَ ٱلْمَلَـٰٓئِكَةِ مُسَوِّمِينَ ۝١٢٥

    کیوں نہیں، بلکہ اگر تم صبر و پرہیزگاری کرو اور یہ لوگ اسی دم تمہارے پاس آجائیں تو تمہارا رب تمہاری امداد پانچ ہزار فرشتوں سے کرے گا جو نشاندار ہوں گے

  126. وَمَا جَعَلَهُ ٱللَّهُ إِلَّا بُشْرَىٰ لَكُمْ وَلِتَطْمَئِنَّ قُلُوبُكُم بِهِۦ‌ۗ وَمَا ٱلنَّصْرُ إِلَّا مِنْ عِندِ ٱللَّهِ ٱلْعَزِيزِ ٱلْحَكِيمِ ۝١٢٦

    اور یہ تو محض تمہارے دل کی خوشی اور اطمینان قلب کے لئے ہے، ورنہ مدد تو اللہ ہی کی طرف سے ہے جو غالب اور حکمتوں واﻻ ہے

  127. لِيَقْطَعَ طَرَفًا مِّنَ ٱلَّذِينَ كَفَرُوٓاْ أَوْ يَكْبِتَهُمْ فَيَنقَلِبُواْ خَآئِبِينَ ۝١٢٧

    (اس امداد الٰہی کا مقصد یہ تھا کہ اللہ) کافروں کی ایک جماعت کو کاٹ دے یا انہیں ذلیل کر ڈالے اور (سارے کے سارے) نامراد ہو کر واپس چلے جائیں

  128. لَيْسَ لَكَ مِنَ ٱلْأَمْرِ شَىْءٌ أَوْ يَتُوبَ عَلَيْهِمْ أَوْ يُعَذِّبَهُمْ فَإِنَّهُمْ ظَـٰلِمُونَ ۝١٢٨

    اے پیغمبر! آپ کے اختیار میں کچھ نہیں، اللہ تعالیٰ چاہے تو ان کی توبہ قبول کرے یا عذاب دے، کیونکہ وه ﻇالم ہیں

  129. وَلِلَّهِ مَا فِى ٱلسَّمَـٰوَٲتِ وَمَا فِى ٱلْأَرْضِ‌ۚ يَغْفِرُ لِمَن يَشَآءُ وَيُعَذِّبُ مَن يَشَآءُ‌ۚ وَٱللَّهُ غَفُورٌ رَّحِيمٌ ۝١٢٩

    آسمانوں اور زمین میں جو کچھ ہے سب اللہ ہی کا ہے، وه جسے چاہے بخشے جسے چاہے عذاب کرے، اللہ تعالیٰ بخشش کرنے واﻻ مہربان ہے

  130. يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ لَا تَأْكُلُواْ ٱلرِّبَ[s[وٲٓاْ] أَضْعَـٰفًا مُّضَـٰعَفَةً‌ۖ وَٱتَّقُواْ ٱللَّهَ لَعَلَّكُمْ تُفْلِحُونَ ۝١٣٠

    اے ایمان والو! بڑھا چڑھا کر سود نہ کھاؤ، اور اللہ تعالیٰ سے ڈرو تاکہ تمہیں نجات ملے

  131. وَٱتَّقُواْ ٱلنَّارَ ٱلَّتِىٓ أُعِدَّتْ لِلْكَـٰفِرِينَ ۝١٣١

    اور اس آگ سے ڈرو جو کافروں کے لئے تیار کی گئی ہے

  132. وَأَطِيعُواْ ٱللَّهَ وَٱلرَّسُولَ لَعَلَّكُمْ تُرْحَمُونَ ۝١٣٢

    اور اللہ اور اس کے رسول کی فرمانبرداری کرو تاکہ تم پر رحم کیا جائے

  133. ۞ وَسَارِعُوٓاْ إِلَىٰ مَغْفِرَةٍ مِّن رَّبِّكُمْ وَجَنَّةٍ عَرْضُهَا ٱلسَّمَـٰوَٲتُ وَٱلْأَرْضُ أُعِدَّتْ لِلْمُتَّقِينَ ۝١٣٣

    اور اپنے رب کی بخشش کی طرف اور اس جنت کی طرف دوڑو جس کا عرض آسمانوں اور زمین کے برابر ہے، جو پرہیزگاروں کے لئے تیار کی گئی ہے

  134. ٱلَّذِينَ يُنفِقُونَ فِى ٱلسَّرَّآءِ وَٱلضَّرَّآءِ وَٱلْكَـٰظِمِينَ ٱلْغَيْظَ وَٱلْعَافِينَ عَنِ ٱلنَّاسِ‌ۗ وَٱللَّهُ يُحِبُّ ٱلْمُحْسِنِينَ ۝١٣٤

    جو لوگ آسانی میں سختی کے موقع پر بھی اللہ کے راستے میں خرچ کرتے ہیں، غصہ پینے والے اور لوگوں سے درگزر کرنے والے ہیں، اللہ تعالیٰ ان نیک کاروں سے محبت کرتا ہے

  135. وَٱلَّذِينَ إِذَا فَعَلُواْ فَـٰحِشَةً أَوْ ظَلَمُوٓاْ أَنفُسَهُمْ ذَكَرُواْ ٱللَّهَ فَٱسْتَغْفَرُواْ لِذُنُوبِهِمْ وَمَن يَغْفِرُ ٱلذُّنُوبَ إِلَّا ٱللَّهُ وَلَمْ يُصِرُّواْ عَلَىٰ مَا فَعَلُواْ وَهُمْ يَعْلَمُونَ ۝١٣٥

    جب ان سے کوئی ناشائستہ کام ہو جائے یا کوئی گناه کر بیٹھیں تو فوراً اللہ کا ذکر اور اپنے گناہوں کے لئے استغفار کرتے ہیں، فیالواقع اللہ تعالیٰ کے سوا اور کون گناہوں کو بخش سکتا ہے؟ اور وه لوگ باوجود علم کے کسی برے کام پر اڑ نہیں جاتے

  136. أُوْلَـٰٓئِكَ جَزَآؤُهُم مَّغْفِرَةٌ مِّن رَّبِّهِمْ وَجَنَّـٰتٌ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ خَـٰلِدِينَ فِيهَا‌ۚ وَنِعْمَ أَجْرُ ٱلْعَـٰمِلِينَ ۝١٣٦

    انہیں کا بدلہ ان کے رب کی طرف سے مغفرت ہے اور جنتیں ہیں جن کے نیچے نہریں بہتی ہیں، جن میں وه ہمیشہ رہیں گے، ان نیک کاموں کے کرنے والوں کا ﺛواب کیا ہی اچھا ہے

  137. قَدْ خَلَتْ مِن قَبْلِكُمْ سُنَنٌ فَسِيرُواْ فِى ٱلْأَرْضِ فَٱنظُرُواْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلْمُكَذِّبِينَ ۝١٣٧

    تم سے پہلے بھی ایسے واقعات گزر چکے ہیں، سو زمین میں چل پھر کر دیکھ لو کہ (آسمانی تعلیم کے) جھٹلانے والوں کا کیا انجام ہوا؟

  138. هَـٰذَا بَيَانٌ لِّلنَّاسِ وَهُدًى وَمَوْعِظَةٌ لِّلْمُتَّقِينَ ۝١٣٨

    عام لوگوں کے لئے تو یہ (قرآن) بیان ہے اور پرہیزگاروں کے لئے ہدایت ونصیحت ہے

  139. وَلَا تَهِنُواْ وَلَا تَحْزَنُواْ وَأَنتُمُ ٱلْأَعْلَوْنَ إِن كُنتُم مُّؤْمِنِينَ ۝١٣٩

    تم نہ سستی کرو اور غمگین ہو، تم ہی غالب رہو گے، اگر تم ایماندار ہو

  140. إِن يَمْسَسْكُمْ قَرْحٌ فَقَدْ مَسَّ ٱلْقَوْمَ قَرْحٌ مِّثْلُهُۥ‌ۚ وَتِلْكَ ٱلْأَيَّامُ نُدَاوِلُهَا بَيْنَ ٱلنَّاسِ وَلِيَعْلَمَ ٱللَّهُ ٱلَّذِينَ ءَامَنُواْ وَيَتَّخِذَ مِنكُمْ شُهَدَآءَ‌ۗ وَٱللَّهُ لَا يُحِبُّ ٱلظَّـٰلِمِينَ ۝١٤٠

    اگر تم زخمی ہوئے ہو تو تمہارے مخالف لوگ بھی تو ایسے ہی زخمی ہو چکے ہیں، ہم ان دنوں کو لوگوں کے درمیان ادلتے بدلتے رہتے ہیں۔ (شکست احد) اس لئے تھی کہ اللہ تعالیٰ ایمان والوں کو ﻇاہر کردے اور تم میں سے بعض کو شہادت کا درجہ عطا فرمائے، اللہ تعالیٰ ﻇالموں سے محبت نہیں کرتا

  141. وَلِيُمَحِّصَ ٱللَّهُ ٱلَّذِينَ ءَامَنُواْ وَيَمْحَقَ ٱلْكَـٰفِرِينَ ۝١٤١

    (یہ وجہ بھی تھی) کہ اللہ تعالیٰ ایمان والوں کو بالکل الگ کر دے اور کافروں کو مٹا دے

  142. أَمْ حَسِبْتُمْ أَن تَدْخُلُواْ ٱلْجَنَّةَ وَلَمَّا يَعْلَمِ ٱللَّهُ ٱلَّذِينَ جَـٰهَدُواْ مِنكُمْ وَيَعْلَمَ ٱلصَّـٰبِرِينَ ۝١٤٢

    کیا تم یہ سمجھ بیٹھے ہو کہ تم جنت میں چلے جاؤ گے، حاﻻنکہ اب تک اللہ تعالیٰ نے یہ ﻇاہر نہیں کیا کہ تم سے جہاد کرنے والے کون ہیں اور صبر کرنے والے کون ہیں؟

  143. وَلَقَدْ كُنتُمْ تَمَنَّوْنَ ٱلْمَوْتَ مِن قَبْلِ أَن تَلْقَوْهُ فَقَدْ رَأَيْتُمُوهُ وَأَنتُمْ تَنظُرُونَ ۝١٤٣

    جنگ سے پہلے تو تم شہادت کی آرزو میں تھے اب اسے اپنی آنکھوں سے اپنے سامنے دیکھ لیا

  144. وَمَا مُحَمَّدٌ إِلَّا رَسُولٌ قَدْ خَلَتْ مِن قَبْلِهِ ٱلرُّسُلُ‌ۚ أَفَإِيْن مَّاتَ أَوْ قُتِلَ ٱنقَلَبْتُمْ عَلَىٰٓ أَعْقَـٰبِكُمْ‌ۚ وَمَن يَنقَلِبْ عَلَىٰ عَقِبَيْهِ فَلَن يَضُرَّ ٱللَّهَ شَيْــًٔا‌ۗ وَسَيَجْزِى ٱللَّهُ ٱلشَّـٰكِرِينَ ۝١٤٤

    (حضرت) محمد ﴿صلی اللہ علیہ وسلم﴾ صرف رسول ہی ہیں، ان سے پہلے بہت سے رسول ہو چکے ہیں، کیا اگر ان کا انتقال ہو جائے یا یہ شہید ہو جائیں، تو تم اسلام سے اپنی ایڑیوں کے بل پھر جاؤ گے؟ اور جو کوئی پھر جائے اپنی ایڑیوں پر تو ہرگز اللہ تعالیٰ کا کچھ نہ بگاڑے گا، عنقریب اللہ تعالیٰ شکر گزاروں کو نیک بدلہ دے گا

  145. وَمَا كَانَ لِنَفْسٍ أَن تَمُوتَ إِلَّا بِإِذْنِ ٱللَّهِ كِتَـٰبًا مُّؤَجَّلاً‌ۗ وَمَن يُرِدْ ثَوَابَ ٱلدُّنْيَا نُؤْتِهِۦ مِنْهَا وَمَن يُرِدْ ثَوَابَ ٱلْأَخِرَةِ نُؤْتِهِۦ مِنْهَا‌ۚ وَسَنَجْزِى ٱلشَّـٰكِرِينَ ۝١٤٥

    بغیر اللہ تعالیٰ کے حکم کے کوئی جاندار نہیں مرسکتا، مقرر شده وقت لکھا ہوا ہے، دنیا کی چاہت والوں کو ہم کچھ دنیا دے دیتے ہیں اور آخرت کا ﺛواب چاہنے والوں کو ہم وه بھی دیں گے۔ اوراحسان ماننے والوں کو ہم بہت جلد نیک بدلہ دیں گے

  146. وَكَأَيِّن مِّن نَّبِىٍّ قَـٰتَلَ مَعَهُۥ رِبِّيُّونَ كَثِيرٌ فَمَا وَهَنُواْ لِمَآ أَصَابَهُمْ فِى سَبِيلِ ٱللَّهِ وَمَا ضَعُفُواْ وَمَا ٱسْتَكَانُواْ‌ۗ وَٱللَّهُ يُحِبُّ ٱلصَّـٰبِرِينَ ۝١٤٦

    بہت سے نبیوں کے ہم رکاب ہو کر، بہت سے اللہ والے جہاد کر چکے ہیں، انہیں بھی اللہ کی راه میں تکلیفیں پہنچیں لیکن نہ تو انہوں نے ہمت ہاری نہ سست رہے اور نہ دبے، اور اللہ صبر کرنے والوں کو (ہی) چاہتا ہے

  147. وَمَا كَانَ قَوْلَهُمْ إِلَّآ أَن قَالُواْ رَبَّنَا ٱغْفِرْ لَنَا ذُنُوبَنَا وَإِسْرَافَنَا فِىٓ أَمْرِنَا وَثَبِّتْ أَقْدَامَنَا وَٱنصُرْنَا عَلَى ٱلْقَوْمِ ٱلْكَـٰفِرِينَ ۝١٤٧

    وه یہی کہتے رہے کہ اے پروردگار! ہمارے گناہوں کو بخش دے اور ہم سے ہمارے کاموں میں جو بے جا زیادتی ہوئی ہے اسے بھی معاف فرما اور ہمیں ﺛابت قدمی عطا فرما اور ہمیں کافروں کی قوم پر مدد دے

  148. فَـَٔـاتَـٰهُمُ ٱللَّهُ ثَوَابَ ٱلدُّنْيَا وَحُسْنَ ثَوَابِ ٱلْأَخِرَةِ‌ۗ وَٱللَّهُ يُحِبُّ ٱلْمُحْسِنِينَ ۝١٤٨

    اللہ تعالیٰ نے انہیں دنیا کا ﺛواب بھی دیا اور آخرت کے ﺛواب کی خوبی بھی عطا فرمائی اور اللہ تعالیٰ نیک لوگوں سے محبت کرتا ہے

  149. يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ إِن تُطِيعُواْ ٱلَّذِينَ كَفَرُواْ يَرُدُّوكُمْ عَلَىٰٓ أَعْقَـٰبِكُمْ فَتَنقَلِبُواْ خَـٰسِرِينَ ۝١٤٩

    اے ایمان والو! اگر تم کافروں کی باتیں مانو گے تو وه تمہیں تمہاری ایڑیوں کے بل پلٹا دیں گے، (یعنی تمہیں مرتد بنا دیں گے) پھر تم نامراد ہو جاؤ گے

  150. بَلِ ٱللَّهُ مَوْلَـٰكُمْ‌ۖ وَهُوَ خَيْرُ ٱلنَّـٰصِرِينَ ۝١٥٠

    بلکہ اللہ ہی تمہارا موﻻ ہے اور وہی بہترین مددگار ہے

  151. سَنُلْقِى فِى قُلُوبِ ٱلَّذِينَ كَفَرُواْ ٱلرُّعْبَ بِمَآ أَشْرَكُواْ بِٱللَّهِ مَا لَمْ يُنَزِّلْ بِهِۦ سُلْطَـٰنًا‌ۖ وَمَأْوَٮٰهُمُ ٱلنَّارُ‌ۚ وَبِئْسَ مَثْوَى ٱلظَّـٰلِمِينَ ۝١٥١

    ہم عنقریب کافروں کے دلوں میں رعب ڈال دیں گے، اس وجہ سے کہ یہ اللہ کے ساتھ ان چیزوں کو شریک کرتے ہیں جس کی کوئی دلیل اللہ نے نہیں اتاری، ان کا ٹھکانہ جہنم ہے، اور ان ﻇالموں کی بری جگہ ہے

  152. وَلَقَدْ صَدَقَكُمُ ٱللَّهُ وَعْدَهُۥٓ إِذْ تَحُسُّونَهُم بِإِذْنِهِۦ‌ۖ حَتَّىٰٓ إِذَا فَشِلْتُمْ وَتَنَـٰزَعْتُمْ فِى ٱلْأَمْرِ وَعَصَيْتُم مِّنۢ بَعْدِ مَآ أَرَٮٰكُم مَّا تُحِبُّونَ‌ۚ مِنكُم مَّن يُرِيدُ ٱلدُّنْيَا وَمِنكُم مَّن يُرِيدُ ٱلْأَخِرَةَ‌ۚ ثُمَّ صَرَفَكُمْ عَنْهُمْ لِيَبْتَلِيَكُمْ‌ۖ وَلَقَدْ عَفَا عَنكُمْ‌ۗ وَٱللَّهُ ذُو فَضْلٍ عَلَى ٱلْمُؤْمِنِينَ ۝١٥٢

    اللہ تعالیٰ نے تم سے اپنا وعده سچا کر دکھایا جبکہ تم اس کے حکم سے انہیں کاٹ رہے تھے۔ یہاں تک کہ جب تم نے پست ہمتی اختیار کی اور کام میں جھگڑنے لگے اور نافرمانی کی، اس کے بعد کہ اس نے تمہاری چاہت کی چیز تمہیں دکھا دی، تم میں سے بعض دنیا چاہتے تھے اور بعض کا اراده آخرت کا تھا تو پھر اس نے تمہیں ان سے پھیر دیا تاکہ تم کو آزمائے اور یقیناً اس نے تمہاری لغزش سے درگزر فرما دیا اور ایمان والوں پر اللہ تعالیٰ بڑے فضل واﻻ ہے

  153. ۞ إِذْ تُصْعِدُونَ وَلَا تَلْوُۥنَ عَلَىٰٓ أَحَدٍ وَٱلرَّسُولُ يَدْعُوكُمْ فِىٓ أُخْرَٮٰكُمْ فَأَثَـٰبَكُمْ غَمَّۢا بِغَ[g[مٍّ ل]ِّكَيْلَا تَحْزَنُواْ عَلَىٰ مَا فَاتَكُمْ وَلَا مَآ أَصَـٰبَكُمْ‌ۗ وَٱللَّهُ خَبِيرُۢ بِمَا تَعْمَلُونَ ۝١٥٣

    جب کہ تم چڑھے چلے جا رہے تھے اور کسی طرف توجہ تک نہیں کرتے تھے اور اللہ کے رسول تمہیں تمہارے پیچھے سے آوازیں دے رہے تھے، بس تمہیں غم پر غم پہنچا تاکہ تم فوت شده چیز پر غمگین نہ ہو اور نہ پہنچنے والی (تکلیف) پر اداس ہو، اللہ تعالیٰ تمہارے تمام اعمال سے خبردار ہے

  154. ثُمَّ أَنزَلَ عَلَيْكُم مِّنۢ بَعْدِ ٱلْغَمِّ أَمَنَةً نُّعَاسًا يَغْشَىٰ طَآئِفَةً مِّنكُمْ‌ۖ وَطَآئِفَةٌ قَدْ أَهَمَّتْهُمْ أَنفُسُهُمْ يَظُنُّونَ بِٱللَّهِ غَيْرَ ٱلْحَقِّ ظَنَّ ٱلْجَـٰهِلِيَّةِ‌ۖ يَقُولُونَ هَل لَّنَا مِنَ ٱلْأَمْرِ مِن شَىْءٍ‌ۗ قُلْ إِنَّ ٱلْأَمْرَ كُلَّهُۥ لِلَّهِ‌ۗ يُخْفُونَ فِىٓ أَنفُسِهِم مَّا لَا يُبْدُونَ لَكَ‌ۖ يَقُولُونَ لَوْ كَانَ لَنَا مِنَ ٱلْأَمْرِ شَىْءٌ مَّا قُتِلْنَا هَـٰهُنَا‌ۗ قُل لَّوْ كُنتُمْ فِى بُيُوتِكُمْ لَبَرَزَ ٱلَّذِينَ كُتِبَ عَلَيْهِمُ ٱلْقَتْلُ إِلَىٰ مَضَاجِعِهِمْ‌ۖ وَلِيَبْتَلِىَ ٱللَّهُ مَا فِى صُدُورِكُمْ وَلِيُمَحِّصَ مَا فِى قُلُوبِكُمْ‌ۗ وَٱللَّهُ عَلِيمُۢ بِذَاتِ ٱلصُّدُورِ ۝١٥٤

    پھر اس نے اس غم کے بعد تم پر امن نازل فرمایا اور تم میں سے ایک جماعت کو امن کی نیند آنے لگی۔ ہاں کچھ وه لوگ بھی تھے کہ انہیں اپنی جانوں کی پڑی ہوئی تھی، وه اللہ تعالیٰ کے ساتھ ناحق جہالت بھری بدگمانیاں کر رہے تھے اور کہتے تھے کیا ہمیں بھی کسی چیز کا اختیار ہے؟ آپ کہہ دیجئے کہ کام کل کا کل اللہ کے اختیار میں ہے، یہ لوگ اپنے دلوں کے بھید آپ کو نہیں بتاتے، کہتے ہیں کہ اگر ہمیں کچھ بھی اختیار ہوتا تو یہاں قتل نہ کئے جاتے۔ آپ کہہ دیجئے کہ گو تم اپنے گھروں میں ہوتے پھر بھی جن کی قمست میں قتل ہونا تھا وه تو مقتل کی طرف چل کھڑے ہوتے، اللہ تعالیٰ کو تمہارے سینوں کے اندر کی چیز کا آزمانا اور جو کچھ تمہارے دلوں میں ہے، اس کو پاک کرنا تھا، اور اللہ تعالی سینوں کے بھید سے آگاه ہے

  155. إِنَّ ٱلَّذِينَ تَوَلَّوْاْ مِنكُمْ يَوْمَ ٱلْتَقَى ٱلْجَمْعَانِ إِنَّمَا ٱسْتَزَلَّهُمُ ٱلشَّيْطَـٰنُ بِبَعْضِ مَا كَسَبُواْ‌ۖ وَلَقَدْ عَفَا ٱللَّهُ عَنْهُمْ‌ۗ إِنَّ ٱللَّهَ غَفُورٌ حَلِيمٌ ۝١٥٥

    تم میں سے جن لوگوں نے اس دن پیٹھ دکھائی جس دن دونوں جماعتوں کی مدبھیڑ ہوئی تھی یہ لوگ اپنے بعض کرتوتوں کے باعﺚ شیطان کے پھسلانے میں آگئے لیکن یقین جانو کہ اللہ تعالیٰ نے انہیں معاف کر دیا اللہ تعالیٰ ہے بخشنے واﻻ اور تحمل واﻻ

  156. يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ لَا تَكُونُواْ كَٱلَّذِينَ كَفَرُواْ وَقَالُواْ لِإِخْوَٲنِهِمْ إِذَا ضَرَبُواْ فِى ٱلْأَرْضِ أَوْ كَانُواْ غُزًّى لَّوْ كَانُواْ عِندَنَا مَا مَاتُواْ وَمَا قُتِلُواْ لِيَجْعَلَ ٱللَّهُ ذَٲلِكَ حَسْرَةً فِى قُلُوبِهِمْ‌ۗ وَٱللَّهُ يُحْىِۦ وَيُمِيتُ‌ۗ وَٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرٌ ۝١٥٦

    اے ایمان والو! تم ان لوگوں کی طرح نہ ہو جانا جنہوں نے کفر کیا اور اپنے بھائیوں کے حق میں جب کہ وه سفر میں ہوں یا جہاد میں ہوں، کہا کہ اگر یہ ہمارے پاس ہوتے تو نہ مرتے اور نہ مارے جاتے، اس کی وجہ یہ تھی کہ اس خیال کو اللہ تعالیٰ ان کی دلی حسرت کا سبب بنا دے، اللہ تعالیٰ جلاتا ہے اور مارتا ہے اور اللہ تمہارے عمل کو دیکھ رہا ہے

  157. وَلَئِن قُتِلْتُمْ فِى سَبِيلِ ٱللَّهِ أَوْ مُتُّمْ لَمَغْفِرَةٌ مِّنَ ٱللَّهِ وَرَحْمَةٌ خَيْرٌ مِّمَّا يَجْمَعُونَ ۝١٥٧

    قسم ہے اگر اللہ تعالیٰ کی راه میں شہید کئے جاؤ یا اپنی موت مرو تو بے شک اللہ تعالیٰ کی بخشش ورحمت اس سے بہتر ہے جسے یہ جمع کر رہے ہیں

  158. وَلَئِن مُّتُّمْ أَوْ قُتِلْتُمْ لَإِلَى ٱللَّهِ تُحْشَرُونَ ۝١٥٨

    بالیقین خواہ تم مر جاؤ یا مار ڈالے جاؤ جمع تو اللہ تعالیٰ کی طرف ہی کئے جاؤ گے۔

  159. فَبِمَا رَحْمَةٍ مِّنَ ٱللَّهِ لِنتَ لَهُمْ‌ۖ وَلَوْ كُنتَ فَظًّا غَلِيظَ ٱلْقَلْبِ لَٱنفَضُّواْ مِنْ حَوْلِكَ‌ۖ فَٱعْفُ عَنْهُمْ وَٱسْتَغْفِرْ لَهُمْ وَشَاوِرْهُمْ فِى ٱلْأَمْرِ‌ۖ فَإِذَا عَزَمْتَ فَتَوَكَّلْ عَلَى ٱللَّهِ‌ۚ إِنَّ ٱللَّهَ يُحِبُّ ٱلْمُتَوَكِّلِينَ ۝١٥٩

    اللہ تعالیٰ کی رحمت کے باعﺚ آپ ان پر نرم دل ہیں اور اگر آپ بد زبان اور سخت دل ہوتے تو یہ سب آپ کے پاس سے چھٹ جاتے، سو آپ ان سے درگزر کریں اور ان کے لئے استغفار کریں اور کام کا مشوره ان سے کیا کریں، پھر جب آپ کا پختہ اراده ہو جائے تو اللہ تعالیٰ پر بھروسہ کریں، بے شک اللہ تعالیٰ توکل کرنے والوں سے محبت کرتا ہے

  160. إِن يَنصُرْكُمُ ٱللَّهُ فَلَا غَالِبَ لَكُمْ‌ۖ وَإِن يَخْذُلْكُمْ فَمَن ذَا ٱلَّذِى يَنصُرُكُم مِّنۢ بَعْدِهِۦ‌ۗ وَعَلَى ٱللَّهِ فَلْيَتَوَكَّلِ ٱلْمُؤْمِنُونَ ۝١٦٠

    اگر اللہ تعالیٰ تمہاری مدد کرے تو تم پر کوئی غالب نہیں آسکتا اور اگر وہ تمہیں چھوڑ دے تو اس کے بعد کون ہے جو تمہاری مدد کرے؟ ایمان والوں کو اللہ تعالیٰ ہی پر بھروسہ رکھنا چاہئے

  161. وَمَا كَانَ لِنَبِىٍّ أَن يَغُلَّ‌ۚ وَمَن يَغْلُلْ يَأْتِ بِمَا غَلَّ يَوْمَ ٱلْقِيَـٰمَةِ‌ۚ ثُمَّ تُوَفَّىٰ كُلُّ نَفْسٍ مَّا كَسَبَتْ وَهُمْ لَا يُظْلَمُونَ ۝١٦١

    ناممکن ہے کہ نبی سے خیانت ہو جائے ہر خیانت کرنے واﻻ خیانت کو لئے ہوئے قیامت کے دن حاضر ہوگا، پھر ہر شخص کو اپنے اعمال کا پورا پورا بدلہ دیا جائے گا، اور وه ﻇلم نہ کئے جائیں گے

  162. أَفَمَنِ ٱتَّبَعَ رِضْوَٲنَ ٱللَّهِ كَمَنۢ بَآءَ بِسَخَطٍ مِّنَ ٱللَّهِ وَمَأْوَٮٰهُ جَهَنَّمُ‌ۚ وَبِئْسَ ٱلْمَصِيرُ ۝١٦٢

    کیا پس وه شخص جو اللہ تعالیٰ کی خوشنودی کے درپے ہے، اس شخص جیسا ہے جو اللہ تعالیٰ کی ناراضگی لے کر لوٹتا ہے؟ اور جس کی جگہ جہنم ہے جو بدترین جگہ ہے

  163. هُمْ دَرَجَـٰتٌ عِندَ ٱللَّهِ‌ۗ وَٱللَّهُ بَصِيرُۢ بِمَا يَعْمَلُونَ ۝١٦٣

    اللہ تعالیٰ کے پاس ان کے الگ الگ درجے ہیں اور ان کے تمام اعمال کو اللہ بخوبی دیکھ رہا ہے

  164. لَقَدْ مَنَّ ٱللَّهُ عَلَى ٱلْمُؤْمِنِينَ إِذْ بَعَثَ فِيهِمْ رَسُولاً مِّنْ أَنفُسِهِمْ يَتْلُواْ عَلَيْهِمْ ءَايَـٰتِهِۦ وَيُزَكِّيهِمْ وَيُعَلِّمُهُمُ ٱلْكِتَـٰبَ وَٱلْحِكْمَةَ وَإِن كَانُواْ مِن قَبْلُ لَفِى ضَلَـٰلٍ مُّبِينٍ ۝١٦٤

    بے شک مسلمانوں پر اللہ تعالیٰ کا بڑا احسان ہے کہ ان ہی میں سے ایک رسول ان میں بھیجا، جو انہیں اس کی آیتیں پڑھ کر سناتا ہے اور انہیں پاک کرتا ہے اور انہیں کتاب اور حکمت سکھاتا ہے، یقیناً یہ سب اس سے پہلے کھلی گمراہی میں تھے

  165. أَوَلَمَّآ أَصَـٰبَتْكُم مُّصِيبَةٌ قَدْ أَصَبْتُم مِّثْلَيْهَا قُلْتُمْ أَنَّىٰ هَـٰذَا‌ۖ قُلْ هُوَ مِنْ عِندِ أَنفُسِكُمْ‌ۗ إِنَّ ٱللَّهَ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ۝١٦٥

    (کیا بات ہے) کہ جب تمہیں ایک ایسی تکلیف پہنچی کہ تم اس جیسی دو چند پہنچا چکے، تو یہ کہنے لگے کہ یہ کہاں سے آگئی؟ آپ کہہ دیجئے کہ یہ خود تمہاری طرف سے ہے، بے شک اللہ تعالیٰ ہر چیز پر قادر ہے

  166. وَمَآ أَصَـٰبَكُمْ يَوْمَ ٱلْتَقَى ٱلْجَمْعَانِ فَبِإِذْنِ ٱللَّهِ وَلِيَعْلَمَ ٱلْمُؤْمِنِينَ ۝١٦٦

    اور تمہیں جو کچھ اس دن پہنچا جس دن دو جماعتوں میں مڈ بھیڑ ہوئی تھی، وه سب اللہ کے حکم سے تھا اور اس لئے کہ اللہ تعالیٰ ایمان والوں کو ﻇاہری طور پر جان لے

  167. وَلِيَعْلَمَ ٱلَّذِينَ نَافَقُواْ‌ۚ وَقِيلَ لَهُمْ تَعَالَوْاْ قَـٰتِلُواْ فِى سَبِيلِ ٱللَّهِ أَوِ ٱدْفَعُواْ‌ۖ قَالُواْ لَوْ نَعْلَمُ قِتَالاً لَّٱتَّبَعْنَـٰكُمْ‌ۗ هُمْ لِلْكُفْرِ يَوْمَئِذٍ أَقْرَبُ مِنْهُمْ لِلْإِيمَـٰنِ‌ۚ يَقُولُونَ بِأَفْوَٲهِهِم مَّا لَيْسَ فِى قُلُوبِهِمْ‌ۗ وَٱللَّهُ أَعْلَمُ بِمَا يَكْتُمُونَ ۝١٦٧

    اور منافقوں کو بھی معلوم کر لے جن سے کہا گیا کہ آؤ اللہ کی راه میں جہاد کرو، یا کافروں کو ہٹاو، ٴ تو وه کہنے لگے کہ اگر ہم لڑائی جانتے ہوتے تو ضرور ساتھ دیتے، وه اس دن بہ نسبت ایمان کے کفر سے بہت قریب تھے، اپنے منھ سے وه باتیں بناتے ہیں جو ان کے دلوں میں نہیں، اور اللہ تعالیٰ خوب جانتا ہے جسے وه چھپاتے ہیں

  168. ٱلَّذِينَ قَالُواْ لِإِخْوَٲنِهِمْ وَقَعَدُواْ لَوْ أَطَاعُونَا مَا قُتِلُواْ‌ۗ قُلْ فَٱدْرَءُواْ عَنْ أَنفُسِكُمُ ٱلْمَوْتَ إِن كُنتُمْ صَـٰدِقِينَ ۝١٦٨

    یہ وه لوگ ہیں جو خود بھی بیٹھے رہے اور اپنے بھائیوں کی بابت کہا کہ اگر وه بھی ہماری بات مان لیتے تو قتل نہ کئے جاتے۔ کہہ دیجئے! کہ اگر تم سچے ہو تو اپنی جانوں سے موت کو ہٹا دو

  169. وَلَا تَحْسَبَنَّ ٱلَّذِينَ قُتِلُواْ فِى سَبِيلِ ٱللَّهِ أَمْوَٲتَۢا‌ۚ بَلْ أَحْيَآءٌ عِندَ رَبِّهِمْ يُرْزَقُونَ ۝١٦٩

    جو لوگ اللہ کی راه میں شہید کئے گئے ہیں ان کو ہرگز مرده نہ سمجھیں، بلکہ وه زنده ہیں اپنے رب کے پاس روزیاں دیئے جاتے ہیں

  170. فَرِحِينَ بِمَآ ءَاتَـٰهُمُ ٱللَّهُ مِن فَضْلِهِۦ وَيَسْتَبْشِرُونَ بِٱلَّذِينَ لَمْ يَلْحَقُواْ بِهِم مِّنْ خَلْفِهِمْ أَلَّا خَوْفٌ عَلَيْهِمْ وَلَا هُمْ يَحْزَنُونَ ۝١٧٠

    اللہ تعالیٰ نے اپنا فضل جو انہیں دے رکھا ہے اس سے بہت خوش ہیں اور خوشیاں منا رہے ہیں ان لوگوں کی بابت جو اب تک ان سے نہیں ملے ان کے پیچھے ہیں، اس پر کہ انہیں نہ کوئی خوف ہے اور نہ وه غمگین ہوں گے

  171. ۞ يَسْتَبْشِرُونَ بِنِعْمَةٍ مِّنَ ٱللَّهِ وَفَضْلٍ وَأَنَّ ٱللَّهَ لَا يُضِيعُ أَجْرَ ٱلْمُؤْمِنِينَ ۝١٧١

    وه خوش ہوتے ہیں اللہ کی نعمت اور فضل سے اور اس سے بھی کہ اللہ تعالیٰ ایمان والوں کے اجر کو برباد نہیں کرتا

  172. ٱلَّذِينَ ٱسْتَجَابُواْ لِلَّهِ وَٱلرَّسُولِ مِنۢ بَعْدِ مَآ أَصَابَهُمُ ٱلْقَرْحُ‌ۚ لِلَّذِينَ أَحْسَنُواْ مِنْهُمْ وَٱتَّقَوْاْ أَجْرٌ عَظِيمٌ ۝١٧٢

    جن لوگوں نے اللہ اور رسول کے حکم کو قبول کیا اس کے بعد کہ انہیں پورے زخم لگ چکے تھے، ان میں سے جنہوں نے نیکی کی اور پرہیزگاری برتی ان کے لئے بہت زیاده اجر ہے

  173. ٱلَّذِينَ قَالَ لَهُمُ ٱلنَّاسُ إِنَّ ٱلنَّاسَ قَدْ جَمَعُواْ لَكُمْ فَٱخْشَوْهُمْ فَزَادَهُمْ إِيمَـٰنًا وَقَالُواْ حَسْبُنَا ٱللَّهُ وَنِعْمَ ٱلْوَكِيلُ ۝١٧٣

    وه لوگ کہ جب ان سے لوگوں نے کہا کہ کافروں نے تمہارے مقابلے پر لشکر جمع کر لئے ہیں، تم ان سے خوف کھاؤ تو اس بات نے انہیں ایمان میں اور بڑھا دیا اور کہنے لگے ہمیں اللہ کافی ہے اور وه بہت اچھا کارساز ہے

  174. فَٱنقَلَبُواْ بِنِعْمَةٍ مِّنَ ٱللَّهِ وَفَضْلٍ لَّمْ يَمْسَسْهُمْ سُوٓءٌ وَٱتَّبَعُواْ رِضْوَٲنَ ٱللَّهِ‌ۗ وَٱللَّهُ ذُو فَضْلٍ عَظِيمٍ ۝١٧٤

    (نتیجہ یہ ہوا کہ) اللہ کی نعمت وفضل کے ساتھ یہ لوٹے، انہیں کوئی برائی نہ پہنچی، انہوں نے اللہ تعالیٰ کی رضامندی کی پیروی کی، اللہ بہت بڑے فضل واﻻ ہے

  175. إِنَّمَا ذَٲلِكُمُ ٱلشَّيْطَـٰنُ يُخَوِّفُ أَوْلِيَآءَهُۥ فَلَا تَخَافُوهُمْ وَخَافُونِ إِن كُنتُم مُّؤْمِنِينَ ۝١٧٥

    یہ خبر دینے واﻻ صرف شیطان ہی ہے جو اپنے دوستوں سے ڈراتا ہے تم ان کافروں سے نہ ڈرو اور میرا خوف رکھو، اگر تم مومن ہو

  176. وَلَا يَحْزُنكَ ٱلَّذِينَ يُسَـٰرِعُونَ فِى ٱلْكُفْرِ‌ۚ إِنَّهُمْ لَن يَضُرُّواْ ٱللَّهَ شَيْــًٔا‌ۗ يُرِيدُ ٱللَّهُ أَلَّا يَجْعَلَ لَهُمْ حَظًّا فِى ٱلْأَخِرَةِ‌ۖ وَلَهُمْ عَذَابٌ عَظِيمٌ ۝١٧٦

    کفر میں آگے بڑھنے والے لوگ تجھے غمناک نہ کریں، یقین مانو کہ یہ اللہ تعالیٰ کا کچھ نہ بگاڑ سکیں گے، اللہ تعالیٰ کا اراده ہے کہ ان کے لئے آخرت کا کوئی حصہ عطا نہ کرے، اور ان کے لئے بڑا عذاب ہے

  177. إِنَّ ٱلَّذِينَ ٱشْتَرَوُاْ ٱلْكُفْرَ بِٱلْإِيمَـٰنِ لَن يَضُرُّواْ ٱللَّهَ شَيْــًٔا وَلَهُمْ عَذَابٌ أَلِيمٌ ۝١٧٧

    کفر کو ایمان کے بدلے خریدنے والے ہرگز ہرگز اللہ تعالیٰ کو کوئی نقصان نہیں پہنچا سکتے اور ان ہی کے لئے المناک عذاب ہے

  178. وَلَا يَحْسَبَنَّ ٱلَّذِينَ كَفَرُوٓاْ أَنَّمَا نُمْلِى لَهُمْ خَيْرٌ لِّأَنفُسِهِمْ‌ۚ إِنَّمَا نُمْلِى لَهُمْ لِيَزْدَادُوٓاْ إِثْمًا‌ۚ وَلَهُمْ عَذَابٌ مُّهِينٌ ۝١٧٨

    کافر لوگ ہماری دی ہوئی مہلت کو اپنے حق میں بہتر نہ سمجھیں، یہ مہلت تو اس لئے ہے کہ وه گناہوں میں اور بڑھ جائیں، ان ہی کے لئے ذلیل کرنے واﻻ عذاب ہے

  179. مَّا كَانَ ٱللَّهُ لِيَذَرَ ٱلْمُؤْمِنِينَ عَلَىٰ مَآ أَنتُمْ عَلَيْهِ حَتَّىٰ يَمِيزَ ٱلْخَبِيثَ مِنَ ٱلطَّيِّبِ‌ۗ وَمَا كَانَ ٱللَّهُ لِيُطْلِعَكُمْ عَلَى ٱلْغَيْبِ وَلَـٰكِنَّ ٱللَّهَ يَجْتَبِى مِن رُّسُلِهِۦ مَن يَشَآءُ‌ۖ فَـَٔـامِنُواْ بِٱللَّهِ وَرُسُلِهِۦ‌ۚ وَإِن تُؤْمِنُواْ وَتَتَّقُواْ فَلَكُمْ أَجْرٌ عَظِيمٌ ۝١٧٩

    جس حال میں تم ہو اسی پر اللہ ایمان والوں کو نہ چھوڑ دے گا جب تک کہ پاک اور ناپاک کو الگ الگ نہ کر دے، اور نہ اللہ تعالیٰ ایسا ہے کہ تمہیں غیب سے آگاه کردے، بلکہ اللہ تعالیٰ اپنے رسولوں میں سے جس کا چاہے انتخاب کر لیتا ہے، اس لئے تم اللہ تعالیٰ پر اور اس کے رسولوں پر ایمان رکھو، اگر تم ایمان ﻻؤ اور تقویٰ کرو تو تمہارے لئے بڑا بھاری اجر ہے

  180. وَلَا يَحْسَبَنَّ ٱلَّذِينَ يَبْخَلُونَ بِمَآ ءَاتَـٰهُمُ ٱللَّهُ مِن فَضْلِهِۦ هُوَ خَيْرًا لَّهُم‌ۖ بَلْ هُوَ شَرٌّ لَّهُمْ‌ۖ سَيُطَوَّقُونَ مَا بَخِلُواْ بِهِۦ يَوْمَ ٱلْقِيَـٰمَةِ‌ۗ وَلِلَّهِ مِيرَٲثُ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ‌ۗ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌ ۝١٨٠

    جنہیں اللہ تعالیٰ نے اپنے فضل سے کچھ دے رکھا ہے وه اس میں اپنی کنجوسی کو اپنے لئے بہتر خیال نہ کریں بلکہ وه ان کے لئے نہایت بدتر ہے، عنقریب قیامت والے دن یہ اپنی کنجوسی کی چیز کے طوق ڈالے جائیں گے، آسمانوں اور زمین کی میراث اللہ تعالیٰ ہی کے لئے اور جو کچھ تم کر رہے ہو، اس سے اللہ تعالیٰ آگاه ہے

  181. لَّقَدْ سَمِعَ ٱللَّهُ قَوْلَ ٱلَّذِينَ قَالُوٓاْ إِنَّ ٱللَّهَ فَقِيرٌ وَنَحْنُ أَغْنِيَآءُ‌ۘ سَنَكْتُبُ مَا قَالُواْ وَقَتْلَهُمُ ٱلْأَنۢبِيَآءَ بِغَيْرِ حَقٍّ وَنَقُولُ ذُوقُواْ عَذَابَ ٱلْحَرِيقِ ۝١٨١

    یقیناً اللہ تعالیٰ نے ان لوگوں کا قول بھی سنا جنہوں نے کہا کہ اللہ تعالیٰ فقیر ہے اور ہم تونگر ہیں ان کے اس قول کو ہم لکھ لیں گے۔ اور ان کا انبیا کو بلا وجہ قتل کرنا بھی، اور ہم ان سے کہیں گے کہ جلنے واﻻ عذاب چکھو!

  182. ذَٲلِكَ بِمَا قَدَّمَتْ أَيْدِيكُمْ وَأَنَّ ٱللَّهَ لَيْسَ بِظَلَّامٍ لِّلْعَبِيدِ ۝١٨٢

    یہ تمہارے پیش کرده اعمال کا بدلہ ہے اور اللہ تعالیٰ اپنے بندوں پر ﻇلم کرنے واﻻ نہیں

  183. ٱلَّذِينَ قَالُوٓاْ إِنَّ ٱللَّهَ عَهِدَ إِلَيْنَآ أَلَّا نُؤْمِنَ لِرَسُولٍ حَتَّىٰ يَأْتِيَنَا بِقُرْبَانٍ تَأْكُلُهُ ٱلنَّارُ‌ۗ قُلْ قَدْ جَآءَكُمْ رُسُلٌ مِّن قَبْلِى بِٱلْبَيِّنَـٰتِ وَبِٱلَّذِى قُلْتُمْ فَلِمَ قَتَلْتُمُوهُمْ إِن كُنتُمْ صَـٰدِقِينَ ۝١٨٣

    یہ وه لوگ ہیں جنہوں نے کہا کہ اللہ تعالیٰ نے ہمیں حکم دیا ہے کہ کسی رسول کو نہ مانیں جب تک وه ہمارے پاس ایسی قربانی نہ ﻻئے جسے آگ کھا جائے۔ آپ کہہ دیجئے کہ اگر تم سچے ہو تو مجھ سے پہلے تمہارے پاس جو رسول دیگر معجزوں کے ساتھ یہ بھی ﻻئے جسے تم کہہ رہے ہو تو پھر تم نے انہیں کیوں مار ڈاﻻ؟

  184. فَإِن كَذَّبُوكَ فَقَدْ كُذِّبَ رُسُلٌ مِّن قَبْلِكَ جَآءُو بِٱلْبَيِّنَـٰتِ وَٱلزُّبُرِ وَٱلْكِتَـٰبِ ٱلْمُنِيرِ ۝١٨٤

    پھر بھی اگر یہ لوگ آپ کو جھٹلائیں تو آپ سے پہلے بھی بہت سے رسول جھٹلائے گئے ہیں جو روشن دلیلیں صحیفے اور منور کتاب لے کر آئے

  185. كُلُّ نَفْسٍ ذَآئِقَةُ ٱلْمَوْتِ‌ۗ وَإِنَّمَا تُوَفَّوْنَ أُجُورَكُمْ يَوْمَ ٱلْقِيَـٰمَةِ‌ۖ فَمَن زُحْزِحَ عَنِ ٱلنَّارِ وَأُدْخِلَ ٱلْجَنَّةَ فَقَدْ فَازَ‌ۗ وَمَا ٱلْحَيَوٲةُ ٱلدُّنْيَآ إِلَّا مَتَـٰعُ ٱلْغُرُورِ ۝١٨٥

    ہر جان موت کا مزه چکھنے والی ہے اور قیامت کے دن تم اپنے بدلے پورے پورے دیئے جاؤ گے، پس جو شخص آگ سے ہٹا دیا جائے اور جنت میں داخل کر دیا جائے بے شک وه کامیاب ہوگیا، اور دنیا کی زندگی تو صرف دھوکے کی جنس ہے

  186. ۞ لَتُبْلَوُنَّ فِىٓ أَمْوَٲلِكُمْ وَأَنفُسِكُمْ وَلَتَسْمَعُنَّ مِنَ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ مِن قَبْلِكُمْ وَمِنَ ٱلَّذِينَ أَشْرَكُوٓاْ أَذًى كَثِيرًا‌ۚ وَإِن تَصْبِرُواْ وَتَتَّقُواْ فَإِنَّ ذَٲلِكَ مِنْ عَزْمِ ٱلْأُمُورِ ۝١٨٦

    یقیناً تمہارے مالوں اور جانوں سے تمہاری آزمائش کی جائے گی اور یہ بھی یقین ہے کہ تمہیں ان لوگوں کی جو تم سے پہلے کتاب دیئے گئے اور مشرکوں کی بہت سی دکھ دینے والی باتیں بھی سننی پڑیں گی اور اگر تم صبر کر لو اور پرہیزگاری اختیار کرو تو یقیناً یہ بہت بڑی ہمت کا کام ہے

  187. وَإِذْ أَخَذَ ٱللَّهُ مِيثَـٰقَ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ لَتُبَيِّنُنَّهُۥ لِلنَّاسِ وَلَا تَكْتُمُونَهُۥ فَنَبَذُوهُ وَرَآءَ ظُهُورِهِمْ وَٱشْتَرَوْاْ بِهِۦ ثَمَنًا قَلِيلاً‌ۖ فَبِئْسَ مَا يَشْتَرُونَ ۝١٨٧

    اور اللہ تعالیٰ نے جب اہل کتاب سے عہد لیا کہ تم اسے سب لوگوں سے ضرور بیان کرو گے اور اسے چھپاؤ گے نہیں، تو پھر بھی ان لوگوں نے اس عہد کو اپنی پیٹھ پیچھے ڈال دیا اور اسے بہت کم قیمت پر بیچ ڈاﻻ۔ ان کا یہ بیوپار بہت برا ہے

  188. لَا تَحْسَبَنَّ ٱلَّذِينَ يَفْرَحُونَ بِمَآ أَتَواْ وَّيُحِبُّونَ أَن يُحْمَدُواْ بِمَا لَمْ يَفْعَلُواْ فَلَا تَحْسَبَنَّهُم بِمَفَازَةٍ مِّنَ ٱلْعَذَابِ‌ۖ وَلَهُمْ عَذَابٌ أَلِيمٌ ۝١٨٨

    وه لوگ جو اپنے کرتوتوں پر خوش ہیں اور چاہتے ہیں کہ جو انہوں نے نہیں کیا اس پر بھی ان کی تعریفیں کی جائیں آپ انہیں عذاب سے چھٹکارا میں نہ سمجھئے ان کے لئے تو دردناک عذاب ہے

  189. وَلِلَّهِ مُلْكُ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ‌ۗ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ۝١٨٩

    آسمانوں اور زمین کی بادشاہی اللہ ہی کے لئے ہے اور اللہ تعالیٰ ہر چیز پر قادر ہے

  190. إِنَّ فِى خَلْقِ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ وَٱخْتِلَـٰفِ ٱلَّيْلِ وَٱلنَّهَارِ لَأَيَـٰتٍ لِّأُوْلِى ٱلْأَلْبَـٰبِ ۝١٩٠

    آسمانوں اور زمین کی پیدائش میں اور رات دن کے ہیر پھیر میں یقیناً عقلمندوں کے لئے نشانیاں ہیں

  191. ٱلَّذِينَ يَذْكُرُونَ ٱللَّهَ قِيَـٰمًا وَقُعُودًا وَعَلَىٰ جُنُوبِهِمْ وَيَتَفَكَّرُونَ فِى خَلْقِ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ رَبَّنَا مَا خَلَقْتَ هَـٰذَا بَـٰطِلاً سُبْحَـٰنَكَ فَقِنَا عَذَابَ ٱلنَّارِ ۝١٩١

    جو اللہ تعالیٰ کا ذکر کھڑے اور بیٹھے اور اپنی کروٹوں پر لیٹے ہوئے کرتے ہیں اور آسمانوں وزمین کی پیدائش میں غوروفکر کرتے ہیں اور کہتے ہیں اے ہمارے پروردگار! تو نے یہ بے فائده نہیں بنایا، تو پاک ہے پس ہمیں آگ کے عذاب سے بچا لے

  192. رَبَّنَآ إِنَّكَ مَن تُدْخِلِ ٱلنَّارَ فَقَدْ أَخْزَيْتَهُۥ‌ۖ وَمَا لِلظَّـٰلِمِينَ مِنْ أَنصَارٍ ۝١٩٢

    اے ہمارے پالنے والے! تو جسے جہنم میں ڈالے یقیناً تو نے اسے رسوا کیا، اور ﻇالموں کا مددگار کوئی نہیں

  193. رَّبَّنَآ إِنَّنَا سَمِعْنَا مُنَادِيًا يُنَادِى لِلْإِيمَـٰنِ أَنْ ءَامِنُواْ بِرَبِّكُمْ فَـَٔـامَنَّا‌ۚ رَبَّنَا فَٱغْفِرْ لَنَا ذُنُوبَنَا وَكَفِّرْ عَنَّا سَيِّـَٔـاتِنَا وَتَوَفَّنَا مَعَ ٱلْأَبْرَارِ ۝١٩٣

    اے ہمارے رب! ہم نے سنا کہ منادی کرنے واﻻ بآواز بلند ایمان کی طرف بلا رہا ہے کہ لوگو! اپنے رب پر ایمان لاؤ، پس ہم ایمان ﻻئے۔ یا الٰہی اب تو ہمارے گناه معاف فرما اور ہماری برائیاں ہم سے دور کر دے اور ہماری موت نیکوں کے ساتھ کر

  194. رَبَّنَا وَءَاتِنَا مَا وَعَدتَّنَا عَلَىٰ رُسُلِكَ وَلَا تُخْزِنَا يَوْمَ ٱلْقِيَـٰمَةِ‌ۗ إِنَّكَ لَا تُخْلِفُ ٱلْمِيعَادَ ۝١٩٤

    اے ہمارے پالنے والے معبود! ہمیں وه دے جس کا وعده تو نے ہم سے اپنے رسولوں کی زبانی کیا ہے اور ہمیں قیامت کے دن رسوا نہ کر، یقیناً تو وعده خلافی نہیں کرتا

  195. فَٱسْتَجَابَ لَهُمْ رَبُّهُمْ أَنِّى لَآ أُضِيعُ عَمَلَ عَـٰمِلٍ مِّنكُم مِّن ذَكَرٍ أَوْ أُنثَىٰ‌ۖ بَعْضُكُم مِّنۢ بَعْضٍ‌ۖ فَٱلَّذِينَ هَاجَرُواْ وَأُخْرِجُواْ مِن دِيَـٰرِهِمْ وَأُوذُواْ فِى سَبِيلِى وَقَـٰتَلُواْ وَقُتِلُواْ لَأُكَفِّرَنَّ عَنْهُمْ سَيِّـَٔـاتِهِمْ وَلَأُدْخِلَنَّهُمْ جَنَّـٰتٍ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ ثَوَابًا مِّنْ عِندِ ٱللَّهِ‌ۗ وَٱللَّهُ عِندَهُۥ حُسْنُ ٱلثَّوَابِ ۝١٩٥

    پس ان کے رب نے ان کی دعا قبول فرمالی کہ تم میں سے کسی کام کرنے والے کے کام کو خواه وه مرد ہو یا عورت میں ہرگز ضائع نہیں کرتا، تم آپس میں ایک دوسرے کے ہم جنس ہو، اس لئے وه لوگ جنہوں نے ہجرت کی اور اپنے گھروں سے نکال دیئے گئے اور جنہیں میری راه میں ایذا دی گئی اور جنہوں نے جہاد کیا اور شہید کئے گئے، میں ضرور ضرور ان کی برائیاں ان سے دور کر دوں گا اور بالیقین انہیں ان جنتوں میں لے جاؤں گا جن کے نیچے نہریں بہہ رہی ہیں، یہ ہے ﺛواب اللہ تعالیٰ کی طرف سے اور اللہ تعالیٰ ہی کے پاس بہترین ﺛواب ہے

  196. لَا يَغُرَّنَّكَ تَقَلُّبُ ٱلَّذِينَ كَفَرُواْ فِى ٱلْبِلَـٰدِ ۝١٩٦

    تجھے کافروں کا شہروں میں چلنا پھرنا فریب میں نہ ڈال دے،

  197. مَتَـٰعٌ قَلِيلٌ ثُمَّ مَأْوَٮٰهُمْ جَهَنَّمُ‌ۚ وَبِئْسَ ٱلْمِهَادُ ۝١٩٧

    یہ تو بہت ہی تھوڑا فائده ہے، اس کے بعد ان کا ٹھکانہ تو جہنم ہے اور وه بری جگہ ہے

  198. لَـٰكِنِ ٱلَّذِينَ ٱتَّقَوْاْ رَبَّهُمْ لَهُمْ جَنَّـٰتٌ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ خَـٰلِدِينَ فِيهَا نُزُلاً مِّنْ عِندِ ٱللَّهِ‌ۗ وَمَا عِندَ ٱللَّهِ خَيْرٌ لِّلْأَبْرَارِ ۝١٩٨

    لیکن جو لوگ اپنے رب سے ڈرتے رہے ان کے لئے جنتیں ہیں جن کے نیچے نہریں جاری ہیں، ان میں وه ہمیشہ رہیں گے یہ مہمانی ہے اللہ کی طرف سے اور نیک کاروں کے لئے جو کچھ اللہ تعالیٰ کے پاس ہے وه بہت ہی بہتر ہے

  199. وَإِنَّ مِنْ أَهْلِ ٱلْكِتَـٰبِ لَمَن يُؤْمِنُ بِٱللَّهِ وَمَآ أُنزِلَ إِلَيْكُمْ وَمَآ أُنزِلَ إِلَيْهِمْ خَـٰشِعِينَ لِلَّهِ لَا يَشْتَرُونَ بِـَٔـايَـٰتِ ٱللَّهِ ثَمَنًا قَلِيلاً‌ۗ أُوْلَـٰٓئِكَ لَهُمْ أَجْرُهُمْ عِندَ رَبِّهِمْ‌ۗ إِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ ۝١٩٩

    یقیناًاہل کتاب میں سے بعض ایسے بھی ہیں جو اللہ تعالیٰ پر ایمان ﻻتے ہیں اور تمہاری طرف جو اتارا گیا ہے اور ان کی جانب جو نازل ہوا اس پر بھی، اللہ تعالیٰ سے ڈرتے ہیں اور اللہ تعالیٰ کی آیتوں کو تھوڑی تھوڑی قیمت پر بیچتے بھی نہیں، ان کا بدلہ ان کے رب کے پاس ہے، یقیناً اللہ تعالیٰ جلد حساب لینے واﻻ ہے

  200. يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ ٱصْبِرُواْ وَصَابِرُواْ وَرَابِطُواْ وَٱتَّقُواْ ٱللَّهَ لَعَلَّكُمْ تُفْلِحُونَ ۝٢٠٠

    اے ایمان والو! تم ﺛابت قدم رہو اور ایک دوسرے کو تھامے رکھو اور جہاد کے لئے تیار رہو اور اللہ تعالیٰ سے ڈرتے رہو تاکہ تم مراد کو پہنچو
