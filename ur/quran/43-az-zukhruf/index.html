بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ

  1. حمٓ ۝١

    حٰم

  2. وَٱلْكِتَـٰبِ ٱلْمُبِينِ ۝٢

    قسم ہے اس واضح کتاب کی

  3. إِنَّا جَعَلْنَـٰهُ قُرْءَٲنًا عَرَبِيًّا لَّعَلَّكُمْ تَعْقِلُونَ ۝٣

    ہم نے اس کو عربی زبان کا قرآن بنایا ہے کہ تم سمجھ لو

  4. وَإِنَّهُۥ فِىٓ أُمِّ ٱلْكِتَـٰبِ لَدَيْنَا لَعَلِىٌّ حَكِيمٌ ۝٤

    یقیناً یہ لوح محفوظ میں ہے اور ہمارے نزدیک بلند مرتبہ حکمت والی ہے

  5. أَفَنَضْرِبُ عَنكُمُ ٱلذِّكْرَ صَفْحًا أَن كُنتُمْ قَوْمًا مُّسْرِفِينَ ۝٥

    کیا ہم اس نصیحت کو تم سے اس بنا پر ہٹالیں کہ تم حد سے گزر جانے والے لوگ ہو

  6. وَكَمْ أَرْسَلْنَا مِن نَّبِىٍّ فِى ٱلْأَوَّلِينَ ۝٦

    اور ہم نے اگلے لوگوں میں بھی کتنے ہی نبی بھیجے

  7. وَمَا يَأْتِيهِم مِّن نَّبِىٍّ إِلَّا كَانُواْ بِهِۦ يَسْتَهْزِءُونَ ۝٧

    جو نبی ان کے پاس آیا انہوں نے اس کا مذاق اڑایا

  8. فَأَهْلَكْنَآ أَشَدَّ مِنْهُم بَطْشًا وَمَضَىٰ مَثَلُ ٱلْأَوَّلِينَ ۝٨

    پس ہم نے ان سے زیاده زور آوروں کو تباه کر ڈاﻻ اور اگلوں کی مثال گزر چکی ہے

  9. وَلَئِن سَأَلْتَهُم مَّنْ خَلَقَ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضَ لَيَقُولُنَّ خَلَقَهُنَّ ٱلْعَزِيزُ ٱلْعَلِيمُ ۝٩

    اگر آپ ان سے دریافت کریں کہ آسمانوں اور زمین کو کس نے پیدا کیا تو یقیناً ان کا جواب یہی ہوگا کہ انہیں غالب و دانا (اللہ) نے ہی پیدا کیا ہے

  10. ٱلَّذِى جَعَلَ لَكُمُ ٱلْأَرْضَ مَهْدًا وَجَعَلَ لَكُمْ فِيهَا سُبُلاً لَّعَلَّكُمْ تَهْتَدُونَ ۝١٠

    وہی ہے جس نے تمہارے لیے زمین کو فرش (بچھونا) بنایا اور اس میں تمہارے لیے راستے کردیے تاکہ تم راه پالیا کرو

  11. وَٱلَّذِى نَزَّلَ مِنَ ٱلسَّمَآءِ مَآءَۢ بِقَدَرٍ فَأَنشَرْنَا بِهِۦ بَلْدَةً مَّيْتًا‌ۚ كَذَٲلِكَ تُخْرَجُونَ ۝١١

    اسی نے آسمان سے ایک اندازے کے مطابق پانی نازل فرمایا، پس ہم نے اس سے مرده شہر کو زنده کردیا۔ اسی طرح تم نکالے جاؤ گے

  12. وَٱلَّذِى خَلَقَ ٱلْأَزْوَٲجَ كُلَّهَا وَجَعَلَ لَكُم مِّنَ ٱلْفُلْكِ وَٱلْأَنْعَـٰمِ مَا تَرْكَبُونَ ۝١٢

    جس نے تمام چیزوں کے جوڑے بنائے اور تمہارے لیے کشتیاں بنائیں اور چوپائے جانور (پیدا کیے) جن پر تم سوار ہوتے ہو

  13. لِتَسْتَوُۥاْ عَلَىٰ ظُهُورِهِۦ ثُمَّ تَذْكُرُواْ نِعْمَةَ رَبِّكُمْ إِذَا ٱسْتَوَيْتُمْ عَلَيْهِ وَتَقُولُواْ سُبْحَـٰنَ ٱلَّذِى سَخَّرَ لَنَا هَـٰذَا وَمَا كُنَّا لَهُۥ مُقْرِنِينَ ۝١٣

    تاکہ تم ان کی پیٹھ پر جم کر سوار ہوا کرو پھر اپنے رب کی نعمت کو یاد کرو جب اس پر ٹھیک ٹھاک بیٹھ جاؤ، اور کہو پاک ذات ہے اس کی جس نے اسے ہمارے بس میں کردیا حاﻻنکہ ہمیں اسے قابو کرنے کی طاقت نہ تھی

  14. وَإِنَّآ إِلَىٰ رَبِّنَا لَمُنقَلِبُونَ ۝١٤

    اور بالیقین ہم اپنے رب کی طرف لوٹ کر جانے والے ہیں

  15. وَجَعَلُواْ لَهُۥ مِنْ عِبَادِهِۦ جُزْءًا‌ۚ إِنَّ ٱلْإِنسَـٰنَ لَكَفُورٌ مُّبِينٌ ۝١٥

    اور انہوں نے اللہ کے بعض بندوں کو اس کا جز ٹھہرا دیا یقیناً انسان کھلم کھلا ناشکرا ہے

  16. أَمِ ٱتَّخَذَ مِمَّا يَخْلُقُ بَنَاتٍ وَأَصْفَـٰكُم بِٱلْبَنِينَ ۝١٦

    کیا اللہ تعالیٰ نے اپنی مخلوق میں سے بیٹیاں تو خود رکھ لیں اور تمہیں بیٹوں سے نوازا

  17. وَإِذَا بُشِّرَ أَحَدُهُم بِمَا ضَرَبَ لِلرَّحْمَـٰنِ مَثَلاً ظَلَّ وَجْهُهُۥ مُسْوَدًّا وَهُوَ كَظِيمٌ ۝١٧

    (حاﻻنکہ) ان میں سے کسی کو جب اس چیز کی خبر دی جائے جس کی مثال اس نے (اللہ) رحمٰن کے لیے بیان کی ہے تو اس کا چہره سیاه پڑ جاتا ہے اور وه غمگین ہو جاتا ہے

  18. أَوَمَن يُنَشَّؤُاْ فِى ٱلْحِلْيَةِ وَهُوَ فِى ٱلْخِصَامِ غَيْرُ مُبِينٍ ۝١٨

    کیا (اللہ کی اوﻻد لڑکیاں ہیں) جو زیورات میں پلیں اور جھگڑے میں (اپنی بات) واضح نہ کرسکیں؟

  19. وَجَعَلُواْ ٱلْمَلَـٰٓئِكَةَ ٱلَّذِينَ هُمْ عِبَـٰدُ ٱلرَّحْمَـٰنِ إِنَـٰثًا‌ۚ أَشَهِدُواْ خَلْقَهُمْ‌ۚ سَتُكْتَبُ شَهَـٰدَتُهُمْ وَيُسْــَٔلُونَ ۝١٩

    اور انہوں نے فرشتوں کو جو رحمٰن کے عبادت گزار ہیں عورتیں قرار دے لیا۔ کیا ان کی پیدائش کے موقع پر یہ موجود تھے؟ ان کی یہ گواہی لکھ لی جائے گی اور ان سے (اس چیز کی) باز پرس کی جائے گی

  20. وَقَالُواْ لَوْ شَآءَ ٱلرَّحْمَـٰنُ مَا عَبَدْنَـٰهُم‌ۗ مَّا لَهُم بِذَٲلِكَ مِنْ عِلْمٍ‌ۖ إِنْ هُمْ إِلَّا يَخْرُصُونَ ۝٢٠

    اور کہتے ہیں اگر اللہ چاہتا تو ہم ان کی عبادت نہ کرتے۔ انہیں اس کی کچھ خبر نہیں، یہ تو صرف اٹکل پچو (جھوٹ باتیں) کہتے ہیں۔

  21. أَمْ ءَاتَيْنَـٰهُمْ كِتَـٰبًا مِّن قَبْلِهِۦ فَهُم بِهِۦ مُسْتَمْسِكُونَ ۝٢١

    کیا ہم نے انہیں اس سے پہلے کوئی (اور) کتاب دی ہے جسے یہ مضبوط تھامے ہوئے ہیں

  22. بَلْ قَالُوٓاْ إِنَّا وَجَدْنَآ ءَابَآءَنَا عَلَىٰٓ أُمَّةٍ وَإِنَّا عَلَىٰٓ ءَاثَـٰرِهِم مُّهْتَدُونَ ۝٢٢

    (نہیں نہیں) بلکہ یہ تو کہتے ہیں کہ ہم نے اپنے باپ دادا کو ایک مذہب پر پایا اور ہم انہی کے نقش قدم پر چل کر راه یافتہ ہیں

  23. وَكَذَٲلِكَ مَآ أَرْسَلْنَا مِن قَبْلِكَ فِى قَرْيَةٍ مِّن نَّذِيرٍ إِلَّا قَالَ مُتْرَفُوهَآ إِنَّا وَجَدْنَآ ءَابَآءَنَا عَلَىٰٓ أُمَّةٍ وَإِنَّا عَلَىٰٓ ءَاثَـٰرِهِم مُّقْتَدُونَ ۝٢٣

    اسی طرح آپ سے پہلے بھی ہم نے جس بستی میں کوئی ڈرانے واﻻ بھیجا وہاں کے آسوده حال لوگوں نے یہی جواب دیا کہ ہم نے اپنے باپ دادا کو (ایک راه پر اور) ایک دین پر پایا اور ہم تو انہی کے نقش پا کی پیروی کرنے والے ہیں

  24. ۞ قَـٰلَ أَوَلَوْ جِئْتُكُم بِأَهْدَىٰ مِمَّا وَجَدتُّمْ عَلَيْهِ ءَابَآءَكُمْ‌ۖ قَالُوٓاْ إِنَّا بِمَآ أُرْسِلْتُم بِهِۦ كَـٰفِرُونَ ۝٢٤

    (نبی نے) کہا بھی کہ اگرچہ میں تمہارے پاس اس سے بہت بہتر (مقصود تک پہنچانے واﻻ) طریقہ لے آیا ہوں جس پر تم نے اپنے باپ دادوں کو پایا، تو انہوں نے جواب دیا کہ ہم اس کے منکر ہیں جسے دے کر تمہیں بھیجا گیا ہے

  25. فَٱنتَقَمْنَا مِنْهُمْ‌ۖ فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلْمُكَذِّبِينَ ۝٢٥

    پس ہم نے ان سے انتقام لیا اور دیکھ لے جھٹلانے والوں کا کیسا انجام ہوا؟

  26. وَإِذْ قَالَ إِبْرَٲهِيمُ لِأَبِيهِ وَقَوْمِهِۦٓ إِنَّنِى بَرَآءٌ مِّمَّا تَعْبُدُونَ ۝٢٦

    اور جبکہ ابراہیم (علیہ السلام) نے اپنے والد سے اور اپنی قوم سے فرمایا کہ میں ان چیزوں سے بیزار ہوں جن کی تم عبادت کرتے ہو

  27. إِلَّا ٱلَّذِى فَطَرَنِى فَإِنَّهُۥ سَيَهْدِينِ ۝٢٧

    بجز اس ذات کے جس نے مجھے پیدا کیا ہے اور وہی مجھے ہدایت بھی کرے گا

  28. وَجَعَلَهَا كَلِمَةَۢ بَاقِيَةً فِى عَقِبِهِۦ لَعَلَّهُمْ يَرْجِعُونَ ۝٢٨

    اور (ابراہیم علیہ السلام) اسی کو اپنی اوﻻد میں بھی باقی رہنے والی بات قائم کر گئے تاکہ لوگ (شرک سے) باز آتے رہیں

  29. بَلْ مَتَّعْتُ هَـٰٓؤُلَآءِ وَءَابَآءَهُمْ حَتَّىٰ جَآءَهُمُ ٱلْحَقُّ وَرَسُولٌ مُّبِينٌ ۝٢٩

    بلکہ میں نے ان لوگوں کو اور ان کے باپ دادوں کو سامان (اور اسباب) دیا، یہاں تک کہ ان کے پاس حق اور صاف صاف سنانے واﻻ رسول آگیا

  30. وَلَمَّا جَآءَهُمُ ٱلْحَقُّ قَالُواْ هَـٰذَا سِحْرٌ وَإِنَّا بِهِۦ كَـٰفِرُونَ ۝٣٠

    اور حق کے پہنچتے ہی یہ بول پڑے کہ یہ تو جادو ہے اور ہم اس کے منکر ہیں

  31. وَقَالُواْ لَوْلَا نُزِّلَ هَـٰذَا ٱلْقُرْءَانُ عَلَىٰ رَجُلٍ مِّنَ ٱلْقَرْيَتَيْنِ عَظِيمٍ ۝٣١

    اور کہنے لگے، یہ قرآن ان دونوں بستیوں میں سے کسی بڑے آدمی پر کیوں نہ نازل کیا گیا

  32. أَهُمْ يَقْسِمُونَ رَحْمَتَ رَبِّكَ‌ۚ نَحْنُ قَسَمْنَا بَيْنَهُم مَّعِيشَتَهُمْ فِى ٱلْحَيَوٲةِ ٱلدُّنْيَا‌ۚ وَرَفَعْنَا بَعْضَهُمْ فَوْقَ بَعْضٍ دَرَجَـٰتٍ لِّيَتَّخِذَ بَعْضُهُم بَعْضًا سُخْرِيًّا‌ۗ وَرَحْمَتُ رَبِّكَ خَيْرٌ مِّمَّا يَجْمَعُونَ ۝٣٢

    کیا آپ کے رب کی رحمت کو یہ تقسیم کرتے ہیں؟ ہم نے ہی ان کی زندگانیٴ دنیا کی روزی ان میں تقسیم کی ہے اور ایک کو دوسرے سے بلند کیا ہے تاکہ ایک دوسرے کو ماتحت کر لے جسے یہ لوگ سمیٹتے پھرتے ہیں اس سے آپ کے رب کی رحمت بہت ہی بہتر ہے

  33. وَلَوْلَآ أَن يَكُونَ ٱلنَّاسُ أُمَّةً وَٲحِدَةً لَّجَعَلْنَا لِمَن يَكْفُرُ بِٱلرَّحْمَـٰنِ لِبُيُوتِهِمْ سُقُفًا مِّن فِضَّةٍ وَمَعَارِجَ عَلَيْهَا يَظْهَرُونَ ۝٣٣

    اور اگر یہ بات نہ ہوتی کہ تمام لوگ ایک ہی طریقہ پر ہو جائیں گے تو رحمٰن کے ساتھ کفر کرنے والوں کے گھروں کی چھتوں کو ہم چاندی کی بنادیتے۔ اور زینوں کو (بھی) جن پر چڑھا کرتے

  34. وَلِبُيُوتِهِمْ أَبْوَٲبًا وَسُرُرًا عَلَيْهَا يَتَّكِــُٔونَ ۝٣٤

    اور ان کے گھروں کے دروازے اور تخت بھی جن پر وه تکیہ لگا لگا کر بیٹھتے

  35. وَزُخْرُفًا‌ۚ وَإِن كُلُّ ذَٲلِكَ لَمَّا مَتَـٰعُ ٱلْحَيَوٲةِ ٱلدُّنْيَا‌ۚ وَٱلْأَخِرَةُ عِندَ رَبِّكَ لِلْمُتَّقِينَ ۝٣٥

    اور سونے کے بھی، اور یہ سب کچھ یونہی سا دنیا کی زندگی کا فائده ہے اور آخرت تو آپ کے رب کے نزدیک (صرف) پرہیزگاروں کے لیے (ہی) ہے۔

  36. وَمَن يَعْشُ عَن ذِكْرِ ٱلرَّحْمَـٰنِ نُقَيِّضْ لَهُۥ شَيْطَـٰنًا فَهُوَ لَهُۥ قَرِينٌ ۝٣٦

    اور جو شخص رحمٰن کی یاد سے غفلت کرے ہم اس پر ایک شیطان مقرر کردیتے ہیں وہی اس کا ساتھی رہتا ہے

  37. وَإِنَّهُمْ لَيَصُدُّونَهُمْ عَنِ ٱلسَّبِيلِ وَيَحْسَبُونَ أَنَّهُم مُّهْتَدُونَ ۝٣٧

    اور وه انہیں راه سے روکتے ہیں اور یہ اسی خیال میں رہتے ہیں کہ یہ ہدایت یافتہ ہیں

  38. حَتَّىٰٓ إِذَا جَآءَنَا قَالَ يَـٰلَيْتَ بَيْنِى وَبَيْنَكَ بُعْدَ ٱلْمَشْرِقَيْنِ فَبِئْسَ ٱلْقَرِينُ ۝٣٨

    یہاں تک کہ جب وه ہمارے پاس آئے گا کہے گا کاش! میرے اور تیرے درمیان مشرق اور مغرب کی دوری ہوتی (تو) بڑا برا ساتھی ہے

  39. وَلَن يَنفَعَكُمُ ٱلْيَوْمَ إِذ ظَّلَمْتُمْ أَنَّكُمْ فِى ٱلْعَذَابِ مُشْتَرِكُونَ ۝٣٩

    اور جب کہ تم ﻇالم ٹھہر چکے تو تمہیں آج ہرگز تم سب کا عذاب میں شریک ہونا کوئی نفع نہ دے گا

  40. أَفَأَنتَ تُسْمِعُ ٱلصُّمَّ أَوْ تَهْدِى ٱلْعُمْىَ وَمَن كَانَ فِى ضَلَـٰلٍ مُّبِينٍ ۝٤٠

    کیا پس تو بہرے کو سنا سکتا ہے یا اندھے کو راه دکھا سکتا ہے اور اسے جو کھلی گمراہی میں ہو

  41. فَإِمَّا نَذْهَبَنَّ بِكَ فَإِنَّا مِنْهُم مُّنتَقِمُونَ ۝٤١

    پس اگر ہم تجھے یہاں سے لے بھی جائیں تو بھی ہم ان سے بدلہ لینے والے ہیں

  42. أَوْ نُرِيَنَّكَ ٱلَّذِى وَعَدْنَـٰهُمْ فَإِنَّا عَلَيْهِم مُّقْتَدِرُونَ ۝٤٢

    یا جو کچھ ان سے وعده کیا ہے وه تجھے دکھا دیں ہم ان پر بھی قدرت رکھتے ہیں

  43. فَٱسْتَمْسِكْ بِٱلَّذِىٓ أُوحِىَ إِلَيْكَ‌ۖ إِنَّكَ عَلَىٰ صِرَٲطٍ مُّسْتَقِيمٍ ۝٤٣

    پس جو وحی آپ کی طرف کی گئی ہے اسے مضبوط تھامے رہیں بیشک آپ راه راست پر ہیں

  44. وَإِنَّهُۥ لَذِكْرٌ لَّكَ وَلِقَوْمِكَ‌ۖ وَسَوْفَ تُسْــَٔلُونَ ۝٤٤

    اور یقیناً یہ (خود) آپ کے لیے اور آپ کی قوم کے لیے نصیحت ہے اور عنقریب تم لوگ پوچھے جاؤ گے

  45. وَسْــَٔلْ مَنْ أَرْسَلْنَا مِن قَبْلِكَ مِن رُّسُلِنَآ أَجَعَلْنَا مِن دُونِ ٱلرَّحْمَـٰنِ ءَالِهَةً يُعْبَدُونَ ۝٤٥

    اور ہمارے ان نبیوں سے پوچھو! جنہیں ہم نے آپ سے پہلے بھیجا تھا کہ کیا ہم نے سوائے رحمٰن کے اور معبود مقرر کیے تھے جن کی عبادت کی جائے؟

  46. وَلَقَدْ أَرْسَلْنَا مُوسَىٰ بِـَٔـايَـٰتِنَآ إِلَىٰ فِرْعَوْنَ وَمَلَإِيْهِۦ فَقَالَ إِنِّى رَسُولُ رَبِّ ٱلْعَـٰلَمِينَ ۝٤٦

    اور ہم نے موسیٰ (علیہ السلام) کو اپنی نشانیاں دے کر فرعون اور اس کے امراء کے پاس بھیجا تو (موسیٰ علیہ السلام نے جاکر) کہا کہ میں تمام جہانوں کے رب کا رسول ہوں

  47. فَلَمَّا جَآءَهُم بِـَٔـايَـٰتِنَآ إِذَا هُم مِّنْهَا يَضْحَكُونَ ۝٤٧

    پس جب وه ہماری نشانیاں لے کر ان کے پاس آئے تو وه بےساختہ ان پر ہنسنے لگے

  48. وَمَا نُرِيهِم مِّنْ ءَايَةٍ إِلَّا هِىَ أَكْبَرُ مِنْ أُخْتِهَا‌ۖ وَأَخَذْنَـٰهُم بِٱلْعَذَابِ لَعَلَّهُمْ يَرْجِعُونَ ۝٤٨

    اور ہم انہیں جو نشانی دکھاتے تھے وه دوسری سے بڑھی چڑھی ہوتی تھی اور ہم نے انہیں عذاب میں پکڑا تاکہ وه باز آجائیں

  49. وَقَالُواْ يَـٰٓأَيُّهَ ٱلسَّاحِرُ ٱدْعُ لَنَا رَبَّكَ بِمَا عَهِدَ عِندَكَ إِنَّنَا لَمُهْتَدُونَ ۝٤٩

    اور انہوں نے کہا اے جادوگر! ہمارے لیے اپنے رب سے اس کی دعا کر جس کا اس نے تجھ سے وعده کر رکھا ہے، یقین مان کہ ہم راه پر لگ جائیں گے

  50. فَلَمَّا كَشَفْنَا عَنْهُمُ ٱلْعَذَابَ إِذَا هُمْ يَنكُثُونَ ۝٥٠

    پھر جب ہم نے وه عذاب ان سے ہٹالیا انہوں نے اسی وقت اپنا قول وقرار توڑ دیا

  51. وَنَادَىٰ فِرْعَوْنُ فِى قَوْمِهِۦ قَالَ يَـٰقَوْمِ أَلَيْسَ لِى مُلْكُ مِصْرَ وَهَـٰذِهِ ٱلْأَنْهَـٰرُ تَجْرِى مِن تَحْتِىٓ‌ۖ أَفَلَا تُبْصِرُونَ ۝٥١

    اور فرعون نے اپنی قوم میں منادی کرائی اور کہا اے میری قوم! کیا مصر کا ملک میرا نہیں؟ اور میرے (محلوں کے) نیچے یہ نہریں بہہ رہی ہیں، کیا تم دیکھتے نہیں؟

  52. أَمْ أَنَا۟ خَيْرٌ مِّنْ هَـٰذَا ٱلَّذِى هُوَ مَهِينٌ وَلَا يَكَادُ يُبِينُ ۝٥٢

    بلکہ میں بہتر ہوں بہ نسبت اس کے جو بےتوقیر ہے اور صاف بول بھی نہیں سکتا

  53. فَلَوْلَآ أُلْقِىَ عَلَيْهِ أَسْوِرَةٌ مِّن ذَهَبٍ أَوْ جَآءَ مَعَهُ ٱلْمَلَـٰٓئِكَةُ مُقْتَرِنِينَ ۝٥٣

    اچھا اس پر سونے کے کنگن کیوں نہیں آپڑے یا اس کے ساتھ پر باندھ کر فرشتے ہی آجاتے

  54. فَٱسْتَخَفَّ قَوْمَهُۥ فَأَطَاعُوهُ‌ۚ إِنَّهُمْ كَانُواْ قَوْمًا فَـٰسِقِينَ ۝٥٤

    اس نے اپنی قوم کو بہلایا پھسلایا اور انہوں نے اسی کی مان لی، یقیناً یہ سارے ہی نافرمان لوگ تھے

  55. فَلَمَّآ ءَاسَفُونَا ٱنتَقَمْنَا مِنْهُمْ فَأَغْرَقْنَـٰهُمْ أَجْمَعِينَ ۝٥٥

    پھر جب انہوں نے ہمیں غصہ دﻻیا تو ہم نے ان سے انتقام لیا اور سب کو ڈبو دیا

  56. فَجَعَلْنَـٰهُمْ سَلَفًا وَمَثَلاً لِّلْأَخِرِينَ ۝٥٦

    پس ہم نے انہیں گیا گزرا کردیا اور پچھلوں کے لیے مثال بنادی

  57. ۞ وَلَمَّا ضُرِبَ ٱبْنُ مَرْيَمَ مَثَلاً إِذَا قَوْمُكَ مِنْهُ يَصِدُّونَ ۝٥٧

    اور جب ابن مریم کی مثال بیان کی گئی تو اس سے تیری قوم (خوشی سے) چیخنے لگی ہے

  58. وَقَالُوٓاْ ءَأَٲلِهَتُنَا خَيْرٌ أَمْ هُوَ‌ۚ مَا ضَرَبُوهُ لَكَ إِلَّا جَدَلاَۢ‌ۚ بَلْ هُمْ قَوْمٌ خَصِمُونَ ۝٥٨

    اور انہوں نے کہا کہ ہمارے معبود اچھے ہیں یا وه؟ تجھ سے ان کا یہ کہنا محض جھگڑے کی غرض سے ہے، بلکہ یہ لوگ ہیں ہی جھگڑالو

  59. إِنْ هُوَ إِلَّا عَبْدٌ أَنْعَمْنَا عَلَيْهِ وَجَعَلْنَـٰهُ مَثَلاً لِّبَنِىٓ إِسْرَٲٓءِيلَ ۝٥٩

    عیسیٰ (علیہ السلام) بھی صرف بنده ہی ہے جس پر ہم نے احسان کیا اور اسے بنی اسرائیل کے لیے نشان قدرت بنایا

  60. وَلَوْ نَشَآءُ لَجَعَلْنَا مِنكُم مَّلَـٰٓئِكَةً فِى ٱلْأَرْضِ يَخْلُفُونَ ۝٦٠

    اگر ہم چاہتے تو تمہارے عوض فرشتے کردیتے جو زمین میں جانشینی کرتے

  61. وَإِنَّهُۥ لَعِلْمٌ لِّلسَّاعَةِ فَلَا تَمْتَرُنَّ بِهَا وَٱتَّبِعُونِ‌ۚ هَـٰذَا صِرَٲطٌ مُّسْتَقِيمٌ ۝٦١

    اور یقیناً عیسیٰ (علیہ السلام) قیامت کی علامت ہے پس تم (قیامت) کے بارے میں شک نہ کرو اور میری تابعداری کرو، یہی سیدھی راه ہے

  62. وَلَا يَصُدَّنَّكُمُ ٱلشَّيْطَـٰنُ‌ۖ إِنَّهُۥ لَكُمْ عَدُوٌّ مُّبِينٌ ۝٦٢

    اور شیطان تمہیں روک نہ دے، یقیناً وه تمہارا صریح دشمن ہے

  63. وَلَمَّا جَآءَ عِيسَىٰ بِٱلْبَيِّنَـٰتِ قَالَ قَدْ جِئْتُكُم بِٱلْحِكْمَةِ وَلِأُبَيِّنَ لَكُم بَعْضَ ٱلَّذِى تَخْتَلِفُونَ فِيهِ‌ۖ فَٱتَّقُواْ ٱللَّهَ وَأَطِيعُونِ ۝٦٣

    اور جب عیسیٰ (علیہ السلام) معجزے کو ﻻئے تو کہا۔ کہ میں تمہارے پاس حکمت ﻻیا ہوں اور اس لیے آیا ہوں کہ جن بعض چیزوں میں تم مختلف ہو، انہیں واضح کردوں، پس تم اللہ تعالیٰ سے ڈرو اور میرا کہا مانو

  64. إِنَّ ٱللَّهَ هُوَ رَبِّى وَرَبُّكُمْ فَٱعْبُدُوهُ‌ۚ هَـٰذَا صِرَٲطٌ مُّسْتَقِيمٌ ۝٦٤

    میرا اور تمہارا رب فقط اللہ تعالیٰ ہی ہے۔ پس تم سب اس کی عبادت کرو۔ راه راست (یہی) ہے

  65. فَٱخْتَلَفَ ٱلْأَحْزَابُ مِنۢ بَيْنِهِمْ‌ۖ فَوَيْلٌ لِّلَّذِينَ ظَلَمُواْ مِنْ عَذَابِ يَوْمٍ أَلِيمٍ ۝٦٥

    پھر (بنی اسرائیل کی) جماعتوں نے آپس میں اختلاف کیا، پس ﻇالموں کے لیے خرابی ہے دکھ والے دن کی آفت سے

  66. هَلْ يَنظُرُونَ إِلَّا ٱلسَّاعَةَ أَن تَأْتِيَهُم بَغْتَةً وَهُمْ لَا يَشْعُرُونَ ۝٦٦

    یہ لوگ صرف قیامت کے منتظر ہیں کہ وه اچانک ان پر آپڑے اور انہیں خبر بھی نہ ہو

  67. ٱلْأَخِلَّآءُ يَوْمَئِذِۭ بَعْضُهُمْ لِبَعْضٍ عَدُوٌّ إِلَّا ٱلْمُتَّقِينَ ۝٦٧

    اس دن (گہرے) دوست بھی ایک دوسرے کے دشمن بن جائیں گے سوائے پرہیزگاروں کے

  68. يَـٰعِبَادِ لَا خَوْفٌ عَلَيْكُمُ ٱلْيَوْمَ وَلَآ أَنتُمْ تَحْزَنُونَ ۝٦٨

    میرے بندو! آج تو تم پر کوئی خوف (و ہراس) ہے اور نہ تم (بد دل اور) غمزده ہوگے

  69. ٱلَّذِينَ ءَامَنُواْ بِـَٔـايَـٰتِنَا وَكَانُواْ مُسْلِمِينَ ۝٦٩

    جو ہماری آیتوں پر ایمان ﻻئے اور تھے بھی وه (فرماں بردار) مسلمان

  70. ٱدْخُلُواْ ٱلْجَنَّةَ أَنتُمْ وَأَزْوَٲجُكُمْ تُحْبَرُونَ ۝٧٠

    تم اور تمہاری بیویاں ہشاش بشاش (راضی خوشی) جنت میں چلے جاؤ

  71. يُطَافُ عَلَيْهِم بِصِحَافٍ مِّن ذَهَبٍ وَأَكْوَابٍ‌ۖ وَفِيهَا مَا تَشْتَهِيهِ ٱلْأَنفُسُ وَتَلَذُّ ٱلْأَعْيُنُ‌ۖ وَأَنتُمْ فِيهَا خَـٰلِدُونَ ۝٧١

    ان کے چاروں طرف سے سونے کی رکابیاں اور سونے کے گلاسوں کا دور چلایا جائے گا، ان کے جی جس چیز کی خواہش کریں اور جس سے ان کی آنکھیں لذت پائیں، سب وہاں ہوگا اور تم اس میں ہمیشہ رہو گے

  72. وَتِلْكَ ٱلْجَنَّةُ ٱلَّتِىٓ أُورِثْتُمُوهَا بِمَا كُنتُمْ تَعْمَلُونَ ۝٧٢

    یہی وه بہشت ہے کہ تم اپنے اعمال کے بدلے اس کے وارث بنائے گئے ہو

  73. لَكُمْ فِيهَا فَـٰكِهَةٌ كَثِيرَةٌ مِّنْهَا تَأْكُلُونَ ۝٧٣

    یہاں تمہارے لیے بکثرت میوے ہیں جنہیں تم کھاتے رہو گے

  74. إِنَّ ٱلْمُجْرِمِينَ فِى عَذَابِ جَهَنَّمَ خَـٰلِدُونَ ۝٧٤

    بیشک گنہگار لوگ عذاب دوزخ میں ہمیشہ رہیں گے

  75. لَا يُفَتَّرُ عَنْهُمْ وَهُمْ فِيهِ مُبْلِسُونَ ۝٧٥

    یہ عذاب کبھی بھی ان سے ہلکا نہ کیا جائے گا اور وه اسی میں مایوس پڑے رہیں گے

  76. وَمَا ظَلَمْنَـٰهُمْ وَلَـٰكِن كَانُواْ هُمُ ٱلظَّـٰلِمِينَ ۝٧٦

    اور ہم نے ان پر ﻇلم نہیں کیا بلکہ یہ خود ہی ﻇالم تھے

  77. وَنَادَوْاْ يَـٰمَـٰلِكُ لِيَقْضِ عَلَيْنَا رَبُّكَ‌ۖ قَالَ إِنَّكُم مَّـٰكِثُونَ ۝٧٧

    اور پکار پکار کر کہیں گے کہ اے مالک! تیرا رب ہمارا کام ہی تمام کردے، وه کہے گا کہ تمہیں تو (ہمیشہ) رہنا ہے

  78. لَقَدْ جِئْنَـٰكُم بِٱلْحَقِّ وَلَـٰكِنَّ أَكْثَرَكُمْ لِلْحَقِّ كَـٰرِهُونَ ۝٧٨

    ہم تو تمہارے پاس حق لے آئے لیکن تم میں سے اکثر لوگ حق سے نفرت رکھنے والے تھے؟

  79. أَمْ أَبْرَمُوٓاْ أَمْرًا فَإِنَّا مُبْرِمُونَ ۝٧٩

    کیا انہوں نے کسی کام کا پختہ اراده کرلیا ہے تو یقین مانو کہ ہم بھی پختہ کام کرنے والے ہیں

  80. أَمْ يَحْسَبُونَ أَنَّا لَا نَسْمَعُ سِرَّهُمْ وَنَجْوَٮٰهُم‌ۚ بَلَىٰ وَرُسُلُنَا لَدَيْهِمْ يَكْتُبُونَ ۝٨٠

    کیا ان کا یہ خیال ہے کہ ہم ان کی پوشیده باتوں کو اور ان کی سرگوشیوں کو نہیں سنتے، (یقیناً ہم برابر سن رہے ہیں) بلکہ ہمارے بھیجے ہوئے ان کے پاس ہی لکھ رہے ہیں

  81. قُلْ إِن كَانَ لِلرَّحْمَـٰنِ وَلَدٌ فَأَنَا۟ أَوَّلُ ٱلْعَـٰبِدِينَ ۝٨١

    آپ کہہ دیجئے! کہ اگر بالفرض رحمٰن کی اوﻻد ہو تو میں سب سے پہلے عبادت کرنے واﻻ ہوتا

  82. سُبْحَـٰنَ رَبِّ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ رَبِّ ٱلْعَرْشِ عَمَّا يَصِفُونَ ۝٨٢

    آسمانوں اور زمین اور عرش کا رب جو کچھ یہ بیان کرتے ہیں اس سے (بہت) پاک ہے

  83. فَذَرْهُمْ يَخُوضُواْ وَيَلْعَبُواْ حَتَّىٰ يُلَـٰقُواْ يَوْمَهُمُ ٱلَّذِى يُوعَدُونَ ۝٨٣

    اب آپ انہیں اسی بحﺚ مباحثہ اور کھیل کود میں چھوڑ دیجئے، یہاں تک کہ انہیں اس دن سے سابقہ پڑجائے جن کا یہ وعده دیے جاتے ہیں

  84. وَهُوَ ٱلَّذِى فِى ٱلسَّمَآءِ إِلَـٰهٌ وَفِى ٱلْأَرْضِ إِلَـٰهٌ‌ۚ وَهُوَ ٱلْحَكِيمُ ٱلْعَلِيمُ ۝٨٤

    وہی آسمانوں میں معبود ہے اور زمین میں بھی وہی قابل عبادت ہے اور وه بڑی حکمت واﻻ اور پورے علم واﻻ ہے

  85. وَتَبَارَكَ ٱلَّذِى لَهُۥ مُلْكُ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا وَعِندَهُۥ عِلْمُ ٱلسَّاعَةِ وَإِلَيْهِ تُرْجَعُونَ ۝٨٥

    اور وه بہت برکتوں واﻻ ہے جس کے پاس آسمان وزمین اور ان کے درمیان کی بادشاہت ہے، اور قیامت کا علم بھی اسی کے پاس ہے اور اسی کی جانب تم سب لوٹائے جاؤ گے

  86. وَلَا يَمْلِكُ ٱلَّذِينَ يَدْعُونَ مِن دُونِهِ ٱلشَّفَـٰعَةَ إِلَّا مَن شَهِدَ بِٱلْحَقِّ وَهُمْ يَعْلَمُونَ ۝٨٦

    جنہیں یہ لوگ اللہ کے سوا پکارتے ہیں وه شفاعت کرنے کا اختیار نہیں رکھتے، ہاں (مستحق شفاعت وه ہیں) جو حق بات کا اقرار کریں اور انہیں علم بھی ہو

  87. وَلَئِن سَأَلْتَهُم مَّنْ خَلَقَهُمْ لَيَقُولُنَّ ٱللَّهُ‌ۖ فَأَنَّىٰ يُؤْفَكُونَ ۝٨٧

    اگر آپ ان سے دریافت کریں کہ انہیں کس نے پیدا کیا ہے؟ تو یقیناً یہ جواب دیں گے کہ اللہ نے، پھر یہ کہاں الٹے جاتے ہیں؟

  88. وَقِيلِهِۦ يَـٰرَبِّ إِنَّ هَـٰٓؤُلَآءِ قَوْمٌ لَّا يُؤْمِنُونَ ۝٨٨

    اور ان کا (پیغمبر کا اکثر) یہ کہنا کہ اے میرے رب! یقیناً یہ وه لوگ ہیں جو ایمان نہیں ﻻتے

  89. فَٱصْفَحْ عَنْهُمْ وَقُلْ سَلَـٰمٌ‌ۚ فَسَوْفَ يَعْلَمُونَ ۝٨٩

    پس آپ ان سے منھ پھیر لیں اور کہہ دیں۔ (اچھا بھائی) سلام! انہیں عنقریب (خود ہی) معلوم ہوجائے گا
