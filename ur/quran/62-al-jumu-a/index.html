بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ

  1. يُسَبِّحُ لِلَّهِ مَا فِى ٱلسَّمَـٰوَٲتِ وَمَا فِى ٱلْأَرْضِ ٱلْمَلِكِ ٱلْقُدُّوسِ ٱلْعَزِيزِ ٱلْحَكِيمِ ۝١

    (ساری چیزیں) جو آسمانوں اور زمین میں ہیں اللہ تعالیٰ کی پاکی بیان کرتی ہیں (جو) بادشاه نہایت پاک (ہے) غالب وباحکمت ہے

  2. هُوَ ٱلَّذِى بَعَثَ فِى ٱلْأُمِّيِّــۧنَ رَسُولاً مِّنْهُمْ يَتْلُواْ عَلَيْهِمْ ءَايَـٰتِهِۦ وَيُزَكِّيهِمْ وَيُعَلِّمُهُمُ ٱلْكِتَـٰبَ وَٱلْحِكْمَةَ وَإِن كَانُواْ مِن قَبْلُ لَفِى ضَلَـٰلٍ مُّبِينٍ ۝٢

    وہی ہے جس نے ناخوانده لوگوں میں ان ہی میں سے ایک رسول بھیجا جو انہیں اس کی آیتیں پڑھ کر سناتا ہے اور ان کو پاک کرتا ہے اور انہیں کتاب وحکمت سکھاتا ہے۔ یقیناً یہ اس سے پہلے کھلی گمراہی میں تھے

  3. وَءَاخَرِينَ مِنْهُمْ لَمَّا يَلْحَقُواْ بِهِمْ‌ۚ وَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ۝٣

    اور دوسروں کے لیے بھی انہیں میں سے جو اب تک ان سے نہیں ملے۔ اور وہی غالب باحکمت ہے

  4. ذَٲلِكَ فَضْلُ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ‌ۚ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ۝٤

    یہ اللہ کا فضل ہے جسے چاہے اپنا فضل دے اور اللہ تعالیٰ بہت بڑے فضل کا مالک ہے

  5. مَثَلُ ٱلَّذِينَ حُمِّلُواْ ٱلتَّوْرَٮٰةَ ثُمَّ لَمْ يَحْمِلُوهَا كَمَثَلِ ٱلْحِمَارِ يَحْمِلُ أَسْفَارَۢا‌ۚ بِئْسَ مَثَلُ ٱلْقَوْمِ ٱلَّذِينَ كَذَّبُواْ بِـَٔـايَـٰتِ ٱللَّهِ‌ۚ وَٱللَّهُ لَا يَهْدِى ٱلْقَوْمَ ٱلظَّـٰلِمِينَ ۝٥

    جن لوگوں کو تورات پر عمل کرنے کا حکم دیا گیا پھر انہوں نے اس پر عمل نہیں کیا ان کی مثال اس گدھے کی سی ہے جو بہت سی کتابیں ﻻدے ہو۔ اللہ کی باتوں کو جھٹلانے والوں کی بڑی بری مثال ہے اور اللہ (ایسے) ﻇالم قوم کو ہدایت نہیں دیتا

  6. قُلْ يَـٰٓأَيُّهَا ٱلَّذِينَ هَادُوٓاْ إِن زَعَمْتُمْ أَنَّكُمْ أَوْلِيَآءُ لِلَّهِ مِن دُونِ ٱلنَّاسِ فَتَمَنَّوُاْ ٱلْمَوْتَ إِن كُنتُمْ صَـٰدِقِينَ ۝٦

    کہہ دیجئے کہ اے یہودیو! اگر تمہارا دعویٰ ہے کہ تم اللہ کے دوست ہو دوسرے لوگوں کے سوا تو تم موت کی تمنا کرو اگر تم سچے ہو

  7. وَلَا يَتَمَنَّوْنَهُۥٓ أَبَدَۢا بِمَا قَدَّمَتْ أَيْدِيهِمْ‌ۚ وَٱللَّهُ عَلِيمُۢ بِٱلظَّـٰلِمِينَ ۝٧

    یہ کبھی بھی موت کی تمنا نہ کریں گے بوجہ ان اعمال کے جو اپنے آگے اپنے ہاتھوں بھیج رکھے ہیں اور اللہ ﻇالموں کو خوب جانتا ہے

  8. قُلْ إِنَّ ٱلْمَوْتَ ٱلَّذِى تَفِرُّونَ مِنْهُ فَإِنَّهُۥ مُلَـٰقِيكُمْ‌ۖ ثُمَّ تُرَدُّونَ إِلَىٰ عَـٰلِمِ ٱلْغَيْبِ وَٱلشَّهَـٰدَةِ فَيُنَبِّئُكُم بِمَا كُنتُمْ تَعْمَلُونَ ۝٨

    کہہ دیجئے! کہ جس موت سے تم بھاگتے پھرتے ہو وه تو تمہیں پہنچ کر رہے گی پھر تم سب چھپے کھلے کے جاننے والے (اللہ) کی طرف لوٹائے جاؤ گے اور وه تمہیں تمہارے کیے ہوئے تمام کام بتلا دے گا

  9. يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ إِذَا نُودِىَ لِلصَّلَوٲةِ مِن يَوْمِ ٱلْجُمُعَةِ فَٱسْعَوْاْ إِلَىٰ ذِكْرِ ٱللَّهِ وَذَرُواْ ٱلْبَيْعَ‌ۚ ذَٲلِكُمْ خَيْرٌ لَّكُمْ إِن كُنتُمْ تَعْلَمُونَ ۝٩

    اے وه لوگو جو ایمان ﻻئے ہو! جمعہ کے دن نماز کی اذان دی جائے تو تم اللہ کے ذکر کی طرف دوڑ پڑو اور خرید وفروخت چھوڑ دو۔ یہ تمہارے حق میں بہت ہی بہتر ہے اگر تم جانتے ہو

  10. فَإِذَا قُضِيَتِ ٱلصَّلَوٲةُ فَٱنتَشِرُواْ فِى ٱلْأَرْضِ وَٱبْتَغُواْ مِن فَضْلِ ٱللَّهِ وَٱذْكُرُواْ ٱللَّهَ كَثِيرًا لَّعَلَّكُمْ تُفْلِحُونَ ۝١٠

    پھر جب نماز ہو چکے تو زمین میں پھیل جاؤ اور اللہ کا فضل تلاش کرو اور بکثرت اللہ کا ذکر کیا کرو تاکہ تم فلاح پالو

  11. وَإِذَا رَأَوْاْ تِجَـٰرَةً أَوْ لَهْوًا ٱنفَضُّوٓاْ إِلَيْهَا وَتَرَكُوكَ قَآئِمًا‌ۚ قُلْ مَا عِندَ ٱللَّهِ خَيْرٌ مِّنَ ٱللَّهْوِ وَمِنَ ٱلتِّجَـٰرَةِ‌ۚ وَٱللَّهُ خَيْرُ ٱلرَّٲزِقِينَ ۝١١

    اور جب کوئی سودا بکتا دیکھیں یا کوئی تماشا نظر آجائے تو اس کی طرف دوڑ جاتے ہیں اور آپ کو کھڑا ہی چھوڑ دیتے ہیں۔ آپ کہہ دیجئے کہ اللہ کے پاس جو ہے وه کھیل اور تجارت سے بہتر ہے۔ اور اللہ تعالیٰ بہترین روزی رساں ہے
