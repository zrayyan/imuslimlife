بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ

  1. قَدْ أَفْلَحَ ٱلْمُؤْمِنُونَ ۝١

    یقیناً ایمان والوں نے فلاح حاصل کرلی

  2. ٱلَّذِينَ هُمْ فِى صَلَاتِهِمْ خَـٰشِعُونَ ۝٢

    جو اپنی نماز میں خشوع کرتے ہیں

  3. وَٱلَّذِينَ هُمْ عَنِ ٱللَّغْوِ مُعْرِضُونَ ۝٣

    جو لغویات سے منھ موڑ لیتے ہیں

  4. وَٱلَّذِينَ هُمْ لِلزَّكَوٲةِ فَـٰعِلُونَ ۝٤

    جو زکوٰة ادا کرنے والے ہیں

  5. وَٱلَّذِينَ هُمْ لِفُرُوجِهِمْ حَـٰفِظُونَ ۝٥

    جو اپنی شرمگاہوں کی حفاﻇت کرنے والے ہیں

  6. إِلَّا عَلَىٰٓ أَزْوَٲجِهِمْ أَوْ مَا مَلَكَتْ أَيْمَـٰنُهُمْ فَإِنَّهُمْ غَيْرُ مَلُومِينَ ۝٦

    بجز اپنی بیویوں اور ملکیت کی لونڈیوں کے یقیناً یہ ملامتیوں میں سے نہیں ہیں

  7. فَمَنِ ٱبْتَغَىٰ وَرَآءَ ذَٲلِكَ فَأُوْلَـٰٓئِكَ هُمُ ٱلْعَادُونَ ۝٧

    جو اس کے سوا کچھ اور چاہیں وہی حد سے تجاوز کرجانے والے ہیں

  8. وَٱلَّذِينَ هُمْ لِأَمَـٰنَـٰتِهِمْ وَعَهْدِهِمْ رَٲعُونَ ۝٨

    جو اپنی امانتوں اور وعدے کی حفاﻇت کرنے والے ہیں

  9. وَٱلَّذِينَ هُمْ عَلَىٰ صَلَوَٲتِهِمْ يُحَافِظُونَ ۝٩

    جو اپنی نمازوں کی نگہبانی کرتے ہیں

  10. أُوْلَـٰٓئِكَ هُمُ ٱلْوَٲرِثُونَ ۝١٠

    یہی وارث ہیں

  11. ٱلَّذِينَ يَرِثُونَ ٱلْفِرْدَوْسَ هُمْ فِيهَا خَـٰلِدُونَ ۝١١

    جو فردوس کے وارث ہوں گے جہاں وه ہمیشہ رہیں گے

  12. وَلَقَدْ خَلَقْنَا ٱلْإِنسَـٰنَ مِن سُلَـٰلَةٍ مِّن طِينٍ ۝١٢

    یقیناً ہم نے انسان کو مٹی کے جوہر سے پیدا کیا

  13. ثُمَّ جَعَلْنَـٰهُ نُطْفَةً فِى قَرَارٍ مَّكِينٍ ۝١٣

    پھر اسے نطفہ بنا کر محفوظ جگہ میں قرار دے دیا

  14. ثُمَّ خَلَقْنَا ٱلنُّطْفَةَ عَلَقَةً فَخَلَقْنَا ٱلْعَلَقَةَ مُضْغَةً فَخَلَقْنَا ٱلْمُضْغَةَ عِظَـٰمًا فَكَسَوْنَا ٱلْعِظَـٰمَ لَحْمًا ثُمَّ أَنشَأْنَـٰهُ خَلْقًا ءَاخَرَ‌ۚ فَتَبَارَكَ ٱللَّهُ أَحْسَنُ ٱلْخَـٰلِقِينَ ۝١٤

    پھر نطفہ کو ہم نے جما ہوا خون بنا دیا، پھر اس خون کے لوتھڑے کو گوشت کا ٹکڑا کردیا۔ پھر گوشت کے ٹکڑے کو ہڈیاں بنا دیں، پھر ہڈیوں کو ہم نے گوشت پہنا دیا، پھر دوسری بناوٹ میں اس کو پیدا کردیا۔ برکتوں واﻻ ہے وه اللہ جو سب سے بہترین پیدا کرنے واﻻ ہے

  15. ثُمَّ إِنَّكُم بَعْدَ ذَٲلِكَ لَمَيِّتُونَ ۝١٥

    اس کے بعد پھر تم سب یقیناً مر جانے والے ہو

  16. ثُمَّ إِنَّكُمْ يَوْمَ ٱلْقِيَـٰمَةِ تُبْعَثُونَ ۝١٦

    پھر قیامت کے دن بلا شبہ تم سب اٹھائے جاؤ گے

  17. وَلَقَدْ خَلَقْنَا فَوْقَكُمْ سَبْعَ طَرَآئِقَ وَمَا كُنَّا عَنِ ٱلْخَلْقِ غَـٰفِلِينَ ۝١٧

    ہم نے تمہارے اوپر سات آسمان بنائے ہیں اور ہم مخلوقات سے غافل نہیں ہیں

  18. وَأَنزَلْنَا مِنَ ٱلسَّمَآءِ مَآءَۢ بِقَدَرٍ فَأَسْكَنَّـٰهُ فِى ٱلْأَرْضِ‌ۖ وَإِنَّا عَلَىٰ ذَهَابِۭ بِهِۦ لَقَـٰدِرُونَ ۝١٨

    ہم ایک صحیح انداز سے آسمان سے پانی برساتے ہیں، پھر اسے زمین میں ٹھہرا دیتے ہیں، اور ہم اس کے لے جانے پر یقیناً قادر ہیں

  19. فَأَنشَأْنَا لَكُم بِهِۦ جَنَّـٰتٍ مِّن نَّخِيلٍ وَأَعْنَـٰبٍ لَّكُمْ فِيهَا فَوَٲكِهُ كَثِيرَةٌ وَمِنْهَا تَأْكُلُونَ ۝١٩

    اسی پانی کے ذریعہ سے ہم تمہارے لئے کھجوروں اور انگوروں کے باغات پیدا کردیتے ہیں، کہ تمہارے لئے ان میں بہت سے میوے ہوتے ہیں ان ہی میں سے تم کھاتے بھی ہو

  20. وَشَجَرَةً تَخْرُجُ مِن طُورِ سَيْنَآءَ تَنۢبُتُ بِٱلدُّهْنِ وَصِبْغٍ لِّلْأَكِلِينَ ۝٢٠

    اور وه درخت جو طور سینا پہاڑ سے نکلتا ہے جو تیل نکالتا ہے اور کھانے والے کے لئے سالن ہے

  21. وَإِنَّ لَكُمْ فِى ٱلْأَنْعَـٰمِ لَعِبْرَةً‌ۖ نُّسْقِيكُم مِّمَّا فِى بُطُونِهَا وَلَكُمْ فِيهَا مَنَـٰفِعُ كَثِيرَةٌ وَمِنْهَا تَأْكُلُونَ ۝٢١

    تمہارے لئے چوپایوں میں بھی بڑی بھاری عبرت ہے۔ ان کے پیٹوں میں سے ہم تمہیں دودھ پلاتے ہیں اور بھی بہت سے نفع تمہارے لئے ان میں ہیں ان میں سے بعض بعض کو تم کھاتے بھی ہو

  22. وَعَلَيْهَا وَعَلَى ٱلْفُلْكِ تُحْمَلُونَ ۝٢٢

    اور ان پر اور کشتیوں پر تم سوار کرائے جاتے ہو

  23. وَلَقَدْ أَرْسَلْنَا نُوحًا إِلَىٰ قَوْمِهِۦ فَقَالَ يَـٰقَوْمِ ٱعْبُدُواْ ٱللَّهَ مَا لَكُم مِّنْ إِلَـٰهٍ غَيْرُهُۥٓ‌ۖ أَفَلَا تَتَّقُونَ ۝٢٣

    یقیناً ہم نے نوح (علیہ السلام) کو اس کی قوم کی طرف رسول بنا کر بھیجا، اس نے کہا کہ اے میری قوم کے لوگو! اللہ کی عبادت کرو اور اس کے سوا تمہارا کوئی معبود نہیں، کیا تم (اس سے) نہیں ڈرتے

  24. فَقَالَ ٱلْمَلَؤُاْ ٱلَّذِينَ كَفَرُواْ مِن قَوْمِهِۦ مَا هَـٰذَآ إِلَّا بَشَرٌ مِّثْلُكُمْ يُرِيدُ أَن يَتَفَضَّلَ عَلَيْكُمْ وَلَوْ شَآءَ ٱللَّهُ لَأَنزَلَ مَلَـٰٓئِكَةً مَّا سَمِعْنَا بِهَـٰذَا فِىٓ ءَابَآئِنَا ٱلْأَوَّلِينَ ۝٢٤

    اس کی قوم کے کافر سرداروں نے صاف کہہ دیا کہ یہ تو تم جیسا ہی انسان ہے، یہ تم پر فضیلت اور بڑائی حاصل کرنا چاہتا ہے۔ اگر اللہ ہی کو منظور ہوتا تو کسی فرشتے کو اتارتا، ہم نے تو اسے اپنے اگلے باپ دادوں کے زمانے میں سنا ہی نہیں

  25. إِنْ هُوَ إِلَّا رَجُلُۢ بِهِۦ جِنَّةٌ فَتَرَبَّصُواْ بِهِۦ حَتَّىٰ حِينٍ ۝٢٥

    یقیناً اس شخص کو جنون ہے، پس تم اسے ایک وقت مقرر تک ڈھیل دو

  26. قَالَ رَبِّ ٱنصُرْنِى بِمَا كَذَّبُونِ ۝٢٦

    نوح (علیہ السلام) نے دعا کی اے میرے رب! ان کے جھٹلانے پر تو میری مدد کر

  27. فَأَوْحَيْنَآ إِلَيْهِ أَنِ ٱصْنَعِ ٱلْفُلْكَ بِأَعْيُنِنَا وَوَحْيِنَا فَإِذَا جَآءَ أَمْرُنَا وَفَارَ ٱلتَّنُّورُ‌ۙ فَٱسْلُكْ فِيهَا مِن كُلٍّ زَوْجَيْنِ ٱثْنَيْنِ وَأَهْلَكَ إِلَّا مَن سَبَقَ عَلَيْهِ ٱلْقَوْلُ مِنْهُمْ‌ۖ وَلَا تُخَـٰطِبْنِى فِى ٱلَّذِينَ ظَلَمُوٓ[s[اْ‌ۖ] إِنَّهُم مُّغْرَقُونَ ۝٢٧

    تو ہم نے ان کی طرف وحی بھیجی کہ تو ہماری آنکھوں کے سامنے ہماری وحی کے مطابق ایک کشتی بنا۔ جب ہمارا حکم آجائے اور تنور ابل پڑے تو، تو ہر قسم کا ایک ایک جوڑا اس میں رکھ لے اور اپنے اہل کو بھی، مگر ان میں سے جن کی بابت ہماری بات پہلے گزر چکی ہے۔ خبردار جن لوگوں نے ﻇلم کیا ہے ان کے بارے میں مجھ سے کچھ کلام نہ کرنا وه تو سب ڈبوئے جائیں گے

  28. فَإِذَا ٱسْتَوَيْتَ أَنتَ وَمَن مَّعَكَ عَلَى ٱلْفُلْكِ فَقُلِ ٱلْحَمْدُ لِلَّهِ ٱلَّذِى نَجَّـٰنَا مِنَ ٱلْقَوْمِ ٱلظَّـٰلِمِينَ ۝٢٨

    جب تو اور تیرے ساتھی کشتی پر باطمینان بیٹھ جاؤ تو کہنا کہ سب تعریف اللہ کے لئے ہی ہے جس نے ہمیں ﻇالم لوگوں سے نجات عطا فرمائی

  29. وَقُل رَّبِّ أَنزِلْنِى مُنزَلاً مُّبَارَكًا وَأَنتَ خَيْرُ ٱلْمُنزِلِينَ ۝٢٩

    اور کہنا کہ اے میرے رب! مجھے بابرکت اتارنا اتار اور تو ہی بہتر ہے اتارنے والوں میں

  30. إِنَّ فِى ذَٲلِكَ لَأَيَـٰتٍ وَإِن كُنَّا لَمُبْتَلِينَ ۝٣٠

    یقیناً اس میں بڑی بڑی نشانیاں ہیں اور ہم بےشک آزمائش کرنے والے ہیں

  31. ثُمَّ أَنشَأْنَا مِنۢ بَعْدِهِمْ قَرْنًا ءَاخَرِينَ ۝٣١

    ان کے بعد ہم نے اور بھی امت پیدا کی

  32. فَأَرْسَلْنَا فِيهِمْ رَسُولاً مِّنْهُمْ أَنِ ٱعْبُدُواْ ٱللَّهَ مَا لَكُم مِّنْ إِلَـٰهٍ غَيْرُهُۥٓ‌ۖ أَفَلَا تَتَّقُونَ ۝٣٢

    پھر ان میں خود ان میں سے (ہی) رسول بھی بھیجا کہ تم سب اللہ کی عبادت کرو اس کے سوا تمہارا کوئی معبود نہیں، تم کیوں نہیں ڈرتے؟

  33. وَقَالَ ٱلْمَلَأُ مِن قَوْمِهِ ٱلَّذِينَ كَفَرُواْ وَكَذَّبُواْ بِلِقَآءِ ٱلْأَخِرَةِ وَأَتْرَفْنَـٰهُمْ فِى ٱلْحَيَوٲةِ ٱلدُّنْيَا مَا هَـٰذَآ إِلَّا بَشَرٌ مِّثْلُكُمْ يَأْكُلُ مِمَّا تَأْكُلُونَ مِنْهُ وَيَشْرَبُ مِمَّا تَشْرَبُونَ ۝٣٣

    اور سرداران قوم نے جواب دیا، جو کفر کرتے تھے اور آخرت کی ملاقات کو جھٹلاتے تھے اور ہم نے انہیں دنیوی زندگی میں خوشحال کر رکھا تھا، کہ یہ تو تم جیسا ہی انسان ہے، تمہاری ہی خوراک یہ بھی کھاتا ہے اور تمہارے پینے کا پانی ہی یہ پیتا ہے

  34. وَلَئِنْ أَطَعْتُم بَشَرًا مِّثْلَكُمْ إِنَّكُمْ إِذًا لَّخَـٰسِرُونَ ۝٣٤

    اگر تم نے اپنے جیسے ہی انسان کی تابعداری کر لی ہے تو بےشک تم سخت خسارے والے ہو

  35. أَيَعِدُكُمْ أَنَّكُمْ إِذَا مِتُّمْ وَكُنتُمْ تُرَابًا وَعِظَـٰمًا أَنَّكُم مُّخْرَجُونَ ۝٣٥

    کیا یہ تمہیں اس بات کا وعده کرتا ہے کہ جب تم مرکر صرف خاک اور ہڈی ره جاؤ گے تو تم پھر زنده کئے جاؤ گے

  36. ۞ هَيْهَاتَ هَيْهَاتَ لِمَا تُوعَدُونَ ۝٣٦

    نہیں نہیں دور اور بہت دور ہے وه جس کا تم وعده دیئے جاتے ہو

  37. إِنْ هِىَ إِلَّا حَيَاتُنَا ٱلدُّنْيَا نَمُوتُ وَنَحْيَا وَمَا نَحْنُ بِمَبْعُوثِينَ ۝٣٧

    (زندگی) تو صرف دنیا کی زندگی ہے ہم مرتے جیتے رہتے ہیں اور یہ نہیں کہ ہم پھر اٹھائے جائیں گے

  38. إِنْ هُوَ إِلَّا رَجُلٌ ٱفْتَرَىٰ عَلَى ٱللَّهِ كَذِبًا وَمَا نَحْنُ لَهُۥ بِمُؤْمِنِينَ ۝٣٨

    یہ تو بس ایسا شخص ہے جس نے اللہ پر جھوٹ (بہتان) باندھ لیا ہے، ہم تو اس پر ایمان ﻻنے والے نہیں ہیں

  39. قَالَ رَبِّ ٱنصُرْنِى بِمَا كَذَّبُونِ ۝٣٩

    نبی نے دعا کی کہ پروردگار! ان کے جھٹلانے پر تو میری مدد کر

  40. قَالَ عَمَّا قَلِيلٍ لَّيُصْبِحُنَّ نَـٰدِمِينَ ۝٤٠

    جواب ملا کہ یہ تو بہت ہی جلد اپنے کئے پر پچھتانے لگیں گے

  41. فَأَخَذَتْهُمُ ٱلصَّيْحَةُ بِٱلْحَقِّ فَجَعَلْنَـٰهُمْ غُثَآءً‌ۚ فَبُعْدًا لِّلْقَوْمِ ٱلظَّـٰلِمِينَ ۝٤١

    بالﺂخر عدل کے تقاضے کے مطابق چیﺦ نے پکڑ لیا اور ہم نے انہیں کوڑا کرکٹ کر ڈاﻻ، پس ﻇالموں کے لئے دوری ہو

  42. ثُمَّ أَنشَأْنَا مِنۢ بَعْدِهِمْ قُرُونًا ءَاخَرِينَ ۝٤٢

    ان کے بعد ہم نے اور بھی بہت سی امتیں پیدا کیں

  43. مَا تَسْبِقُ مِنْ أُمَّةٍ أَجَلَهَا وَمَا يَسْتَــْٔخِرُونَ ۝٤٣

    نہ تو کوئی امت اپنے وقت مقرره سے آگے بڑھی اور نہ پیچھے رہی

  44. ثُمَّ أَرْسَلْنَا رُسُلَنَا تَتْرَا‌ۖ كُلَّ مَا جَآءَ أُمَّةً رَّسُولُهَا كَذَّبُوهُ‌ۚ فَأَتْبَعْنَا بَعْضَهُم بَعْضًا وَجَعَلْنَـٰهُمْ أَحَادِيثَ‌ۚ فَبُعْدًا لِّقَوْمٍ لَّا يُؤْمِنُونَ ۝٤٤

    پھر ہم نے لگاتار رسول بھیجے، جب جب اس امت کے پاس اس کا رسول آیا اس نے جھٹلایا، پس ہم نے ایک دوسرے کے پیچھے لگا دیا اور انہیں افسانہ بنا دیا۔ ان لوگوں کو دوری ہے جو ایمان قبول نہیں کرتے

  45. ثُمَّ أَرْسَلْنَا مُوسَىٰ وَأَخَاهُ هَـٰرُونَ بِـَٔـايَـٰتِنَا وَسُلْطَـٰنٍ مُّبِينٍ ۝٤٥

    پھر ہم نے موسیٰ (علیہ السلام) کو اور ان کے بھائی ہارون (علیہ السلام) کو اپنی آیتوں اور کھلی دلیل کے ساتھ بھیجا

  46. إِلَىٰ فِرْعَوْنَ وَمَلَإِيْهِۦ فَٱسْتَكْبَرُواْ وَكَانُواْ قَوْمًا عَالِينَ ۝٤٦

    فرعون اور اس کے لشکروں کی طرف، پس انہوں نے تکبر کیا اور تھے ہی وه سرکش لوگ

  47. فَقَالُوٓاْ أَنُؤْمِنُ لِبَشَرَيْنِ مِثْلِنَا وَقَوْمُهُمَا لَنَا عَـٰبِدُونَ ۝٤٧

    کہنے لگے کہ کیا ہم اپنے جیسے دو شخصوں پر ایمان ﻻئیں؟ حاﻻنکہ خود ان کی قوم (بھی) ہمارے ماتحت ہے

  48. فَكَذَّبُوهُمَا فَكَانُواْ مِنَ ٱلْمُهْلَكِينَ ۝٤٨

    پس انہوں نے ان دونوں کو جھٹلایا آخر وه بھی ہلاک شده لوگوں میں مل گئے

  49. وَلَقَدْ ءَاتَيْنَا مُوسَى ٱلْكِتَـٰبَ لَعَلَّهُمْ يَهْتَدُونَ ۝٤٩

    ہم نے تو موسیٰ (علیہ السلام) کو کتاب (بھی) دی کہ لوگ راه راست پر آجائیں

  50. وَجَعَلْنَا ٱبْنَ مَرْيَمَ وَأُمَّهُۥٓ ءَايَةً وَءَاوَيْنَـٰهُمَآ إِلَىٰ رَبْوَةٍ ذَاتِ قَرَارٍ وَمَعِينٍ ۝٥٠

    ہم نے ابن مریم اور اس کی والده کو ایک نشانی بنایا اور ان دونوں کو بلند صاف قرار والی اور جاری پانی والی جگہ میں پناه دی

  51. يَـٰٓأَيُّهَا ٱلرُّسُلُ كُلُواْ مِنَ ٱلطَّيِّبَـٰتِ وَٱعْمَلُواْ صَـٰلِحًا‌ۖ إِنِّى بِمَا تَعْمَلُونَ عَلِيمٌ ۝٥١

    اے پیغمبرو! حلال چیزیں کھاؤ اور نیک عمل کرو تم جو کچھ کر رہے ہو اس سے میں بخوبی واقف ہوں

  52. وَإِنَّ هَـٰذِهِۦٓ أُمَّتُكُمْ أُمَّةً وَٲحِدَةً وَأَنَا۟ رَبُّكُمْ فَٱتَّقُونِ ۝٥٢

    یقیناً تمہارا یہ دین ایک ہی دین ہے اور میں ہی تم سب کا رب ہوں، پس تم مجھ سے ڈرتے رہو

  53. فَتَقَطَّعُوٓاْ أَمْرَهُم بَيْنَهُمْ زُبُرًا‌ۖ كُلُّ حِزْبِۭ بِمَا لَدَيْهِمْ فَرِحُونَ ۝٥٣

    پھر انہوں نے خود (ہی) اپنے امر (دین) کے آپس میں ٹکڑے ٹکڑے کر لیئے، ہر گروه جو کچھ اس کے پاس ہے اسی پر اترا رہا ہے

  54. فَذَرْهُمْ فِى غَمْرَتِهِمْ حَتَّىٰ حِينٍ ۝٥٤

    پس آپ (بھی) انہیں ان کی غفلت میں ہی کچھ مدت پڑا رہنے دیں

  55. أَيَحْسَبُونَ أَنَّمَا نُمِدُّهُم بِهِۦ مِن مَّالٍ وَبَنِينَ ۝٥٥

    کیا یہ (یوں) سمجھ بیٹھے ہیں؟ کہ ہم جو بھی ان کے مال و اوﻻد بڑھا رہے ہیں

  56. نُسَارِعُ لَهُمْ فِى ٱلْخَيْرَٲتِ‌ۚ بَل لَّا يَشْعُرُونَ ۝٥٦

    وه ان کے لئے بھلائیوں میں جلدی کر رہے ہیں (نہیں نہیں) بلکہ یہ سمجھتے ہی نہیں

  57. إِنَّ ٱلَّذِينَ هُم مِّنْ خَشْيَةِ رَبِّهِم مُّشْفِقُونَ ۝٥٧

    یقیناً جو لوگ اپنے رب کی ہیبت سے ڈرتے ہیں

  58. وَٱلَّذِينَ هُم بِـَٔـايَـٰتِ رَبِّهِمْ يُؤْمِنُونَ ۝٥٨

    اور جو اپنے رب کی آیتوں پر ایمان رکھتے ہیں

  59. وَٱلَّذِينَ هُم بِرَبِّهِمْ لَا يُشْرِكُونَ ۝٥٩

    اور جو اپنے رب کے ساتھ کسی کو شریک نہیں کرتے

  60. وَٱلَّذِينَ يُؤْتُونَ مَآ ءَاتَواْ وَّقُلُوبُهُمْ وَجِلَةٌ أَنَّهُمْ إِلَىٰ رَبِّهِمْ رَٲجِعُونَ ۝٦٠

    اور جو لوگ دیتے ہیں جو کچھ دیتے ہیں اور ان کے دل کپکپاتے ہیں کہ وه اپنے رب کی طرف لوٹنے والے ہیں

  61. أُوْلَـٰٓئِكَ يُسَـٰرِعُونَ فِى ٱلْخَيْرَٲتِ وَهُمْ لَهَا سَـٰبِقُونَ ۝٦١

    یہی ہیں جو جلدی جلدی بھلائیاں حاصل کر رہے ہیں اور یہی ہیں جو ان کی طرف دوڑ جانے والے ہیں

  62. وَلَا نُكَلِّفُ نَفْسًا إِلَّا وُسْعَهَا‌ۖ وَلَدَيْنَا كِتَـٰبٌ يَنطِقُ بِٱلْحَقِّ‌ۚ وَهُمْ لَا يُظْلَمُونَ ۝٦٢

    ہم کسی نفس کو اس کی طاقت سے زیاده تکلیف نہیں دیتے، اور ہمارے پاس ایسی کتاب ہے جو حق کے ساتھ بولتی ہے، ان کے اوپر کچھ ﻇلم نہ کیا جائے گا

  63. بَلْ قُلُوبُهُمْ فِى غَمْرَةٍ مِّنْ هَـٰذَا وَلَهُمْ أَعْمَـٰلٌ مِّن دُونِ ذَٲلِكَ هُمْ لَهَا عَـٰمِلُونَ ۝٦٣

    بلکہ ان کے دل اس طرف سے غفلت میں ہیں اور ان کے لئے اس کے سوا بھی بہت سے اعمال ہیں جنہیں وه کرنے والے ہیں

  64. حَتَّىٰٓ إِذَآ أَخَذْنَا مُتْرَفِيهِم بِٱلْعَذَابِ إِذَا هُمْ يَجْــَٔرُونَ ۝٦٤

    یہاں تک کہ جب ہم نے ان کے آسوده حال لوگوں کو عذاب میں پکڑ لیا تو وه بلبلانے لگے

  65. لَا تَجْــَٔرُواْ ٱلْيَوْمَ‌ۖ إِنَّكُم مِّنَّا لَا تُنصَرُونَ ۝٦٥

    آج مت بلبلاؤ یقیناً تم ہمارے مقابلہ پر مدد نہ کئے جاؤ گے۔

  66. قَدْ كَانَتْ ءَايَـٰتِى تُتْلَىٰ عَلَيْكُمْ فَكُنتُمْ عَلَىٰٓ أَعْقَـٰبِكُمْ تَنكِصُونَ ۝٦٦

    میری آیتیں تو تمہارے سامنے پڑھی جاتی تھیں پھر بھی تم اپنی ایڑیوں کے بل الٹے بھاگتے تھے

  67. مُسْتَكْبِرِينَ بِهِۦ سَـٰمِرًا تَهْجُرُونَ ۝٦٧

    اکڑتے اینٹھتے افسانہ گوئی کرتے اسے چھوڑ دیتے تھے

  68. أَفَلَمْ يَدَّبَّرُواْ ٱلْقَوْلَ أَمْ جَآءَهُم مَّا لَمْ يَأْتِ ءَابَآءَهُمُ ٱلْأَوَّلِينَ ۝٦٨

    کیا انہوں نے اس بات میں غور وفکر ہی نہیں کیا؟ بلکہ ان کے پاس وه آیا جو ان کے اگلے باپ دادوں کے پاس نہیں آیا تھا؟

  69. أَمْ لَمْ يَعْرِفُواْ رَسُولَهُمْ فَهُمْ لَهُۥ مُنكِرُونَ ۝٦٩

    یا انہوں نے اپنے پیغمبر کو پہچانا نہیں کہ اس کے منکر ہو رہے ہیں؟

  70. أَمْ يَقُولُونَ بِهِۦ جِنَّةُۢ‌ۚ بَلْ جَآءَهُم بِٱلْحَقِّ وَأَكْثَرُهُمْ لِلْحَقِّ كَـٰرِهُونَ ۝٧٠

    یا یہ کہتے ہیں کہ اسے جنون ہے؟ بلکہ وه تو ان کے پاس حق ﻻیا ہے۔ ہاں ان میں اکثر حق سے چڑنے والے ہیں

  71. وَلَوِ ٱتَّبَعَ ٱلْحَقُّ أَهْوَآءَهُمْ لَفَسَدَتِ ٱلسَّمَـٰوَٲتُ وَٱلْأَرْضُ وَمَن فِيهِنَّ‌ۚ بَلْ أَتَيْنَـٰهُم بِذِكْرِهِمْ فَهُمْ عَن ذِكْرِهِم مُّعْرِضُونَ ۝٧١

    اگر حق ہی ان کی خواہشوں کا پیرو ہوجائے تو زمین وآسمان اور ان کے درمیان کی ہر چیز درہم برہم ہو جائے۔ حق تو یہ ہے کہ ہم نے انہیں ان کی نصیحت پہنچا دی ہے لیکن وه اپنی نصیحت سے منھ موڑنے والے ہیں

  72. أَمْ تَسْــَٔلُهُمْ خَرْجًا فَخَرَاجُ رَبِّكَ خَيْرٌ‌ۖ وَهُوَ خَيْرُ ٱلرَّٲزِقِينَ ۝٧٢

    کیا آپ ان سے کوئی اجرت چاہتے ہیں؟ یاد رکھیئے کہ آپ کے رب کی اجرت بہت ہی بہتر ہے اور وه سب سے بہتر روزی رساں ہے

  73. وَإِنَّكَ لَتَدْعُوهُمْ إِلَىٰ صِرَٲطٍ مُّسْتَقِيمٍ ۝٧٣

    یقیناً آپ تو انہیں راه راست کی طرف بلا رہے ہیں

  74. وَإِنَّ ٱلَّذِينَ لَا يُؤْمِنُونَ بِٱلْأَخِرَةِ عَنِ ٱلصِّرَٲطِ لَنَـٰكِبُونَ ۝٧٤

    بےشک جو لوگ آخرت پر یقین نہیں رکھتے وه سیدھے راستے سے مڑ جانے والے ہیں

  75. ۞ وَلَوْ رَحِمْنَـٰهُمْ وَكَشَفْنَا مَا بِهِم مِّن ضُرٍّ لَّلَجُّواْ فِى طُغْيَـٰنِهِمْ يَعْمَهُونَ ۝٧٥

    اور اگر ہم ان پر رحم فرمائیں اور ان کی تکلیفیں دور کردیں تو یہ تو اپنی اپنی سرکشی میں جم کر اور بہکنے لگیں

  76. وَلَقَدْ أَخَذْنَـٰهُم بِٱلْعَذَابِ فَمَا ٱسْتَكَانُواْ لِرَبِّهِمْ وَمَا يَتَضَرَّعُونَ ۝٧٦

    اور ہم نے انہیں عذاب میں بھی پکڑا تاہم یہ لوگ نہ تو اپنے پروردگار کے سامنے جھکے اور نہ ہی عاجزی اختیار کی

  77. حَتَّىٰٓ إِذَا فَتَحْنَا عَلَيْهِم بَابًا ذَا عَذَابٍ شَدِيدٍ إِذَا هُمْ فِيهِ مُبْلِسُونَ ۝٧٧

    یہاں تک کہ جب ہم نے ان پر سخت عذاب کا دروازه کھول دیا تو اسی وقت فوراً مایوس ہوگئے

  78. وَهُوَ ٱلَّذِىٓ أَنشَأَ لَكُمُ ٱلسَّمْعَ وَٱلْأَبْصَـٰرَ وَٱلْأَفْــِٔدَةَ‌ۚ قَلِيلاً مَّا تَشْكُرُونَ ۝٧٨

    وه اللہ ہے جس نے تمہارے لئے کان اور آنکھیں اور دل پیدا کئے، مگر تم بہت (ہی) کم شکر کرتے ہو

  79. وَهُوَ ٱلَّذِى ذَرَأَكُمْ فِى ٱلْأَرْضِ وَإِلَيْهِ تُحْشَرُونَ ۝٧٩

    اور وہی ہے جس نے تمہیں پیدا کرکے زمین میں پھیلا دیا اور اسی کی طرف تم جمع کئے جاؤ گے

  80. وَهُوَ ٱلَّذِى يُحْىِۦ وَيُمِيتُ وَلَهُ ٱخْتِلَـٰفُ ٱلَّيْلِ وَٱلنَّهَارِ‌ۚ أَفَلَا تَعْقِلُونَ ۝٨٠

    اور یہ وہی ہے جو جلاتا اور مارتا ہے اور رات دن کے ردوبدل کا مختار بھی وہی ہے۔ کیا تم کو سمجھ بوجھ نہیں؟

  81. بَلْ قَالُواْ مِثْلَ مَا قَالَ ٱلْأَوَّلُونَ ۝٨١

    بلکہ ان لوگوں نے بھی ویسی ہی بات کہی جو اگلے کہتے چلے آئے

  82. قَالُوٓاْ أَءِذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَـٰمًا أَءِنَّا لَمَبْعُوثُونَ ۝٨٢

    کہ کیا جب ہم مر کر مٹی اور ہڈی ہوجائیں گے کیا پھر بھی ہم ضرور اٹھائے جائیں گے؟

  83. لَقَدْ وُعِدْنَا نَحْنُ وَءَابَآؤُنَا هَـٰذَا مِن قَبْلُ إِنْ هَـٰذَآ إِلَّآ أَسَـٰطِيرُ ٱلْأَوَّلِينَ ۝٨٣

    ہم سے اور ہمارے باپ دادوں سے پہلے ہی یہ وعده ہوتا چلا آیا ہے کچھ نہیں یہ تو صرف اگلے لوگوں کے افسانے ہیں

  84. قُل لِّمَنِ ٱلْأَرْضُ وَمَن فِيهَآ إِن كُنتُمْ تَعْلَمُونَ ۝٨٤

    پوچھیئے تو سہی کہ زمین اور اس کی کل چیزیں کس کی ہیں؟ بتلاؤ اگر جانتے ہو؟

  85. سَيَقُولُونَ لِلَّهِ‌ۚ قُلْ أَفَلَا تَذَكَّرُونَ ۝٨٥

    فوراً جواب دیں گے کہ اللہ کی، کہہ دیجیئے کہ پھر تم نصیحت کیوں نہیں حاصل کرتے

  86. قُلْ مَن رَّبُّ ٱلسَّمَـٰوَٲتِ ٱلسَّبْعِ وَرَبُّ ٱلْعَرْشِ ٱلْعَظِيمِ ۝٨٦

    دریافت کیجیئے کہ ساتوں آسمان کا اور بہت باعظمت عرش کا رب کون ہے؟

  87. سَيَقُولُونَ لِلَّهِ‌ۚ قُلْ أَفَلَا تَتَّقُونَ ۝٨٧

    وه لوگ جواب دیں گے کہ اللہ ہی ہے۔ کہہ دیجیئے کہ پھر تم کیوں نہیں ڈرتے؟

  88. قُلْ مَنۢ بِيَدِهِۦ مَلَكُوتُ كُلِّ شَىْءٍ وَهُوَ يُجِيرُ وَلَا يُجَارُ عَلَيْهِ إِن كُنتُمْ تَعْلَمُونَ ۝٨٨

    پوچھیئے کہ تمام چیزوں کا اختیار کس کے ہاتھ میں ہے؟ جو پناه دیتا ہے اور جس کے مقابلے میں کوئی پناه نہیں دیا جاتا، اگر تم جانتے ہو تو بتلا دو؟

  89. سَيَقُولُونَ لِلَّهِ‌ۚ قُلْ فَأَنَّىٰ تُسْحَرُونَ ۝٨٩

    یہی جواب دیں گے کہ اللہ ہی ہے، کہہ دیجیئے پھر تم کدھر سے جادو کر دیئے جاتے ہو؟

  90. بَلْ أَتَيْنَـٰهُم بِٱلْحَقِّ وَإِنَّهُمْ لَكَـٰذِبُونَ ۝٩٠

    حق یہ ہے کہ ہم نے انہیں حق پہنچا دیا ہے اور یہ بےشک جھوٹے ہیں

  91. مَا ٱتَّخَذَ ٱللَّهُ مِن وَلَدٍ وَمَا كَانَ مَعَهُۥ مِنْ إِلَـٰهٍ‌ۚ إِذًا لَّذَهَبَ كُلُّ إِلَـٰهِۭ بِمَا خَلَقَ وَلَعَلَا بَعْضُهُمْ عَلَىٰ بَعْضٍ‌ۚ سُبْحَـٰنَ ٱللَّهِ عَمَّا يَصِفُونَ ۝٩١

    نہ تو اللہ نے کسی کو بیٹا بنایا اور نہ اس کے ساتھ اور کوئی معبود ہے، ورنہ ہر معبود اپنی مخلوق کو لئے لئے پھرتا اور ہر ایک دوسرے پر چڑھ دوڑتا۔ جو اوصاف یہ بتلاتے ہیں ان سے اللہ پاک (اور بےنیاز) ہے

  92. عَـٰلِمِ ٱلْغَيْبِ وَٱلشَّهَـٰدَةِ فَتَعَـٰلَىٰ عَمَّا يُشْرِكُونَ ۝٩٢

    وه غائب حاضر کا جاننے واﻻ ہے اور جو شرک یہ کرتے ہیں اس سے باﻻتر ہے

  93. قُل رَّبِّ إِمَّا تُرِيَنِّى مَا يُوعَدُونَ ۝٩٣

    آپ دعا کریں کہ اے میرے پروردگار! اگر تو مجھے وه دکھائے جس کا وعده انہیں دیا جا رہا ہے

  94. رَبِّ فَلَا تَجْعَلْنِى فِى ٱلْقَوْمِ ٱلظَّـٰلِمِينَ ۝٩٤

    تو اے رب! تو مجھے ان ﻇالموں کے گروه میں نہ کرنا

  95. وَإِنَّا عَلَىٰٓ أَن نُّرِيَكَ مَا نَعِدُهُمْ لَقَـٰدِرُونَ ۝٩٥

    ہم جو کچھ وعدے انہیں دے رہے ہیں سب آپ کو دکھا دینے پر یقیناً قادر ہیں

  96. ٱدْفَعْ بِٱلَّتِى هِىَ أَحْسَنُ ٱلسَّيِّئَةَ‌ۚ نَحْنُ أَعْلَمُ بِمَا يَصِفُونَ ۝٩٦

    برائی کو اس طریقے سے دور کریں جو سراسر بھلائی واﻻ ہو، جو کچھ یہ بیان کرتے ہیں ہم بخوبی واقف ہیں

  97. وَقُل رَّبِّ أَعُوذُ بِكَ مِنْ هَمَزَٲتِ ٱلشَّيَـٰطِينِ ۝٩٧

    اور دعا کریں کہ اے میرے پروردگار! میں شیطانوں کے وسوسوں سے تیری پناه چاہتا ہوں

  98. وَأَعُوذُ بِكَ رَبِّ أَن يَحْضُرُونِ ۝٩٨

    اور اے رب! میں تیری پناه چاہتا ہوں کہ وه میرے پاس آجائیں

  99. حَتَّىٰٓ إِذَا جَآءَ أَحَدَهُمُ ٱلْمَوْتُ قَالَ رَبِّ ٱرْجِعُونِ ۝٩٩

    یہاں تک کہ جب ان میں کسی کو موت آنے لگتی ہے تو کہتا ہے اے میرے پروردگار! مجھے واپس لوٹا دے

  100. لَعَلِّىٓ أَعْمَلُ صَـٰلِحًا فِيمَا تَرَكْتُ‌ۚ كَلَّآ‌ۚ إِنَّهَا كَلِمَةٌ هُوَ قَآئِلُهَا‌ۖ وَمِن وَرَآئِهِم بَرْزَخٌ إِلَىٰ يَوْمِ يُبْعَثُونَ ۝١٠٠

    کہ اپنی چھوڑی ہوئی دنیا میں جا کر نیک اعمال کر لوں، ہرگز ایسا نہیں ہوگا، یہ تو صرف ایک قول ہے جس کا یہ قائل ہے، ان کے پس پشت تو ایک حجاب ہے، ان کے دوباره جی اٹھنے کے دن تک

  101. فَإِذَا نُفِخَ فِى ٱلصُّورِ فَلَآ أَنسَابَ بَيْنَهُمْ يَوْمَئِذٍ وَلَا يَتَسَآءَلُونَ ۝١٠١

    پس جب کہ صور پھونک دیا جائے گا اس دن نہ تو آپس کے رشتے دار ہی رہیں گے، نہ آپس کی پوچھ گچھ

  102. فَمَن ثَقُلَتْ مَوَٲزِينُهُۥ فَأُوْلَـٰٓئِكَ هُمُ ٱلْمُفْلِحُونَ ۝١٠٢

    جن کی ترازو کا پلہ بھاری ہوگیا وه تو نجات والے ہوگئے

  103. وَمَنْ خَفَّتْ مَوَٲزِينُهُۥ فَأُوْلَـٰٓئِكَ ٱلَّذِينَ خَسِرُوٓاْ أَنفُسَهُمْ فِى جَهَنَّمَ خَـٰلِدُونَ ۝١٠٣

    اور جن کے ترازو کا پلہ ہلکا ہوگیا یہ ہیں وه جنہوں نے اپنا نقصان آپ کر لیا جو ہمیشہ کے لئے جہنم واصل ہوئے

  104. تَلْفَحُ وُجُوهَهُمُ ٱلنَّارُ وَهُمْ فِيهَا كَـٰلِحُونَ ۝١٠٤

    ان کے چہروں کو آگ جھلستی رہے گی اور وه وہاں بدشکل بنے ہوئے ہوں گے

  105. أَلَمْ تَكُنْ ءَايَـٰتِى تُتْلَىٰ عَلَيْكُمْ فَكُنتُم بِهَا تُكَذِّبُونَ ۝١٠٥

    کیا میری آیتیں تمہارے سامنے تلاوت نہیں کی جاتی تھیں؟ پھر بھی تم انہیں جھٹلاتے تھے

  106. قَالُواْ رَبَّنَا غَلَبَتْ عَلَيْنَا شِقْوَتُنَا وَكُنَّا قَوْمًا ضَآلِّينَ ۝١٠٦

    کہیں گے کہ اے پروردگار! ہماری بدبختی ہم پر غالب آگئی (واقعی) ہم تھے ہی گمراه

  107. رَبَّنَآ أَخْرِجْنَا مِنْهَا فَإِنْ عُدْنَا فَإِنَّا ظَـٰلِمُونَ ۝١٠٧

    اے ہمارے پروردگار! ہمیں یہاں سے نجات دے اگر اب بھی ہم ایسا ہی کریں تو بےشک ہم ﻇالم ہیں

  108. قَالَ ٱخْسَــُٔواْ فِيهَا وَلَا تُكَلِّمُونِ ۝١٠٨

    اللہ تعالیٰ فرمائے گا پھٹکارے ہوئے یہیں پڑے رہو اور مجھ سے کلام نہ کرو

  109. إِنَّهُۥ كَانَ فَرِيقٌ مِّنْ عِبَادِى يَقُولُونَ رَبَّنَآ ءَامَنَّا فَٱغْفِرْ لَنَا وَٱرْحَمْنَا وَأَنتَ خَيْرُ ٱلرَّٲحِمِينَ ۝١٠٩

    میرے بندوں کی ایک جماعت تھی جو برابر یہی کہتی رہی کہ اے ہمارے پروردگار! ہم ایمان ﻻچکے ہیں تو ہمیں بخش اور ہم پر رحم فرما تو سب مہربانوں سے زیاده مہربان ہے

  110. فَٱتَّخَذْتُمُوهُمْ سِخْرِيًّا حَتَّىٰٓ أَنسَوْكُمْ ذِكْرِى وَكُنتُم مِّنْهُمْ تَضْحَكُونَ ۝١١٠

    (لیکن) تم انہیں مذاق میں ہی اڑاتے رہے یہاں تک کہ (اس مشغلے نے) تم کو میری یاد (بھی) بھلا دی اور تم ان سے مذاق ہی کرتے رہے

  111. إِنِّى جَزَيْتُهُمُ ٱلْيَوْمَ بِمَا صَبَرُوٓاْ أَنَّهُمْ هُمُ ٱلْفَآئِزُونَ ۝١١١

    میں نے آج انہیں ان کے اس صبر کا بدلہ دے دیا ہے کہ وه خاطر خواه اپنی مراد کو پہنچ چکے ہیں

  112. قَـٰلَ كَمْ لَبِثْتُمْ فِى ٱلْأَرْضِ عَدَدَ سِنِينَ ۝١١٢

    اللہ تعالیٰ دریافت فرمائے گا کہ تم زمین میں باعتبار برسوں کی گنتی کے کس قدر رہے؟

  113. قَالُواْ لَبِثْنَا يَوْمًا أَوْ بَعْضَ يَوْمٍ فَسْــَٔلِ ٱلْعَآدِّينَ ۝١١٣

    وه کہیں گے ایک دن یا ایک دن سے بھی کم، گنتی گننے والوں سے بھی پوچھ لیجیئے

  114. قَـٰلَ إِن لَّبِثْتُمْ إِلَّا قَلِيلاً‌ۖ لَّوْ أَنَّكُمْ كُنتُمْ تَعْلَمُونَ ۝١١٤

    اللہ تعالیٰ فرمائے گا فی الواقع تم وہاں بہت ہی کم رہے ہو اے کاش! تم اسے پہلے ہی سے جان لیتے؟

  115. أَفَحَسِبْتُمْ أَنَّمَا خَلَقْنَـٰكُمْ عَبَثًا وَأَنَّكُمْ إِلَيْنَا لَا تُرْجَعُونَ ۝١١٥

    کیا تم یہ گمان کئے ہوئے ہو کہ ہم نے تمہیں یوں ہی بیکار پیدا کیا ہے اور یہ کہ تم ہماری طرف لوٹائے ہی نہ جاؤ گے

  116. فَتَعَـٰلَى ٱللَّهُ ٱلْمَلِكُ ٱلْحَقُّ‌ۖ لَآ إِلَـٰهَ إِلَّا هُوَ رَبُّ ٱلْعَرْشِ ٱلْكَرِيمِ ۝١١٦

    اللہ تعالیٰ سچا بادشاه ہے وه بڑی بلندی واﻻ ہے۔ اس کے سوا کوئی معبود نہیں، وہی بزرگ عرش کا مالک ہے

  117. وَمَن يَدْعُ مَعَ ٱللَّهِ إِلَـٰهًا ءَاخَرَ لَا بُرْهَـٰنَ لَهُۥ بِهِۦ فَإِنَّمَا حِسَابُهُۥ عِندَ رَبِّهِۦٓ‌ۚ إِنَّهُۥ لَا يُفْلِحُ ٱلْكَـٰفِرُونَ ۝١١٧

    جو شخص اللہ کے ساتھ کسی دوسرے معبود کو پکارے جس کی کوئی دلیل اس کے پاس نہیں، پس اس کا حساب تو اس کے رب کے اوپر ہی ہے۔ بےشک کافر لوگ نجات سے محروم ہیں

  118. وَقُل رَّبِّ ٱغْفِرْ وَٱرْحَمْ وَأَنتَ خَيْرُ ٱلرَّٲحِمِينَ ۝١١٨

    اور کہو کہ اے میرے رب! تو بخش اور رحم کر اور تو سب مہربانوں سے بہتر مہربانی کرنے واﻻ ہے
