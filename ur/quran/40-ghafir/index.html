بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ

  1. حمٓ ۝١

    حٰم

  2. تَنزِيلُ ٱلْكِتَـٰبِ مِنَ ٱللَّهِ ٱلْعَزِيزِ ٱلْعَلِيمِ ۝٢

    اس کتاب کا نازل فرمانا اس اللہ کی طرف سے ہے جو غالب اور دانا ہے

  3. غَافِرِ ٱلذَّنۢبِ وَقَابِلِ ٱلتَّوْبِ شَدِيدِ ٱلْعِقَابِ ذِى ٱلطَّوْلِۖ لَآ إِلَـٰهَ إِلَّا هُوَۖ إِلَيْهِ ٱلْمَصِيرُ ۝٣

    گناه کا بخشنے واﻻ اور توبہ کا قبول فرمانے واﻻ سخت عذاب واﻻ انعام و قدرت واﻻ، جس کے سوا کوئی معبود نہیں۔ اسی کی طرف واپس لوٹنا ہے

  4. مَا يُجَـٰدِلُ فِىٓ ءَايَـٰتِ ٱللَّهِ إِلَّا ٱلَّذِينَ كَفَرُواْ فَلَا يَغْرُرْكَ تَقَلُّبُهُمْ فِى ٱلْبِلَـٰدِ ۝٤

    اللہ تعالیٰ کی آیتوں میں وہی لوگ جھگڑتے ہیں جو کافر ہیں پس ان لوگوں کا شہروں میں چلنا پھرنا آپ کو دھوکے میں نہ ڈالے

  5. كَذَّبَتْ قَبْلَهُمْ قَوْمُ نُوحٍ وَٱلْأَحْزَابُ مِنۢ بَعْدِهِمْۖ وَهَمَّتْ كُلُّ أُمَّةِۭ بِرَسُولِهِمْ لِيَأْخُذُوهُۖ وَجَـٰدَلُواْ بِٱلْبَـٰطِلِ لِيُدْحِضُواْ بِهِ ٱلْحَقَّ فَأَخَذْتُهُمْۖ فَكَيْفَ كَانَ عِقَابِ ۝٥

    قوم نوح نے اور ان کے بعد کے گروہوں نے بھی جھٹلایا تھا۔ اور ہر امت نے اپنے رسول کو گرفتار کر لینے کا اراده کیا اور باطل کے ذریعہ کج بحثیاں کیں، تاکہ ان سے حق کو بگاڑ دیں پس میں نے ان کو پکڑ لیا، سو میری طرف سے کیسی سزا ہوئی

  6. وَكَذَٲلِكَ حَقَّتْ كَلِمَتُ رَبِّكَ عَلَى ٱلَّذِينَ كَفَرُوٓاْ أَنَّهُمْ أَصْحَـٰبُ ٱلنَّارِ ۝٦

    اور اسی طرح آپ کے رب کا حکم کافروں پر ﺛابت ہو گیا کہ وه دوزخی ہیں

  7. ٱلَّذِينَ يَحْمِلُونَ ٱلْعَرْشَ وَمَنْ حَوْلَهُۥ يُسَبِّحُونَ بِحَمْدِ رَبِّهِمْ وَيُؤْمِنُونَ بِهِۦ وَيَسْتَغْفِرُونَ لِلَّذِينَ ءَامَنُواْ رَبَّنَا وَسِعْتَ كُلَّ شَىْءٍ رَّحْمَةً وَعِلْمًا فَٱغْفِرْ لِلَّذِينَ تَابُواْ وَٱتَّبَعُواْ سَبِيلَكَ وَقِهِمْ عَذَابَ ٱلْجَحِيمِ ۝٧

    عرش کے اٹھانے والے اور اسکے آس پاس کے (فرشتے) اپنے رب کی تسبیح حمد کے ساتھ ساتھ کرتے ہیں اور اس پر ایمان رکھتے ہیں اور ایمان والوں کے لیے استفغار کرتے ہیں، کہتے ہیں کہ اے ہمارے پروردگار! تو نے ہر چیز کو اپنی بخشش اور علم سے گھیر رکھا ہے، پس تو انہیں بخش دے جو توبہ کریں اور تیری راه کی پیروی کریں اور تو انہیں دوزخ کے عذاب سے بھی بچا لے

  8. رَبَّنَا وَأَدْخِلْهُمْ جَنَّـٰتِ عَدْنٍ ٱلَّتِى وَعَدتَّهُمْ وَمَن صَلَحَ مِنْ ءَابَآئِهِمْ وَأَزْوَٲجِهِمْ وَذُرِّيَّـٰتِهِمْۚ إِنَّكَ أَنتَ ٱلْعَزِيزُ ٱلْحَكِيمُ ۝٨

    اے ہمارے رب! تو انہیں ہمیشگی والی جنتوں میں لے جا جن کا تو نے ان سے وعده کیا ہے اور ان کے باپ دادوں اور بیویوں اور اوﻻد میں سے (بھی) ان (سب) کو جو نیک عمل ہیں۔ یقیناً تو تو غالب و باحکمت ہے

  9. وَقِهِمُ ٱلسَّيِّـَٔـاتِۚ وَمَن تَقِ ٱلسَّيِّـَٔـاتِ يَوْمَئِذٍ فَقَدْ رَحِمْتَهُۥۚ وَذَٲلِكَ هُوَ ٱلْفَوْزُ ٱلْعَظِيمُ ۝٩

    انہیں برائیوں سے بھی محفوظ رکھ، حق تو یہ ہے کہ اس دن تونے جسے برائیوں سے بچا لیا اس پر تو نے رحمت کر دی اور بہت بڑی کامیابی تو یہی ہے

  10. إِنَّ ٱلَّذِينَ كَفَرُواْ يُنَادَوْنَ لَمَقْتُ ٱللَّهِ أَكْبَرُ مِن مَّقْتِكُمْ أَنفُسَكُمْ إِذْ تُدْعَوْنَ إِلَى ٱلْإِيمَـٰنِ فَتَكْفُرُونَ ۝١٠

    بےشک جن لوگوں نے کفر کیا انہیں یہ آواز دی جائے گی کہ یقیناً اللہ کا تم پر غصہ ہونااس سے بہت زیاده ہے جو تم غصہ ہوتے تھے اپنے جی سے، جب تم ایمان کی طرف بلائے جاتے تھے پھر کفر کرنے لگتے تھے

  11. قَالُواْ رَبَّنَآ أَمَتَّنَا ٱثْنَتَيْنِ وَأَحْيَيْتَنَا ٱثْنَتَيْنِ فَٱعْتَرَفْنَا بِذُنُوبِنَا فَهَلْ إِلَىٰ خُرُوجٍ مِّن سَبِيلٍ ۝١١

    وه کہیں گے اے ہمارے پروردگار! تو نے ہمیں دوبار مارا اور دو بار ہی جلایا، اب ہم اپنے گناہوں کے اقراری ہیں، تو کیا اب کوئی راه نکلنے کی بھی ہے؟

  12. ذَٲلِكُم بِأَنَّهُۥٓ إِذَا دُعِىَ ٱللَّهُ وَحْدَهُۥ كَفَرْتُمْۖ وَإِن يُشْرَكْ بِهِۦ تُؤْمِنُواْۚ فَٱلْحُكْمُ لِلَّهِ ٱلْعَلِىِّ ٱلْكَبِيرِ ۝١٢

    یہ (عذاب) تمہیں اس لیے ہے کہ جب صرف اکیلے اللہ کا ذکر کیا جاتا تو تم انکار کر جاتے تھے اور اگر اس کے ساتھ کسی کو شریک کیا جاتا تھا تو تم مان لیتے تھے پس اب فیصلہ اللہ بلند و بزرگ ہی کا ہے

  13. هُوَ ٱلَّذِى يُرِيكُمْ ءَايَـٰتِهِۦ وَيُنَزِّلُ لَكُم مِّنَ ٱلسَّمَآءِ رِزْقًاۚ وَمَا يَتَذَكَّرُ إِلَّا مَن يُنِيبُ ۝١٣

    وہی ہے جو تمہیں اپنی نشانیاں دکھلاتا ہے اور تمہارے لیےآسمان سے روزی اتارتا ہے، نصیحت تو صرف وہی حاصل کرتے ہیں جو (اللہ کی طرف) رجوع کرتے ہیں

  14. فَٱدْعُواْ ٱللَّهَ مُخْلِصِينَ لَهُ ٱلدِّينَ وَلَوْ كَرِهَ ٱلْكَـٰفِرُونَ ۝١٤

    تم اللہ کو پکارتے رہو اس کے لیے دین کو خالص کر کے گو کافر برا مانیں

  15. رَفِيعُ ٱلدَّرَجَـٰتِ ذُو ٱلْعَرْشِ يُلْقِى ٱلرُّوحَ مِنْ أَمْرِهِۦ عَلَىٰ مَن يَشَآءُ مِنْ عِبَادِهِۦ لِيُنذِرَ يَوْمَ ٱلتَّلَاقِ ۝١٥

    بلند درجوں واﻻ عرش کامالک وه اپنے بندوں میں سے جس پر چاہتا ہے وحی نازل فرماتا ہے، تاکہ وه ملاقات کے دن سے ڈرائے

  16. يَوْمَ هُم بَـٰرِزُونَۖ لَا يَخْفَىٰ عَلَى ٱللَّهِ مِنْهُمْ شَىْءٌۚ لِّمَنِ ٱلْمُلْكُ ٱلْيَوْمَۖ لِلَّهِ ٱلْوَٲحِدِ ٱلْقَهَّارِ ۝١٦

    جس دن سب لوگ ﻇاہر ہو جائیں گے، ان کی کوئی چیز اللہ سے پوشیده نہ رہے گی۔ آج کس کی بادشاہی ہے؟ فقط اللہ واحد و قہار کی

  17. ٱلْيَوْمَ تُجْزَىٰ كُلُّ نَفْسِۭ بِمَا كَسَبَتْۚ لَا ظُلْمَ ٱلْيَوْمَۚ إِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ ۝١٧

    آج ہر نفس کو اس کی کمائی کا بدلہ دیا جائے گا۔ آج (کسی قسم کا) ﻇلم نہیں، یقیناً اللہ تعالیٰ بہت جلد حساب کرنے واﻻ ہے

  18. وَأَنذِرْهُمْ يَوْمَ ٱلْأَزِفَةِ إِذِ ٱلْقُلُوبُ لَدَى ٱلْحَنَاجِرِ كَـٰظِمِينَۚ مَا لِلظَّـٰلِمِينَ مِنْ حَمِيمٍ وَلَا شَفِيعٍ يُطَاعُ ۝١٨

    اور انہیں بہت ہی قریب آنے والی (قیامت سے) آگاه کر دیجئے، جب کہ دل حلق تک پہنچ جائیں گے اور سب خاموش ہوں گے، ﻇالموں کا نہ کوئی دلی دوست ہوگا نہ سفارشی، کہ جس کی بات مانی جائے گی

  19. يَعْلَمُ خَآئِنَةَ ٱلْأَعْيُنِ وَمَا تُخْفِى ٱلصُّدُورُ ۝١٩

    وه آنکھوں کی خیانت کو اور سینوں کی پوشیده باتوں کو (خوب) جانتا ہے

  20. وَٱللَّهُ يَقْضِى بِٱلْحَقِّۖ وَٱلَّذِينَ يَدْعُونَ مِن دُونِهِۦ لَا يَقْضُونَ بِشَىْءٍۗ إِنَّ ٱللَّهَ هُوَ ٱلسَّمِيعُ ٱلْبَصِيرُ ۝٢٠

    اور اللہ تعالیٰ ٹھیک ٹھیک فیصلہ کر دے گا اس کے سوا جنہیں یہ لوگ پکارتے ہیں وه کسی چیز کا بھی فیصلہ نہیں کرسکتے، بیشک اللہ تعالیٰ خوب سنتا خوب دیکھتا ہے

  21. ۞ أَوَلَمْ يَسِيرُواْ فِى ٱلْأَرْضِ فَيَنظُرُواْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلَّذِينَ كَانُواْ مِن قَبْلِهِمْۚ كَانُواْ هُمْ أَشَدَّ مِنْهُمْ قُوَّةً وَءَاثَارًا فِى ٱلْأَرْضِ فَأَخَذَهُمُ ٱللَّهُ بِذُنُوبِهِمْ وَمَا كَانَ لَهُم مِّنَ ٱللَّهِ مِن وَاقٍ ۝٢١

    کیا یہ لوگ زمین میں چلے پھرے نہیں کہ دیکھتے کہ جو لوگ ان سے پہلے تھے ان کا نتیجہ کیسا کچھ ہوا؟ وه باعتبار قوت و طاقت کے اور باعتبار زمین میں اپنی یادگاروں کے ان سے بہت زیاده تھے، پس اللہ نے انہیں ان کے گناہوں پر پکڑ لیا اور کوئی نہ ہوا جو انہیں اللہ کے عذاب سے بچا لیتا

  22. ذَٲلِكَ بِأَنَّهُمْ كَانَت تَّأْتِيهِمْ رُسُلُهُم بِٱلْبَيِّنَـٰتِ فَكَفَرُواْ فَأَخَذَهُمُ ٱللَّهُۚ إِنَّهُۥ قَوِىٌّ شَدِيدُ ٱلْعِقَابِ ۝٢٢

    یہ اس وجہ سے کہ ان کے پاس ان کے پیغمبر معجزے لے لے کرآتے تھے تو وه انکار کر دیتے تھے، پس اللہ انہیں پکڑ لیتا تھا۔ یقیناً وه طاقتور اور سخت عذاب واﻻ ہے

  23. وَلَقَدْ أَرْسَلْنَا مُوسَىٰ بِـَٔـايَـٰتِنَا وَسُلْطَـٰنٍ مُّبِينٍ ۝٢٣

    اور ہم نے موسیٰ (علیہ السلام) کو اپنی آیتوں اور کھلی دلیلوں کے ساتھ بھیجا

  24. إِلَىٰ فِرْعَوْنَ وَهَـٰمَـٰنَ وَقَـٰرُونَ فَقَالُواْ سَـٰحِرٌ كَذَّابٌ ۝٢٤

    فرعون ہامان اور قارون کی طرف تو انہوں نے کہا (یہ تو) جادوگر اور جھوٹا ہے

  25. فَلَمَّا جَآءَهُم بِٱلْحَقِّ مِنْ عِندِنَا قَالُواْ ٱقْتُلُوٓاْ أَبْنَآءَ ٱلَّذِينَ ءَامَنُواْ مَعَهُۥ وَٱسْتَحْيُواْ نِسَآءَهُمْۚ وَمَا كَيْدُ ٱلْكَـٰفِرِينَ إِلَّا فِى ضَلَـٰلٍ ۝٢٥

    پس جب ان کے پاس (موسیٰ علیہ السلام) ہماری طرف سے (دین) حق کولے کر آئے توانہوں نے کہا کہ اس کے ساتھ جوایمان والے ہیں ان کے لڑکوں کو تو مار ڈالو اور ان کی لڑکیوں کو زنده رکھو اور کافروں کی جو حیلہ سازی ہے وه غلطی میں ہی ہے

  26. وَقَالَ فِرْعَوْنُ ذَرُونِىٓ أَقْتُلْ مُوسَىٰ وَلْيَدْعُ رَبَّهُۥٓ‌ۖ إِنِّىٓ أَخَافُ أَن يُبَدِّلَ دِينَكُمْ أَوْ أَن يُظْهِرَ فِى ٱلْأَرْضِ ٱلْفَسَادَ ۝٢٦

    اور فرعون نے کہا مجھے چھوڑ دو کہ میں موسیٰ (علیہ السلام) کو مار ڈالوں اور اسے چاہئے کہ اپنے رب کو پکارے، مجھے تو ڈر ہے کہ یہ کہیں تمہارا دین نہ بدل ڈالے یا ملک میں کوئی (بہت بڑا) فساد برپا نہ کردے

  27. وَقَالَ مُوسَىٰٓ إِنِّى عُذْتُ بِرَبِّى وَرَبِّكُم مِّن كُلِّ مُتَكَبِّرٍ لَّا يُؤْمِنُ بِيَوْمِ ٱلْحِسَابِ ۝٢٧

    موسیٰ (علیہ السلام) نے کہا میں اپنے اور تمہارے رب کی پناه میں آتا ہوں ہر اس تکبر کرنے والے شخص (کی برائی) سے جو روز حساب پر ایمان نہیں رکھتا

  28. وَقَالَ رَجُلٌ مُّؤْمِنٌ مِّنْ ءَالِ فِرْعَوْنَ يَكْتُمُ إِيمَـٰنَهُۥٓ أَتَقْتُلُونَ رَجُلاً أَن يَقُولَ رَبِّىَ ٱللَّهُ وَقَدْ جَآءَكُم بِٱلْبَيِّنَـٰتِ مِن رَّبِّكُمْۖ وَإِن يَكُ كَـٰذِبًا فَعَلَيْهِ كَذِبُهُۥۖ وَإِن يَكُ صَادِقًا يُصِبْكُم بَعْضُ ٱلَّذِى يَعِدُكُمْۖ إِنَّ ٱللَّهَ لَا يَهْدِى مَنْ هُوَ مُسْرِفٌ كَذَّابٌ ۝٢٨

    اور ایک مومن شخص نے، جو فرعون کے خاندان میں سے تھا اور اپنا ایمان چھپائے ہوئے تھا، کہا کہ کیا تم ایک شحض کو محض اس بات پر قتل کرتے ہو کہ وه کہتا ہے میرا رب اللہ ہے اور تمہارے رب کی طرف سے دلیلیں لے کر آیا ہے، اگر وه جھوٹا ہو تو اس کا جھوٹ اسی پر ہے اور اگر وه سچا ہو، تو جس (عذاب) کا وه تم سے وعده کر رہا ہے اس میں سے کچھ نہ کچھ تو تم پر آ پڑے گا، اللہ تعالیٰ اس کی رہبری نہیں کرتا جو حد سے گزر جانے والے اور جھوٹے ہیں

  29. يَـٰقَوْمِ لَكُمُ ٱلْمُلْكُ ٱلْيَوْمَ ظَـٰهِرِينَ فِى ٱلْأَرْضِ فَمَن يَنصُرُنَا مِنۢ بَأْسِ ٱللَّهِ إِن جَآءَنَاۚ قَالَ فِرْعَوْنُ مَآ أُرِيكُمْ إِلَّا مَآ أَرَىٰ وَمَآ أَهْدِيكُمْ إِلَّا سَبِيلَ ٱلرَّشَادِ ۝٢٩

    اے میری قوم کے لوگو! آج تو بادشاہت تمہاری ہے کہ اس زمین پر تم غالب ہو لیکن اگر اللہ کا عذاب ہم پر آ گیا تو کون ہماری مدد کرے گا؟ فرعون بوﻻ، میں تو تمہیں وہی رائے دے رہا ہوں جو خود دیکھ رہا ہوں اور میں تو تمہیں بھلائی کی راه ہی بتلا رہا ہوں

  30. وَقَالَ ٱلَّذِىٓ ءَامَنَ يَـٰقَوْمِ إِنِّىٓ أَخَافُ عَلَيْكُم مِّثْلَ يَوْمِ ٱلْأَحْزَابِ ۝٣٠

    اس مومن نے کہا اے میری قوم! (کے لوگو) مجھے تو اندیشہ ہے کہ تم پر بھی ویسا ہی روز (بد عذاب) نہ آئے جو اور امتوں پر آیا

  31. مِثْلَ دَأْبِ قَوْمِ نُوحٍ وَعَادٍ وَثَمُودَ وَٱلَّذِينَ مِنۢ بَعْدِهِمْۚ وَمَا ٱللَّهُ يُرِيدُ ظُلْمًا لِّلْعِبَادِ ۝٣١

    جیسے امت نو ح اور عاد وﺛمود اور ان کے بعد والوں کا (حال ہوا)، اللہ اپنے بندوں پر کسی طرح کا ﻇلم کرنا نہیں چاہتا

  32. وَيَـٰقَوْمِ إِنِّىٓ أَخَافُ عَلَيْكُمْ يَوْمَ ٱلتَّنَادِ ۝٣٢

    اور مجھے تم پر ہانک پکار کے دن کابھی ڈر ہے

  33. يَوْمَ تُوَلُّونَ مُدْبِرِينَ مَا لَكُم مِّنَ ٱللَّهِ مِنْ عَاصِمٍۗ وَمَن يُضْلِلِ ٱللَّهُ فَمَا لَهُۥ مِنْ هَادٍ ۝٣٣

    جس دن تم پیٹھ پھیر کر لوٹو گے، تمہیں اللہ سے بچانے واﻻ کوئی نہ ہوگا اور جسے اللہ گمراه کر دے اس کا ہادی کوئی نہیں

  34. وَلَقَدْ جَآءَكُمْ يُوسُفُ مِن قَبْلُ بِٱلْبَيِّنَـٰتِ فَمَا زِلْتُمْ فِى شَكٍّ مِّمَّا جَآءَكُم بِهِۦۖ حَتَّىٰٓ إِذَا هَلَكَ قُلْتُمْ لَن يَبْعَثَ ٱللَّهُ مِنۢ بَعْدِهِۦ رَسُوۚلاً كَذَٲلِكَ يُضِلُّ ٱللَّهُ مَنْ هُوَ مُسْرِفٌ مُّرْتَابٌ ۝٣٤

    اور اس سے پہلے تمہارے پاس (حضرت) یوسف دلیلیں لے کر آئے، پھر بھی تم ان کی ﻻئی ہوئی (دلیل) میں شک و شبہ ہی کرتے رہے یہاں تک کہ جب ان کی وفات ہو گئی تو کہنے لگے ان کے بعد تو اللہ کسی رسول کو بھیجے گا ہی نہیں اسی طرح اللہ گمراه کرتا ہے ہر اس شخص کو جو حد سے بڑھ جانے واﻻ شک و شبہ کرنے واﻻ ہو

  35. ٱلَّذِينَ يُجَـٰدِلُونَ فِىٓ ءَايَـٰتِ ٱللَّهِ بِغَيْرِ سُلْطَـٰنٍ أَتَـٰهُمْۖ كَبُرَ مَقْتًا عِندَ ٱللَّهِ وَعِندَ ٱلَّذِينَ ءَامَنُواْۚ كَذَٲلِكَ يَطْبَعُ ٱللَّهُ عَلَىٰ كُلِّ قَلْبِ مُتَكَبِّرٍ جَبَّارٍ ۝٣٥

    جو بغیر کسی سند کے جو ان کے پاس آئی ہو اللہ کی آیتوں میں جھگڑتے ہیں، اللہ کے نزدیک اور مومنوں کے نزدیک یہ تو بہت بڑی ناراضگی کی چیز ہے، اللہ تعالیٰ اسی طرح ہر ایک مغرور سرکش کے دل پر مہر کردیتا ہے

  36. وَقَالَ فِرْعَوْنُ يَـٰهَـٰمَـٰنُ ٱبْنِ لِى صَرْحًا لَّعَلِّىٓ أَبْلُغُ ٱلْأَسْبَـٰبَ ۝٣٦

    فرعون نے کہا اے ہامان! میرے لیے ایک باﻻخانہ بنا شاید کہ میں آسمان کے جو دروازے ہیں

  37. أَسْبَـٰبَ ٱلسَّمَـٰوَٲتِ فَأَطَّلِعَ إِلَىٰٓ إِلَـٰهِ مُوسَىٰ وَإِنِّى لَأَظُنُّهُۥ كَـٰذِبًاۚ وَكَذَٲلِكَ زُيِّنَ لِفِرْعَوْنَ سُوٓءُ عَمَلِهِۦ وَصُدَّ عَنِ ٱلسَّبِيلِۚ وَمَا كَيْدُ فِرْعَوْنَ إِلَّا فِى تَبَابٍ ۝٣٧

    (ان) دروازوں تک پہنچ جاؤں اور موسیٰ کے معبود کو جھانک لوں اور بیشک میں سمجھتا ہوں وه جھوٹا ہے اور اسی طرح فرعون کی بدکرداریاں اسے بھلی دکھائی گئیں اور راه سے روک دیا گیا اور فرعون کی (ہر) حیلہ سازی تباہی میں ہی رہی

  38. وَقَالَ ٱلَّذِىٓ ءَامَنَ يَـٰقَوْمِ ٱتَّبِعُونِ أَهْدِكُمْ سَبِيلَ ٱلرَّشَادِ ۝٣٨

    اور اس مومن شخص نے کہا کہ اے میری قوم! (کے لوگو) تم (سب) میری پیروی کرو میں نیک راه کی طرف تمہاری رہبری کروں گا

  39. يَـٰقَوْمِ إِنَّمَا هَـٰذِهِ ٱلْحَيَوٲةُ ٱلدُّنْيَا مَتَـٰعٌ وَإِنَّ ٱلْأَخِرَةَ هِىَ دَارُ ٱلْقَرَارِ ۝٣٩

    اے میری قوم! یہ حیات دنیا متاع فانی ہے، (یقین مانو کہ قرار) اور ہمیشگی کا گھر تو آخرت ہی ہے

  40. مَنْ عَمِلَ سَيِّئَةً فَلَا يُجْزَىٰٓ إِلَّا مِثْلَهَاۖ وَمَنْ عَمِلَ صَـٰلِحًا مِّن ذَكَرٍ أَوْ أُنثَىٰ وَهُوَ مُؤْمِنٌ فَأُوْلَـٰٓئِكَ يَدْخُلُونَ ٱلْجَنَّةَ يُرْزَقُونَ فِيهَا بِغَيْرِ حِسَابٍ ۝٤٠

    جس نے گناه کیا ہے اسے تو برابر برابر کا بدلہ ہی ہے اور جس نے نیکی کی ہے خواه وه مرد ہو یا عورت اور وه ایمان واﻻ ہو تو یہ لوگ جنت میں جائیں گے اور وہاں بےشمار روزی پائیں گے

  41. ۞ وَيَـٰقَوْمِ مَا لِىٓ أَدْعُوكُمْ إِلَى ٱلنَّجَوٲةِ وَتَدْعُونَنِىٓ إِلَى ٱلنَّارِ ۝٤١

    اے میری قوم! یہ کیا بات ہے کہ میں تمہیں نجات کی طرف بلا رہا ہوں اور تم مجھے دوزخ کی طرف بلا رہے ہو

  42. تَدْعُونَنِى لِأَكْفُرَ بِٱللَّهِ وَأُشْرِكَ بِهِۦ مَا لَيْسَ لِى بِهِۦ عِلْمٌ وَأَنَا۟ أَدْعُوكُمْ إِلَى ٱلْعَزِيزِ ٱلْغَفَّـٰرِ ۝٤٢

    تم مجھے یہ دعوت دے رہے ہوکہ میں اللہ کے ساتھ کفر کروں اور اس کے ساتھ شرک کروں جس کا کوئی علم مجھے نہیں اور میں تمہیں غالب بخشنے والے (معبود) کی طرف دعوت دے رہا ہوں

  43. لَا جَرَمَ أَنَّمَا تَدْعُونَنِىٓ إِلَيْهِ لَيْسَ لَهُۥ دَعْوَةٌ فِى ٱلدُّنْيَا وَلَا فِى ٱلْأَخِرَةِ وَأَنَّ مَرَدَّنَآ إِلَى ٱللَّهِ وَأَنَّ ٱلْمُسْرِفِينَ هُمْ أَصْحَـٰبُ ٱلنَّارِ ۝٤٣

    یہ یقینی امر ہے کہ تم مجھے جس کی طرف بلا رہے ہو وه تو نہ دنیا میں پکارے جانے کے قابل ہے نہ آخرت میں، اور یہ (بھی یقینی بات ہے) کہ ہم سب کا لوٹنا اللہ کی طرف ہے اور حد سے گزر جانے والے ہی (یقیناً) اہل دوزخ ہیں

  44. فَسَتَذْكُرُونَ مَآ أَقُولُ لَكُمْۚ وَأُفَوِّضُ أَمْرِىٓ إِلَى ٱللَّهِۚ إِنَّ ٱللَّهَ بَصِيرُۢ بِٱلْعِبَادِ ۝٤٤

    پس آگے چل کر تم میری باتوں کو یاد کرو گے میں اپنا معاملہ اللہ کے سپرد کرتا ہوں، یقیناً اللہ تعالیٰ بندوں کا نگراں ہے

  45. فَوَقَـٰهُ ٱللَّهُ سَيِّـَٔـاتِ مَا مَكَرُواْۖ وَحَاقَ بِـَٔـالِ فِرْعَوْنَ سُوٓءُ ٱلْعَذَابِ ۝٤٥

    پس اسے اللہ تعالیٰ نے تمام بدیوں سے محفوظ رکھ لیا جو انہوں نے سوچ رکھی تھیں اور فرعون والوں پر بری طرح کا عذاب الٹ پڑا

  46. ٱلنَّارُ يُعْرَضُونَ عَلَيْهَا غُدُوًّا وَعَشِيًّاۖ وَيَوْمَ تَقُومُ ٱلسَّاعَةُ أَدْخِلُوٓاْ ءَالَ فِرْعَوْنَ أَشَدَّ ٱلْعَذَابِ ۝٤٦

    آگ ہے جس کے سامنے یہ ہر صبح شام ﻻئے جاتے ہیں اور جس دن قیامت قائم ہوگی (فرمان ہوگا کہ) فرعونیوں کو سخت ترین عذاب میں ڈالو

  47. وَإِذْ يَتَحَآجُّونَ فِى ٱلنَّارِ فَيَقُولُ ٱلضُّعَفَـٰٓؤُاْ لِلَّذِينَ ٱسْتَكْبَرُوٓاْ إِنَّا كُنَّا لَكُمْ تَبَعًا فَهَلْ أَنتُم مُّغْنُونَ عَنَّا نَصِيبًا مِّنَ ٱلنَّارِ ۝٤٧

    اور جب کہ دوزخ میں ایک دوسرے سے جھگڑیں گے تو کمزور لوگ تکبر والوں سے (جن کے یہ تابع تھے) کہیں گے کہ ہم تو تمہارے پیرو تھے تو کیا اب تم ہم سے اس آگ کا کوئی حصہ ہٹا سکتے ہو؟

  48. قَالَ ٱلَّذِينَ ٱسْتَكْبَرُوٓاْ إِنَّا كُلٌّ فِيهَآ إِنَّ ٱللَّهَ قَدْ حَكَمَ بَيْنَ ٱلْعِبَادِ ۝٤٨

    وه بڑے لوگ جواب دیں گے ہم تو سبھی اس آگ میں ہیں، اللہ تعالیٰ اپنے بندوں کے درمیان فیصلے کر چکا ہے

  49. وَقَالَ ٱلَّذِينَ فِى ٱلنَّارِ لِخَزَنَةِ جَهَنَّمَ ٱدْعُواْ رَبَّكُمْ يُخَفِّفْ عَنَّا يَوْمًا مِّنَ ٱلْعَذَابِ ۝٤٩

    اور (تمام) جہنمی مل کر جہنم کے داروغوں سے کہیں گے کہ تم ہی اپنے پروردگار سے دعا کرو کہ وه کسی دن تو ہمارے عذاب میں کمی کردے

  50. قَالُوٓاْ أَوَلَمْ تَكُ تَأْتِيكُمْ رُسُلُكُم بِٱلْبَيِّنَـٰتِۖ قَالُواْ بَلَىٰۚ قَالُواْ فَٱدْعُواْۗ وَمَا دُعَـٰٓؤُاْ ٱلْكَـٰفِرِينَ إِلَّا فِى ضَلَـٰلٍ ۝٥٠

    وه جواب دیں گے کہ کیا تمہارے پاس تمہارے رسول معجزے لے کر نہیں آئے تھے؟ وه کہیں گے کیوں نہیں، وه کہیں گے کہ پھر تم ہی دعا کرو اور کافروں کی دعا محض بے اﺛر اور بےراه ہے

  51. إِنَّا لَنَنصُرُ رُسُلَنَا وَٱلَّذِينَ ءَامَنُواْ فِى ٱلْحَيَوٲةِ ٱلدُّنْيَا وَيَوْمَ يَقُومُ ٱلْأَشْهَـٰدُ ۝٥١

    یقیناً ہم اپنے رسولوں کی اور ایمان والوں کی مدد زندگانیٴ دنیا میں بھی کریں گے اور اس دن بھی جب گواہی دینے والے کھڑے ہوں گے

  52. يَوْمَ لَا يَنفَعُ ٱلظَّـٰلِمِينَ مَعْذِرَتُهُمْۖ وَلَهُمُ ٱللَّعْنَةُ وَلَهُمْ سُوٓءُ ٱلدَّارِ ۝٥٢

    جس دن ﻇالموں کو ان کی (عذر) معذرت کچھ نفع نہ دے گی ان کے لیے لعنت ہی ہوگی اور ان کے لیے برا گھر ہو گا

  53. وَلَقَدْ ءَاتَيْنَا مُوسَى ٱلْهُدَىٰ وَأَوْرَثْنَا بَنِىٓ إِسْرَٲٓءِيلَ ٱلْكِتَـٰبَ ۝٥٣

    ہم نے موسیٰ (علیہ السلام) کو ہدایت نامہ عطا فرمایا اور بنو اسرائیل کو اس کتاب کا وارث بنایا

  54. هُدًى وَذِكْرَىٰ لِأُوْلِى ٱلْأَلْبَـٰبِ ۝٥٤

    کہ وه ہدایت و نصیحت تھی عقل مندوں کے لیے

  55. فَٱصْبِرْ إِنَّ وَعْدَ ٱللَّهِ حَقٌّ وَٱسْتَغْفِرْ لِذَنۢبِكَ وَسَبِّحْ بِحَمْدِ رَبِّكَ بِٱلْعَشِىِّ وَٱلْإِبْكَـٰرِ ۝٥٥

    پس اے نبی! تو صبر کر اللہ کا وعده بلاشک (وشبہ) سچا ہی ہے تو اپنے گناه کی معافی مانگتا ره اور صبح شام اپنے پروردگار کی تسبیح اور حمد بیان کرتا ره

  56. إِنَّ ٱلَّذِينَ يُجَـٰدِلُونَ فِىٓ ءَايَـٰتِ ٱللَّهِ بِغَيْرِ سُلْطَـٰنٍ أَتَـٰهُمْۙ إِن فِى صُدُورِهِمْ إِلَّا كِبْرٌ مَّا هُم بِبَـٰلِغِيهِۚ فَٱسْتَعِذْ بِٱللَّهِۖ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلْبَصِيرُ ۝٥٦

    جو لوگ باوجود اپنے پاس کسی سند کے نہ ہونے کے آیات الٰہی میں جھگڑا کرتے ہیں ان کے دلوں میں بجز نری بڑائی کے اور کچھ نہیں وه اس تک پہنچنے والے ہی نہیں، سو تو اللہ کی پناه مانگتا ره بیشک وه پورا سننے واﻻ اور سب سے زیاده دیکھنے واﻻ ہے

  57. لَخَلْقُ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ أَكْبَرُ مِنْ خَلْقِ ٱلنَّاسِ وَلَـٰكِنَّ أَكْثَرَ ٱلنَّاسِ لَا يَعْلَمُونَ ۝٥٧

    آسمان و زمین کی پیدائش یقیناً انسان کی پیدائش سے بہت بڑا کام ہے، لیکن (یہ اور بات ہے کہ) اکثر لوگ بےعلم ہیں

  58. وَمَا يَسْتَوِى ٱلْأَعْمَىٰ وَٱلْبَصِيرُ وَٱلَّذِينَ ءَامَنُواْ وَعَمِلُواْ ٱلصَّـٰلِحَـٰتِ وَلَا ٱلْمُسِىٓءُۚ قَلِيلاً مَّا تَتَذَكَّرُونَ ۝٥٨

    اندھا اور بینا برابر نہیں نہ وه لوگ جو ایمان ﻻئے اور بھلے کام کیے بدکاروں کے (برابر ہیں)، تم (بہت) کم نصیحت حاصل کر رہے ہو

  59. إِنَّ ٱلسَّاعَةَ لَأَتِيَةٌ لَّا رَيْبَ فِيهَا وَلَـٰكِنَّ أَكْثَرَ ٱلنَّاسِ لَا يُؤْمِنُونَ ۝٥٩

    قیامت بالیقین اور بےشبہ آنے والی ہے، لیکن (یہ اور بات ہے کہ) بہت سے لوگ ایمان نہیں ﻻتے

  60. وَقَالَ رَبُّكُمُ ٱدْعُونِىٓ أَسْتَجِبْ لَكُمْۚ إِنَّ ٱلَّذِينَ يَسْتَكْبِرُونَ عَنْ عِبَادَتِى سَيَدْخُلُونَ جَهَنَّمَ دَاخِرِينَ ۝٦٠

    اور تمہارے رب کا فرمان (سرزد ہوچکا ہے) کہ مجھ سے دعا کرو میں تمہاری دعاؤں کو قبول کروں گا یقین مانو کہ جو لوگ میری عبادت سے خودسری کرتے ہیں وه ابھی ابھی ذلیل ہوکر جہنم میں پہنچ جائیں گے

  61. ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلَّيْلَ لِتَسْكُنُواْ فِيهِ وَٱلنَّهَارَ مُبْصِرًاۚ إِنَّ ٱللَّهَ لَذُو فَضْلٍ عَلَى ٱلنَّاسِ وَلَـٰكِنَّ أَكْثَرَ ٱلنَّاسِ لَا يَشْكُرُونَ ۝٦١

    اللہ تعالیٰ نے تمھارے لیے رات بنادی کہ تم اس میں آرام حاصل کرو اور دن کو دیکھنے واﻻ بنا دیا، بیشک اللہ تعالیٰ لوگوں پر فضل وکرم واﻻ ہے لیکن اکثر لوگ شکر گزاری نہیں کرتے

  62. ذَٲلِكُمُ ٱللَّهُ رَبُّكُمْ خَـٰلِقُ كُلِّ شَىْءٍ لَّآ إِلَـٰهَ إِلَّا هُوَۖ فَأَنَّىٰ تُؤْفَكُونَ ۝٦٢

    یہی اللہ ہے تم سب کا رب ہر چیز کا خالق اس کے سوا کوئی معبود نہیں پھر کہاں تم پھرے جاتے ہو

  63. كَذَٲلِكَ يُؤْفَكُ ٱلَّذِينَ كَانُواْ بِـَٔـايَـٰتِ ٱللَّهِ يَجْحَدُونَ ۝٦٣

    اسی طرح وه لوگ بھی پھیرے جاتے رہے جو اللہ کی آیتوں کا انکار کرتے تھے

  64. ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلْأَرْضَ قَرَارًا وَٱلسَّمَآءَ بِنَآءً وَصَوَّرَكُمْ فَأَحْسَنَ صُوَرَكُمْ وَرَزَقَكُم مِّنَ ٱلطَّيِّبَـٰتِۚ ذَٲلِكُمُ ٱللَّهُ رَبُّكُمْۖ فَتَبَارَكَ ٱللَّهُ رَبُّ ٱلْعَـٰلَمِينَ ۝٦٤

    اللہ ہی ہے جس نے تمہارے لیے زمین کو ٹھہرنے کی جگہ اور آسمان کو چھت بنا دیا اور تمہاری صورتیں بنائیں اور بہت اچھی بنائیں اور تمہیں عمده عمده چیزیں کھانے کو عطا فرمائیں، یہی اللہ تمہارا پروردگار ہے، پس بہت ہی برکتوں واﻻ اللہ ہے سارے جہان کا پرورش کرنے واﻻ

  65. هُوَ ٱلْحَىُّ لَآ إِلَـٰهَ إِلَّا هُوَ فَٱدْعُوهُ مُخْلِصِينَ لَهُ ٱلدِّينَۗ ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ۝٦٥

    وه زنده ہے جس کے سوا کوئی معبود نہیں پس تم خالص اس کی عبادت کرتے ہوئے اسے پکارو، تمام خوبیاں اللہ ہی کے لیے ہیں جو تمام جہانوں کا رب ہے

  66. ۞ قُلْ إِنِّى نُهِيتُ أَنْ أَعْبُدَ ٱلَّذِينَ تَدْعُونَ مِن دُونِ ٱللَّهِ لَمَّا جَآءَنِىَ ٱلْبَيِّنَـٰتُ مِن رَّبِّى وَأُمِرْتُ أَنْ أُسْلِمَ لِرَبِّ ٱلْعَـٰلَمِينَ ۝٦٦

    آپ کہہ دیجئے! کہ مجھے ان کی عبادت سے روک دیا گیا ہے جنہیں تم اللہ کے سوا پکار رہے ہو اس بنا پر کہ میرے پاس میرے رب کی دلیلیں پہنچ چکی ہیں، مجھے یہ حکم دیا گیا ہے کہ میں تمام جہانوں کے رب کا تابع فرمان ہو جاؤں

  67. هُوَ ٱلَّذِى خَلَقَكُم مِّن تُرَابٍ ثُمَّ مِن نُّطْفَةٍ ثُمَّ مِنْ عَلَقَةٍ ثُمَّ يُخْرِجُكُمْ طِفْلاً ثُمَّ لِتَبْلُغُوٓاْ أَشُدَّكُمْ ثُمَّ لِتَكُونُواْ شُيُوخًاۚ وَمِنكُم مَّن يُتَوَفَّىٰ مِن قَبْلُۖ وَلِتَبْلُغُوٓاْ أَجَلاً مُّسَمًّى وَلَعَلَّكُمْ تَعْقِلُونَ ۝٦٧

    وه وہی ہے جس نے تمہیں مٹی سے پھر نطفے سے پھر خون کے لوتھڑے سے پیدا کیا پھر تمہیں بچہ کی صورت میں نکالتا ہے، پھر (تمہیں بڑھاتا ہے کہ) تم اپنی پوری قوت کو پہنچ جاؤ پھر بوڑھے ہو جاؤ، تم میں سے بعض اس سے پہلے ہی فوت ہو جاتے ہیں، (وه تمہیں چھوڑ دیتا ہے) تاکہ تم مدت معین تک پہنچ جاؤ اور تاکہ تم سوچ سمجھ لو

  68. هُوَ ٱلَّذِى يُحْىِۦ وَيُمِيتُۖ فَإِذَا قَضَىٰٓ أَمْرًا فَإِنَّمَا يَقُولُ لَهُۥ كُن فَيَكُونُ ۝٦٨

    وہی ہے جو جلاتا ہے اور مار ڈالتا ہے، پھر جب وه کسی کام کا کرنا مقرر کرتا ہے تو اسے صرف یہ کہتا ہے کہ ہو جا پس وه ہو جاتا ہے

  69. أَلَمْ تَرَ إِلَى ٱلَّذِينَ يُجَـٰدِلُونَ فِىٓ ءَايَـٰتِ ٱللَّهِ أَنَّىٰ يُصْرَفُونَ ۝٦٩

    کیا تو نے انہیں دیکھا جو اللہ کی آیتوں میں جھگڑتے ہیں، وه کہاں پھیر دیے جاتے ہیں

  70. ٱلَّذِينَ كَذَّبُواْ بِٱلْكِتَـٰبِ وَبِمَآ أَرْسَلْنَا بِهِۦ رُسُلَنَاۖ فَسَوْفَ يَعْلَمُونَ ۝٧٠

    جن لوگوں نے کتاب کو جھٹلایا اور اسے بھی جو ہم نے اپنے رسولوں کے ساتھ بھیجا انہیں ابھی ابھی حقیقت حال معلوم ہو جائے گی

  71. إِذِ ٱلْأَغْلَـٰلُ فِىٓ أَعْنَـٰقِهِمْ وَٱلسَّلَـٰسِلُ يُسْحَبُونَ ۝٧١

    جب کہ ان کی گردنوں میں طوق ہوں گے اور زنجیریں ہوں گی گھسیٹے جائیں گے

  72. فِى ٱلْحَمِيمِ ثُمَّ فِى ٱلنَّارِ يُسْجَرُونَ ۝٧٢

    کھولتے ہوئے پانی میں اور پھر جہنم کی آگ میں جلائے جائیں گے

  73. ثُمَّ قِيلَ لَهُمْ أَيْنَ مَا كُنتُمْ تُشْرِكُونَ ۝٧٣

    پھر ان سے پوچھا جائے گا کہ جنہیں تم شریک کرتے تھے وه کہاں ہیں۔

  74. مِن دُونِ ٱللَّهِۖ قَالُواْ ضَلُّواْ عَنَّا بَل لَّمْ نَكُن نَّدْعُواْ مِن قَبْلُ شَيْــًٔاۚ كَذَٲلِكَ يُضِلُّ ٱللَّهُ ٱلْكَـٰفِرِينَ ۝٧٤

    جو اللہ کے سوا تھے وه کہیں گے کہ وه تو ہم سے بہک گئے بلکہ ہم تو اس سے پہلے کسی کو بھی پکارتے ہی نہ تھے۔ اللہ تعالیٰ کافروں کو اسی طرح گمراه کرتا ہے

  75. ذَٲلِكُم بِمَا كُنتُمْ تَفْرَحُونَ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ وَبِمَا كُنتُمْ تَمْرَحُونَ ۝٧٥

    یہ بدلہ ہے اس چیز کا جو تم زمین میں ناحق پھولے نہ سماتے تھے۔ اور (بےجا) اتراتے پھرتے تھے

  76. ٱدْخُلُوٓاْ أَبْوَٲبَ جَهَنَّمَ خَـٰلِدِينَ فِيهَاۖ فَبِئْسَ مَثْوَى ٱلْمُتَكَبِّرِينَ ۝٧٦

    (اب آؤ) جہنم میں ہمیشہ رہنے کے لیے (اس کے) دروازوں میں داخل ہو جاؤ، کیا ہی بری جگہ ہے تکبر کرنے والے کی

  77. فَٱصْبِرْ إِنَّ وَعْدَ ٱللَّهِ حَۚقٌّ فَإِمَّا نُرِيَنَّكَ بَعْضَ ٱلَّذِى نَعِدُهُمْ أَوْ نَتَوَفَّيَنَّكَ فَإِلَيْنَا يُرْجَعُونَ ۝٧٧

    پس آپ صبر کریں اللہ کا وعده قطعاً سچا ہے، انہیں ہم نے جو وعدے دے رکھے ہیں ان میں سے کچھ ہم آپ کو دکھائیں یا (اس سے پہلے) ہم آپ کو وفات دے دیں، ان کا لوٹایا جانا تو ہماری ہی طرف ہے

  78. وَلَقَدْ أَرْسَلْنَا رُسُلاً مِّن قَبْلِكَ مِنْهُم مَّن قَصَصْنَا عَلَيْكَ وَمِنْهُم مَّن لَّمْ نَقْصُصْ عَلَيْكَۗ وَمَا كَانَ لِرَسُولٍ أَن يَأْتِىَ بِـَٔـايَةٍ إِلَّا بِإِذْنِ ٱللَّهِۚ فَإِذَا جَآءَ أَمْرُ ٱللَّهِ قُضِىَ بِٱلْحَقِّ وَخَسِرَ هُنَالِكَ ٱلْمُبْطِلُونَ ۝٧٨

    یقیناً ہم آپ سے پہلے بھی بہت سے رسول بھیج چکے ہیں جن میں سے بعض کے (واقعات) ہم آپ کو بیان کر چکے ہیں اور ان میں سے بعض کے (قصے) تو ہم نے آپ کو بیان ہی نہیں کیے اور کسی رسول کا یہ (مقدور) نہ تھا کہ کوئی معجزه اللہ کی اجازت کے بغیر ﻻ سکے پھر جس وقت اللہ کا حکم آئےگا حق کے ساتھ فیصلہ کر دیا جائے گا اور اس جگہ اہل باطل خسارے میں ره جائیں گے

  79. ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلْأَنْعَـٰمَ لِتَرْكَبُواْ مِنْهَا وَمِنْهَا تَأْكُلُونَ ۝٧٩

    اللہ وه ہے جس نے تمہارے لیے چوپائے پیدا کیے جن میں سے بعض پر تم سوار ہوتے ہو اور بعض کو تم کھاتے ہو

  80. وَلَكُمْ فِيهَا مَنَـٰفِعُ وَلِتَبْلُغُواْ عَلَيْهَا حَاجَةً فِى صُدُورِكُمْ وَعَلَيْهَا وَعَلَى ٱلْفُلْكِ تُحْمَلُونَ ۝٨٠

    اور بھی تمہارے لیے ان میں بہت سے نفع ہیں اور تاکہ اپنے سینوں میں چھپی ہوئی حاجتوں کو انہی پر سواری کر کے تم حاصل کر لو اور ان چوپایوں پر اور کشتیوں پر سوار کیے جاتے ہو

  81. وَيُرِيكُمْ ءَايَـٰتِهِۦ فَأَىَّ ءَايَـٰتِ ٱللَّهِ تُنكِرُونَ ۝٨١

    اللہ تمہیں اپنی نشانیاں دکھاتا جارہا ہے، پس تم اللہ کی کن کن نشانیوں کے منکر بنتے رہو گے

  82. أَفَلَمْ يَسِيرُواْ فِى ٱلْأَرْضِ فَيَنظُرُواْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلَّذِينَ مِن قَبْلِهِمْۚ كَانُوٓاْ أَكْثَرَ مِنْهُمْ وَأَشَدَّ قُوَّةً وَءَاثَارًا فِى ٱلْأَرْضِ فَمَآ أَغْنَىٰ عَنْهُم مَّا كَانُواْ يَكْسِبُونَ ۝٨٢

    کیا انہوں نے زمین میں چل پھر کر اپنے سے پہلوں کا انجام نہیں دیکھا؟ جو ان سے تعداد میں زیاده تھے قوت میں سخت اور زمین میں بہت ساری یادگاریں چھوڑی تھیں، ان کے کیے کاموں نے انہیں کچھ بھی فائده نہیں پہنچایا

  83. فَلَمَّا جَآءَتْهُمْ رُسُلُهُم بِٱلْبَيِّنَـٰتِ فَرِحُواْ بِمَا عِندَهُم مِّنَ ٱلْعِلْمِ وَحَاقَ بِهِم مَّا كَانُواْ بِهِۦ يَسْتَهْزِءُونَ ۝٨٣

    پس جب بھی ان کے پاس ان کے رسول کھلی نشانیاں لے کر آئے تو یہ اپنے پاس کے علم پر اترانے لگے، بالﺂخر جس چیز کو مذاق میں اڑا رہے تھے وه ان پر الٹ پڑی

  84. فَلَمَّا رَأَوْاْ بَأْسَنَا قَالُوٓاْ ءَامَنَّا بِٱللَّهِ وَحْدَهُۥ وَكَفَرْنَا بِمَا كُنَّا بِهِۦ مُشْرِكِينَ ۝٨٤

    ہمارا عذاب دیکھتے ہی کہنے لگے کہ اللہ واحد پر ہم ایمان ﻻئے اور جن جن کو ہم اس کا شریک بنا رہے تھے ہم نے ان سب سے انکار کیا

  85. فَلَمْ يَكُ يَنفَعُهُمْ إِيمَـٰنُهُمْ لَمَّا رَأَوْاْ بَأْسَنَاۖ سُنَّتَ ٱللَّهِ ٱلَّتِى قَدْ خَلَتْ فِى عِبَادِهِۖۦ وَخَسِرَ هُنَالِكَ ٱلْكَـٰفِرُونَ ۝٨٥

    لیکن ہمارے عذاب کو دیکھ لینے کے بعد ان کے ایمان نے انہیں نفع نہ دیا۔ اللہ نے اپنا معمول یہی مقرر کر رکھا ہے جو اس کے بندوں میں برابر چلا آ رہا ہے۔ اور اس جگہ کافر خراب وخستہ ہوئے
