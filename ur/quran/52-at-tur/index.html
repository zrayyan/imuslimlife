بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ

  1. وَٱلطُّورِ ۝١

    قسم ہے طور کی

  2. وَكِتَـٰبٍ مَّسْطُورٍ ۝٢

    اور لکھی ہوئی کتاب کی

  3. فِى رَقٍّ مَّنشُورٍ ۝٣

    جو جھلی کے کھلے ہوئے ورق میں ہے

  4. وَٱلْبَيْتِ ٱلْمَعْمُورِ ۝٤

    اورآباد گھر کی

  5. وَٱلسَّقْفِ ٱلْمَرْفُوعِ ۝٥

    اور اونچی چھت کی

  6. وَٱلْبَحْرِ ٱلْمَسْجُورِ ۝٦

    اور بھڑکائے ہوئے سمندر کی

  7. إِنَّ عَذَابَ رَبِّكَ لَوَٲقِعٌ ۝٧

    بیشک آپ کے رب کا عذاب ہو کر رہنے واﻻ ہے

  8. مَّا لَهُۥ مِن دَافِعٍ ۝٨

    اسے کوئی روکنے واﻻ نہیں

  9. يَوْمَ تَمُورُ ٱلسَّمَآءُ مَوْرًا ۝٩

    جس دن آسمان تھرتھرانے لگے گا

  10. وَتَسِيرُ ٱلْجِبَالُ سَيْرًا ۝١٠

    اور پہاڑ چلنے پھرنے لگیں گے

  11. فَوَيْلٌ يَوْمَئِذٍ لِّلْمُكَذِّبِينَ ۝١١

    اس دن جھٹلانے والوں کو (پوری) خرابی ہے

  12. ٱلَّذِينَ هُمْ فِى خَوْضٍ يَلْعَبُونَ ۝١٢

    جو اپنی بیہوده گوئی میں اچھل کود کر رہے ہیں

  13. يَوْمَ يُدَعُّونَ إِلَىٰ نَارِ جَهَنَّمَ دَعًّا ۝١٣

    جس دن وه دھکے دے دے کر آتش جہنم کی طرف ﻻئیں جائیں گے

  14. هَـٰذِهِ ٱلنَّارُ ٱلَّتِى كُنتُم بِهَا تُكَذِّبُونَ ۝١٤

    یہی وه آتش دوزخ ہے جسے تم جھوٹ بتلاتے تھے

  15. أَفَسِحْرٌ هَـٰذَآ أَمْ أَنتُمْ لَا تُبْصِرُونَ ۝١٥

    (اب بتاؤ) کیا یہ جادو ہے؟ یا تم دیکھتے ہی نہیں ہو

  16. ٱصْلَوْهَا فَٱصْبِرُوٓاْ أَوْ لَا تَصْبِرُواْ سَوَآءٌ عَلَيْكُمْ‌ۖ إِنَّمَا تُجْزَوْنَ مَا كُنتُمْ تَعْمَلُونَ ۝١٦

    جاؤ دوزخ میں اب تمہارا صبر کرنا اور نہ کرنا تمہارے لیے یکساں ہے تمہیں فقط تمہارے کیے کا بدلہ دیا جائے گا

  17. إِنَّ ٱلْمُتَّقِينَ فِى جَنَّـٰتٍ وَنَعِيمٍ ۝١٧

    یقیناً پرہیزگار لوگ جنتوں میں اور نعمتوں میں ہیں

  18. فَـٰكِهِينَ بِمَآ ءَاتَـٰهُمْ رَبُّهُمْ وَوَقَـٰهُمْ رَبُّهُمْ عَذَابَ ٱلْجَحِيمِ ۝١٨

    جو انہیں ان کے رب نے دے رکھی ہیں اس پر خوش خوش ہیں، اوران کے پروردگار نے انہیں جہنم کے عذاب سے بھی بچا لیا ہے

  19. كُلُواْ وَٱشْرَبُواْ هَنِيٓـــَٔۢا بِمَا كُنتُمْ تَعْمَلُونَ ۝١٩

    تم مزے سے کھاتے پیتے رہو ان اعمال کے بدلے جو تم کرتے تھے

  20. مُتَّكِــِٔينَ عَلَىٰ سُرُرٍ مَّصْفُوفَةٍ‌ۖ وَزَوَّجْنَـٰهُم بِحُورٍ عِينٍ ۝٢٠

    برابر بچھے ہوئے شاندار تختے پر تکیے لگائے ہوئے۔ اور ہم نے ان کے نکاح بڑی بڑی آنکھوں والی (حوروں) سے کر دیئے ہیں

  21. وَٱلَّذِينَ ءَامَنُواْ وَٱتَّبَعَتْهُمْ ذُرِّيَّتُهُم بِإِيمَـٰنٍ أَلْحَقْنَا بِهِمْ ذُرِّيَّتَهُمْ وَمَآ أَلَتْنَـٰهُم مِّنْ عَمَلِهِم مِّن شَىْءٍ‌ۚ كُلُّ ٱمْرِىِٕۭ بِمَا كَسَبَ رَهِينٌ ۝٢١

    اور جو لوگ ایمان ﻻئے اور ان کی اوﻻد نے بھی ایمان میں ان کی پیروی کی ہم ان کی اوﻻد کو ان تک پہنچا دیں گے اور ان کے عمل سے ہم کچھ کم نہ کریں گے، ہر شخص اپنے اپنے اعمال کا گروی ہے

  22. وَأَمْدَدْنَـٰهُم بِفَـٰكِهَةٍ وَلَحْمٍ مِّمَّا يَشْتَهُونَ ۝٢٢

    ہم ان کے لیے میوے اورمرغوب گوشت کی ریل پیل کردیں گے

  23. يَتَنَـٰزَعُونَ فِيهَا كَأْسًا لَّا لَغْوٌ فِيهَا وَلَا تَأْثِيمٌ ۝٢٣

    (خوش طبعی کے ساتھ) ایک دوسرے سے جام (شراب) کی چھینا جھپٹی کریں گے جس شراب کے سرور میں تو بیہوده گوئی ہوگی نہ گناه

  24. ۞ وَيَطُوفُ عَلَيْهِمْ غِلْمَانٌ لَّهُمْ كَأَنَّهُمْ لُؤْلُؤٌ مَّكْنُونٌ ۝٢٤

    اور ان کے اردگرد ان کے نو عمر غلام چل پھر رہے ہوں گے، گویا کہ وه موتی تھے جو ڈھکے رکھے تھے

  25. وَأَقْبَلَ بَعْضُهُمْ عَلَىٰ بَعْضٍ يَتَسَآءَلُونَ ۝٢٥

    اور آپس میں ایک دوسرے کی طرف متوجہ ہو کر سوال کریں گے

  26. قَالُوٓاْ إِنَّا كُنَّا قَبْلُ فِىٓ أَهْلِنَا مُشْفِقِينَ ۝٢٦

    کہیں گے کہ اس سے پہلے ہم اپنے گھر والوں کے درمیان بہت ڈرا کرتے تھے

  27. فَمَنَّ ٱللَّهُ عَلَيْنَا وَوَقَـٰنَا عَذَابَ ٱلسَّمُومِ ۝٢٧

    پس اللہ تعالیٰ نے ہم پر بڑا احسان کیا اور ہمیں تیز وتند گرم ہواؤں کے عذاب سے بچا لیا

  28. إِنَّا كُنَّا مِن قَبْلُ نَدْعُوهُ‌ۖ إِنَّهُۥ هُوَ ٱلْبَرُّ ٱلرَّحِيمُ ۝٢٨

    ہم اس سے پہلے ہی اس کی عبادت کیا کرتے تھے، بیشک وه محسن اور مہربان ہے

  29. فَذَكِّرْ فَمَآ أَنتَ بِنِعْمَتِ رَبِّكَ بِكَاهِنٍ وَلَا مَجْنُونٍ ۝٢٩

    تو آپ سمجھاتے رہیں کیونکہ آپ اپنے رب کے فضل سے نہ تو کاہن ہیں نہ دیوانہ

  30. أَمْ يَقُولُونَ شَاعِرٌ نَّتَرَبَّصُ بِهِۦ رَيْبَ ٱلْمَنُونِ ۝٣٠

    کیا کافر یوں کہتے ہیں کہ یہ شاعر ہے ہم اس پر زمانے کے حوادث (یعنی موت) کا انتظار کر رہے ہیں

  31. قُلْ تَرَبَّصُواْ فَإِنِّى مَعَكُم مِّنَ ٱلْمُتَرَبِّصِينَ ۝٣١

    کہہ دیجئے! تم منتظر رہو میں بھی تمہارے ساتھ انتظار کرنے والوں میں ہوں

  32. أَمْ تَأْمُرُهُمْ أَحْلَـٰمُهُم بِهَـٰذَآ‌ۚ أَمْ هُمْ قَوْمٌ طَاغُونَ ۝٣٢

    کیا ان کی عقلیں انہیں یہی سکھاتی ہیں؟ یا یہ لوگ ہی سرکش ہیں

  33. أَمْ يَقُولُونَ تَقَوَّلَهُۥ‌ۚ بَل لَّا يُؤْمِنُونَ ۝٣٣

    کیا یہ کہتے ہیں کہ اس نبی نے (قرآن) خود گھڑ لیا ہے، واقعہ یہ ہے کہ وه ایمان نہیں ﻻتے

  34. فَلْيَأْتُواْ بِحَدِيثٍ مِّثْلِهِۦٓ إِن كَانُواْ صَـٰدِقِينَ ۝٣٤

    اچھا اگر یہ سچے ہیں تو بھلا اس جیسی ایک (ہی) بات یہ (بھی) تو لے آئیں

  35. أَمْ خُلِقُواْ مِنْ غَيْرِ شَىْءٍ أَمْ هُمُ ٱلْخَـٰلِقُونَ ۝٣٥

    کیا یہ بغیر کسی (پیدا کرنے والے) کے خود بخود پیدا ہوگئے ہیں؟ یا یہ خود پیدا کرنے والے ہیں؟

  36. أَمْ خَلَقُواْ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضَ‌ۚ بَل لَّا يُوقِنُونَ ۝٣٦

    کیا انہوں نے ہی آسمانوں اور زمین کو پیدا کیا ہے؟ بلکہ یہ یقین نہ کرنے والے لوگ ہیں

  37. أَمْ عِندَهُمْ خَزَآئِنُ رَبِّكَ أَمْ هُمُ ٱلْمُصَۣيْطِرُونَ ۝٣٧

    یا کیا ان کے پاس تیرے رب کے خزانے ہیں؟ یا (ان خزانوں کے) یہ داروغہ ہیں

  38. أَمْ لَهُمْ سُلَّمٌ يَسْتَمِعُونَ فِيهِ‌ۖ فَلْيَأْتِ مُسْتَمِعُهُم بِسُلْطَـٰنٍ مُّبِينٍ ۝٣٨

    یا کیا ان کے پاس کوئی سیڑھی ہے جس پر چڑھ کر سنتے ہیں؟ (اگر ایسا ہے) تو ان کا سننے واﻻ کوئی روشن دلیل پیش کرے

  39. أَمْ لَهُ ٱلْبَنَـٰتُ وَلَكُمُ ٱلْبَنُونَ ۝٣٩

    کیا اللہ کی تو سب لڑکیاں ہیں اور تمہارے ہاں لڑکے ہیں؟

  40. أَمْ تَسْــَٔلُهُمْ أَجْرًا فَهُم مِّن مَّغْرَمٍ مُّثْقَلُونَ ۝٤٠

    کیا تو ان سے کوئی اجرت طلب کرتا ہے کہ یہ اس کے تاوان سے بوجھل ہو رہے ہیں

  41. أَمْ عِندَهُمُ ٱلْغَيْبُ فَهُمْ يَكْتُبُونَ ۝٤١

    کیا ان کے پاس علم غیب ہے جسے یہ لکھ لیتے ہیں؟

  42. أَمْ يُرِيدُونَ كَيْدًا‌ۖ فَٱلَّذِينَ كَفَرُواْ هُمُ ٱلْمَكِيدُونَ ۝٤٢

    کیا یہ لوگ کوئی فریب کرنا چاہتے ہیں؟ تو یقین کرلیں کہ فریب خورده کافر ہی ہیں

  43. أَمْ لَهُمْ إِلَـٰهٌ غَيْرُ ٱللَّهِ‌ۚ سُبْحَـٰنَ ٱللَّهِ عَمَّا يُشْرِكُونَ ۝٤٣

    کیا اللہ کے سوا ان کا کوئی معبود ہے؟ (ہرگز نہیں) اللہ تعالیٰ ان کے شرک سے پاک ہے

  44. وَإِن يَرَوْاْ كِسْفًا مِّنَ ٱلسَّمَآءِ سَاقِطًا يَقُولُواْ سَحَابٌ مَّرْكُومٌ ۝٤٤

    اگر یہ لوگ آسمان کے کسی ٹکڑے کو گرتا ہوا دیکھ لیں تب بھی کہہ دیں کہ یہ تہ بہ تہ بادل ہے

  45. فَذَرْهُمْ حَتَّىٰ يُلَـٰقُواْ يَوْمَهُمُ ٱلَّذِى فِيهِ يُصْعَقُونَ ۝٤٥

    تو انہیں چھوڑ دے یہاں تک کہ انہیں اس دن سے سابقہ پڑے جس میں یہ بے ہوش کر دیئے جائیں گے

  46. يَوْمَ لَا يُغْنِى عَنْهُمْ كَيْدُهُمْ شَيْــًٔا وَلَا هُمْ يُنصَرُونَ ۝٤٦

    جس دن انہیں ان کا مکر کچھ کام نہ دے گا اور نہ وه مدد کیے جائیں گے

  47. وَإِنَّ لِلَّذِينَ ظَلَمُواْ عَذَابًا دُونَ ذَٲلِكَ وَلَـٰكِنَّ أَكْثَرَهُمْ لَا يَعْلَمُونَ ۝٤٧

    بیشک ﻇالموں کے لیے اس کے علاوه اور عذاب بھی ہیں۔ لیکن ان لوگوں میں سے اکثر بےعلم ہیں

  48. وَٱصْبِرْ لِحُكْمِ رَبِّكَ فَإِنَّكَ بِأَعْيُنِنَا‌ۖ وَسَبِّحْ بِحَمْدِ رَبِّكَ حِينَ تَقُومُ ۝٤٨

    تو اپنے رب کے حکم کے انتظار میں صبر سے کام لے، بیشک تو ہماری آنکھوں کے سامنے ہے۔ صبح کو جب تو اٹھے اپنے رب کی پاکی اور حمد بیان کر

  49. وَمِنَ ٱلَّيْلِ فَسَبِّحْهُ وَإِدْبَـٰرَ ٱلنُّجُومِ ۝٤٩

    اور رات کو بھی اس کی تسبیح پڑھ اور ستاروں کے ڈوبتے وقت بھی
