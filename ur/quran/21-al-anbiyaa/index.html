بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ

  1. ٱقْتَرَبَ لِلنَّاسِ حِسَابُهُمْ وَهُمْ فِى غَفْلَةٍ مُّعْرِضُونَ ۝١

    لوگوں کے حساب کا وقت قریب آگیا پھر بھی وه بے خبری میں منھ پھیرے ہوئے ہیں

  2. مَا يَأْتِيهِم مِّن ذِكْرٍ مِّن رَّبِّهِم مُّحْدَثٍ إِلَّا ٱسْتَمَعُوهُ وَهُمْ يَلْعَبُونَ ۝٢

    ان کے پاس ان کے رب کی طرف سے جو بھی نئی نئی نصیحت آتی ہے اسے وه کھیل کود میں ہی سنتے ہیں

  3. لَاهِيَةً قُلُوبُهُمْ‌ۗ وَأَسَرُّواْ ٱلنَّجْوَى ٱلَّذِينَ ظَلَمُواْ هَلْ هَـٰذَآ إِلَّا بَشَرٌ مِّثْلُكُمْ‌ۖ أَفَتَأْتُونَ ٱلسِّحْرَ وَأَنتُمْ تُبْصِرُونَ ۝٣

    ان کے دل بالکل غافل ہیں اور ان ﻇالموں نے چپکے چپکے سرگوشیاں کیں کہ وه تم ہی جیسا انسان ہے، پھر کیا وجہ ہے جو تم آنکھوں دیکھتے جادو میں آجاتے ہو

  4. قَالَ رَبِّى يَعْلَمُ ٱلْقَوْلَ فِى ٱلسَّمَآءِ وَٱلْأَرْضِ‌ۖ وَهُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ۝٤

    پیغمبر نے کہا میرا پروردگار ہر اس بات کو جو زمین وآسمان میں ہے بخوبی جانتا ہے، وه بہت ہی سننے واﻻ اور جاننے واﻻ ہے

  5. بَلْ قَالُوٓاْ أَضْغَـٰثُ أَحْلَـٰمِۭ بَلِ ٱفْتَرَٮٰهُ بَلْ هُوَ شَاعِرٌ فَلْيَأْتِنَا بِـَٔـايَةٍ كَمَآ أُرْسِلَ ٱلْأَوَّلُونَ ۝٥

    اتنا ہی نہیں بلکہ یہ تو کہتے ہیں کہ یہ قرآن پراگنده خوابوں کا مجموعہ ہے بلکہ اس نے از خود اسے گھڑ لیا ہے بلکہ یہ شاعر ہے، ورنہ ہمارے سامنے یہ کوئی ایسی نشانی ﻻتے جیسے کہ اگلے پیغمبر بھیجے گئے تھے

  6. مَآ ءَامَنَتْ قَبْلَهُم مِّن قَرْيَةٍ أَهْلَكْنَـٰهَآ‌ۖ أَفَهُمْ يُؤْمِنُونَ ۝٦

    ان سے پہلے جتنی بستیاں ہم نے اجاڑیں سب ایمان سے خالی تھیں۔ تو کیا اب یہ ایمان ﻻئیں گے

  7. وَمَآ أَرْسَلْنَا قَبْلَكَ إِلَّا رِجَالاً نُّوحِىٓ إِلَيْهِمْ‌ۖ فَسْــَٔلُوٓاْ أَهْلَ ٱلذِّكْرِ إِن كُنتُمْ لَا تَعْلَمُونَ ۝٧

    تجھ سے پہلے جتنے پیغمبر ہم نے بھیجے سبھی مرد تھے جن کی طرف ہم وحی اتارتے تھے پس تم اہل کتاب سے پوچھ لو اگر خود تمہیں علم نہ ہو

  8. وَمَا جَعَلْنَـٰهُمْ جَسَدًا لَّا يَأْكُلُونَ ٱلطَّعَامَ وَمَا كَانُواْ خَـٰلِدِينَ ۝٨

    ہم نے ان کے ایسے جسم نہیں بنائے تھے کہ وه کھانا نہ کھائیں اور نہ وه ہمیشہ رہنے والے تھے

  9. ثُمَّ صَدَقْنَـٰهُمُ ٱلْوَعْدَ فَأَنجَيْنَـٰهُمْ وَمَن نَّشَآءُ وَأَهْلَكْنَا ٱلْمُسْرِفِينَ ۝٩

    پھر ہم نے ان سے کیے ہوئے سب وعدے سچے کیے انہیں اور جن جن کو ہم نے چاہا نجات عطا فرمائی اور حد سے نکل جانے والوں کو غارت کر دیا

  10. لَقَدْ أَنزَلْنَآ إِلَيْكُمْ كِتَـٰبًا فِيهِ ذِكْرُكُمْ‌ۖ أَفَلَا تَعْقِلُونَ ۝١٠

    یقیناً ہم نے تمہاری جانب کتاب نازل فرمائی ہے جس میں تمہارے لئے ذکر ہے، کیا پھر بھی تم عقل نہیں رکھتے؟

  11. وَكَمْ قَصَمْنَا مِن قَرْيَةٍ كَانَتْ ظَالِمَةً وَأَنشَأْنَا بَعْدَهَا قَوْمًا ءَاخَرِينَ ۝١١

    اور بہت سی بستیاں ہم نے تباه کر دیں جو ﻇالم تھیں اور ان کے بعد ہم نے دوسری قوم کو پیدا کر دیا

  12. فَلَمَّآ أَحَسُّواْ بَأْسَنَآ إِذَا هُم مِّنْهَا يَرْكُضُونَ ۝١٢

    جب انہوں نے ہمارے عذاب کا احساس کر لیا، تو لگے اس سے بھاگنے

  13. لَا تَرْكُضُواْ وَٱرْجِعُوٓاْ إِلَىٰ مَآ أُتْرِفْتُمْ فِيهِ وَمَسَـٰكِنِكُمْ لَعَلَّكُمْ تُسْــَٔلُونَ ۝١٣

    بھاگ دوڑ نہ کرو اور جہاں تمہیں آسودگی دی گئی تھی وہیں واپس لوٹو اور اپنے مکانات کی طرف جاؤ تاکہ تم سے سوال تو کر لیا جائے

  14. قَالُواْ يَـٰوَيْلَنَآ إِنَّا كُنَّا ظَـٰلِمِينَ ۝١٤

    کہنے لگے ہائے ہماری خرابی! بیشک ہم ﻇالم تھے

  15. فَمَا زَالَت تِّلْكَ دَعْوَٮٰهُمْ حَتَّىٰ جَعَلْنَـٰهُمْ حَصِيدًا خَـٰمِدِينَ ۝١٥

    پھر تو ان کا یہی قول تھا یہاں تک کہ ہم نے انہیں جڑ سے کٹی ہوئی کھیتی اور بجھی پڑی آگ (کی طرح) کر دیا

  16. وَمَا خَلَقْنَا ٱلسَّمَآءَ وَٱلْأَرْضَ وَمَا بَيْنَهُمَا لَـٰعِبِينَ ۝١٦

    ہم نے آسمان وزمین اور ان کے درمیان کی چیزوں کو کھیلتے ہوئے نہیں بنایا

  17. لَوْ أَرَدْنَآ أَن نَّتَّخِذَ لَهْوًا لَّٱتَّخَذْنَـٰهُ مِن لَّدُنَّآ إِن كُنَّا فَـٰعِلِينَ ۝١٧

    اگر ہم یوں ہی کھیل تماشے کا اراده کرتے تو اسے اپنے پاس سے ہی بنا لیتے، اگر ہم کرنے والے ہی ہوتے

  18. بَلْ نَقْذِفُ بِٱلْحَقِّ عَلَى ٱلْبَـٰطِلِ فَيَدْمَغُهُۥ فَإِذَا هُوَ زَاهِقٌ‌ۚ وَلَكُمُ ٱلْوَيْلُ مِمَّا تَصِفُونَ ۝١٨

    بلکہ ہم سچ کو جھوٹ پر پھینک مارتے ہیں پس سچ جھوٹ کا سر توڑ دیتا ہے اور وه اسی وقت نابود ہو جاتا ہے، تم جو باتیں بناتے ہو وه تمہاری لئے باعﺚ خرابی ہیں

  19. وَلَهُۥ مَن فِى ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ‌ۚ وَمَنْ عِندَهُۥ لَا يَسْتَكْبِرُونَ عَنْ عِبَادَتِهِۦ وَلَا يَسْتَحْسِرُونَ ۝١٩

    آسمانوں اور زمین میں جو ہے اسی اللہ کا ہے اور جو اس کے پاس ہیں وه اس کی عبادت سے نہ سرکشی کرتے ہیں اور نہ تھکتے ہیں

  20. يُسَبِّحُونَ ٱلَّيْلَ وَٱلنَّهَارَ لَا يَفْتُرُونَ ۝٢٠

    وه دن رات تسبیح بیان کرتے ہیں اور ذرا سی بھی سستی نہیں کرتے

  21. أَمِ ٱتَّخَذُوٓاْ ءَالِهَةً مِّنَ ٱلْأَرْضِ هُمْ يُنشِرُونَ ۝٢١

    کیا ان لوگوں نے زمین (کی مخلوقات میں) سے جنہیں معبود بنا رکھا ہے وه زنده کردیتے ہیں

  22. لَوْ كَانَ فِيهِمَآ ءَالِهَةٌ إِلَّا ٱللَّهُ لَفَسَدَتَا‌ۚ فَسُبْحَـٰنَ ٱللَّهِ رَبِّ ٱلْعَرْشِ عَمَّا يَصِفُونَ ۝٢٢

    اگر آسمان وزمین میں سوائے اللہ تعالیٰ کے اور بھی معبود ہوتے تو یہ دونوں درہم برہم ہوجاتے پس اللہ تعالیٰ عرش کا رب ہر اس وصف سے پاک ہے جو یہ مشرک بیان کرتے ہیں

  23. لَا يُسْــَٔلُ عَمَّا يَفْعَلُ وَهُمْ يُسْــَٔلُونَ ۝٢٣

    وه اپنے کاموں کے لئے (کسی کےآگے) جواب ده نہیں اور سب (اس کےآگے) جواب ده ہیں

  24. أَمِ ٱتَّخَذُواْ مِن دُونِهِۦٓ ءَالِهَةً‌ۖ قُلْ هَاتُواْ بُرْهَـٰنَكُمْ‌ۖ هَـٰذَا ذِكْرُ مَن مَّعِىَ وَذِكْرُ مَن قَبْلِى‌ۗ بَلْ أَكْثَرُهُمْ لَا يَعْلَمُونَ ٱلْحَقَّ‌ۖ فَهُم مُّعْرِضُونَ ۝٢٤

    کیا ان لوگوں نے اللہ کے سوا اور معبود بنا رکھے ہیں، ان سے کہہ دو ﻻؤ اپنی دلیل پیش کرو۔ یہ ہے میرے ساتھ والوں کی کتاب اور مجھ سے اگلوں کی دلیل۔ بات یہ ہے کہ ان میں کے اکثر لوگ حق کو نہیں جانتے اسی وجہ سے منھ موڑے ہوئے ہیں

  25. وَمَآ أَرْسَلْنَا مِن قَبْلِكَ مِن رَّسُولٍ إِلَّا نُوحِىٓ إِلَيْهِ أَنَّهُۥ لَآ إِلَـٰهَ إِلَّآ أَنَا۟ فَٱعْبُدُونِ ۝٢٥

    تجھ سے پہلے بھی جو رسول ہم نے بھیجا اس کی طرف یہی وحی نازل فرمائی کہ میرے سوا کوئی معبود برحق نہیں پس تم سب میری ہی عبادت کرو

  26. وَقَالُواْ ٱتَّخَذَ ٱلرَّحْمَـٰنُ وَلَدًا‌ۗ سُبْحَـٰنَهُۥ‌ۚ بَلْ عِبَادٌ مُّكْرَمُونَ ۝٢٦

    (مشرک لوگ) کہتے ہیں کہ رحمٰن اوﻻد واﻻ ہے (غلط ہے) اس کی ذات پاک ہے، بلکہ وه سب اس کے باعزت بندے ہیں

  27. لَا يَسْبِقُونَهُۥ بِٱلْقَوْلِ وَهُم بِأَمْرِهِۦ يَعْمَلُونَ ۝٢٧

    کسی بات میں اللہ پر پیش دستی نہیں کرتے بلکہ اس کے فرمان پر کاربند ہیں

  28. يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَلَا يَشْفَعُونَ إِلَّا لِمَنِ ٱرْتَضَىٰ وَهُم مِّنْ خَشْيَتِهِۦ مُشْفِقُونَ ۝٢٨

    وه ان کے آگے پیچھے تمام امور سے واقف ہے وه کسی کی بھی سفارش نہیں کرتے بجز ان کے جن سے اللہ خوش ہو وه تو خود ہیبت الٰہی سے لرزاں وترساں ہیں

  29. ۞ وَمَن يَقُلْ مِنْهُمْ إِنِّىٓ إِلَـٰهٌ مِّن دُونِهِۦ فَذَٲلِكَ نَجْزِيهِ جَهَنَّمَ‌ۚ كَذَٲلِكَ نَجْزِى ٱلظَّـٰلِمِينَ ۝٢٩

    ان میں سے اگر کوئی بھی کہہ دے کہ اللہ کے سوا میں ﻻئق عبادت ہوں تو ہم اسے دوزخ کی سزا دیں ہم ﻇالموں کو اسی طرح سزا دیتے ہیں

  30. أَوَلَمْ يَرَ ٱلَّذِينَ كَفَرُوٓاْ أَنَّ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضَ كَانَتَا رَتْقًا فَفَتَقْنَـٰهُمَا‌ۖ وَجَعَلْنَا مِنَ ٱلْمَآءِ كُلَّ شَىْءٍ حَىٍّ‌ۖ أَفَلَا يُؤْمِنُونَ ۝٣٠

    کیا کافر لوگوں نے یہ نہیں دیکھا کہ آسمان وزمین باہم ملے جلے تھے پھر ہم نے انہیں جدا کیا اور ہر زنده چیز کو ہم نے پانی سے پیدا کیا، کیا یہ لوگ پھر بھی ایمان نہیں ﻻتے

  31. وَجَعَلْنَا فِى ٱلْأَرْضِ رَوَٲسِىَ أَن تَمِيدَ بِهِمْ وَجَعَلْنَا فِيهَا فِجَاجًا سُبُلاً لَّعَلَّهُمْ يَهْتَدُونَ ۝٣١

    اور ہم نے زمین میں پہاڑ بنا دیئے تاکہ وه مخلوق کو ہلا نہ سکے، اور ہم نے اس میں کشاده راہیں بنا دیں تاکہ وه راستہ حاصل کریں

  32. وَجَعَلْنَا ٱلسَّمَآءَ سَقْفًا مَّحْفُوظًا‌ۖ وَهُمْ عَنْ ءَايَـٰتِهَا مُعْرِضُونَ ۝٣٢

    آسمان کو محفوظ چھت بھی ہم نے ہی بنایا ہے۔ لیکن لوگ اس کی قدرت کے نمونوں پر دھیان نہیں دھرتے

  33. وَهُوَ ٱلَّذِى خَلَقَ ٱلَّيْلَ وَٱلنَّهَارَ وَٱلشَّمْسَ وَٱلْقَمَرَ‌ۖ كُلٌّ فِى فَلَكٍ يَسْبَحُونَ ۝٣٣

    وہی اللہ ہے جس نے رات اور دن اور سورج اور چاند کو پیدا کیا ہے۔ ان میں سے ہر ایک اپنے اپنے مدار میں تیرتے پھرتے ہیں

  34. وَمَا جَعَلْنَا لِبَشَرٍ مِّن قَبْلِكَ ٱلْخُلْدَ‌ۖ أَفَإِيْن مِّتَّ فَهُمُ ٱلْخَـٰلِدُونَ ۝٣٤

    آپ سے پہلے کسی انسان کو بھی ہم نے ہمیشگی نہیں دی، کیا اگر آپ مرگئے تو وه ہمیشہ کے لئے ره جائیں گے

  35. كُلُّ نَفْسٍ ذَآئِقَةُ ٱلْمَوْتِ‌ۗ وَنَبْلُوكُم بِٱلشَّرِّ وَٱلْخَيْرِ فِتْنَةً‌ۖ وَإِلَيْنَا تُرْجَعُونَ ۝٣٥

    ہر جان دار موت کا مزه چکھنے واﻻ ہے۔ ہم بطریق امتحان تم میں سے ہر ایک کو برائی بھلائی میں مبتلا کرتے ہیں اور تم سب ہماری ہی طرف لوٹائے جاؤ گے

  36. وَإِذَا رَءَاكَ ٱلَّذِينَ كَفَرُوٓاْ إِن يَتَّخِذُونَكَ إِلَّا هُزُوًا أَهَـٰذَا ٱلَّذِى يَذْكُرُ ءَالِهَتَكُمْ وَهُم بِذِكْرِ ٱلرَّحْمَـٰنِ هُمْ كَـٰفِرُونَ ۝٣٦

    یہ منکرین تجھے جب بھی دیکھتے ہیں تو تمہارا مذاق ہی اڑاتے ہیں کہ کیا یہی وه ہے جو تمہارے معبودوں کا ذکر برائی سے کرتا ہے، اور وه خود ہی رحمٰن کی یاد کے بالکل ہی منکر ہیں

  37. خُلِقَ ٱلْإِنسَـٰنُ مِنْ عَجَلٍ‌ۚ سَأُوْرِيكُمْ ءَايَـٰتِى فَلَا تَسْتَعْجِلُونِ ۝٣٧

    انسان جلد باز مخلوق ہے۔ میں تمہیں اپنی نشانیاں ابھی ابھی دکھاؤں گا تم مجھ سے جلد بازی نہ کرو

  38. وَيَقُولُونَ مَتَىٰ هَـٰذَا ٱلْوَعْدُ إِن كُنتُمْ صَـٰدِقِينَ ۝٣٨

    کہتے ہیں کہ اگر سچے ہو تو بتا دو کہ یہ وعده کب ہے

  39. لَوْ يَعْلَمُ ٱلَّذِينَ كَفَرُواْ حِينَ لَا يَكُفُّونَ عَن وُجُوهِهِمُ ٱلنَّارَ وَلَا عَن ظُهُورِهِمْ وَلَا هُمْ يُنصَرُونَ ۝٣٩

    کاش! یہ کافر جانتے کہ اس وقت نہ تو یہ کافر آگ کو اپنے چہروں سے ہٹا سکیں گے اور نہ اپنی پیٹھوں سے اور نہ ان کی مدد کی جائے گی

  40. بَلْ تَأْتِيهِم بَغْتَةً فَتَبْهَتُهُمْ فَلَا يَسْتَطِيعُونَ رَدَّهَا وَلَا هُمْ يُنظَرُونَ ۝٤٠

    (ہاں ہاں!) وعدے کی گھڑی ان کے پاس اچانک آجائے گی اور انہیں ہکا بکا کر دے گی، پھر نہ تو یہ لوگ اسے ٹال سکیں گے اور نہ ذرا سی بھی مہلت دیئے جائیں گے

  41. وَلَقَدِ ٱسْتُهْزِئَ بِرُسُلٍ مِّن قَبْلِكَ فَحَاقَ بِٱلَّذِينَ سَخِرُواْ مِنْهُم مَّا كَانُواْ بِهِۦ يَسْتَهْزِءُونَ ۝٤١

    اور تجھ سے پہلے رسولوں کے ساتھ بھی ہنسی مذاق کیا گیا پس ہنسی کرنے والوں کو ہی اس چیز نے گھیر لیا جس کی وه ہنسی اڑاتے تھے

  42. قُلْ مَن يَكْلَؤُكُم بِٱلَّيْلِ وَٱلنَّهَارِ مِنَ ٱلرَّحْمَـٰنِ‌ۗ بَلْ هُمْ عَن ذِكْرِ رَبِّهِم مُّعْرِضُونَ ۝٤٢

    ان سے پوچھئے کہ رحمٰن سے، دن اور رات تمہاری حفاﻇت کون کر سکتا ہے؟ بات یہ ہے کہ یہ لوگ اپنے رب کے ذکر سے پھرے ہوئے ہیں

  43. أَمْ لَهُمْ ءَالِهَةٌ تَمْنَعُهُم مِّن دُونِنَا‌ۚ لَا يَسْتَطِيعُونَ نَصْرَ أَنفُسِهِمْ وَلَا هُم مِّنَّا يُصْحَبُونَ ۝٤٣

    کیا ہمارے سوا ان کے اور معبود ہیں جو انہیں مصیبتوں سے بچا لیں۔ کوئی بھی خود اپنی مدد کی طاقت نہیں رکھتا اور نہ کوئی ہماری طرف سے رفاقت دیا جاتا ہے

  44. بَلْ مَتَّعْنَا هَـٰٓؤُلَآءِ وَءَابَآءَهُمْ حَتَّىٰ طَالَ عَلَيْهِمُ ٱلْعُمُرُ‌ۗ أَفَلَا يَرَوْنَ أَنَّا نَأْتِى ٱلْأَرْضَ نَنقُصُهَا مِنْ أَطْرَافِهَآ‌ۚ أَفَهُمُ ٱلْغَـٰلِبُونَ ۝٤٤

    بلکہ ہم نے انہیں اور ان کے باپ دادوں کو زندگی کے سروسامان دیے یہاں تک کہ ان کی مدت عمر گزر گئی۔ کیا وه نہیں دیکھتے کہ ہم زمین کو اس کے کناروں سے گھٹاتے چلے آرہے ہیں، اب کیا وہی غالب ہیں؟

  45. قُلْ إِنَّمَآ أُنذِرُكُم بِٱلْوَحْىِ‌ۚ وَلَا يَسْمَعُ ٱلصُّمُّ ٱلدُّعَآءَ إِذَا مَا يُنذَرُونَ ۝٤٥

    کہہ دیجئے! میں تو تمہیں اللہ کی وحی کے ذریعہ آگاه کر رہا ہوں مگر بہرے لوگ بات نہیں سنتے جبکہ انہیں آگاه کیا جائے

  46. وَلَئِن مَّسَّتْهُمْ نَفْحَةٌ مِّنْ عَذَابِ رَبِّكَ لَيَقُولُنَّ يَـٰوَيْلَنَآ إِنَّا كُنَّا ظَـٰلِمِينَ ۝٤٦

    اگر انہیں تیرے رب کے کسی عذاب کا جھونکا بھی لگ جائے تو پکار اٹھیں کہ ہائے ہماری بدبختی! یقیناً ہم گنہگار تھے

  47. وَنَضَعُ ٱلْمَوَٲزِينَ ٱلْقِسْطَ لِيَوْمِ ٱلْقِيَـٰمَةِ فَلَا تُظْلَمُ نَفْسٌ شَيْــًٔا‌ۖ وَإِن كَانَ مِثْقَالَ حَبَّةٍ مِّنْ خَرْدَلٍ أَتَيْنَا بِهَا‌ۗ وَكَفَىٰ بِنَا حَـٰسِبِينَ ۝٤٧

    قیامت کے دن ہم درمیان میں ﻻ رکھیں گے ٹھیک ٹھیک تولنے والی ترازو کو۔ پھر کسی پر کچھ بھی ﻇلم نہ کیا جائے گا۔ اور اگر ایک رائی کے دانے کے برابر بھی عمل ہوگا ہم اسے ﻻ حاضر کریں گے، اور ہم کافی ہیں حساب کرنے والے

  48. وَلَقَدْ ءَاتَيْنَا مُوسَىٰ وَهَـٰرُونَ ٱلْفُرْقَانَ وَضِيَآءً وَذِكْرًا لِّلْمُتَّقِينَ ۝٤٨

    یہ بالکل سچ ہے کہ ہم نے موسیٰ وہارون کو فیصلے کرنے والی نورانی اور پرہیزگاروں کے لئے وعﻆ ونصیحت والی کتاب عطا فرمائی

  49. ٱلَّذِينَ يَخْشَوْنَ رَبَّهُم بِٱلْغَيْبِ وَهُم مِّنَ ٱلسَّاعَةِ مُشْفِقُونَ ۝٤٩

    وه لوگ جو اپنے رب سے بن دیکھے خوف کھاتے ہیں اور قیامت (کے تصور) سے کانپتے رہتے ہیں

  50. وَهَـٰذَا ذِكْرٌ مُّبَارَكٌ أَنزَلْنَـٰهُ‌ۚ أَفَأَنتُمْ لَهُۥ مُنكِرُونَ ۝٥٠

    اور یہ نصیحت وبرکت واﻻ قرآن بھی ہم ہی نے نازل فرمایا ہے کیا پھر بھی تم اس کے منکر ہو

  51. ۞ وَلَقَدْ ءَاتَيْنَآ إِبْرَٲهِيمَ رُشْدَهُۥ مِن قَبْلُ وَكُنَّا بِهِۦ عَـٰلِمِينَ ۝٥١

    یقیناً ہم نے اس سے پہلے ابراہیم کو اس کی سمجھ بوجھ بخشی تھی اور ہم اس کے احوال سے بخوبی واقف تھے

  52. إِذْ قَالَ لِأَبِيهِ وَقَوْمِهِۦ مَا هَـٰذِهِ ٱلتَّمَاثِيلُ ٱلَّتِىٓ أَنتُمْ لَهَا عَـٰكِفُونَ ۝٥٢

    جبکہ اس نے اپنے باپ سے اور اپنی قوم سے کہا کہ یہ مورتیاں جن کے تم مجاور بنے بیٹھے ہو کیا ہیں؟

  53. قَالُواْ وَجَدْنَآ ءَابَآءَنَا لَهَا عَـٰبِدِينَ ۝٥٣

    سب نے جواب دیا کہ ہم نے اپنے باپ دادا کو انہی کی عبادت کرتے ہوئے پایا

  54. قَالَ لَقَدْ كُنتُمْ أَنتُمْ وَءَابَآؤُكُمْ فِى ضَلَـٰلٍ مُّبِينٍ ۝٥٤

    آپ نے فرمایا! پھر تو تم اور تمہارے باپ دادا سبھی یقیناً کھلی گمراہی میں مبتلا رہے

  55. قَالُوٓاْ أَجِئْتَنَا بِٱلْحَقِّ أَمْ أَنتَ مِنَ ٱللَّـٰعِبِينَ ۝٥٥

    کہنے لگے کیا آپ ہمارے پاس سچ مچ حق ﻻئے ہیں یا یوں ہی مذاق کر رہے ہیں

  56. قَالَ بَل رَّبُّكُمْ رَبُّ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ ٱلَّذِى فَطَرَهُنَّ وَأَنَا۟ عَلَىٰ ذَٲلِكُم مِّنَ ٱلشَّـٰهِدِينَ ۝٥٦

    آپ نے فرمایا نہیں درحقیقت تم سب کا پروردگار تو وه ہے جو آسمانوں اور زمین کا مالک ہے جس نے انہیں پیدا کیا ہے، میں تو اسی بات کا گواه اور قائل ہوں

  57. وَتَٱللَّهِ لَأَكِيدَنَّ أَصْنَـٰمَكُم بَعْدَ أَن تُوَلُّواْ مُدْبِرِينَ ۝٥٧

    اور اللہ کی قسم! میں تمہارے ان معبودوں کے ساتھ جب تم علیحده پیٹھ پھیر کر چل دو گے ایک چال چلوں گا

  58. فَجَعَلَهُمْ جُذَٲذًا إِلَّا كَبِيرًا لَّهُمْ لَعَلَّهُمْ إِلَيْهِ يَرْجِعُونَ ۝٥٨

    پس اس نے ان سب کے ٹکڑے ٹکڑے کر دیئے ہاں صرف بڑے بت کو چھوڑ دیا یہ بھی اس لئے کہ وه سب اس کی طرف ہی لوٹیں

  59. قَالُواْ مَن فَعَلَ هَـٰذَا بِـَٔـالِهَتِنَآ إِنَّهُۥ لَمِنَ ٱلظَّـٰلِمِينَ ۝٥٩

    کہنے لگے کہ ہمارے خداؤں کے ساتھ یہ کس نے کیا؟ ایسا شخص تو یقیناً ﻇالموں میں سے ہے

  60. قَالُواْ سَمِعْنَا فَتًى يَذْكُرُهُمْ يُقَالُ لَهُۥٓ إِبْرَٲهِيمُ ۝٦٠

    بولے ہم نے ایک نوجوان کو ان کا تذکره کرتے ہوئے سنا تھا جسے ابراہیم (علیہ السلام) کہا جاتا ہے

  61. قَالُواْ فَأْتُواْ بِهِۦ عَلَىٰٓ أَعْيُنِ ٱلنَّاسِ لَعَلَّهُمْ يَشْهَدُونَ ۝٦١

    سب نے کہا اچھا اسے مجمع میں لوگوں کی نگاہوں کے سامنے ﻻؤ تاکہ سب دیکھیں

  62. قَالُوٓاْ ءَأَنتَ فَعَلْتَ هَـٰذَا بِـَٔـالِهَتِنَا يَـٰٓإِبْرَٲهِيمُ ۝٦٢

    کہنے لگے! اے ابراہیم (علیہ السلام) کیا تو نے ہی ہمارے خداؤں کے ساتھ یہ حرکت کی ہے

  63. قَالَ بَلْ فَعَلَهُۥ كَبِيرُهُمْ هَـٰذَا فَسْــَٔلُوهُمْ إِن كَانُواْ يَنطِقُونَ ۝٦٣

    آپ نے جواب دیا بلکہ اس کام کو ان کے بڑے نے کیا ہے تم اپنے خداؤں سے ہی پوچھ لو اگر یہ بولتے چالتے ہوں

  64. فَرَجَعُوٓاْ إِلَىٰٓ أَنفُسِهِمْ فَقَالُوٓاْ إِنَّكُمْ أَنتُمُ ٱلظَّـٰلِمُونَ ۝٦٤

    پس یہ لوگ اپنے دلوں میں قائل ہوگئے اور کہنے لگے واقعی ﻇالم تو تم ہی ہو

  65. ثُمَّ نُكِسُواْ عَلَىٰ رُءُوسِهِمْ لَقَدْ عَلِمْتَ مَا هَـٰٓؤُلَآءِ يَنطِقُونَ ۝٦٥

    پھر اپنے سروں کے بل اوندھے ہوگئے (اور کہنے لگے کہ) یہ تو تجھے بھی معلوم ہے کہ یہ بولنے چالنے والے نہیں

  66. قَالَ أَفَتَعْبُدُونَ مِن دُونِ ٱللَّهِ مَا لَا يَنفَعُكُمْ شَيْــًٔا وَلَا يَضُرُّكُمْ ۝٦٦

    اللہ کے خلیل نے اسی وقت فرمایا افسوس! کیا تم اللہ کے علاوه ان کی عبادت کرتے ہو جو نہ تمہیں کچھ بھی نفع پہنچا سکیں نہ نقصان

  67. أُفٍّ لَّكُمْ وَلِمَا تَعْبُدُونَ مِن دُونِ ٱللَّهِ‌ۖ أَفَلَا تَعْقِلُونَ ۝٦٧

    تف ہے تم پر اور ان پر جن کی تم اللہ کے سوا عبادت کرتے ہو۔ کیا تمہیں اتنی سی عقل بھی نہیں؟

  68. قَالُواْ حَرِّقُوهُ وَٱنصُرُوٓاْ ءَالِهَتَكُمْ إِن كُنتُمْ فَـٰعِلِينَ ۝٦٨

    کہنے لگے کہ اسے جلا دو اور اپنے خداؤں کی مدد کرو اگر تمہیں کچھ کرنا ہی ہے

  69. قُلْنَا يَـٰنَارُ كُونِى بَرْدًا وَسَلَـٰمًا عَلَىٰٓ إِبْرَٲهِيمَ ۝٦٩

    ہم نے فرما دیا اے آگ! تو ٹھنڈی پڑ جا اور ابراہیم (علیہ السلام) کے لئے سلامتی (اور آرام کی چیز) بن جا!

  70. وَأَرَادُواْ بِهِۦ كَيْدًا فَجَعَلْنَـٰهُمُ ٱلْأَخْسَرِينَ ۝٧٠

    گو انہوں نے ابراہیم (علیہ السلام) کا برا چاہا، لیکن ہم نے انہیں ناکام بنا دیا

  71. وَنَجَّيْنَـٰهُ وَلُوطًا إِلَى ٱلْأَرْضِ ٱلَّتِى بَـٰرَكْنَا فِيهَا لِلْعَـٰلَمِينَ ۝٧١

    اور ہم ابراہیم اور لوط کو بچاکر اس زمین کی طرف لے چلے جس میں ہم نے تمام جہان والوں کے لئے برکت رکھی تھی

  72. وَوَهَبْنَا لَهُۥٓ إِسْحَـٰقَ وَيَعْقُوبَ نَافِلَةً‌ۖ وَكُلاًّ جَعَلْنَا صَـٰلِحِينَ ۝٧٢

    اور ہم نے اسےاسحاق عطا فرمایااور یعقوب اس پر مزید۔ اور ہرایک کو ہم نے صالح بنایا

  73. وَجَعَلْنَـٰهُمْ أَئِمَّةً يَهْدُونَ بِأَمْرِنَا وَأَوْحَيْنَآ إِلَيْهِمْ فِعْلَ ٱلْخَيْرَٲتِ وَإِقَامَ ٱلصَّلَوٲةِ وَإِيتَآءَ ٱلزَّكَوٲةِ‌ۖ وَكَانُواْ لَنَا عَـٰبِدِينَ ۝٧٣

    اور ہم نے انہیں پیشوا بنا دیا کہ ہمارے حکم سے لوگوں کی رہبری کریں اور ہم نے ان کی طرف نیک کاموں کے کرنے اور نمازوں کے قائم رکھنے اور زکوٰة دینے کی وحی (تلقین) کی، اور وه سب کے سب ہمارے عبادت گزار بندے تھے

  74. وَلُوطًا ءَاتَيْنَـٰهُ حُكْمًا وَعِلْمًا وَنَجَّيْنَـٰهُ مِنَ ٱلْقَرْيَةِ ٱلَّتِى كَانَت تَّعْمَلُ ٱلْخَبَـٰٓئِثَ‌ۗ إِنَّهُمْ كَانُواْ قَوْمَ سَوْءٍ فَـٰسِقِينَ ۝٧٤

    ہم نے لوط (علیہ السلام) کو بھی حکم اور علم دیا اور اسے اسی بستی سے نجات دی جہاں کے لوگ گندے کاموں میں مبتلا تھے۔ اور تھے بھی بدترین گنہگار

  75. وَأَدْخَلْنَـٰهُ فِى رَحْمَتِنَآ‌ۖ إِنَّهُۥ مِنَ ٱلصَّـٰلِحِينَ ۝٧٥

    اور ہم نے لوط (علیہ السلام) کو اپنی رحمت میں داخل کر لیا بے شک وه نیکو کار لوگوں میں سے تھا

  76. وَنُوحًا إِذْ نَادَىٰ مِن قَبْلُ فَٱسْتَجَبْنَا لَهُۥ فَنَجَّيْنَـٰهُ وَأَهْلَهُۥ مِنَ ٱلْكَرْبِ ٱلْعَظِيمِ ۝٧٦

    نوح کے اس وقت کو یاد کیجئے جبکہ اس نے اس سے پہلے دعا کی ہم نے اس کی دعا قبول فرمائی اور اسے اور اس کے گھر والوں کو بڑے کرب سے نجات دی

  77. وَنَصَرْنَـٰهُ مِنَ ٱلْقَوْمِ ٱلَّذِينَ كَذَّبُواْ بِـَٔـايَـٰتِنَآ‌ۚ إِنَّهُمْ كَانُواْ قَوْمَ سَوْءٍ فَأَغْرَقْنَـٰهُمْ أَجْمَعِينَ ۝٧٧

    اور جو لوگ ہماری آیتوں کو جھٹلا رہے تھے ان کے مقابلے میں ہم نے اس کی مدد کی، یقیناً وه برے لوگ تھے پس ہم نےان سب کو ڈبو دیا

  78. وَدَاوُۥدَ وَسُلَيْمَـٰنَ إِذْ يَحْكُمَانِ فِى ٱلْحَرْثِ إِذْ نَفَشَتْ فِيهِ غَنَمُ ٱلْقَوْمِ وَكُنَّا لِحُكْمِهِمْ شَـٰهِدِينَ ۝٧٨

    اور داؤد اور سلیمان (علیہما السلام) کو یاد کیجئے جبکہ وه کھیت کے معاملہ میں فیصلہ کر رہے تھے کہ کچھ لوگوں کی بکریاں رات کو اس میں چر چگ گئی تھیں، اور ان کے فیصلے میں ہم موجود تھے

  79. فَفَهَّمْنَـٰهَا سُلَيْمَـٰنَ‌ۚ وَكُلاًّ ءَاتَيْنَا حُكْمًا وَعِلْمًا‌ۚ وَسَخَّرْنَا مَعَ دَاوُۥدَ ٱلْجِبَالَ يُسَبِّحْنَ وَٱلطَّيْرَ‌ۚ وَكُنَّا فَـٰعِلِينَ ۝٧٩

    ہم نے اس کا صحیح فیصلہ سلیمان کو سمجھا دیا۔ ہاں ہر ایک کو ہم نے حکم وعلم دے رکھا تھا اور داؤد کے تابع ہم نے پہاڑ کر دیئے تھے جو تسبیح کرتے تھے اور پرند بھی۔ ہم کرنے والے ہی تھے

  80. وَعَلَّمْنَـٰهُ صَنْعَةَ لَبُوسٍ لَّكُمْ لِتُحْصِنَكُم مِّنۢ بَأْسِكُمْ‌ۖ فَهَلْ أَنتُمْ شَـٰكِرُونَ ۝٨٠

    اور ہم نے اسے تمہارے لئے لباس بنانے کی کاریگری سکھائی تاکہ لڑائی کے ضرر سے تمہارا بچاؤ ہو۔ کیا تم شکر گزار بنو گے؟

  81. وَلِسُلَيْمَـٰنَ ٱلرِّيحَ عَاصِفَةً تَجْرِى بِأَمْرِهِۦٓ إِلَى ٱلْأَرْضِ ٱلَّتِى بَـٰرَكْنَا فِيهَا‌ۚ وَكُنَّا بِكُلِّ شَىْءٍ عَـٰلِمِينَ ۝٨١

    ہم نے تند وتیز ہوا کو سلیمان (علیہ السلام) کے تابع کر دیا جو اس کے فرمان کے مطابق اس زمین کی طرف چلتی تھی جہاں ہم نے برکت دے رکھی تھی، اور ہم ہر چیز سے باخبر اور دانا ہیں

  82. وَمِنَ ٱلشَّيَـٰطِينِ مَن يَغُوصُونَ لَهُۥ وَيَعْمَلُونَ عَمَلاً دُونَ ذَٲلِكَ‌ۖ وَكُنَّا لَهُمْ حَـٰفِظِينَ ۝٨٢

    اسی طرح سے بہت سے شیاطین بھی ہم نے اس کے تابع کیے تھے جو اس کے فرمان سے غوطے لگاتے تھے اور اس کے سوا بھی بہت سے کام کرتے تھے، ان کے نگہبان ہم ہی تھے

  83. ۞ وَأَيُّوبَ إِذْ نَادَىٰ رَبَّهُۥٓ أَنِّى مَسَّنِىَ ٱلضُّرُّ وَأَنتَ أَرْحَمُ ٱلرَّٲحِمِينَ ۝٨٣

    ایوب (علیہ السلام) کی اس حالت کو یاد کرو جبکہ اس نے اپنے پروردگار کو پکارا کہ مجھے یہ بیماری لگ گئی ہے اور تو رحم کرنے والوں سے زیاده رحم کرنے واﻻ ہے

  84. فَٱسْتَجَبْنَا لَهُۥ فَكَشَفْنَا مَا بِهِۦ مِن ضُرٍّ‌ۖ وَءَاتَيْنَـٰهُ أَهْلَهُۥ وَمِثْلَهُم مَّعَهُمْ رَحْمَةً مِّنْ عِندِنَا وَذِكْرَىٰ لِلْعَـٰبِدِينَ ۝٨٤

    تو ہم نے اس کی سن لی اور جو دکھ انہیں تھا اسے دور کر دیا اور اس کو اہل وعیال عطا فرمائے بلکہ ان کے ساتھ ویسے ہی اور اپنی خاص مہربانی سے تاکہ سچے بندوں کے لئے سبب نصیحت ہو

  85. وَإِسْمَـٰعِيلَ وَإِدْرِيسَ وَذَا ٱلْكِفْلِ‌ۖ كُلٌّ مِّنَ ٱلصَّـٰبِرِينَ ۝٨٥

    اور اسماعیل اور ادریس اور ذوالکفل، (علیہم السلام) یہ سب صابر لوگ تھے

  86. وَأَدْخَلْنَـٰهُمْ فِى رَحْمَتِنَآ‌ۖ إِنَّهُم مِّنَ ٱلصَّـٰلِحِينَ ۝٨٦

    ہم نے انہیں اپنی رحمت میں داخل کر لیا۔ یہ سب لوگ نیک تھے

  87. وَذَا ٱلنُّونِ إِذ ذَّهَبَ مُغَـٰضِبًا فَظَنَّ أَن لَّن نَّقْدِرَ عَلَيْهِ فَنَادَىٰ فِى ٱلظُّلُمَـٰتِ أَن لَّآ إِلَـٰهَ إِلَّآ أَنتَ سُبْحَـٰنَكَ إِنِّى كُنتُ مِنَ ٱلظَّـٰلِمِينَ ۝٨٧

    مچھلی والے (حضرت یونس علیہ السلام) کو یاد کرو! جبکہ وه غصہ سے چل دیا اور خیال کیا کہ ہم اسے نہ پکڑ سکیں گے۔ بالﺂخر وه اندھیروں کے اندر سے پکار اٹھا کہ الٰہی تیرے سوا کوئی معبود نہیں تو پاک ہے، بیشک میں ﻇالموں میں ہو گیا

  88. فَٱسْتَجَبْنَا لَهُۥ وَنَجَّيْنَـٰهُ مِنَ ٱلْغَمِّ‌ۚ وَكَذَٲلِكَ نُـۨجِى ٱلْمُؤْمِنِينَ ۝٨٨

    تو ہم نے اس کی پکار سن لی اور اسے غم سے نجات دے دی اور ہم ایمان والوں کو اسی طرح بچا لیا کرتے ہیں

  89. وَزَكَرِيَّآ إِذْ نَادَىٰ رَبَّهُۥ رَبِّ لَا تَذَرْنِى فَرْدًا وَأَنتَ خَيْرُ ٱلْوَٲرِثِينَ ۝٨٩

    اور زکریا (علیہ السلام) کو یاد کرو جب اس نے اپنے رب سے دعا کی کہ اے میرے پروردگار! مجھے تنہا نہ چھوڑ، تو سب سے بہتر وارث ہے

  90. فَٱسْتَجَبْنَا لَهُۥ وَوَهَبْنَا لَهُۥ يَحْيَىٰ وَأَصْلَحْنَا لَهُۥ زَوْجَهُۥٓ‌ۚ إِنَّهُمْ كَانُواْ يُسَـٰرِعُونَ فِى ٱلْخَيْرَٲتِ وَيَدْعُونَنَا رَغَبًا وَرَهَبًا‌ۖ وَكَانُواْ لَنَا خَـٰشِعِينَ ۝٩٠

    ہم نے اس کی دعا کو قبول فرما کر اسے یحيٰ (علیہ السلام) عطا فرمایا اور ان کی بیوی کو ان کے لئے درست کر دیا۔ یہ بزرگ لوگ نیک کاموں کی طرف جلدی کرتے تھے اور ہمیں ﻻلچ طمع اور ڈر خوف سے پکارتے تھے۔ اور ہمارے سامنے عاجزی کرنے والے تھے

  91. وَٱلَّتِىٓ أَحْصَنَتْ فَرْجَهَا فَنَفَخْنَا فِيهَا مِن رُّوحِنَا وَجَعَلْنَـٰهَا وَٱبْنَهَآ ءَايَةً لِّلْعَـٰلَمِينَ ۝٩١

    اور وه پاک دامن بی بی جس نے اپنی عصمت کی حفاﻇت کی ہم نے اس کے اندر روح سے پھونک دی اور خود انہیں اور ان کے لڑکے کو تمام جہان کے لئے نشانی بنا دیا

  92. إِنَّ هَـٰذِهِۦٓ أُمَّتُكُمْ أُمَّةً وَٲحِدَةً وَأَنَا۟ رَبُّكُمْ فَٱعْبُدُونِ ۝٩٢

    یہ تمہاری امت ہے جو حقیقت میں ایک ہی امت ہے، اور میں تم سب کا پروردگار ہوں پس تم میری ہی عبادت کرو

  93. وَتَقَطَّعُوٓاْ أَمْرَهُم بَيْنَهُمْ‌ۖ كُلٌّ إِلَيْنَا رَٲجِعُونَ ۝٩٣

    مگر لوگوں نےآپس میں اپنے دین میں فرقہ بندیاں کر لیں، سب کے سب ہماری ہی طرف لوٹنے والے ہیں

  94. فَمَن يَعْمَلْ مِنَ ٱلصَّـٰلِحَـٰتِ وَهُوَ مُؤْمِنٌ فَلَا كُفْرَانَ لِسَعْيِهِۦ وَإِنَّا لَهُۥ كَـٰتِبُونَ ۝٩٤

    پھر جو بھی نیک عمل کرے اور وه مومن (بھی) ہو تو اس کی کوشش کی بے قدری نہیں کی جائے گی۔ ہم تو اس کے لکھنے والے ہیں

  95. وَحَرَٲمٌ عَلَىٰ قَرْيَةٍ أَهْلَكْنَـٰهَآ أَنَّهُمْ لَا يَرْجِعُونَ ۝٩٥

    اور جس بستی کو ہم نے ہلاک کر دیا اس پر ﻻزم ہے کہ وہاں کے لوگ پلٹ کر نہیں آئیں گے

  96. حَتَّىٰٓ إِذَا فُتِحَتْ يَأْجُوجُ وَمَأْجُوجُ وَهُم مِّن كُلِّ حَدَبٍ يَنسِلُونَ ۝٩٦

    یہاں تک کہ یاجوج اور ماجوج کھول دیئے جائیں گے اور وه ہر بلندی سے دوڑتے ہوئے آئیں گے

  97. وَٱقْتَرَبَ ٱلْوَعْدُ ٱلْحَقُّ فَإِذَا هِىَ شَـٰخِصَةٌ أَبْصَـٰرُ ٱلَّذِينَ كَفَرُواْ يَـٰوَيْلَنَا قَدْ كُنَّا فِى غَفْلَةٍ مِّنْ هَـٰذَا بَلْ كُنَّا ظَـٰلِمِينَ ۝٩٧

    اور سچا وعده قریب آلگے گا اس وقت کافروں کی نگاہیں پھٹی کی پھٹی ره جائیں گی، کہ ہائے افسوس! ہم اس حال سے غافل تھے بلکہ فی الواقع ہم قصور وار تھے

  98. إِنَّكُمْ وَمَا تَعْبُدُونَ مِن دُونِ ٱللَّهِ حَصَبُ جَهَنَّمَ أَنتُمْ لَهَا وَٲرِدُونَ ۝٩٨

    تم اور اللہ کے سوا جن جن کی تم عبادت کرتے ہو، سب دوزخ کا ایندھن بنو گے، تم سب دوزخ میں جانے والے ہو

  99. لَوْ كَانَ هَـٰٓؤُلَآءِ ءَالِهَةً مَّا وَرَدُوهَا‌ۖ وَكُلٌّ فِيهَا خَـٰلِدُونَ ۝٩٩

    اگر یہ (سچے) معبود ہوتے تو جہنم میں داخل نہ ہوتے، اور سب کے سب اسی میں ہمیشہ رہنے والے ہیں

  100. لَهُمْ فِيهَا زَفِيرٌ وَهُمْ فِيهَا لَا يَسْمَعُونَ ۝١٠٠

    وه وہاں چلا رہے ہوں گے اور وہاں کچھ بھی نہ سن سکیں گے

  101. إِنَّ ٱلَّذِينَ سَبَقَتْ لَهُم مِّنَّا ٱلْحُسْنَىٰٓ أُوْلَـٰٓئِكَ عَنْهَا مُبْعَدُونَ ۝١٠١

    البتہ بے شک جن کے لئے ہماری طرف سے نیکی پہلے ہی ٹھہر چکی ہے۔ وه سب جہنم سے دور ہی رکھے جائیں گے

  102. لَا يَسْمَعُونَ حَسِيسَهَا‌ۖ وَهُمْ فِى مَا ٱشْتَهَتْ أَنفُسُهُمْ خَـٰلِدُونَ ۝١٠٢

    وه تو دوزخ کی آہٹ تک نہ سنیں گے اور اپنی من بھاتی چیزوں میں ہمیشہ رہنے والے ہوں گے

  103. لَا يَحْزُنُهُمُ ٱلْفَزَعُ ٱلْأَكْبَرُ وَتَتَلَقَّـٰهُمُ ٱلْمَلَـٰٓئِكَةُ هَـٰذَا يَوْمُكُمُ ٱلَّذِى كُنتُمْ تُوعَدُونَ ۝١٠٣

    وه بڑی گھبراہٹ (بھی) انہیں غمگین نہ کر سکے گی اور فرشتے انہیں ہاتھوں ہاتھ لیں گے، کہ یہی تمہارا وه دن ہے جس کا تم وعده دیئے جاتے رہے

  104. يَوْمَ نَطْوِى ٱلسَّمَآءَ كَطَىِّ ٱلسِّجِلِّ لِلْكُتُبِ‌ۚ كَمَا بَدَأْنَآ أَوَّلَ خَلْقٍ نُّعِيدُهُۥ‌ۚ وَعْدًا عَلَيْنَآ‌ۚ إِنَّا كُنَّا فَـٰعِلِينَ ۝١٠٤

    جس دن ہم آسمان کو یوں لپیٹ لیں گے جیسے طومار میں اوراق لپیٹ دیئے جاتے ہیں، جیسے کہ ہم نے اول دفعہ پیدائش کی تھی اسی طرح دوباره کریں گے۔ یہ ہمارے ذمے وعده ہے اور ہم اسے ضرور کر کے (ہی) رہیں گے

  105. وَلَقَدْ كَتَبْنَا فِى ٱلزَّبُورِ مِنۢ بَعْدِ ٱلذِّكْرِ أَنَّ ٱلْأَرْضَ يَرِثُهَا عِبَادِىَ ٱلصَّـٰلِحُونَ ۝١٠٥

    ہم زبور میں پند ونصیحت کے بعد یہ لکھ چکے ہیں کہ زمین کے وارث میرے نیک بندے (ہی) ہوں گے

  106. إِنَّ فِى هَـٰذَا لَبَلَـٰغًا لِّقَوْمٍ عَـٰبِدِينَ ۝١٠٦

    عبادت گزار بندوں کے لئے تو اس میں ایک بڑا پیغام ہے

  107. وَمَآ أَرْسَلْنَـٰكَ إِلَّا رَحْمَةً لِّلْعَـٰلَمِينَ ۝١٠٧

    اور ہم نے آپ کو تمام جہان والوں کے لئے رحمت بنا کر ہی بھیجا ہے

  108. قُلْ إِنَّمَا يُوحَىٰٓ إِلَىَّ أَنَّمَآ إِلَـٰهُكُمْ إِلَـٰهٌ وَٲحِدٌ‌ۖ فَهَلْ أَنتُم مُّسْلِمُونَ ۝١٠٨

    کہہ دیجئے! میرے پاس تو پس وحی کی جاتی ہے کہ تم سب کا معبود ایک ہی ہے، تو کیا تم بھی اس کی فرمانبرداری کرنے والے ہو؟

  109. فَإِن تَوَلَّوْاْ فَقُلْ ءَاذَنتُكُمْ عَلَىٰ سَوَآءٍ‌ۖ وَإِنْ أَدْرِىٓ أَقَرِيبٌ أَم بَعِيدٌ مَّا تُوعَدُونَ ۝١٠٩

    پھر اگر یہ منھ موڑ لیں تو کہہ دیجئے کہ میں نے تمہیں یکساں طور پر خبردار کر دیا ہے۔ مجھے علم نہیں کہ جس کا وعده تم سے کیا جا رہا ہے وه قریب ہے یا دور

  110. إِنَّهُۥ يَعْلَمُ ٱلْجَهْرَ مِنَ ٱلْقَوْلِ وَيَعْلَمُ مَا تَكْتُمُونَ ۝١١٠

    البتہ اللہ تعالیٰ تو کھلی اور ﻇاہر بات کو بھی جانتا ہے اور جو تم چھپاتے ہو اسے بھی جانتا ہے

  111. وَإِنْ أَدْرِى لَعَلَّهُۥ فِتْنَةٌ لَّكُمْ وَمَتَـٰعٌ إِلَىٰ حِينٍ ۝١١١

    مجھے اس کا بھی علم نہیں، ممکن ہے یہ تمہاری آزمائش ہو اور ایک مقرره وقت تک کا فائده (پہنچانا) ہے

  112. قَـٰلَ رَبِّ ٱحْكُم بِٱلْحَقِّ‌ۗ وَرَبُّنَا ٱلرَّحْمَـٰنُ ٱلْمُسْتَعَانُ عَلَىٰ مَا تَصِفُونَ ۝١١٢

    خود نبی نے کہا اے رب! انصاف کے ساتھ فیصلہ فرما اور ہمارا رب بڑا مہربان ہے جس سے مدد طلب کی جاتی ہے ان باتوں پر جو تم بیان کرتے ہو
