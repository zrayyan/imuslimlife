بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ

  1. ٱلْحَمْدُ لِلَّهِ ٱلَّذِى لَهُۥ مَا فِى ٱلسَّمَـٰوَٲتِ وَمَا فِى ٱلْأَرْضِ وَلَهُ ٱلْحَمْدُ فِى ٱلْأَخِرَةِۚ وَهُوَ ٱلْحَكِيمُ ٱلْخَبِيرُ ۝١

    تمام تعریفیں اس اللہ کے لئے سزاوار ہیں جس کی ملکیت میں وه سب کچھ ہے جو آسمانوں اور زمین میں ہے آخرت میں بھی تعریف اسی کے لئے ہے، وه (بڑی) حکمتوں واﻻ اور (پورا) خبردار ہے

  2. يَعْلَمُ مَا يَلِجُ فِى ٱلْأَرْضِ وَمَا يَخْرُجُ مِنْهَا وَمَا يَنزِلُ مِنَ ٱلسَّمَآءِ وَمَا يَعْرُجُ فِيهَاۚ وَهُوَ ٱلرَّحِيمُ ٱلْغَفُورُ ۝٢

    جو زمین میں جائے اور جو اس سے نکلے جو آسمان سے اترے اور جو چڑھ کر اس میں جائے وه سب سے باخبر ہے۔ اور وه مہربان نہایت بخشش واﻻ ہے

  3. وَقَالَ ٱلَّذِينَ كَفَرُواْ لَا تَأْتِينَا ٱلسَّاعَةُۖ قُلْ بَلَىٰ وَرَبِّى لَتَأْتِيَنَّكُمْ عَـٰلِمِ ٱلْغَيْبِۖ لَا يَعْزُبُ عَنْهُ مِثْقَالُ ذَرَّةٍ فِى ٱلسَّمَـٰوَٲتِ وَلَا فِى ٱلْأَرْضِ وَلَآ أَصْغَرُ مِن ذَٲلِكَ وَلَآ أَكْبَرُ إِلَّا فِى كِتَـٰبٍ مُّبِينٍ ۝٣

    کفار کہتے ہیں کہ ہم پر قیامت نہیں آئے گی۔ آپ کہہ دیجیئے! کہ مجھے میرے رب کی قسم! جو عالم الغیب ہے کہ وه یقیناً تم پر آئے گی اللہ تعالیٰ سے ایک ذرے کے برابر کی چیز بھی پوشیده نہیں نہ آسمانوں میں اور نہ زمین میں بلکہ اس سے بھی چھوٹی اور بڑی چیز کھلی کتاب میں موجود ہے

  4. لِّيَجْزِىَ ٱلَّذِينَ ءَامَنُواْ وَعَمِلُواْ ٱلصَّـٰلِحَـٰتِۚ أُوْلَـٰٓئِكَ لَهُم مَّغْفِرَةٌ وَرِزْقٌ كَرِيمٌ ۝٤

    تاکہ وه ایمان والوں اور نیکوکاروں کو بھلائی عطا فرمائے، یہی لوگ ہیں جن کے لئے مغفرت اور عزت کی روزی ہے

  5. وَٱلَّذِينَ سَعَوْ فِىٓ ءَايَـٰتِنَا مُعَـٰجِزِينَ أُوْلَـٰٓئِكَ لَهُمْ عَذَابٌ مِّن رِّجْزٍ أَلِيمٌ ۝٥

    اور ہماری آیتوں کو نیچا دکھانے کی جنہوں نے کوشش کی ہے یہ وه لوگ ہیں جن کے لئے بدترین قسم کا دردناک عذاب ہے

  6. وَيَرَى ٱلَّذِينَ أُوتُواْ ٱلْعِلْمَ ٱلَّذِىٓ أُنزِلَ إِلَيْكَ مِن رَّبِّكَ هُوَ ٱلْحَقَّ وَيَهْدِىٓ إِلَىٰ صِرَٲطِ ٱلْعَزِيزِ ٱلْحَمِيدِ ۝٦

    اور جنہیں علم ہے وه دیکھ لیں گے کہ جو کچھ آپ کی جانب آپ کے رب کی طرف سے نازل ہوا ہے وه (سراسر) حق ہے اور اللہ غالب خوبیوں والے کی راه کی رہبری کرتا ہے

  7. وَقَالَ ٱلَّذِينَ كَفَرُواْ هَلْ نَدُلُّكُمْ عَلَىٰ رَجُلٍ يُنَبِّئُكُمْ إِذَا مُزِّقْتُمْ كُلَّ مُمَزَّقٍ إِنَّكُمْ لَفِى خَلْقٍ جَدِيدٍ ۝٧

    اور کافروں نے کہا (آؤ) ہم تمہیں ایک ایسا شخص بتلائیں جو تمہیں یہ خبر پہنچا رہا ہے کہ جب تم بالکل ہی ریزه ریزه ہوجاؤ گے تو تم پھر سے ایک نئی پیدائش میں آؤ گے

  8. أَفْتَرَىٰ عَلَى ٱللَّهِ كَذِبًا أَم بِهِۦ جِنَّۗةُۢ بَلِ ٱلَّذِينَ لَا يُؤْمِنُونَ بِٱلْأَخِرَةِ فِى ٱلْعَذَابِ وَٱلضَّلَـٰلِ ٱلْبَعِيدِ ۝٨

    (ہم نہیں کہہ سکتے) کہ خود اس نے (ہی) اللہ پر جھوٹ باندھ لیا ہے یا اسے دیوانگی ہے بلکہ (حقیقت یہ ہے) کہ آخرت پر یقین نہ رکھنے والے ہی عذاب میں اور دور کی گمراہی میں ہیں

  9. أَفَلَمْ يَرَوْاْ إِلَىٰ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُم مِّنَ ٱلسَّمَآءِ وَٱلْأَرْضِۚ إِن نَّشَأْ نَخْسِفْ بِهِمُ ٱلْأَرْضَ أَوْ نُسْقِطْ عَلَيْهِمْ كِسَفًا مِّنَ ٱلسَّمَآءِۚ إِنَّ فِى ذَٲلِكَ لَأَيَةً لِّكُلِّ عَبْدٍ مُّنِيبٍ ۝٩

    کیا پس وه اپنے آگے پیچھے آسمان وزمین کو دیکھ نہیں رہے ہیں؟ اگر ہم چاہیں تو انہیں زمین میں دھنسا دیں یا ان پر آسمان کے ٹکڑے گرادیں، یقیناً اس میں پوری دلیل ہے ہر اس بندے کے لئے جو (دل سے) متوجہ ہو

  10. ۞ وَلَقَدْ ءَاتَيْنَا دَاوُۥدَ مِنَّا فَضْلاًۖ يَـٰجِبَالُ أَوِّبِى مَعَهُۥ وَٱلطَّيْرَۖ وَأَلَنَّا لَهُ ٱلْحَدِيدَ ۝١٠

    اور ہم نے داؤد پر اپنا فضل کیا، اے پہاڑو! اس کے ساتھ رغبت سے تسبیح پڑھا کرو اور پرندوں کو بھی (یہی حکم ہے) اور ہم نے اس کے لئے لوہا نرم کردیا

  11. أَنِ ٱعْمَلْ سَـٰبِغَـٰتٍ وَقَدِّرْ فِى ٱلسَّرْدِۖ وَٱعْمَلُواْ صَـٰلِحًاۖ إِنِّى بِمَا تَعْمَلُونَ بَصِيرٌ ۝١١

    کہ تو پوری پوری زرہیں بنا اور جوڑوں میں اندازه رکھ تم سب نیک کام کیا کرو۔ (یقین مانو) کہ میں تمہارے اعمال دیکھ رہا ہوں

  12. وَلِسُلَيْمَـٰنَ ٱلرِّيحَ غُدُوُّهَا شَهْرٌ وَرَوَاحُهَا شَهْرٌۖ وَأَسَلْنَا لَهُۥ عَيْنَ ٱلْقِطْرِۖ وَمِنَ ٱلْجِنِّ مَن يَعْمَلُ بَيْنَ يَدَيْهِ بِإِذْنِ رَبِّهِۖۦ وَمَن يَزِغْ مِنْهُمْ عَنْ أَمْرِنَا نُذِقْهُ مِنْ عَذَابِ ٱلسَّعِيرِ ۝١٢

    اور ہم نے سلیمان کے لئے ہوا کو مسخر کردیا کہ صبح کی منزل اس کی مہینہ بھر کی ہوتی تھی اور شام کی منزل بھی اور ہم نے ان کے لئے تانبے کا چشمہ بہا دیا۔ اور اس کے رب کے حکم سے بعض جنات اس کی ماتحتی میں اس کے سامنے کام کرتے تھے اور ان میں سے جو بھی ہمارے حکم سے سرتابی کرے ہم اسے بھڑکتی ہوئی آگ کے عذاب کا مزه چکھائیں گے

  13. يَعْمَلُونَ لَهُۥ مَا يَشَآءُ مِن مَّحَـٰرِيبَ وَتَمَـٰثِيلَ وَجِفَانٍ كَٱلْجَوَابِ وَقُدُورٍ رَّاسِيَـٰتٍۚ ٱعْمَلُوٓاْ ءَالَ دَاوُۥدَ شُكْرًاۚ وَقَلِيلٌ مِّنْ عِبَادِىَ ٱلشَّكُورُ ۝١٣

    جو کچھ سلیمان چاہتے وه جنات تیار کردیتے مثلا قلعے اور مجسمے اور حوضوں کے برابر لگن اور چولہوں پر جمی ہوئی مضبوط دیگیں، اے آل داؤد اس کے شکریہ میں نیک عمل کرو، میرے بندوں میں سے شکرگزار بندے کم ہی ہوتے ہیں

  14. فَلَمَّا قَضَيْنَا عَلَيْهِ ٱلْمَوْتَ مَا دَلَّهُمْ عَلَىٰ مَوْتِهِۦٓ إِلَّا دَآبَّةُ ٱلْأَرْضِ تَأْكُلُ مِنسَأَتَهُۖۥ فَلَمَّا خَرَّ تَبَيَّنَتِ ٱلْجِنُّ أَن لَّوْ كَانُواْ يَعْلَمُونَ ٱلْغَيْبَ مَا لَبِثُواْ فِى ٱلْعَذَابِ ٱلْمُهِينِ ۝١٤

    پھر جب ہم نے ان پر موت کا حکم بھیج دیا تو ان کی خبر جنات کو کسی نے نہ دی سوائے گھن کے کیڑے کے جو ان کی عصا کو کھا رہا تھا۔ پس جب (سلیمان) گر پڑے اس وقت جنوں نے جان لیا کہ اگر وه غیب دان ہوتے تو اس کے ذلت کے عذاب میں مبتلا نہ رہتے

  15. لَقَدْ كَانَ لِسَبَإٍ فِى مَسْكَنِهِمْ ءَايَةٌۖ جَنَّتَانِ عَن يَمِينٍ وَشِمَالٍۖ كُلُواْ مِن رِّزْقِ رَبِّكُمْ وَٱشْكُرُواْ لَهُۚۥ بَلْدَةٌ طَيِّبَةٌ وَرَبٌّ غَفُورٌ ۝١٥

    قوم سبا کے لئے اپنی بستیوں میں (قدرت الٰہی کی) نشانی تھی ان کے دائیں بائیں دو باغ تھے (ہم نے ان کو حکم دیا تھا کہ) اپنے رب کی دی ہوئی روزی کھاؤ اور اس کا شکر ادا کرو، یہ عمده شہر اور وه بخشنے واﻻ رب ہے

  16. فَأَعْرَضُواْ فَأَرْسَلْنَا عَلَيْهِمْ سَيْلَ ٱلْعَرِمِ وَبَدَّلْنَـٰهُم بِجَنَّتَيْهِمْ جَنَّتَيْنِ ذَوَاتَىْ أُكُلٍ خَمْطٍ وَأَثْلٍ وَشَىْءٍ مِّن سِدْرٍ قَلِيلٍ ۝١٦

    لیکن انہوں نے روگردانی کی تو ہم نے ان پر زور کے سیلاب (کا پانی) بھیج دیا اور ہم نے ان کے (ہرے بھرے) باغوں کے بدلے دو (ایسے) باغ دیئے جو بدمزه میوؤں والے اور (بکثرت) جھاؤ اور کچھ بیری کے درختوں والے تھے

  17. ذَٲلِكَ جَزَيْنَـٰهُم بِمَا كَفَرُواْۖ وَهَلْ نُجَـٰزِىٓ إِلَّا ٱلْكَفُورَ ۝١٧

    ہم نے ان کی ناشکری کا یہ بدلہ انہیں دیا۔ ہم (ایسی) سخت سزا بڑے بڑے ناشکروں ہی کو دیتے ہیں

  18. وَجَعَلْنَا بَيْنَهُمْ وَبَيْنَ ٱلْقُرَى ٱلَّتِى بَـٰرَكْنَا فِيهَا قُرًى ظَـٰهِرَةً وَقَدَّرْنَا فِيهَا ٱلسَّيْرَۖ سِيرُواْ فِيهَا لَيَالِىَ وَأَيَّامًا ءَامِنِينَ ۝١٨

    اور ہم نے ان کے اور ان بستیوں کے درمیان جن میں ہم نے برکت دے رکھی تھی چند بستیاں اور (آباد) رکھی تھیں جو برسرراه ﻇاہر تھیں، اور ان میں چلنے کی منزلیں مقرر کردی تھیں ان میں راتوں اور دنوں کو بہ امن وامان چلتے پھرتے رہو

  19. فَقَالُواْ رَبَّنَا بَـٰعِدْ بَيْنَ أَسْفَارِنَا وَظَلَمُوٓاْ أَنفُسَهُمْ فَجَعَلْنَـٰهُمْ أَحَادِيثَ وَمَزَّقْنَـٰهُمْ كُلَّ مُمَزَّقٍۚ إِنَّ فِى ذَٲلِكَ لَأَيَـٰتٍ لِّكُلِّ صَبَّارٍ شَكُورٍ ۝١٩

    لیکن انہوں نے پھر کہا کہ اے ہمارے پروردگار! ہمارے سفر دور دراز کردے چونکہ خود انہوں نے اپنے ہاتھوں اپنا برا کیا اس لئے ہم نے انہیں (گزشتہ) فسانوں کی صورت میں کردیا اور ان کے ٹکڑے ٹکڑے اڑا دیئے۔ بلاشبہ ہر ایک صبروشکر کرنے والے کے لئے اس (ماجرے) میں بہت سی عبرتیں ہیں

  20. وَلَقَدْ صَدَّقَ عَلَيْهِمْ إِبْلِيسُ ظَنَّهُۥ فَٱتَّبَعُوهُ إِلَّا فَرِيقًا مِّنَ ٱلْمُؤْمِنِينَ ۝٢٠

    اور شیطان نے ان کے بارے میں اپنا گمان سچا کر دکھایا یہ لوگ سب کے سب اس کے تابعدار بن گئے سوائے مومنوں کی ایک جماعت کے

  21. وَمَا كَانَ لَهُۥ عَلَيْهِم مِّن سُلْطَـٰنٍ إِلَّا لِنَعْلَمَ مَن يُؤْمِنُ بِٱلْأَخِرَةِ مِمَّنْ هُوَ مِنْهَا فِى شَكٍّۗ وَرَبُّكَ عَلَىٰ كُلِّ شَىْءٍ حَفِيظٌ ۝٢١

    شیطان کا ان پر کوئی زور (اور دباؤ) نہ تھا مگر اس لئے کہ ہم ان لوگوں کو جو آخرت پر ایمان رکھتے ہیں ﻇاہر کردیں ان لوگوں میں سے جو اس سے شک میں ہیں۔ اور آپ کا رب (ہر) ہر چیز پر نگہبان ہے

  22. قُلِ ٱدْعُواْ ٱلَّذِينَ زَعَمْتُم مِّن دُونِ ٱللَّهِۖ لَا يَمْلِكُونَ مِثْقَالَ ذَرَّةٍ فِى ٱلسَّمَـٰوَٲتِ وَلَا فِى ٱلْأَرْضِ وَمَا لَهُمْ فِيهِمَا مِن شِرْكٍ وَمَا لَهُۥ مِنْهُم مِّن ظَهِيرٍ ۝٢٢

    کہہ دیجیئے! کہ اللہ کے سوا جن جن کا تمہیں گمان ہے (سب) کو پکار لو، نہ ان میں سے کسی کو آسمانوں اور زمینوں میں سے ایک ذره کا اختیار ہے نہ ان کا ان میں کوئی حصہ ہے نہ ان میں سے کوئی اللہ کا مددگار ہے

  23. وَلَا تَنفَعُ ٱلشَّفَـٰعَةُ عِندَهُۥٓ إِلَّا لِمَنْ أَذِنَ لَهُۚۥ حَتَّىٰٓ إِذَا فُزِّعَ عَن قُلُوبِهِمْ قَالُواْ مَاذَا قَالَ رَبُّكُمْۖ قَالُواْ ٱلْحَقَّۖ وَهُوَ ٱلْعَلِىُّ ٱلْكَبِيرُ ۝٢٣

    شفاعت (سفارش) بھی اس کے پاس کچھ نفع نہیں دیتی بجز ان کے جن کے لئے اجازت ہوجائے۔ یہاں تک کہ جب ان کے دلوں سے گھبراہٹ دور کردی جاتی ہے تو پوچھتے ہیں تمہارے پرودگار نے کیا فرمایا؟ جواب دیتے ہیں کہ حق فرمایا اور وه بلند وباﻻ اور بہت بڑا ہے

  24. ۞ قُلْ مَن يَرْزُقُكُم مِّنَ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِۖ قُلِ ٱللَّهُۖ وَإِنَّآ أَوْ إِيَّاكُمْ لَعَلَىٰ هُدًى أَوْ فِى ضَلَـٰلٍ مُّبِينٍ ۝٢٤

    پوچھیئے کہ تمہیں آسمانوں اور زمین سے روزی کون پہنچاتا ہے؟ (خود) جواب دیجیئے! کہ اللہ تعالیٰ۔ (سنو) ہم یا تم۔ یا تو یقیناً ہدایت پر ہیں یا کھلی گمراہی میں ہیں؟

  25. قُل لَّا تُسْــَٔلُونَ عَمَّآ أَجْرَمْنَا وَلَا نُسْــَٔلُ عَمَّا تَعْمَلُونَ ۝٢٥

    کہہ دیجیئے! کہ ہمارے کئے ہوئے گناہوں کی بابت تم سے کوئی سوال نہ کیا جائے گا نہ تمہارے اعمال کی بازپرس ہم سے کی جائے گی

  26. قُلْ يَجْمَعُ بَيْنَنَا رَبُّنَا ثُمَّ يَفْتَحُ بَيْنَنَا بِٱلْحَقِّ وَهُوَ ٱلْفَتَّاحُ ٱلْعَلِيمُ ۝٢٦

    انہیں خبر دے دیجیئے کہ ہم سب کو ہمارا رب جمع کرکے پھر ہم میں سچے فیصلے کردے گا۔ وه فیصلے چکانے واﻻ ہے اور دانا

  27. قُلْ أَرُونِىَ ٱلَّذِينَ أَلْحَقْتُم بِهِۦ شُرَكَآءَۖ كَلَّاۚ بَلْ هُوَ ٱللَّهُ ٱلْعَزِيزُ ٱلْحَكِيمُ ۝٢٧

    کہہ دیجیئے! کہ اچھا مجھے بھی تو انہیں دکھا دو جنہیں تم اللہ کا شریک ٹھہرا کر اس کے ساتھ ملا رہے ہو، ایسا ہرگز نہیں، بلکہ وہی اللہ ہے غالب باحکمت

  28. وَمَآ أَرْسَلْنَـٰكَ إِلَّا كَآفَّةً لِّلنَّاسِ بَشِيرًا وَنَذِيرًا وَلَـٰكِنَّ أَكْثَرَ ٱلنَّاسِ لَا يَعْلَمُونَ ۝٢٨

    ہم نے آپ کو تمام لوگوں کے لئے خوشخبریاں سنانے واﻻ اور ڈرانے واﻻ بنا کر بھیجا ہے ہاں مگر (یہ صحیح ہے) کہ لوگوں کی اکثریت بےعلم ہے

  29. وَيَقُولُونَ مَتَىٰ هَـٰذَا ٱلْوَعْدُ إِن كُنتُمْ صَـٰدِقِينَ ۝٢٩

    پوچھتے ہیں کہ وه وعده ہے کب؟ سچے ہو تو بتا دو

  30. قُل لَّكُم مِّيعَادُ يَوْمٍ لَّا تَسْتَــْٔخِرُونَ عَنْهُ سَاعَةً وَلَا تَسْتَقْدِمُونَ ۝٣٠

    جواب دیجیئے کہ وعدے کا دن ٹھیک معین ہے جس سے ایک ساعت نہ تم پیچھے ہٹ سکتے ہو نہ آگے بڑھ سکتے ہو

  31. وَقَالَ ٱلَّذِينَ كَفَرُواْ لَن نُّؤْمِنَ بِهَـٰذَا ٱلْقُرْءَانِ وَلَا بِٱلَّذِى بَيْنَ يَدَيْهِۗ وَلَوْ تَرَىٰٓ إِذِ ٱلظَّـٰلِمُونَ مَوْقُوفُونَ عِندَ رَبِّهِمْ يَرْجِعُ بَعْضُهُمْ إِلَىٰ بَعْضٍ ٱلْقَوْلَ يَقُولُ ٱلَّذِينَ ٱسْتُضْعِفُواْ لِلَّذِينَ ٱسْتَكْبَرُواْ لَوْلَآ أَنتُمْ لَكُنَّا مُؤْمِنِينَ ۝٣١

    اور کافروں نے کہا کہ ہم ہرگز نہ تو اس قرآن کو مانیں نہ اس سے پہلے کی کتابوں کو! اے دیکھنے والے کاش کہ تو ان ﻇالموں کو اس وقت دیکھتا جب کہ یہ اپنے رب کے سامنے کھڑے ہوئے ایک دوسرے کو الزام دے رہے ہوں گے کمزور لوگ بڑے لوگوں سے کہیں گے اگر تم نہ ہوتے تو ہم تو مومن ہوتے

  32. قَالَ ٱلَّذِينَ ٱسْتَكْبَرُواْ لِلَّذِينَ ٱسْتُضْعِفُوٓاْ أَنَحْنُ صَدَدْنَـٰكُمْ عَنِ ٱلْهُدَىٰ بَعْدَ إِذْ جَآءَكُمۖ بَلْ كُنتُم مُّجْرِمِينَ ۝٣٢

    یہ بڑے لوگ ان کمزوروں کو جواب دیں گے کہ کیا تمہارے پاس ہدایت آچکنے کے بعد ہم نے تمہیں اس سے روکا تھا؟ (نہیں) بلکہ تم (خود) ہی مجرم تھے

  33. وَقَالَ ٱلَّذِينَ ٱسْتُضْعِفُواْ لِلَّذِينَ ٱسْتَكْبَرُواْ بَلْ مَكْرُ ٱلَّيْلِ وَٱلنَّهَارِ إِذْ تَأْمُرُونَنَآ أَن نَّكْفُرَ بِٱللَّهِ وَنَجْعَلَ لَهُۥٓ أَندَادًاۚ وَأَسَرُّواْ ٱلنَّدَامَةَ لَمَّا رَأَوُاْ ٱلْعَذَابَ وَجَعَلْنَا ٱلْأَغْلَـٰلَ فِىٓ أَعْنَاقِ ٱلَّذِينَ كَفَرُواْۚ هَلْ يُجْزَوْنَ إِلَّا مَا كَانُواْ يَعْمَلُونَ ۝٣٣

    (اس کے جواب میں) یہ کمزور لوگ ان متکبروں سے کہیں گے، (نہیں نہیں) بلکہ دن رات مکروفریب سے ہمیں اللہ کے ساتھ کفر کرنے اور اس کے شریک مقرر کرنے کا تمہارا حکم دینا ہماری بےایمانی کا باعﺚ ہوا، اور عذاب کو دیکھتے ہی سب کے سب دل میں پشیمان ہو رہے ہوں گے، اور کافروں کی گردنوں میں ہم طوق ڈال دیں گے انہیں صرف ان کے کئے کرائے اعمال کا بدلہ دیا جائے گا

  34. وَمَآ أَرْسَلْنَا فِى قَرْيَةٍ مِّن نَّذِيرٍ إِلَّا قَالَ مُتْرَفُوهَآ إِنَّا بِمَآ أُرْسِلْتُم بِهِۦ كَـٰفِرُونَ ۝٣٤

    اور ہم تو جس بستی میں جو بھی آگاه کرنے واﻻ بھیجا وہاں کے خوشحال لوگوں نے یہی کہا کہ جس چیز کے ساتھ تم بھیجے گئے ہو ہم اس کے ساتھ کفر کرنے والے ہیں

  35. وَقَالُواْ نَحْنُ أَكْثَرُ أَمْوَٲلاً وَأَوْلَـٰدًا وَمَا نَحْنُ بِمُعَذَّبِينَ ۝٣٥

    اور کہا ہم مال واوﻻد میں بہت بڑھے ہوئے ہیں یہ نہیں ہوسکتا کہ ہم عذاب دیئے جائیں

  36. قُلْ إِنَّ رَبِّى يَبْسُطُ ٱلرِّزْقَ لِمَن يَشَآءُ وَيَقْدِرُ وَلَـٰكِنَّ أَكْثَرَ ٱلنَّاسِ لَا يَعْلَمُونَ ۝٣٦

    کہہ دیجیئے! کہ میرا رب جس کے لئے چاہے روزی کشاده کردیتا ہے اور تنگ بھی کردیتا ہے، لیکن اکثر لوگ نہیں جانتے

  37. وَمَآ أَمْوَٲلُكُمْ وَلَآ أَوْلَـٰدُكُم بِٱلَّتِى تُقَرِّبُكُمْ عِندَنَا زُلْفَىٰٓ إِلَّا مَنْ ءَامَنَ وَعَمِلَ صَـٰلِحًا فَأُوْلَـٰٓئِكَ لَهُمْ جَزَآءُ ٱلضِّعْفِ بِمَا عَمِلُواْ وَهُمْ فِى ٱلْغُرُفَـٰتِ ءَامِنُونَ ۝٣٧

    اور تمہارے مال اور اوﻻد ایسے نہیں کہ تمہیں ہمارے پاس (مرتبوں سے) قریب کردیں ہاں جو ایمان ﻻئیں اور نیک عمل کریں ان کے لئے ان کے اعمال کا دوہرا اجر ہے اور وه نڈر و بےخوف ہو کر باﻻ خانوں میں رہیں گے

  38. وَٱلَّذِينَ يَسْعَوْنَ فِىٓ ءَايَـٰتِنَا مُعَـٰجِزِينَ أُوْلَـٰٓئِكَ فِى ٱلْعَذَابِ مُحْضَرُونَ ۝٣٨

    اور جو لوگ ہماری آیتوں کے مقابلے کی تگ ودو میں لگے رہتے ہیں یہی ہیں جو عذاب میں پکڑ کر حاضر رکھے جائیں گے

  39. قُلْ إِنَّ رَبِّى يَبْسُطُ ٱلرِّزْقَ لِمَن يَشَآءُ مِنْ عِبَادِهِۦ وَيَقْدِرُ لَهُۥۚ وَمَآ أَنفَقْتُم مِّن شَىْءٍ فَهُوَ يُخْلِفُهُۖۥ وَهُوَ خَيْرُ ٱلرَّٲزِقِينَ ۝٣٩

    کہہ دیجیئے! کہ میرا رب اپنے بندوں میں جس کے لئے چاہے روزی کشاده کرتا ہے اور جس کے لئے چاہے تنگ کردیتا ہے، تم جو کچھ بھی اللہ کی راه میں خرچ کرو گے اللہ اس کا (پورا پورا) بدلہ دے گا اور وه سب سے بہتر روزی دینے واﻻ ہے

  40. وَيَوْمَ يَحْشُرُهُمْ جَمِيعًا ثُمَّ يَقُولُ لِلْمَلَـٰٓئِكَةِ أَهَـٰٓؤُلَآءِ إِيَّاكُمْ كَانُواْ يَعْبُدُونَ ۝٤٠

    اور ان سب کو اللہ اس دن جمع کرکے فرشتوں سے دریافت فرمائے گا کہ کیا یہ لوگ تمہاری عبادت کرتے تھے

  41. قَالُواْ سُبْحَـٰنَكَ أَنتَ وَلِيُّنَا مِن دُونِهِمۖ بَلْ كَانُواْ يَعْبُدُونَ ٱلْجِنَّۖ أَكْثَرُهُم بِهِم مُّؤْمِنُونَ ۝٤١

    وه کہیں گے تیری ذات پاک ہے اور ہمارا ولی تو تو ہے نہ کہ یہ بلکہ یہ لوگ جنوں کی عبادت کرتے تھے، ان میں کے اکثر کا ان ہی پر ایمان تھا

  42. فَٱلْيَوْمَ لَا يَمْلِكُ بَعْضُكُمْ لِبَعْضٍ نَّفْعًا وَلَا ضَرًّا وَنَقُولُ لِلَّذِينَ ظَلَمُواْ ذُوقُواْ عَذَابَ ٱلنَّارِ ٱلَّتِى كُنتُم بِهَا تُكَذِّبُونَ ۝٤٢

    پس آج تم میں سے کوئی (بھی) کسی کے لئے (بھی کسی قسم کے) نفع نقصان کا مالک نہ ہوگا۔ اور ہم ﻇالموں سے کہہ دیں گے کہ اس آگ کا عذاب چکھو جسے تم جھٹلاتے رہے

  43. وَإِذَا تُتْلَىٰ عَلَيْهِمْ ءَايَـٰتُنَا بَيِّنَـٰتٍ قَالُواْ مَا هَـٰذَآ إِلَّا رَجُلٌ يُرِيدُ أَن يَصُدَّكُمْ عَمَّا كَانَ يَعْبُدُ ءَابَآؤُكُمْ وَقَالُواْ مَا هَـٰذَآ إِلَّآ إِفْكٌ مُّفْتَرًىۚ وَقَالَ ٱلَّذِينَ كَفَرُواْ لِلْحَقِّ لَمَّا جَآءَهُمْ إِنْ هَـٰذَآ إِلَّا سِحْرٌ مُّبِينٌ ۝٤٣

    اور جب ان کے سامنے ہماری صاف صاف آیتیں پڑھی جاتی ہیں تو کہتے ہیں کہ یہ ایسا شخص ہے جو تمہیں تمہارے باپ دادا کے معبودوں سے روک دینا چاہتا ہے۔ (اس کے سوا کوئی بات نہیں)، اور کہتے ہیں کہ یہ تو گھڑا ہوا جھوٹ ہے اور حق ان کے پاس آچکا پھر بھی کافر یہی کہتے رہے کہ یہ تو کھلا ہوا جادو ہے

  44. وَمَآ ءَاتَيْنَـٰهُم مِّن كُتُبٍ يَدْرُسُونَهَاۖ وَمَآ أَرْسَلْنَآ إِلَيْهِمْ قَبْلَكَ مِن نَّذِيرٍ ۝٤٤

    اور ان (مکہ والوں) کو نہ تو ہم نے کتابیں دے رکھی ہیں جنہیں یہ پڑھتے ہوں نہ ان کے پاس آپ سے پہلے کوئی آگاه کرنے واﻻ آیا

  45. وَكَذَّبَ ٱلَّذِينَ مِن قَبْلِهِمْ وَمَا بَلَغُواْ مِعْشَارَ مَآ ءَاتَيْنَـٰهُمْ فَكَذَّبُواْ رُسُلِىۖ فَكَيْفَ كَانَ نَكِيرِ ۝٤٥

    اور ان سے پہلے کے لوگوں نے بھی ہماری باتوں کو جھٹلایا تھا اور انہیں ہم نے جو دے رکھا تھا یہ تو اس کے دسویں حصے کو بھی نہیں پہنچے، پس انہوں نے میرے رسولوں کو جھٹلایا، (پھر دیکھ کہ) میرا عذاب کیسا (سخت) تھا

  46. ۞ قُلْ إِنَّمَآ أَعِظُكُم بِوَٲحِدَةٍۖ أَن تَقُومُواْ لِلَّهِ مَثْنَىٰ وَفُرَٲدَىٰ ثُمَّ تَتَفَكَّرُواْۚ مَا بِصَاحِبِكُم مِّن جِنَّةٍۚ إِنْ هُوَ إِلَّا نَذِيرٌ لَّكُم بَيْنَ يَدَىْ عَذَابٍ شَدِيدٍ ۝٤٦

    کہہ دیجیئے! کہ میں تمہیں صرف ایک ہی بات کی نصیحت کرتا ہوں کہ تم اللہ کے واسطے (ضد چھوڑ کر) دو دو مل کر یا تنہا تنہا کھڑے ہو کر سوچو تو سہی، تمہارے اس رفیق کو کوئی جنون نہیں، وه تو تمہیں ایک بڑے (سخت) عذاب کے آنے سے پہلے ڈرانے واﻻ ہے

  47. قُلْ مَا سَأَلْتُكُم مِّنْ أَجْرٍ فَهُوَ لَكُمْۖ إِنْ أَجْرِىَ إِلَّا عَلَى ٱللَّهِۖ وَهُوَ عَلَىٰ كُلِّ شَىْءٍ شَهِيدٌ ۝٤٧

    کہہ دیجیئے! کہ جو بدلہ میں تم سے مانگوں وه تمہارے لئے ہے میرا بدلہ تو اللہ تعالیٰ ہی کے ذمے ہے۔ وه ہر چیز سے باخبر (اور مطلع) ہے

  48. قُلْ إِنَّ رَبِّى يَقْذِفُ بِٱلْحَقِّ عَلَّـٰمُ ٱلْغُيُوبِ ۝٤٨

    کہہ دیجیئے! کہ میرا رب حق (سچی وحی) نازل فرماتا ہے وه ہر غیب کا جاننے واﻻ ہے

  49. قُلْ جَآءَ ٱلْحَقُّ وَمَا يُبْدِئُ ٱلْبَـٰطِلُ وَمَا يُعِيدُ ۝٤٩

    کہہ دیجیئے کہ حق آچکا باطل نہ تو پہلے کچھ کرسکا ہے اور نہ کرسکے گا

  50. قُلْ إِن ضَلَلْتُ فَإِنَّمَآ أَضِلُّ عَلَىٰ نَفْسِىۖ وَإِنِ ٱهْتَدَيْتُ فَبِمَا يُوحِىٓ إِلَىَّ رَبِّىٓۚ إِنَّهُۥ سَمِيعٌ قَرِيبٌ ۝٥٠

    کہہ دیجیئے کہ اگر میں بہک جاؤں تو میرے بہکنے (کا وبال) مجھ پر ہی ہے اور اگر میں راه ہدایت پر ہوں تو بہ سبب اس وحی کے جو میرا پروردگار مجھے کرتا ہے وه بڑا ہی سننے واﻻ اور بہت ہی قریب ہے

  51. وَلَوْ تَرَىٰٓ إِذْ فَزِعُواْ فَلَا فَوْتَ وَأُخِذُواْ مِن مَّكَانٍ قَرِيبٍ ۝٥١

    اور اگر آپ (وه وقت) ملاحظہ کریں جب کہ یہ کفار گھبرائے پھریں گے پھر نکل بھاگنے کی کوئی صورت نہ ہوگی اور قریب کی جگہ سے گرفتار کر لئے جائیں گے

  52. وَقَالُوٓاْ ءَامَنَّا بِهِۦ وَأَنَّىٰ لَهُمُ ٱلتَّنَاوُشُ مِن مَّكَانِۭ بَعِيدٍ ۝٥٢

    اس وقت کہیں گے کہ ہم اس قرآن پر ایمان ﻻئے لیکن اس قدر دور جگہ سے (مطلوبہ چیز) کیسے ہاتھ آسکتی ہے

  53. وَقَدْ كَفَرُواْ بِهِۦ مِن قَبْلُۖ وَيَقْذِفُونَ بِٱلْغَيْبِ مِن مَّكَانِۭ بَعِيدٍ ۝٥٣

    اس سے پہلے تو انہوں نے اس سے کفر کیا تھا، اور دور دراز سے بن دیکھے ہی پھینکتے رہے

  54. وَحِيلَ بَيْنَهُمْ وَبَيْنَ مَا يَشْتَهُونَ كَمَا فُعِلَ بِأَشْيَاعِهِم مِّن قَبْلُۚ إِنَّهُمْ كَانُواْ فِى شَكٍّ مُّرِيبِۭ ۝٥٤

    ان کی چاہتوں اور ان کے درمیان پرده حائل کردیا گیا جیسے کہ اس سے پہلے بھی ان جیسوں کے ساتھ کیا گیا، وه بھی (ان ہی کی طرح) شک وتردد میں (پڑے ہوئے) تھے
