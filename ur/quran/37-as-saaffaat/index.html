بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ

  1. وَٱلصَّـٰٓفَّـٰتِ صَفًّا ۝١

    قسم ہے صف باندھنے والے (فرشتوں) کی

  2. فَٱلزَّٲجِرَٲتِ زَجْرًا ۝٢

    پھر پوری طرح ڈانٹنے والوں کی

  3. فَٱلتَّـٰلِيَـٰتِ ذِكْرًا ۝٣

    پھر ذکر اللہ کی تلاوت کرنے والوں کی

  4. إِنَّ إِلَـٰهَكُمْ لَوَٲحِدٌ ۝٤

    یقیناً تم سب کا معبود ایک ہی ہے

  5. رَّبُّ ٱلسَّمَـٰوَٲتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا وَرَبُّ ٱلْمَشَـٰرِقِ ۝٥

    آسمانوں اور زمین اور ان کے درمیان کی تمام چیزوں اور مَشرقوں کا رب وہی ہے

  6. إِنَّا زَيَّنَّا ٱلسَّمَآءَ ٱلدُّنْيَا بِزِينَةٍ ٱلْكَوَاكِبِ ۝٦

    ہم نے آسمان دنیا کو ستاروں کی زینت سے آراستہ کیا

  7. وَحِفْظًا مِّن كُلِّ شَيْطَـٰنٍ مَّارِدٍ ۝٧

    اور حفاﻇت کی سرکش شیطان سے

  8. لَّا يَسَّمَّعُونَ إِلَى ٱلْمَلَإِ ٱلْأَعْلَىٰ وَيُقْذَفُونَ مِن كُلِّ جَانِبٍ ۝٨

    عالم باﻻ کے فرشتوں (کی باتوں) کو سننے کے لئے وه کان بھی نہیں لگا سکتے، بلکہ ہر طرف سے وه مارے جاتے ہیں

  9. دُحُورًا‌ۖ وَلَهُمْ عَذَابٌ وَاصِبٌ ۝٩

    بھگانے کے لئے اور ان کے لئے دائمی عذاب ہے

  10. إِلَّا مَنْ خَطِفَ ٱلْخَطْفَةَ فَأَتْبَعَهُۥ شِهَابٌ ثَاقِبٌ ۝١٠

    مگر جو کوئی ایک آدھ بات اچک لے بھاگے تو (فوراً ہی) اس کے پیچھے دہکتا ہوا شعلہ لگ جاتا ہے

  11. فَٱسْتَفْتِهِمْ أَهُمْ أَشَدُّ خَلْقًا أَم مَّنْ خَلَقْنَآ‌ۚ إِنَّا خَلَقْنَـٰهُم مِّن طِينٍ لَّازِبِۭ ۝١١

    ان کافروں سے پوچھو تو کہ آیا ان کا پیدا کرنا زیاده دشوار ہے یا (ان کا) جنہیں ہم نے (ان کے علاوه) پیدا کیا؟ ہم نے (انسانوں) کو لیس دار مٹی سے پیدا کیا ہے

  12. بَلْ عَجِبْتَ وَيَسْخَرُونَ ۝١٢

    بلکہ تو تعجب کر رہا ہے اور یہ مسخرا پن کر رہے ہیں

  13. وَإِذَا ذُكِّرُواْ لَا يَذْكُرُونَ ۝١٣

    اور جب انہیں نصیحت کی جاتی ہے یہ نہیں مانتے

  14. وَإِذَا رَأَوْاْ ءَايَةً يَسْتَسْخِرُونَ ۝١٤

    اور جب کسی معجزے کو دیکھتے ہیں تو مذاق اڑاتے ہیں

  15. وَقَالُوٓاْ إِنْ هَـٰذَآ إِلَّا سِحْرٌ مُّبِينٌ ۝١٥

    اور کہتے ہیں کہ یہ تو بالکل کھلم کھلا جادو ہی ہے

  16. أَءِذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَـٰمًا أَءِنَّا لَمَبْعُوثُونَ ۝١٦

    کیا جب ہم مر جائیں گے اور خاک اور ہڈی ہو جائیں گے پھر کیا (سچ مچ) ہم اٹھائے جائیں گے؟

  17. أَوَءَابَآؤُنَا ٱلْأَوَّلُونَ ۝١٧

    کیا ہم سے پہلے کے ہمارے باپ دادا بھی؟

  18. قُلْ نَعَمْ وَأَنتُمْ دَٲخِرُونَ ۝١٨

    آپ جواب دیجئے! کہ ہاں ہاں اور تم ذلیل (بھی) ہوں گے

  19. فَإِنَّمَا هِىَ زَجْرَةٌ وَٲحِدَةٌ فَإِذَا هُمْ يَنظُرُونَ ۝١٩

    وه تو صرف ایک زور کی جھڑکی ہے کہ یکایک یہ دیکھنے لگیں گے

  20. وَقَالُواْ يَـٰوَيْلَنَا هَـٰذَا يَوْمُ ٱلدِّينِ ۝٢٠

    اور کہیں گے کہ ہائے ہماری خرابی یہی جزا (سزا) کا دن ہے

  21. هَـٰذَا يَوْمُ ٱلْفَصْلِ ٱلَّذِى كُنتُم بِهِۦ تُكَذِّبُونَ ۝٢١

    یہی فیصلہ کا دن ہے جسے تم جھٹلاتے رہے

  22. ۞ ٱحْشُرُواْ ٱلَّذِينَ ظَلَمُواْ وَأَزْوَٲجَهُمْ وَمَا كَانُواْ يَعْبُدُونَ ۝٢٢

    ﻇالموں کو اور ان کے ہمراہیوں کو اور (جن) جن کی وه اللہ کے علاوه پرستش کرتے تھے

  23. مِن دُونِ ٱللَّهِ فَٱهْدُوهُمْ إِلَىٰ صِرَٲطِ ٱلْجَحِيمِ ۝٢٣

    (ان سب کو) جمع کرکے انہیں دوزخ کی راه دکھا دو

  24. وَقِفُوهُمْ‌ۖ إِنَّهُم مَّسْــُٔولُونَ ۝٢٤

    اور انہیں ٹھہرا لو، (اس لئے) کہ ان سے (ضروری) سوال کیے جانے والے ہیں

  25. مَا لَكُمْ لَا تَنَاصَرُونَ ۝٢٥

    تمہیں کیا ہو گیا ہے کہ (اس وقت) تم ایک دوسرے کی مدد نہیں کرتے

  26. بَلْ هُمُ ٱلْيَوْمَ مُسْتَسْلِمُونَ ۝٢٦

    بلکہ وه (سب کے سب) آج فرمانبردار بن گئے

  27. وَأَقْبَلَ بَعْضُهُمْ عَلَىٰ بَعْضٍ يَتَسَآءَلُونَ ۝٢٧

    وه ایک دوسرے کی طرف متوجہ ہو کر سوال وجواب کرنے لگیں گے

  28. قَالُوٓاْ إِنَّكُمْ كُنتُمْ تَأْتُونَنَا عَنِ ٱلْيَمِينِ ۝٢٨

    کہیں گے کہ تم تو ہمارے پاس ہماری دائیں طرف سے آتے تھے

  29. قَالُواْ بَل لَّمْ تَكُونُواْ مُؤْمِنِينَ ۝٢٩

    وه جواب دیں گے کہ نہیں بلکہ تم ہی ایمان والے نہ تھے

  30. وَمَا كَانَ لَنَا عَلَيْكُم مِّن سُلْطَـٰنِۭ‌ۖ بَلْ كُنتُمْ قَوْمًا طَـٰغِينَ ۝٣٠

    اور کچھ ہمارا زور تو تم پر تھا (ہی) نہیں۔ بلکہ تم (خود) سرکش لوگ تھے

  31. فَحَقَّ عَلَيْنَا قَوْلُ رَبِّنَآ‌ۖ إِنَّا لَذَآئِقُونَ ۝٣١

    اب تو ہم (سب) پر ہمارے رب کی یہ بات ﺛابت ہو چکی کہ ہم (عذاب) چکھنے والے ہیں

  32. فَأَغْوَيْنَـٰكُمْ إِنَّا كُنَّا غَـٰوِينَ ۝٣٢

    پس ہم نے تمہیں گمراه کیا ہم تو خود بھی گمراه ہی تھے

  33. فَإِنَّهُمْ يَوْمَئِذٍ فِى ٱلْعَذَابِ مُشْتَرِكُونَ ۝٣٣

    سو اب آج کے دن تو (سب کے سب) عذاب میں شریک ہیں

  34. إِنَّا كَذَٲلِكَ نَفْعَلُ بِٱلْمُجْرِمِينَ ۝٣٤

    ہم گناه گاروں کے ساتھ اسی طرح کیا کرتے ہیں

  35. إِنَّهُمْ كَانُوٓاْ إِذَا قِيلَ لَهُمْ لَآ إِلَـٰهَ إِلَّا ٱللَّهُ يَسْتَكْبِرُونَ ۝٣٥

    یہ وه (لوگ) ہیں کہ جب ان سے کہا جاتا ہے کہ اللہ کے سوا کوئی معبود نہیں تو یہ سرکشی کرتے تھے

  36. وَيَقُولُونَ أَئِنَّا لَتَارِكُوٓاْ ءَالِهَتِنَا لِشَاعِرٍ مَّجْنُونِۭ ۝٣٦

    اور کہتے تھے کہ کیا ہم اپنے معبودوں کو ایک دیوانے شاعر کی بات پر چھوڑ دیں؟

  37. بَلْ جَآءَ بِٱلْحَقِّ وَصَدَّقَ ٱلْمُرْسَلِينَ ۝٣٧

    (نہیں نہیں) بلکہ (نبی) تو حق (سچا دین) ﻻئے ہیں اور سب رسولوں کو سچا جانتے ہیں

  38. إِنَّكُمْ لَذَآئِقُواْ ٱلْعَذَابِ ٱلْأَلِيمِ ۝٣٨

    یقیناً تم دردناک عذاب (کا مزه) چکھنے والے ہو

  39. وَمَا تُجْزَوْنَ إِلَّا مَا كُنتُمْ تَعْمَلُونَ ۝٣٩

    تمہیں اسی کا بدلہ دیا جائے گا جو تم کرتے تھے

  40. إِلَّا عِبَادَ ٱللَّهِ ٱلْمُخْلَصِينَ ۝٤٠

    مگر اللہ تعالیٰ کے خالص برگزیده بندے

  41. أُوْلَـٰٓئِكَ لَهُمْ رِزْقٌ مَّعْلُومٌ ۝٤١

    انہیں کے لئے مقرره روزی ہے

  42. فَوَٲكِهُ‌ۖ وَهُم مُّكْرَمُونَ ۝٤٢

    (ہر طرح کے) میوے، اور وه باعزت واکرام ہوں گے

  43. فِى جَنَّـٰتِ ٱلنَّعِيمِ ۝٤٣

    نعمتوں والی جنتوں میں

  44. عَلَىٰ سُرُرٍ مُّتَقَـٰبِلِينَ ۝٤٤

    تختوں پر ایک دوسرے کے سامنے (بیٹھے) ہوں گے

  45. يُطَافُ عَلَيْهِم بِكَأْسٍ مِّن مَّعِينِۭ ۝٤٥

    جاری شراب کے جام کا ان پر دور چل رہا ہوگا

  46. بَيْضَآءَ لَذَّةٍ لِّلشَّـٰرِبِينَ ۝٤٦

    جو صاف شفاف اور پینے میں لذیذ ہوگی

  47. لَا فِيهَا غَوْلٌ وَلَا هُمْ عَنْهَا يُنزَفُونَ ۝٤٧

    نہ اس سے درد سر ہو اور نہ اس کے پینے سے بہکیں

  48. وَعِندَهُمْ قَـٰصِرَٲتُ ٱلطَّرْفِ عِينٌ ۝٤٨

    اور ان کے پاس نیچی نظروں، بڑی بڑی آنکھوں والی (حوریں) ہوں گی

  49. كَأَنَّهُنَّ بَيْضٌ مَّكْنُونٌ ۝٤٩

    ایسی جیسے چھپائے ہوئے انڈے

  50. فَأَقْبَلَ بَعْضُهُمْ عَلَىٰ بَعْضٍ يَتَسَآءَلُونَ ۝٥٠

    (جنتی) ایک دوسرے کی طرف رخ کرکے پوچھیں گے

  51. قَالَ قَآئِلٌ مِّنْهُمْ إِنِّى كَانَ لِى قَرِينٌ ۝٥١

    ان میں سے ایک کہنے واﻻ کہے گا کہ میرا ایک ساتھی تھا

  52. يَقُولُ أَءِنَّكَ لَمِنَ ٱلْمُصَدِّقِينَ ۝٥٢

    جو (مجھ سے) کہا کرتا تھا کہ کیا تو (قیامت کے آنے کا) یقین کرنے والوں میں سے ہے؟

  53. أَءِذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَـٰمًا أَءِنَّا لَمَدِينُونَ ۝٥٣

    کیا جب کہ ہم مر کر مٹی اور ہڈی ہو جائیں گے کیا اس وقت ہم جزا دیئے جانے والے ہیں؟

  54. قَالَ هَلْ أَنتُم مُّطَّلِعُونَ ۝٥٤

    کہے گا تم چاہتے ہو کہ جھانک کر دیکھ لو؟

  55. فَٱطَّلَعَ فَرَءَاهُ فِى سَوَآءِ ٱلْجَحِيمِ ۝٥٥

    جھانکتے ہی اسے بیچوں بیچ جہنم میں (جلتا ہوا) دیکھے گا

  56. قَالَ تَٱللَّهِ إِن كِدتَّ لَتُرْدِينِ ۝٥٦

    کہے گا واللہ! قریب تھا کہ تو مجھے (بھی) برباد کر دے

  57. وَلَوْلَا نِعْمَةُ رَبِّى لَكُنتُ مِنَ ٱلْمُحْضَرِينَ ۝٥٧

    اگر میرے رب کا احسان نہ ہوتا تو میں بھی دوزخ میں حاضر کئے جانے والوں میں ہوتا

  58. أَفَمَا نَحْنُ بِمَيِّتِينَ ۝٥٨

    کیا (یہ صحیح ہے) کہ ہم مرنے والے ہی نہیں؟

  59. إِلَّا مَوْتَتَنَا ٱلْأُولَىٰ وَمَا نَحْنُ بِمُعَذَّبِينَ ۝٥٩

    بجز پہلی ایک موت کے، اور نہ ہم عذاب کیے جانے والے ہیں

  60. إِنَّ هَـٰذَا لَهُوَ ٱلْفَوْزُ ٱلْعَظِيمُ ۝٦٠

    پھر تو (ﻇاہر بات ہے کہ) یہ بڑی کامیابی ہے

  61. لِمِثْلِ هَـٰذَا فَلْيَعْمَلِ ٱلْعَـٰمِلُونَ ۝٦١

    ایسی (کامیابی) کے لئے عمل کرنے والوں کو عمل کرنا چاہئے

  62. أَذَٲلِكَ خَيْرٌ نُّزُلاً أَمْ شَجَرَةُ ٱلزَّقُّومِ ۝٦٢

    کیا یہ مہمانی اچھی ہے یا سینڈھ (زقوم) کا درخت؟

  63. إِنَّا جَعَلْنَـٰهَا فِتْنَةً لِّلظَّـٰلِمِينَ ۝٦٣

    جسے ہم نے ﻇالموں کے لئے سخت آزمائش بنا رکھا ہے

  64. إِنَّهَا شَجَرَةٌ تَخْرُجُ فِىٓ أَصْلِ ٱلْجَحِيمِ ۝٦٤

    بے شک وه درخت جہنم کی جڑ میں سے نکلتا ہے

  65. طَلْعُهَا كَأَنَّهُۥ رُءُوسُ ٱلشَّيَـٰطِينِ ۝٦٥

    جس کے خوشے شیطانوں کے سروں جیسے ہوتے ہیں

  66. فَإِنَّهُمْ لَأَكِلُونَ مِنْهَا فَمَالِــُٔونَ مِنْهَا ٱلْبُطُونَ ۝٦٦

    (جہنمی) اسی درخت سے کھائیں گے اور اسی سے پیٹ بھریں گے

  67. ثُمَّ إِنَّ لَهُمْ عَلَيْهَا لَشَوْبًا مِّنْ حَمِيمٍ ۝٦٧

    پھر اس پر گرم جلتے جلتے پانی کی ملونی ہوگی

  68. ثُمَّ إِنَّ مَرْجِعَهُمْ لَإِلَى ٱلْجَحِيمِ ۝٦٨

    پھر ان سب کا لوٹنا جہنم کی (آگ کے ڈھیرکی) طرف ہوگا

  69. إِنَّهُمْ أَلْفَوْاْ ءَابَآءَهُمْ ضَآلِّينَ ۝٦٩

    یقین مانو! کہ انہوں نے اپنے باپ دادا کو بہکا ہوا پایا

  70. فَهُمْ عَلَىٰٓ ءَاثَـٰرِهِمْ يُهْرَعُونَ ۝٧٠

    اور یہ انہی کے نشان قدم پر دوڑتے رہے

  71. وَلَقَدْ ضَلَّ قَبْلَهُمْ أَكْثَرُ ٱلْأَوَّلِينَ ۝٧١

    ان سے پہلے بھی بہت سے اگلے بہک چکے ہیں

  72. وَلَقَدْ أَرْسَلْنَا فِيهِم مُّنذِرِينَ ۝٧٢

    جن میں ہم نے ڈرانے والے (رسول) بھیجے تھے

  73. فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلْمُنذَرِينَ ۝٧٣

    اب تو دیکھ لے کہ جنہیں دھمکایا گیا تھا ان کا انجام کیسا کچھ ہوا

  74. إِلَّا عِبَادَ ٱللَّهِ ٱلْمُخْلَصِينَ ۝٧٤

    سوائے اللہ کے برگزیده بندوں کے

  75. وَلَقَدْ نَادَٮٰنَا نُوحٌ فَلَنِعْمَ ٱلْمُجِيبُونَ ۝٧٥

    اور ہمیں نوح (علیہ السلام) نے پکارا تو (دیکھ لو) ہم کیسے اچھے دعا قبول کرنے والے ہیں

  76. وَنَجَّيْنَـٰهُ وَأَهْلَهُۥ مِنَ ٱلْكَرْبِ ٱلْعَظِيمِ ۝٧٦

    ہم نے اسے اور اس کے گھر والوں کو اس زبردست مصیبت سے بچا لیا

  77. وَجَعَلْنَا ذُرِّيَّتَهُۥ هُمُ ٱلْبَاقِينَ ۝٧٧

    اور اس کی اوﻻد کو ہم نے باقی رہنے والی بنا دی

  78. وَتَرَكْنَا عَلَيْهِ فِى ٱلْأَخِرِينَ ۝٧٨

    اور ہم نے اس کا (ذکر خیر) پچھلوں میں باقی رکھا

  79. سَلَـٰمٌ عَلَىٰ نُوحٍ فِى ٱلْعَـٰلَمِينَ ۝٧٩

    نوح (علیہ السلام) پر تمام جہانوں میں سلام ہو

  80. إِنَّا كَذَٲلِكَ نَجْزِى ٱلْمُحْسِنِينَ ۝٨٠

    ہم نیکی کرنے والوں کو اسی طرح بدلہ دیتے ہیں

  81. إِنَّهُۥ مِنْ عِبَادِنَا ٱلْمُؤْمِنِينَ ۝٨١

    وه ہمارے ایمان والے بندوں میں سے تھا

  82. ثُمَّ أَغْرَقْنَا ٱلْأَخَرِينَ ۝٨٢

    پھر ہم نے دوسروں کو ڈبو دیا

  83. ۞ وَإِنَّ مِن شِيعَتِهِۦ لَإِبْرَٲهِيمَ ۝٨٣

    اور اس (نوح علیہ السلام کی) تابعداری کرنے والوں میں سے (ہی) ابراہیم (علیہ السلام بھی) تھے

  84. إِذْ جَآءَ رَبَّهُۥ بِقَلْبٍ سَلِيمٍ ۝٨٤

    جبکہ اپنے رب کے پاس بے عیب دل ﻻئے

  85. إِذْ قَالَ لِأَبِيهِ وَقَوْمِهِۦ مَاذَا تَعْبُدُونَ ۝٨٥

    انہوں نے اپنے باپ اور اپنی قوم سے کہا کہ تم کیا پوج رہے ہو؟

  86. أَئِفْكًا ءَالِهَةً دُونَ ٱللَّهِ تُرِيدُونَ ۝٨٦

    کیا تم اللہ کے سوا گھڑے ہوئے معبود چاہتے ہو؟

  87. فَمَا ظَنُّكُم بِرَبِّ ٱلْعَـٰلَمِينَ ۝٨٧

    تو یہ (بتلاؤ کہ) تم نے رب العالمین کو کیا سمجھ رکھا ہے؟

  88. فَنَظَرَ نَظْرَةً فِى ٱلنُّجُومِ ۝٨٨

    اب ابراہیم (علیہ السلام) نے ایک نگاه ستاروں کی طرف اٹھائی

  89. فَقَالَ إِنِّى سَقِيمٌ ۝٨٩

    اور کہا میں تو بیمار ہوں

  90. فَتَوَلَّوْاْ عَنْهُ مُدْبِرِينَ ۝٩٠

    اس پر وه سب اس سے منھ موڑے ہوئے واپس چلے گئے

  91. فَرَاغَ إِلَىٰٓ ءَالِهَتِهِمْ فَقَالَ أَلَا تَأْكُلُونَ ۝٩١

    آپ (چﭗ چپاتے) ان کے معبودوں کے پاس گئے اور فرمانے لگے تم کھاتے کیوں نہیں؟

  92. مَا لَكُمْ لَا تَنطِقُونَ ۝٩٢

    تمہیں کیا ہو گیا کہ بات تک نہیں کرتے ہو

  93. فَرَاغَ عَلَيْهِمْ ضَرْبَۢا بِٱلْيَمِينِ ۝٩٣

    پھر تو (پوری قوت کے ساتھ) دائیں ہاتھ سے انہیں مارنے پر پل پڑے

  94. فَأَقْبَلُوٓاْ إِلَيْهِ يَزِفُّونَ ۝٩٤

    وه (بت پرست) دوڑے بھاگے آپ کی طرف متوجہ ہوئے

  95. قَالَ أَتَعْبُدُونَ مَا تَنْحِتُونَ ۝٩٥

    تو آپ نے فرمایا تم انہیں پوجتے ہو جنہیں (خود) تم تراشتے ہو

  96. وَٱللَّهُ خَلَقَكُمْ وَمَا تَعْمَلُونَ ۝٩٦

    حاﻻنکہ تمہیں اور تمہاری بنائی ہوئی چیزوں کو اللہ ہی نے پیدا کیا ہے

  97. قَالُواْ ٱبْنُواْ لَهُۥ بُنْيَـٰنًا فَأَلْقُوهُ فِى ٱلْجَحِيمِ ۝٩٧

    وه کہنے لگے اس کے لئے ایک مکان بناؤ اور اس (دہکتی ہوئی) آگ میں اسے ڈال دو

  98. فَأَرَادُواْ بِهِۦ كَيْدًا فَجَعَلْنَـٰهُمُ ٱلْأَسْفَلِينَ ۝٩٨

    انہوں نے تو اس (ابراہیم علیہ السلام) کے ساتھ مکر کرنا چاہا لیکن ہم نے انہی کو نیچا کر دیا

  99. وَقَالَ إِنِّى ذَاهِبٌ إِلَىٰ رَبِّى سَيَهْدِينِ ۝٩٩

    اور اس (ابراہیم علیہ السلام) نے کہا میں تو ہجرت کر کے اپنے پروردگار کی طرف جانے واﻻ ہوں۔ وه ضرور میری رہنمائی کرے گا

  100. رَبِّ هَبْ لِى مِنَ ٱلصَّـٰلِحِينَ ۝١٠٠

    اے میرے رب! مجھے نیک بخت اوﻻد عطا فرما

  101. فَبَشَّرْنَـٰهُ بِغُلَـٰمٍ حَلِيمٍ ۝١٠١

    تو ہم نے اسے ایک بردبار بچے کی بشارت دی

  102. فَلَمَّا بَلَغَ مَعَهُ ٱلسَّعْىَ قَالَ يَـٰبُنَىَّ إِنِّىٓ أَرَىٰ فِى ٱلْمَنَامِ أَنِّىٓ أَذْبَحُكَ فَٱنظُرْ مَاذَا تَرَىٰ‌ۚ قَالَ يَـٰٓأَبَتِ ٱفْعَلْ مَا تُؤْمَرُ‌ۖ سَتَجِدُنِىٓ إِن شَآءَ ٱللَّهُ مِنَ ٱلصَّـٰبِرِينَ ۝١٠٢

    پھر جب وه (بچہ) اتنی عمر کو پہنچا کہ اس کے ساتھ چلے پھرے، تو اس (ابراہیم علیہ السلام) نے کہا میرے پیارے بچے! میں خواب میں اپنے آپ کو تجھے ذبح کرتے ہوئے دیکھ رہا ہوں۔ اب تو بتا کہ تیری کیا رائے ہے؟ بیٹے نے جواب دیا کہ ابا! جو حکم ہوا ہے اسے بجا ﻻئیے انشاءاللہ آپ مجھے صبر کرنے والوں میں سے پائیں گے

  103. فَلَمَّآ أَسْلَمَا وَتَلَّهُۥ لِلْجَبِينِ ۝١٠٣

    غرض جب دونوں مطیع ہوگئے اور اس نے (باپ نے) اس کو (بیٹے کو) پیشانی کے بل گرا دیا

  104. وَنَـٰدَيْنَـٰهُ أَن يَـٰٓإِبْرَٲهِيمُ ۝١٠٤

    تو ہم نے آواز دی کہ اے ابراہیم!

  105. قَدْ صَدَّقْتَ ٱلرُّءْيَآ‌ۚ إِنَّا كَذَٲلِكَ نَجْزِى ٱلْمُحْسِنِينَ ۝١٠٥

    یقیناً تو نے اپنے خواب کو سچا کر دکھایا، بیشک ہم نیکی کرنے والوں کو اسی طرح جزا دیتے ہیں

  106. إِنَّ هَـٰذَا لَهُوَ ٱلْبَلَـٰٓؤُاْ ٱلْمُبِينُ ۝١٠٦

    درحقیقت یہ کھلا امتحان تھا

  107. وَفَدَيْنَـٰهُ بِذِبْحٍ عَظِيمٍ ۝١٠٧

    اور ہم نے ایک بڑا ذبیحہ اس کے فدیہ میں دے دیا

  108. وَتَرَكْنَا عَلَيْهِ فِى ٱلْأَخِرِينَ ۝١٠٨

    اور ہم نے ان کا ذکر خیر پچھلوں میں باقی رکھا

  109. سَلَـٰمٌ عَلَىٰٓ إِبْرَٲهِيمَ ۝١٠٩

    ابراہیم (علیہ السلام) پر سلام ہو

  110. كَذَٲلِكَ نَجْزِى ٱلْمُحْسِنِينَ ۝١١٠

    ہم نیکو کاروں کو اسی طرح بدلہ دیتے ہیں

  111. إِنَّهُۥ مِنْ عِبَادِنَا ٱلْمُؤْمِنِينَ ۝١١١

    بیشک وه ہمارے ایمان دار بندوں میں سے تھا

  112. وَبَشَّرْنَـٰهُ بِإِسْحَـٰقَ نَبِيًّا مِّنَ ٱلصَّـٰلِحِينَ ۝١١٢

    اور ہم نے اس کو اسحاق (علیہ السلام) نبی کی بشارت دی جو صالح لوگوں میں سے ہوگا

  113. وَبَـٰرَكْنَا عَلَيْهِ وَعَلَىٰٓ إِسْحَـٰقَ‌ۚ وَمِن ذُرِّيَّتِهِمَا مُحْسِنٌ وَظَالِمٌ لِّنَفْسِهِۦ مُبِينٌ ۝١١٣

    اور ہم نے ابراہیم واسحاق (علیہ السلام) پر برکتیں نازل فرمائیں، اور ان دونوں کی اوﻻد میں بعضے تو نیک بخت ہیں اور بعض اپنے نفس پر صریح ﻇلم کرنے والے ہیں

  114. وَلَقَدْ مَنَنَّا عَلَىٰ مُوسَىٰ وَهَـٰرُونَ ۝١١٤

    یقیناً ہم نے موسیٰ اور ہارون (علیہ السلام) پر بڑا احسان کیا

  115. وَنَجَّيْنَـٰهُمَا وَقَوْمَهُمَا مِنَ ٱلْكَرْبِ ٱلْعَظِيمِ ۝١١٥

    اور انہیں اور ان کی قوم کو بہت بڑے دکھ درد سے نجات دے دی

  116. وَنَصَرْنَـٰهُمْ فَكَانُواْ هُمُ ٱلْغَـٰلِبِينَ ۝١١٦

    اور ان کی مدد کی تو وہی غالب رہے

  117. وَءَاتَيْنَـٰهُمَا ٱلْكِتَـٰبَ ٱلْمُسْتَبِينَ ۝١١٧

    اور ہم نے انہیں (واضح اور) روشن کتاب دی

  118. وَهَدَيْنَـٰهُمَا ٱلصِّرَٲطَ ٱلْمُسْتَقِيمَ ۝١١٨

    اور انہیں سیدھے راستہ پرقائم رکھا

  119. وَتَرَكْنَا عَلَيْهِمَا فِى ٱلْأَخِرِينَ ۝١١٩

    اور ہم نے ان دونوں کے لئے پیچھے آنے والوں میں یہ بات باقی رکھی

  120. سَلَـٰمٌ عَلَىٰ مُوسَىٰ وَهَـٰرُونَ ۝١٢٠

    کہ موسیٰ اور ہارون (علیہما السلام) پر سلام ہو

  121. إِنَّا كَذَٲلِكَ نَجْزِى ٱلْمُحْسِنِينَ ۝١٢١

    بے شک ہم نیک لوگوں کو اسی طرح بدلے دیا کرتے ہیں

  122. إِنَّهُمَا مِنْ عِبَادِنَا ٱلْمُؤْمِنِينَ ۝١٢٢

    یقیناً یہ دونوں ہمارے مومن بندوں میں سے تھے

  123. وَإِنَّ إِلْيَاسَ لَمِنَ ٱلْمُرْسَلِينَ ۝١٢٣

    بے شک الیاس (علیہ السلام) بھی پیغمبروں میں سے تھے

  124. إِذْ قَالَ لِقَوْمِهِۦٓ أَلَا تَتَّقُونَ ۝١٢٤

    جب کہ انہوں نے اپنی قوم سے فرمایا کہ تم اللہ سے ڈرتے نہیں ہو؟

  125. أَتَدْعُونَ بَعْلاً وَتَذَرُونَ أَحْسَنَ ٱلْخَـٰلِقِينَ ۝١٢٥

    کیا تم بعل (نامی بت) کو پکارتے ہو؟ اور سب سے بہتر خالق کو چھوڑ دیتے ہو؟

  126. ٱللَّهَ رَبَّكُمْ وَرَبَّ ءَابَآئِكُمُ ٱلْأَوَّلِينَ ۝١٢٦

    اللہ جو تمہارا اور تمہارے اگلے تمام باپ دادوں کا رب ہے

  127. فَكَذَّبُوهُ فَإِنَّهُمْ لَمُحْضَرُونَ ۝١٢٧

    لیکن قوم نے انہیں جھٹلایا، پس وه ضرور (عذاب میں) حاضر رکھے جائیں گے

  128. إِلَّا عِبَادَ ٱللَّهِ ٱلْمُخْلَصِينَ ۝١٢٨

    سوائے اللہ تعالی کے مخلص بندوں کے

  129. وَتَرَكْنَا عَلَيْهِ فِى ٱلْأَخِرِينَ ۝١٢٩

    ہم نے (الیاس علیہ السلام) کا ذکر خیر پچھلوں میں بھی باقی رکھا

  130. سَلَـٰمٌ عَلَىٰٓ إِلْ يَاسِينَ ۝١٣٠

    کہ الیاس پر سلام ہو

  131. إِنَّا كَذَٲلِكَ نَجْزِى ٱلْمُحْسِنِينَ ۝١٣١

    ہم نیکی کرنے والوں کو اسی طرح بدلہ دیتے ہیں

  132. إِنَّهُۥ مِنْ عِبَادِنَا ٱلْمُؤْمِنِينَ ۝١٣٢

    بیشک وه ہمارے ایمان دار بندوں میں سے تھے

  133. وَإِنَّ لُوطًا لَّمِنَ ٱلْمُرْسَلِينَ ۝١٣٣

    بیشک لوط (علیہ السلام بھی) پیغمبروں میں سے تھے

  134. إِذْ نَجَّيْنَـٰهُ وَأَهْلَهُۥٓ أَجْمَعِينَ ۝١٣٤

    ہم نے انہیں اور ان کے گھر والوں کو سب کو نجات دی

  135. إِلَّا عَجُوزًا فِى ٱلْغَـٰبِرِينَ ۝١٣٥

    بجز اس بڑھیا کے جو پیچھے ره جانے والوں میں سے ره گئی

  136. ثُمَّ دَمَّرْنَا ٱلْأَخَرِينَ ۝١٣٦

    پھر ہم نے اوروں کو ہلاک کر دیا

  137. وَإِنَّكُمْ لَتَمُرُّونَ عَلَيْهِم مُّصْبِحِينَ ۝١٣٧

    اور تم تو صبح ہونے پر ان کی بستیوں کے پاس سے گزرتے ہو

  138. وَبِٱلَّيْلِ‌ۗ أَفَلَا تَعْقِلُونَ ۝١٣٨

    اور رات کو بھی، کیا پھر بھی نہیں سمجھتے؟

  139. وَإِنَّ يُونُسَ لَمِنَ ٱلْمُرْسَلِينَ ۝١٣٩

    اور بلاشبہ یونس (علیہ السلام) نبیوں میں سے تھے

  140. إِذْ أَبَقَ إِلَى ٱلْفُلْكِ ٱلْمَشْحُونِ ۝١٤٠

    جب بھاگ کر پہنچے بھری کشتی پر

  141. فَسَاهَمَ فَكَانَ مِنَ ٱلْمُدْحَضِينَ ۝١٤١

    پھر قرعہ اندازی ہوئی تو یہ مغلوب ہوگئے

  142. فَٱلْتَقَمَهُ ٱلْحُوتُ وَهُوَ مُلِيمٌ ۝١٤٢

    تو پھر انہیں مچھلی نے نگل لیا اور وه خود اپنے آپ کو ملامت کرنے لگ گئے

  143. فَلَوْلَآ أَنَّهُۥ كَانَ مِنَ ٱلْمُسَبِّحِينَ ۝١٤٣

    پس اگر یہ پاکی بیان کرنے والوں میں سے نہ ہوتے

  144. لَلَبِثَ فِى بَطْنِهِۦٓ إِلَىٰ يَوْمِ يُبْعَثُونَ ۝١٤٤

    تو لوگوں کے اٹھائے جانے کے دن تک اس کے پیٹ میں ہی رہتے

  145. ۞ فَنَبَذْنَـٰهُ بِٱلْعَرَآءِ وَهُوَ سَقِيمٌ ۝١٤٥

    پس انہیں ہم نے چٹیل میدان میں ڈال دیا اور وه اس وقت بیمار تھے

  146. وَأَنۢبَتْنَا عَلَيْهِ شَجَرَةً مِّن يَقْطِينٍ ۝١٤٦

    اور ان پر سایہ کرنے واﻻ ایک بیل دار درخت ہم نے اگا دیا

  147. وَأَرْسَلْنَـٰهُ إِلَىٰ مِاْئَةِ أَلْفٍ أَوْ يَزِيدُونَ ۝١٤٧

    اور ہم نے انہیں ایک لاکھ بلکہ اور زیاده آدمیوں کی طرف بھیجا

  148. فَـَٔـامَنُواْ فَمَتَّعْنَـٰهُمْ إِلَىٰ حِينٍ ۝١٤٨

    پس وه ایمان ﻻئے، اور ہم نے انہیں ایک زمانہ تک عیش وعشرت دی

  149. فَٱسْتَفْتِهِمْ أَلِرَبِّكَ ٱلْبَنَاتُ وَلَهُمُ ٱلْبَنُونَ ۝١٤٩

    ان سے دریافت کیجئے! کہ کیا آپ کے رب کی تو بیٹیاں ہیں اور ان کے بیٹے ہیں؟

  150. أَمْ خَلَقْنَا ٱلْمَلَـٰٓئِكَةَ إِنَـٰثًا وَهُمْ شَـٰهِدُونَ ۝١٥٠

    یا یہ اس وقت موجود تھے جبکہ ہم نے فرشتوں کو مؤنﺚ پیدا کیا

  151. أَلَآ إِنَّهُم مِّنْ إِفْكِهِمْ لَيَقُولُونَ ۝١٥١

    آگاه رہو! کہ یہ لوگ صرف اپنی افترا پردازی سے کہہ رہے ہیں

  152. وَلَدَ ٱللَّهُ وَإِنَّهُمْ لَكَـٰذِبُونَ ۝١٥٢

    کہ اللہ تعالی کی اوﻻد ہے۔ یقیناً یہ محض جھوٹے ہیں

  153. أَصْطَفَى ٱلْبَنَاتِ عَلَى ٱلْبَنِينَ ۝١٥٣

    کیا اللہ تعالی نے اپنے لیے بیٹیوں کو بیٹوں پر ترجیح دی

  154. مَا لَكُمْ كَيْفَ تَحْكُمُونَ ۝١٥٤

    تمہیں کیا ہو گیا ہے کیسے حکم لگاتے پھرتے ہو؟

  155. أَفَلَا تَذَكَّرُونَ ۝١٥٥

    کیا تم اس قدر بھی نہیں سمجھتے؟

  156. أَمْ لَكُمْ سُلْطَـٰنٌ مُّبِينٌ ۝١٥٦

    یا تمہارے پاس اس کی کوئی صاف دلیل ہے

  157. فَأْتُواْ بِكِتَـٰبِكُمْ إِن كُنتُمْ صَـٰدِقِينَ ۝١٥٧

    تو جاؤ اگر سچے ہو تو اپنی ہی کتاب لے آؤ

  158. وَجَعَلُواْ بَيْنَهُۥ وَبَيْنَ ٱلْجِنَّةِ نَسَبًا‌ۚ وَلَقَدْ عَلِمَتِ ٱلْجِنَّةُ إِنَّهُمْ لَمُحْضَرُونَ ۝١٥٨

    اور ان لوگوں نے تو اللہ کے اور جنات کے درمیان بھی قرابت داری ٹھہرائی ہے، اور حاﻻنکہ خود جنات کو معلوم ہے کہ وه (اس عقیده کے لوگ عذاب کے سامنے) پیش کیے جائیں گے

  159. سُبْحَـٰنَ ٱللَّهِ عَمَّا يَصِفُونَ ۝١٥٩

    جو کچھ یہ (اللہ کے بارے میں) بیان کر رہے ہیں اس سے اللہ تعالیٰ بالکل پاک ہے

  160. إِلَّا عِبَادَ ٱللَّهِ ٱلْمُخْلَصِينَ ۝١٦٠

    سوائے! اللہ کے مخلص بندوں کے

  161. فَإِنَّكُمْ وَمَا تَعْبُدُونَ ۝١٦١

    یقین مانو کہ تم سب اور تمہارے معبودان (باطل)

  162. مَآ أَنتُمْ عَلَيْهِ بِفَـٰتِنِينَ ۝١٦٢

    کسی ایک کو بھی بہکا نہیں سکتے

  163. إِلَّا مَنْ هُوَ صَالِ ٱلْجَحِيمِ ۝١٦٣

    بجز اس کے جو جہنمی ہی ہے

  164. وَمَا مِنَّآ إِلَّا لَهُۥ مَقَامٌ مَّعْلُومٌ ۝١٦٤

    (فرشتوں کا قول ہے کہ) ہم میں سے تو ہر ایک کی جگہ مقرر ہے

  165. وَإِنَّا لَنَحْنُ ٱلصَّآفُّونَ ۝١٦٥

    اور ہم تو (بندگیٴ الٰہی میں) صف بستہ کھڑے ہیں

  166. وَإِنَّا لَنَحْنُ ٱلْمُسَبِّحُونَ ۝١٦٦

    اور اس کی تسبیح بیان کر رہے ہیں

  167. وَإِن كَانُواْ لَيَقُولُونَ ۝١٦٧

    کفار تو کہا کرتے تھے

  168. لَوْ أَنَّ عِندَنَا ذِكْرًا مِّنَ ٱلْأَوَّلِينَ ۝١٦٨

    کہ اگر ہمارے سامنے اگلے لوگوں کا ذکر ہوتا

  169. لَكُنَّا عِبَادَ ٱللَّهِ ٱلْمُخْلَصِينَ ۝١٦٩

    تو ہم بھی اللہ کے چیده بندے بن جاتے

  170. فَكَفَرُواْ بِهِۦ‌ۖ فَسَوْفَ يَعْلَمُونَ ۝١٧٠

    لیکن پھر اس قرآن کے ساتھ کفر کر گئے، پس اب عنقریب جان لیں گے

  171. وَلَقَدْ سَبَقَتْ كَلِمَتُنَا لِعِبَادِنَا ٱلْمُرْسَلِينَ ۝١٧١

    اور البتہ ہمارا وعده پہلے ہی اپنے رسولوں کے لئے صادر ہو چکا ہے

  172. إِنَّهُمْ لَهُمُ ٱلْمَنصُورُونَ ۝١٧٢

    کہ یقیناً وه ہی مدد کیے جائیں گے

  173. وَإِنَّ جُندَنَا لَهُمُ ٱلْغَـٰلِبُونَ ۝١٧٣

    اور ہمارا ہی لشکر غالب (اور برتر) رہے گا

  174. فَتَوَلَّ عَنْهُمْ حَتَّىٰ حِينٍ ۝١٧٤

    اب آپ کچھ دنوں تک ان سے منھ پھیر لیجئے

  175. وَأَبْصِرْهُمْ فَسَوْفَ يُبْصِرُونَ ۝١٧٥

    اور انہیں دیکھتے رہیئے، اور یہ بھی آگے چل کر دیکھ لیں گے

  176. أَفَبِعَذَابِنَا يَسْتَعْجِلُونَ ۝١٧٦

    کیا یہ ہمارے عذاب کی جلدی مچا رہے ہیں؟

  177. فَإِذَا نَزَلَ بِسَاحَتِهِمْ فَسَآءَ صَبَاحُ ٱلْمُنذَرِينَ ۝١٧٧

    سنو! جب ہمارا عذاب ان کے میدان میں اتر آئے گا اس وقت ان کی جن کو متنبہ کر دیا گیا تھا بڑی بری صبح ہوگی

  178. وَتَوَلَّ عَنْهُمْ حَتَّىٰ حِينٍ ۝١٧٨

    آپ کچھ وقت تک ان کا خیال چھوڑ دیجئے

  179. وَأَبْصِرْ فَسَوْفَ يُبْصِرُونَ ۝١٧٩

    اور دیکھتے رہئیے یہ بھی ابھی ابھی دیکھ لیں گے

  180. سُبْحَـٰنَ رَبِّكَ رَبِّ ٱلْعِزَّةِ عَمَّا يَصِفُونَ ۝١٨٠

    پاک ہے آپ کا رب جو بہت بڑی عزت واﻻ ہے ہر اس چیز سے (جو مشرک) بیان کرتے ہیں

  181. وَسَلَـٰمٌ عَلَى ٱلْمُرْسَلِينَ ۝١٨١

    پیغمبروں پر سلام ہے

  182. وَٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ۝١٨٢

    اور سب طرح کی تعریف اللہ کے لئے ہے جو سارے جہان کا رب ہے
